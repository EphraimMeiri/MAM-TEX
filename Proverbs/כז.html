מקרא על פי המסורה

ספר משלי

פרק כז

כזאא אַֽל־תִּ֭תְהַלֵּל בְּי֣וֹם מָחָ֑ר
כִּ֤י לֹֽא־תֵ֝דַ֗ע מַה־יֵּ֥לֶד יֽוֹם׃
בב יְהַלֶּלְךָ֣ זָ֣ר וְלֹא־פִ֑יךָ
נׇ֝כְרִ֗י וְאַל־שְׂפָתֶֽיךָ׃
גג כֹּֽבֶד־אֶ֭בֶן וְנֵ֣טֶל הַח֑וֹל
וְכַ֥עַס אֱ֝וִ֗יל כָּבֵ֥ד מִשְּׁנֵיהֶֽם׃
דד אַכְזְרִיּ֣וּת חֵ֭מָה וְשֶׁ֣טֶף אָ֑ף
וּמִ֥י יַ֝עֲמֹ֗ד לִפְנֵ֥י קִנְאָֽה׃
הה ט֭וֹבָה תּוֹכַ֣חַת מְגֻלָּ֑ה
מֵאַהֲבָ֥ה מְסֻתָּֽרֶת׃
וו נֶ֭אֱמָנִים פִּצְעֵ֣י אוֹהֵ֑ב
וְ֝נַעְתָּר֗וֹת נְשִׁיק֥וֹת שׂוֹנֵֽא׃
זז נֶ֣פֶשׁ שְׂ֭בֵעָה תָּב֣וּס נֹ֑פֶת
וְנֶ֥פֶשׁ רְ֝עֵבָ֗ה כׇּל־מַ֥ר מָתֽוֹק׃
חח כְּ֭צִפּוֹר נוֹדֶ֣דֶת מִן־קִנָּ֑הּ
כֵּֽן־אִ֝֗ישׁנכֵּֽן־אִ֝֗ישׁ · א=כֵּֽן־אִ֝ישׁ (השמטת נקודת הרביע) נוֹדֵ֥ד מִמְּקוֹמֽוֹ׃
טט שֶׁ֣מֶן וּ֭קְטֹרֶת יְשַׂמַּֽח־לֵ֑ב
וּמֶ֥תֶק רֵ֝עֵ֗הוּ מֵעֲצַת־נָֽפֶשׁ׃
יי רֵ֥עֲךָ֨ (ורעה)קקקרי/כתיב · (ורעה) וְרֵ֪עַ אָבִ֡יךָ אַֽל־תַּעֲזֹ֗ב
וּבֵ֥ית אָחִ֗יךָ אַל־תָּ֭בוֹא בְּי֣וֹם אֵידֶ֑ךָ
ט֥וֹב שָׁכֵ֥ן קָ֝ר֗וֹבמֵאָ֥ח רָחֽוֹק׃
יאיא חֲכַ֣ם בְּ֭נִי וְשַׂמַּ֣ח לִבִּ֑י
וְאָשִׁ֖יבָה חֹרְפִ֣י דָבָֽר׃
יביב עָר֤וּם ׀ רָאָ֣ה רָעָ֣ה נִסְתָּ֑ר
פְּ֝תָאיִ֗ם עָבְר֥וּ נֶעֱנָֽשׁוּ׃
יגיג קַח־בִּ֭גְדוֹ כִּי־עָ֣רַב זָ֑ר
וּבְעַ֖ד נׇכְרִיָּ֣ה חַבְלֵֽהוּ׃
ידיד מְבָ֘רֵ֤ךְ רֵעֵ֨הוּ ׀ בְּק֣וֹל גָּ֭דוֹל בַּבֹּ֣קֶר הַשְׁכֵּ֑ים
קְ֝לָלָ֗ה תֵּחָ֥שֶׁב לֽוֹ׃
טוטו דֶּ֣לֶף ט֭וֹרֵד בְּי֣וֹם סַגְרִ֑יר
וְאֵ֥שֶׁת מִ֝דְויָנִ֗יםקקקרי/כתיב · (מדונים) מִ֝דְיָנִ֗ים נִשְׁתָּוָֽה׃
טזטז צֹפְנֶ֥יהָ צָפַן־ר֑וּחַ
וְשֶׁ֖מֶן יְמִינ֣וֹ יִקְרָֽא׃
יזיז בַּרְזֶ֣ל בְּבַרְזֶ֣ל יָ֑חַד
וְ֝אִ֗ישׁ יַ֣חַד פְּנֵֽי־רֵעֵֽהוּ׃
יחיח נֹצֵ֣ר תְּ֭אֵנָה יֹאכַ֣ל פִּרְיָ֑הּ
וְשֹׁמֵ֖ר אֲדֹנָ֣יו יְכֻבָּֽד׃
יטיט כַּ֭מַּיִם הַפָּנִ֣ים לַפָּנִ֑ים
כֵּ֤ן לֵֽב־הָ֝אָדָ֗ם לָאָדָֽם׃
ככ שְׁא֣וֹל וַ֭אֲבַדֹּה לֹ֣א תִשְׂבַּ֑עְנָה
וְעֵינֵ֥י הָ֝אָדָ֗ם לֹ֣א תִשְׂבַּֽעְנָה׃
כאכא מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב
וְ֝אִ֗ישׁ לְפִ֣י מַהֲלָלֽוֹ׃
כבכב אִ֥ם תִּכְתּֽוֹשׁ־אֶת־הָאֱוִ֨יל ׀ בַּ֥מַּכְתֵּ֡שׁ
בְּת֣וֹךְ הָ֭רִיפוֹת בַּעֱלִ֑י
לֹא־תָס֥וּר מֵ֝עָלָ֗יואִוַּלְתּֽוֹ׃
{פ}　　
כגכג יָדֹ֣עַ תֵּ֭דַע פְּנֵ֣י צֹאנֶ֑ךָ
שִׁ֥ית לִ֝בְּךָ֗ לַעֲדָרִֽים׃
כדכד כִּ֤י לֹ֣א לְעוֹלָ֣ם חֹ֑סֶן
וְאִם־נֵ֝֗זֶר לְד֣וֹר (דור)קקקרי/כתיב · (דור) וָדֽוֹר׃
כהכה גָּלָ֣ה חָ֭צִיר וְנִרְאָה־דֶ֑שֶׁא
וְ֝נֶאֶסְפ֗וּ עִשְּׂב֥וֹת הָרִֽים׃
כוכו כְּבָשִׂ֥ים לִלְבוּשֶׁ֑ךָ
וּמְחִ֥יר שָׂ֝דֶ֗ה עַתּוּדִֽים׃
כזכז וְדֵ֤י ׀ חֲלֵ֬ב עִזִּ֗ים
לְֽ֭לַחְמְךָ לְלֶ֣חֶם בֵּיתֶ֑ךָ
וְ֝חַיִּ֗יםלְנַעֲרוֹתֶֽיךָ׃
