כגאא כִּֽי־תֵ֭שֵׁב לִלְח֣וֹם אֶת־מוֹשֵׁ֑ל
בִּ֥ין תָּ֝בִ֗ין אֶת־אֲשֶׁ֥ר לְפָנֶֽיךָ׃
בב וְשַׂמְתָּ֣ שַׂכִּ֣ין בְּלֹעֶ֑ךָ
אִם־בַּ֖עַל נֶ֣פֶשׁ אָֽתָּה׃
גג אַל־תִּ֭תְאָו לְמַטְעַמּוֹתָ֑יו
וְ֝ה֗וּא לֶ֣חֶם כְּזָבִֽים׃
דד אַל־תִּיגַ֥ע לְֽהַעֲשִׁ֑יר
מִֽבִּינָתְךָ֥ חֲדָֽל׃
הה הֲתָ֤עִויףקקקרי/כתיב · (התעוף) הֲתָ֤עִיף עֵינֶ֥יךָ בּ֗וֹוְֽאֵ֫ינֶ֥נּוּ
כִּ֤י עָשֹׂ֣ה יַעֲשֶׂה־לּ֣וֹ כְנָפַ֑יִם
כְּ֝נֶ֗שֶׁר (ועיף)קקקרי/כתיב · (ועיף) יָע֥וּף הַשָּׁמָֽיִם׃

חח פִּֽתְּךָ־אָכַ֥לְתָּ תְקִיאֶ֑נָּה
וְ֝שִׁחַ֗תָּ דְּבָרֶ֥יךָ הַנְּעִימִֽים׃
טט בְּאׇזְנֵ֣י כְ֭סִיל אַל־תְּדַבֵּ֑ר
כִּי־יָ֝ב֗וּז לְשֵׂ֣כֶל מִלֶּֽיךָ׃
יי אַל־תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם
וּבִשְׂדֵ֥י יְ֝תוֹמִ֗ים אַל־תָּבֹֽא׃
יאיא כִּֽי־גֹאֲלָ֥ם חָזָ֑ק
הֽוּא־יָרִ֖יב אֶת־רִיבָ֣ם אִתָּֽךְ׃
יביב הָבִ֣יאָה לַמּוּסָ֣ר לִבֶּ֑ךָ
וְ֝אׇזְנֶ֗ךָ לְאִמְרֵי־דָֽעַת׃
יגיג אַל־תִּמְנַ֣ע מִנַּ֣עַר מוּסָ֑ר
כִּֽי־תַכֶּ֥נּוּ בַ֝שֵּׁ֗בֶט לֹ֣א יָמֽוּת׃
ידיד אַ֭תָּה בַּשֵּׁ֣בֶט תַּכֶּ֑נּוּ
וְ֝נַפְשׁ֗וֹ מִשְּׁא֥וֹל תַּצִּֽיל׃
טוטו בְּ֭נִי אִם־חָכַ֣ם לִבֶּ֑ךָ
יִשְׂמַ֖ח לִבִּ֣י גַם־אָֽנִי׃
טזטז וְתַעְלֹ֥זְנָה כִלְיוֹתָ֑י
בְּדַבֵּ֥ר שְׂ֝פָתֶ֗יךָ מֵישָׁרִֽים׃
יזיז אַל־יְקַנֵּ֣א לִ֭בְּךָ בַּחַטָּאִ֑ים
כִּ֥י אִם־בְּיִרְאַת־יְ֝הֹוָ֗ה כׇּל־הַיּֽוֹם׃
יחיח כִּ֭י אִם־יֵ֣שׁ אַחֲרִ֑ית
וְ֝תִקְוָתְךָ֗ לֹ֣א תִכָּרֵֽת׃
יטיט שְׁמַע־אַתָּ֣ה בְנִ֣י וַחֲכָ֑ם
וְאַשֵּׁ֖ר בַּדֶּ֣רֶךְ לִבֶּֽךָ׃
