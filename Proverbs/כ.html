ספר משלי

פרק כ

כאא לֵ֣ץ הַ֭יַּיִן הֹמֶ֣ה שֵׁכָ֑ר
וְכׇל־שֹׁ֥גֶה בּ֝֗וֹנבּ֝֗וֹ · א=בּ֝וֹ (השמטת נקודת הרביע) לֹ֣א יֶחְכָּֽם׃
בב נַ֣הַם כַּ֭כְּפִיר אֵ֣ימַת מֶ֑לֶךְ
מִ֝תְעַבְּר֗וֹ חוֹטֵ֥א נַפְשֽׁוֹ׃
גג כָּב֣וֹד לָ֭אִישׁ שֶׁ֣בֶת מֵרִ֑יב
וְכׇל־אֱ֝וִ֗יל יִתְגַּלָּֽע׃
דד מֵ֭חֹרֶף עָצֵ֣ל לֹא־יַחֲרֹ֑שׁ
יוְשָׁאַ֖לקקקרי/כתיב · (ישאל) וְשָׁאַ֖ל בַּקָּצִ֣יר וָאָֽיִן׃
הה מַ֣יִם עֲ֭מֻקִּים עֵצָ֣ה בְלֶב־אִ֑ישׁ
וְאִ֖ישׁ תְּבוּנָ֣ה יִדְלֶֽנָּה׃
וו רׇב־אָדָ֗ם יִ֭קְרָא אִ֣ישׁ חַסְדּ֑וֹ
וְאִ֥ישׁ אֱ֝מוּנִ֗ים מִ֣י יִמְצָֽא׃
זז מִתְהַלֵּ֣ךְ בְּתֻמּ֣וֹ צַדִּ֑יק
אַשְׁרֵ֖י בָנָ֣יו אַחֲרָֽיו׃
חח מֶ֗לֶךְ יוֹשֵׁ֥ב עַל־כִּסֵּא־דִ֑ין
מְזָרֶ֖ה בְעֵינָ֣יו כׇּל־רָֽע׃
טט מִֽי־יֹ֭אמַר זִכִּ֣יתִי לִבִּ֑י
טָ֝הַ֗רְתִּי מֵחַטָּאתִֽי׃
יי אֶ֣בֶן וָ֭אֶבֶן אֵיפָ֣ה וְאֵיפָ֑ה
תּוֹעֲבַ֥ת יְ֝הֹוָ֗ה גַּם־שְׁנֵיהֶֽם׃
יאיא גַּ֣ם בְּ֭מַעֲלָלָיו יִתְנַכֶּר־נָ֑עַר
אִם־זַ֖ךְ וְאִם־יָשָׁ֣ר ׃
יביב אֹ֣זֶן שֹׁ֭מַעַת וְעַ֣יִן רֹאָ֑ה
יְ֝הֹוָ֗ה עָשָׂ֥ה גַם־שְׁנֵיהֶֽם׃
יגיג אַל־תֶּאֱהַ֣ב שֵׁ֭נָה פֶּן־תִּוָּרֵ֑שׁ
פְּקַ֖ח עֵינֶ֣יךָ שְֽׂבַֽע־לָֽחֶם׃
ידיד רַ֣ע רַ֭ע יֹאמַ֣ר הַקּוֹנֶ֑ה
וְאֹזֵ֥ל ל֝֗וֹנל֝֗וֹ · א=ל֝וֹ (השמטת נקודת הרביע) אָ֣ז יִתְהַלָּֽל׃
טוטו יֵ֣שׁ זָ֭הָב וְרׇב־פְּנִינִ֑ים
וּכְלִ֥י יְ֝קָ֗ר שִׂפְתֵי־דָֽעַת׃
טזטז לְֽקַח־בִּ֭גְדוֹ כִּי־עָ֣רַב זָ֑ר
וּבְעַ֖ד (נכרים)קקקרי/כתיב · (נכרים) נׇכְרִיָּ֣ה חַבְלֵֽהוּ׃
יזיז עָרֵ֣ב לָ֭אִישׁ לֶ֣חֶם שָׁ֑קֶר
וְ֝אַחַ֗ר יִמָּֽלֵא־פִ֥יהוּ חָצָֽץ׃
יחיח מַ֭חֲשָׁבוֹת בְּעֵצָ֣ה תִכּ֑וֹן
וּ֝בְתַחְבֻּל֗וֹת עֲשֵׂ֣ה מִלְחָמָֽה׃
יטיט גּֽוֹלֶה־סּ֭וֹד הוֹלֵ֣ךְ רָכִ֑יל
וּלְפֹתֶ֥ה שְׂ֝פָתָ֗יו לֹ֣א תִתְעָרָֽב׃
ככ מְ֭קַלֵּל אָבִ֣יו וְאִמּ֑וֹ
יִֽדְעַ֥ךְ נֵ֝ר֗וֹ (באישון)קקקרי/כתיב · (באישון) בֶּאֱשׁ֥וּן חֹֽשֶׁךְ׃
כאכא נַ֭חֲלָה (מבחלת)קקקרי/כתיב · (מבחלת) מְבֹהֶ֣לֶת בָּרִאשׁוֹנָ֑ה
וְ֝אַחֲרִיתָ֗הּ לֹ֣א תְבֹרָֽךְ׃
כבכב אַל־תֹּאמַ֥ר אֲשַׁלְּמָה־רָ֑ע
קַוֵּ֥ה לַ֝יהֹוָ֗ה וְיֹ֣שַֽׁע לָֽךְ׃
כגכג תּוֹעֲבַ֣ת יְ֭הֹוָה אֶ֣בֶן וָאָ֑בֶן
וּמֹאזְנֵ֖י מִרְמָ֣ה לֹא־טֽוֹב׃
כדכד מֵיְהֹוָ֥ה מִצְעֲדֵי־גָ֑בֶר
וְ֝אָדָ֗ם מַה־יָּבִ֥ין דַּרְכּֽוֹ׃
כהכה מוֹקֵ֣שׁ אָ֭דָם יָ֣לַע קֹ֑דֶשׁ
וְאַחַ֖ר נְדָרִ֣ים לְבַקֵּֽר׃
כוכו מְזָרֶ֣ה רְ֭שָׁעִים מֶ֣לֶךְ חָכָ֑ם
וַיָּ֖שֶׁב עֲלֵיהֶ֣ם אוֹפָֽן׃
כזכז נֵ֣ר יְ֭הֹוָה נִשְׁמַ֣ת אָדָ֑ם
חֹ֝פֵ֗שׂ כׇּל־חַדְרֵי־בָֽטֶן׃
כחכח חֶ֣סֶד וֶ֭אֱמֶת יִצְּרוּ־מֶ֑לֶךְ
וְסָעַ֖ד בַּחֶ֣סֶד כִּסְאֽוֹ׃
כטכט תִּפְאֶ֣רֶת בַּחוּרִ֣ים כֹּחָ֑ם
וַהֲדַ֖ר זְקֵנִ֣ים שֵׂיבָֽה׃
לל חַבֻּר֣וֹת פֶּ֭צַע תַּמְר֣יוּקקקקרי/כתיב · (תמריק) תַּמְר֣וּק בְּרָ֑ע
וּ֝מַכּ֗וֹת חַדְרֵי־בָֽטֶן׃
