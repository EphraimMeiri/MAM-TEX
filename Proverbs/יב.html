מקרא על פי המסורה

ספר משלי

פרק יב

יבאא אֹהֵ֣ב מ֭וּסָר אֹ֣הֵֽב דָּ֑עַת
וְשׂוֹנֵ֖א תוֹכַ֣חַת בָּֽעַר׃
בב ט֗וֹב יָפִ֣יק רָ֭צוֹן מֵיְהֹוָ֑ה
וְאִ֖ישׁ מְזִמּ֣וֹת יַרְשִֽׁיעַ׃
גג לֹא־יִכּ֣וֹן אָדָ֣ם בְּרֶ֑שַׁע
וְשֹׁ֥רֶשׁ צַ֝דִּיקִ֗ים בַּל־יִמּֽוֹט׃
דד אֵֽשֶׁת־חַ֭יִל עֲטֶ֣רֶת בַּעְלָ֑הּ
וּכְרָקָ֖ב בְּעַצְמוֹתָ֣יו מְבִישָֽׁה׃
הה מַחְשְׁב֣וֹת צַדִּיקִ֣ים מִשְׁפָּ֑ט
תַּחְבֻּל֖וֹת רְשָׁעִ֣ים מִרְמָֽה׃
וו דִּבְרֵ֣י רְשָׁעִ֣ים אֱרׇב־דָּ֑ם
וּפִ֥י יְ֝שָׁרִ֗ים יַצִּילֵֽם׃
זז הָפ֣וֹךְ רְשָׁעִ֣ים וְאֵינָ֑ם
וּבֵ֖ית צַדִּיקִ֣ים יַעֲמֹֽד׃
חח לְֽפִי־שִׂ֭כְלוֹ יְהֻלַּל־אִ֑ישׁ
וְנַעֲוֵה־לֵ֝֗בנוְנַעֲוֵה־לֵ֝֗ב · א=וְנַעֲוֵה־לֵ֝ב (השמטת נקודת הרביע) יִהְיֶ֥ה לָבֽוּז׃
טט ט֣וֹב נִ֭קְלֶה וְעֶ֣בֶד ל֑וֹ
מִ֝מִּתְכַּבֵּ֗ד וַחֲסַר־לָֽחֶם׃
יי יוֹדֵ֣עַ צַ֭דִּיק נֶ֣פֶשׁ בְּהֶמְתּ֑וֹ
וְֽרַחֲמֵ֥י רְ֝שָׁעִ֗ים אַכְזָרִֽי׃
יאיא עֹבֵ֣ד אַ֭דְמָתוֹ יִֽשְׂבַּֽע־לָ֑חֶם
וּמְרַדֵּ֖ף רֵיקִ֣ים חֲסַר־לֵֽב׃
יביב חָמַ֣ד רָ֭שָׁע מְצ֣וֹד רָעִ֑ים
וְשֹׁ֖רֶשׁ צַדִּיקִ֣ים יִתֵּֽן׃
יגיג בְּפֶ֣שַׁע שְׂ֭פָתַיִם מוֹקֵ֣שׁ רָ֑ע
וַיֵּצֵ֖א מִצָּרָ֣ה צַדִּֽיק׃
ידיד מִפְּרִ֣י פִי־אִ֭ישׁ יִשְׂבַּע־ט֑וֹב
וּגְמ֥וּל יְדֵי־אָ֝דָ֗ם יָשִׁ֥ויבקקקרי/כתיב · (ישוב) יָשִׁ֥יב לֽוֹ׃
טוטו דֶּ֣רֶךְ אֱ֭וִיל יָשָׁ֣ר בְּעֵינָ֑יו
וְשֹׁמֵ֖עַ לְעֵצָ֣ה חָכָֽם׃
טזטז אֱוִ֗יל בַּ֭יּוֹם יִוָּדַ֣ע כַּעְס֑וֹ
וְכֹסֶ֖ה קָל֣וֹן עָרֽוּם׃
יזיז יָפִ֣יחַ אֱ֭מוּנָה יַגִּ֣יד צֶ֑דֶק
וְעֵ֖ד שְׁקָרִ֣ים מִרְמָֽה׃
יחיח יֵ֣שׁ בּ֭וֹטֶה כְּמַדְקְר֣וֹת חָ֑רֶב
וּלְשׁ֖וֹן חֲכָמִ֣ים מַרְפֵּֽא׃
יטיט שְֽׂפַת־אֱ֭מֶת תִּכּ֣וֹן לָעַ֑ד
וְעַד־אַ֝רְגִּ֗יעָה לְשׁ֣וֹן שָֽׁקֶר׃
ככ מִ֭רְמָה בְּלֶב־חֹ֣רְשֵׁי רָ֑ע
וּֽלְיֹעֲצֵ֖י שָׁל֣וֹם שִׂמְחָֽה׃
כאכא לֹא־יְאֻנֶּ֣ה לַצַּדִּ֣יק כׇּל־אָ֑וֶן
וּ֝רְשָׁעִ֗ים מָ֣לְאוּ רָֽע׃
כבכב תּוֹעֲבַ֣ת יְ֭הֹוָה שִׂפְתֵי־שָׁ֑קֶר
וְעֹשֵׂ֖י אֱמוּנָ֣ה רְצוֹנֽוֹ׃
כגכג אָדָ֣ם עָ֭רוּם כֹּ֣סֶה דָּ֑עַת
וְלֵ֥ב כְּ֝סִילִ֗יםנכְּ֝סִילִ֗ים=ל ושיטת-א ובדפוסים (וכן הכריעו ברויאר ומג"ה)א!=כְּ֝סִילִים (חסרה נקודת הרביע בלמ"ד); לפי ברויאר ההשמטה רק בספק, אבל מעיון בכתב־היד הוא ודאי. יִקְרָ֥א אִוֶּֽלֶת׃
כדכד יַד־חָרוּצִ֥ים תִּמְשׁ֑וֹל
וּ֝רְמִיָּ֗ה תִּהְיֶ֥ה לָמַֽס׃
כהכה דְּאָגָ֣ה בְלֶב־אִ֣ישׁ יַשְׁחֶ֑נָּה
וְדָבָ֖ר ט֣וֹב יְשַׂמְּחֶֽנָּה׃
כוכו יָתֵ֣ר מֵרֵעֵ֣הוּ צַדִּ֑יק
וְדֶ֖רֶךְ רְשָׁעִ֣ים תַּתְעֵֽם׃
כזכז לֹא־יַחֲרֹ֣ךְ רְמִיָּ֣ה צֵיד֑וֹ
וְהוֹן־אָדָ֖ם יָקָ֣ר חָרֽוּץ׃
כחכח בְּאֹֽרַח־צְדָקָ֥ה חַיִּ֑ים
וְדֶ֖רֶךְ נְתִיבָ֣הֿ אַל־מָֽוֶת׃
