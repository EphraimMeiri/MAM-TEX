ספר משלי

פרק ז


{פ}　　
זאא בְּ֭נִי שְׁמֹ֣ר אֲמָרָ֑י
וּ֝מִצְוֺתַ֗י תִּצְפֹּ֥ן אִתָּֽךְ׃
בב שְׁמֹ֣ר מִצְוֺתַ֣י וֶחְיֵ֑ה
וְ֝תוֹרָתִ֗י כְּאִישׁ֥וֹן עֵינֶֽיךָ׃
גג קׇשְׁרֵ֥ם עַל־אֶצְבְּעֹתֶ֑יךָ
כׇּ֝תְבֵ֗ם עַל־ל֥וּחַ לִבֶּֽךָ׃
דד אֱמֹ֣ר לַ֭חׇכְמָה אֲחֹ֣תִי אָ֑תְּ
וּ֝מֹדָ֗ע לַבִּינָ֥ה תִקְרָֽא׃
הה לִ֭שְׁמׇרְךָ מֵאִשָּׁ֣ה זָרָ֑ה
מִ֝נׇּכְרִיָּ֗ה אֲמָרֶ֥יהָ הֶחֱלִֽיקָה׃
וו כִּ֭י בְּחַלּ֣וֹן בֵּיתִ֑י
בְּעַ֖ד אֶשְׁנַבִּ֣י נִשְׁקָֽפְתִּי׃
זז וָאֵ֤רֶא בַפְּתָאיִ֗ם אָ֘בִ֤ינָה בַבָּנִ֗ים
נַ֣עַר חֲסַר־לֵֽב׃
חח עֹבֵ֣ר בַּ֭שּׁוּק אֵ֣צֶל פִּנָּ֑הּ
וְדֶ֖רֶךְ בֵּיתָ֣הּ יִצְעָֽד׃
טט בְּנֶֽשֶׁף־בְּעֶ֥רֶב י֑וֹם
בְּאִישׁ֥וֹן לַ֝֗יְלָה וַאֲפֵלָֽה׃
יי וְהִנֵּ֣ה אִ֭שָּׁה לִקְרָאת֑וֹ
שִׁ֥ית ז֝וֹנָ֗ה וּנְצֻ֥רַת לֵֽב׃
יאיא הֹמִיָּ֣ה הִ֣יא וְסֹרָ֑רֶת
בְּ֝בֵיתָ֗הּ לֹא־יִשְׁכְּנ֥וּ רַגְלֶֽיהָ׃
יביב פַּ֤עַם ׀ בַּח֗וּץ פַּ֥עַם בָּרְחֹב֑וֹת
וְאֵ֖צֶל כׇּל־פִּנָּ֣ה תֶאֱרֹֽב׃
יגיג וְהֶחֱזִ֣יקָה בּ֭וֹנבּ֭וֹ=א?,ל,ש1,ק13,פטרבורג-EVR-II-B-55 (בטעם דחי) וכך הכריעו ברויאר (בשתי מהדורותיו הראשונות) ומג"ה, וכמו כן בדפוסים וקורןא?=בּ֖וֹ (בטעם טרחא?) וכך הכריע ברויאר בכתר ירושלים; נראה שגם בכתר הכוונה לדחי, אך הקו נדחק קצת שמאלה למרכז האות בגלל ראש הלמ"ד מהשורה למטה. וְנָ֣שְׁקָה לּ֑וֹ
הֵעֵ֥זָה פָ֝נֶ֗יהָ וַתֹּ֣אמַר לֽוֹ׃
ידיד זִבְחֵ֣י שְׁלָמִ֣ים עָלָ֑י
הַ֝יּ֗וֹם שִׁלַּ֥מְתִּי נְדָרָֽי׃
טוטו עַל־כֵּ֭ן יָצָ֣אתִי לִקְרָאתֶ֑ךָ
לְשַׁחֵ֥ר פָּ֝נֶ֗יךָ וָאֶמְצָאֶֽךָּ׃
טזטז מַ֭רְבַדִּים רָבַ֣דְתִּי עַרְשִׂ֑י
חֲ֝טֻב֗וֹת אֵט֥וּן מִצְרָֽיִם׃
יזיז נַ֥פְתִּי מִשְׁכָּבִ֑י
מֹ֥ר אֲ֝הָלִ֗ים וְקִנָּמֽוֹן׃
יחיח לְכָ֤ה נִרְוֶ֣ה דֹ֭דִים עַד־הַבֹּ֑קֶר
נִ֝תְעַלְּסָ֗ה ׃
יטיט כִּ֤י אֵ֣ין הָאִ֣ישׁ בְּבֵית֑וֹ
הָ֝לַ֗ךְ בְּדֶ֣רֶךְ מֵרָחֽוֹק׃
ככ צְֽרוֹר־הַ֭כֶּסֶף לָקַ֣ח בְּיָד֑וֹ
לְי֥וֹם הַ֝כֵּ֗סֶא יָבֹ֥א בֵיתֽוֹ׃
כאכא הִ֭טַּתּוּ בְּרֹ֣ב לִקְחָ֑הּ
בְּחֵ֥לֶק שְׂ֝פָתֶ֗יהָ תַּדִּיחֶֽנּוּ׃
כבכב ה֤וֹלֵ֥ךְ אַחֲרֶ֗יהָפִּ֫תְאֹ֥ם
כְּ֭שׁוֹר אֶל־טֶ֣בַח יָבֹ֑א
וּ֝כְעֶ֗כֶס אֶל־מוּסַ֥ר אֱוִֽיל׃
כגכג עַ֤ד יְפַלַּ֪ח חֵ֡ץכְּֽבֵד֗וֹ
כְּמַהֵ֣ר צִפּ֣וֹר אֶל־פָּ֑ח
וְלֹא־יָ֝דַ֗ע כִּֽי־בְנַפְשׁ֥וֹ הֽוּא׃
כדכד וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־לִ֑י
וְ֝הַקְשִׁ֗יבוּ לְאִמְרֵי־פִֽי׃
כהכה אַל־יֵ֣שְׂטְ אֶל־דְּרָכֶ֣יהָ לִבֶּ֑ךָ
אַל־תֵּ֝֗תַענאַל־תֵּ֝֗תַע · א=א=אַל־תֵּ֝תַע (השמטת נקודת הרביע) בִּנְתִיבוֹתֶֽיהָ׃
כוכו כִּֽי־רַבִּ֣ים חֲלָלִ֣ים הִפִּ֑ילָה
וַ֝עֲצֻמִ֗ים כׇּל־הֲרֻגֶֽיהָ׃
כזכז דַּרְכֵ֣י שְׁא֣וֹל בֵּיתָ֑הּ
יֹ֝רְד֗וֹת אֶל־חַדְרֵי־מָֽוֶת׃
