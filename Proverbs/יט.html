מֵעִקֵּ֥שׁ שְׂ֝פָתָ֗יו וְה֣וּא כְסִֽיל׃
בב גַּ֤ם בְּלֹא־דַ֣עַת נֶ֣פֶשׁ לֹא־ט֑וֹב
וְאָ֖ץ בְּרַגְלַ֣יִם חוֹטֵֽא׃
גג אִוֶּ֣לֶת אָ֭דָם תְּסַלֵּ֣ף דַּרְכּ֑וֹ
וְעַל־יְ֝הֹוָ֗ה יִזְעַ֥ף לִבּֽוֹ׃
דד ה֗וֹן יֹ֭סִיף רֵעִ֣ים רַבִּ֑ים
וְ֝דָ֗ל מֵרֵעֵ֥הוּ יִפָּרֵֽד׃
הה עֵ֣ד שְׁ֭קָרִים לֹ֣א יִנָּקֶ֑ה
וְיָפִ֥יחַ כְּ֝זָבִ֗ים לֹ֣א יִמָּלֵֽט׃
וו רַ֭בִּים יְחַלּ֣וּ פְנֵֽי־נָדִ֑יב
וְכׇל־הָ֝רֵ֗עַ לְאִ֣ישׁ מַתָּֽן׃
זז נ=א (קמץ ומרכא); התיבה מוקפת במ"ג דפוס ונציה <כָּ֥ל־>. ראו עוד במידע על מהדורה זו, פרק שני, לגבי ההכרעה בין קמץ קטן לקמץ גדול בתיבה זו. יש עוד "כל" בקמץ ומרכא בתהלים (לה,י). אֲחֵי־רָ֨שׁ ׀ שְֽׂנֵאֻ֗הוּ

יאיא שֵׂ֣כֶל אָ֭דָם הֶאֱרִ֣יךְ אַפּ֑וֹ
וְ֝תִפְאַרְתּ֗וֹ עֲבֹ֣ר עַל־פָּֽשַׁע׃
יביב נַ֣הַם כַּ֭כְּפִיר זַ֣עַף מֶ֑לֶךְ
וּכְטַ֖ל עַל־עֵ֣שֶׂב רְצוֹנֽוֹ׃
יגיג הַוֺּ֣ת לְ֭אָבִיו בֵּ֣ן כְּסִ֑יל
וְדֶ֥לֶף טֹ֝רֵ֗ד מִדְיְנֵ֥י אִשָּֽׁה׃
ידיד בַּ֣יִת וָ֭הוֹן נַחֲלַ֣ת אָב֑וֹת
וּ֝מֵיְהֹוָ֗ה אִשָּׁ֥ה מַשְׂכָּֽלֶת׃
טוטו עַ֭צְלָה תַּפִּ֣יל תַּרְדֵּמָ֑ה
וְנֶ֖פֶשׁ רְמִיָּ֣ה תִרְעָֽב׃
