פרק א

אאא מִ֭שְׁלֵינמִ֭שְׁלֵי=מ"ש ובדפוסים וקורן (אות מ"ם גדולה)א,ל=מִ֭שְׁלֵי (מ"ם רגילה)מעיון בצילום כתי"ל נראה שאין אות גדולה, וכן דעתם של דותן והמקליד. אמנם ברויאר ציין ספק, כי גובה האות מתחילה טיפה מתחת לשורה, אבל ברור שאין בכך כוונה מיוחדת. שְׁלֹמֹ֣ה בֶן־דָּוִ֑ד
מֶ֝֗לֶךְנמֶ֝֗לֶךְ · א=מֶ֝לֶךְ יִשְׂרָאֵֽל׃
בב לָדַ֣עַת חׇכְמָ֣ה וּמוּסָ֑ר
לְ֝הָבִ֗ין אִמְרֵ֥י בִינָֽה׃
גג לָ֭קַחַת מוּסַ֣ר הַשְׂכֵּ֑ל
צֶ֥דֶק וּ֝מִשְׁפָּ֗ט וּמֵשָׁרִֽים׃
דד לָתֵ֣ת לִפְתָאיִ֣ם עׇרְמָ֑ה
לְ֝נַ֗עַר דַּ֣עַת וּמְזִמָּֽה׃
הה יִשְׁמַ֣ע חָ֭כָם וְי֣וֹסֶף לֶ֑קַח
וְ֝נָב֗וֹן תַּחְבֻּל֥וֹת יִקְנֶֽה׃
וו לְהָבִ֣ין מָ֭שָׁל וּמְלִיצָ֑ה
דִּבְרֵ֥י חֲ֝כָמִ֗ים וְחִידֹתָֽם׃
זז יִרְאַ֣ת יְ֭הֹוָה רֵאשִׁ֣ית דָּ֑עַת
חׇכְמָ֥ה וּ֝מוּסָ֗ר אֱוִילִ֥ים בָּֽזוּ׃
{פ}　　
חח שְׁמַ֣ע בְּ֭נִי מוּסַ֣ר אָבִ֑יךָ
וְאַל־תִּ֝טֹּ֗שׁ תּוֹרַ֥ת אִמֶּֽךָ׃
טט כִּ֤י ׀ לִוְיַ֤ת חֵ֓ן הֵ֬ם לְרֹאשֶׁ֑ךָ
וַ֝עֲנָקִ֗ים לְגַרְגְּרֹתֶֽךָ׃
יי בְּנִ֡י
אִם־יְפַתּ֥וּךָ חַ֝טָּאִ֗ים אַל־תֹּבֵֽא׃
יאיא אִם־יֹאמְרוּ֮לְכָ֢ה אִ֫תָּ֥נוּ
נֶאֶרְבָ֥ה לְדָ֑ם
נִצְפְּנָ֖ה לְנָקִ֣י חִנָּֽם׃
יביב נִ֭בְלָעֵם כִּשְׁא֣וֹל חַיִּ֑ים
וּ֝תְמִימִ֗ים כְּי֣וֹרְדֵי בֽוֹר׃
יגיג כׇּל־ה֣וֹן יָקָ֣ר נִמְצָ֑א
נְמַלֵּ֖א בָתֵּ֣ינוּ שָׁלָֽל׃
ידיד גּ֭וֹרָ֣לְךָ תַּפִּ֣יל בְּתוֹכֵ֑נוּ
כִּ֥יס אֶ֝חָ֗ד יִהְיֶ֥ה לְכֻלָּֽנוּ׃
טוטו בְּנִ֗י אַל־תֵּלֵ֣ךְ בְּדֶ֣רֶךְ אִתָּ֑ם
מְנַ֥ע רַ֝גְלְךָ֗ מִנְּתִיבָתָֽם׃
טזטז כִּ֣י רַ֭גְלֵיהֶם לָרַ֣ע יָר֑וּצוּ
וִ֝ימַהֲר֗וּ לִשְׁפׇּךְ־דָּֽם׃
יזיז כִּֽי־חִ֭נָּם מְזֹרָ֣ה הָרָ֑שֶׁת
בְּ֝עֵינֵ֗י כׇּל־בַּ֥עַל כָּנָֽף׃
יחיח וְ֭הֵם לְדָמָ֣ם יֶאֱרֹ֑בוּ
יִ֝צְפְּנ֗וּ לְנַפְשֹׁתָֽם׃
יטיט כֵּ֗ן אׇ֭רְחוֹת כׇּל־בֹּ֣צֵֽעַ בָּ֑צַע
אֶת־נֶ֖פֶשׁ בְּעָלָ֣יו יִקָּֽח׃
{פ}　　
ככ חׇ֭כְמוֹת בַּח֣וּץ תָּרֹ֑נָּה
בָּ֝רְחֹב֗וֹת תִּתֵּ֥ן קוֹלָֽהּ׃
כאכא בְּרֹ֥אשׁ הֹמִיּ֗וֹתתִּ֫קְרָ֥א
בְּפִתְחֵ֖י שְׁעָרִ֥ים בָּעִ֗יר
אֲמָרֶ֥יהָ תֹאמֵֽר׃נבְּרֹ֥אשׁ הֹמִיּ֗וֹתתִּ֫קְרָ֥אבְּפִתְחֵ֖י שְׁעָרִ֥ים בָּעִ֗יראֲמָרֶ֥יהָ תֹאמֵֽר׃=א,ל וכן הוא במ"ג דפוס ונציה ובחלק מהדפוסים (כגון במבורגר). אך בחלק מהדפוסים (כגון לטריס ומ"ג דפוס וורשה) תיבת "בעיר" שייכת לחלק השלישי של הפסוק ולא השני; ולכן אצלם "בפתחי" במרכא במקום טפחא, ו"שערים" באתנחתא, ואח"כ "בעיר" ברביע מוגרש: <בְּרֹ֥אשׁ הֹֽמִיּ֗וֹת תִּ֫קְרָ֥א בְּפִתְחֵ֥י שְׁעָרִ֑ים בָּ֝עִ֗יר אֲמָרֶ֥יהָ תֹאמֵֽר׃>קורן=<בְּרֹ֥אשׁ הֹֽמִיּ֗וֹת תִּ֫קְרָ֥א בְּפִתְחֵ֖י שְׁעָרִ֥ים בָּ֝עִ֗יר אֲמָרֶ֥יהָ תֹאמֵֽר׃> (!)
כבכב עַד־מָתַ֣י ׀ פְּתָיִם֮תְּֽאֵהֲב֫וּ פֶ֥תִי
וְלֵצִ֗ים לָ֭צוֹן חָמְד֣וּ לָהֶ֑ם
וּ֝כְסִילִ֗ים יִשְׂנְאוּ־דָֽעַת׃
כגכג תָּשׁ֗וּבוּלְֽת֫וֹכַחְתִּ֥י
הִנֵּ֤ה אַבִּ֣יעָה לָכֶ֣ם רוּחִ֑י
אוֹדִ֖יעָה דְבָרַ֣י אֶתְכֶֽם׃
כדכד יַ֣עַן קָ֭רָאתִי וַתְּמָאֵ֑נוּ
נָטִ֥יתִי יָ֝דִ֗י וְאֵ֣ין מַקְשִֽׁיב׃
כהכה וַתִּפְרְע֥וּ כׇל־עֲצָתִ֑י
וְ֝תוֹכַחְתִּ֗י לֹ֣א אֲבִיתֶֽם׃
כוכו גַּם־אֲ֭נִי בְּאֵידְכֶ֣ם אֶשְׂחָ֑ק
אֶ֝לְעַ֗ג בְּבֹ֣א פַחְדְּכֶֽם׃
כזכז בְּבֹ֤א (כשאוה)קקקרי/כתיב · (כשאוה) כְשׁוֹאָ֨ה | פַּחְדְּכֶ֗ם
וְֽ֭אֵידְכֶם כְּסוּפָ֣ה יֶאֱתֶ֑ה
בְּבֹ֥א עֲ֝לֵיכֶ֗םצָרָ֥ה וְצוּקָֽה׃
כחכח אָ֣ז יִ֭קְרָאֻנְנִי וְלֹ֣א אֶעֱנֶ֑ה
יְ֝שַׁחֲרֻ֗נְנִי וְלֹ֣א יִמְצָאֻֽנְנִי׃
כטכט תַּ֭חַת כִּֽי־שָׂ֣נְאוּ דָ֑עַת
וְיִרְאַ֥ת יְ֝הֹוָ֗ה לֹ֣א בָחָֽרוּ׃
לל לֹא־אָב֥וּ לַעֲצָתִ֑י
נָ֝אֲצ֗וּ כׇּל־תּוֹכַחְתִּֽי׃
לאלא וְֽ֭יֹאכְלוּ מִפְּרִ֣י דַרְכָּ֑ם
וּֽמִמֹּעֲצֹ֖תֵיהֶ֣ם יִשְׂבָּֽעוּ׃
לבלב כִּ֤י מְשׁוּבַ֣ת פְּתָיִ֣ם תַּהַרְגֵ֑ם
וְשַׁלְוַ֖ת כְּסִילִ֣ים תְּאַבְּדֵֽם׃
לגלג וְשֹׁמֵ֣עַֽ לִ֭י יִשְׁכׇּן־בֶּ֑טַח
וְ֝שַׁאֲנַ֗ן מִפַּ֥חַד רָעָֽה׃
