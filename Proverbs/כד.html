מקרא על פי המסורה

ספר משלי

פרק כד

כדאא אַל־תְּ֭קַנֵּא בְּאַנְשֵׁ֣י רָעָ֑ה
וְאַל־תִּ֝תְאָ֗ו לִֽהְי֥וֹת אִתָּֽם׃
בב כִּי־שֹׁ֭ד יֶהְגֶּ֣ה לִבָּ֑ם
וְ֝עָמָ֗ל שִׂפְתֵיהֶ֥ם תְּדַבֵּֽרְנָה׃
גג בְּ֭חׇכְמָה יִבָּ֣נֶה בָּ֑יִת
וּ֝בִתְבוּנָ֗ה יִתְכּוֹנָֽן׃
דד וּ֭בְדַעַת חֲדָרִ֣ים יִמָּלְא֑וּ
כׇּל־ה֖וֹן יָקָ֣ר וְנָעִֽים׃
הה גֶּבֶר־חָכָ֥ם בַּע֑וֹז
וְאִֽישׁ־דַּ֝֗עַת מְאַמֶּץ־כֹּֽחַ׃
וו כִּ֣י בְ֭תַחְבֻּלוֹת תַּעֲשֶׂה־לְּךָ֣ מִלְחָמָ֑ה
וּ֝תְשׁוּעָ֗ה בְּרֹ֣ב יוֹעֵֽץ׃
זז רָאמ֣וֹת לֶאֱוִ֣יל חׇכְמ֑וֹת
בַּ֝שַּׁ֗עַר לֹ֣א יִפְתַּח־פִּֽיהוּ׃
חח מְחַשֵּׁ֥ב לְהָרֵ֑עַ
ל֝֗וֹנל֝֗וֹ · א=ל֝וֹ (השמטת נקודת הרביע) בַּעַל־מְזִמּ֥וֹת יִקְרָֽאוּ׃
טט זִמַּ֣ת אִוֶּ֣לֶת חַטָּ֑את
וְתוֹעֲבַ֖ת לְאָדָ֣ם לֵֽץ׃
יי הִ֭תְרַפִּיתָ בְּי֥וֹם צָרָ֗ה
צַ֣ר כֹּחֶֽכָה׃
יאיא הַ֭צֵּל לְקֻחִ֣ים לַמָּ֑וֶת
וּמָטִ֥ים לַ֝הֶ֗רֶג אִם־תַּחְשֽׂוֹךְ׃
יביב כִּֽי־תֹאמַ֗רהֵן֮ לֹא־יָדַ֢עְנ֫וּ זֶ֥ה
הֲֽלֹא־תֹ֘כֵ֤ן לִבּ֨וֹת ׀ הֽוּא־יָבִ֗ין
וְנֹצֵ֣ר נַ֭פְשְׁךָ ה֣וּא יֵדָ֑ע
וְהֵשִׁ֖יב לְאָדָ֣ם ׃
יגיג אֱכׇל־בְּנִ֣י דְבַ֣שׁ כִּי־ט֑וֹב
וְנֹ֥פֶת מָ֝ת֗וֹק עַל־חִכֶּֽךָ׃
ידיד כֵּ֤ן ׀ דְּעֶ֥ה חׇכְמָ֗ה לְנַ֫פְשֶׁ֥ךָ
אִם־מָ֭צָאתָ וְיֵ֣שׁ אַחֲרִ֑ית
וְ֝תִקְוָתְךָ֗לֹ֣א תִכָּרֵֽת׃
{פ}　　
טוטו אַל־תֶּאֱרֹ֣ב רָ֭שָׁע לִנְוֵ֣ה צַדִּ֑יק
אַֽל־תְּשַׁדֵּ֥ד רִבְצֽוֹ׃
טזטז כִּ֤י שֶׁ֨בַע ׀ יִפּ֣וֹל צַדִּ֣יק וָקָ֑ם
וּ֝רְשָׁעִ֗ים יִכָּשְׁל֥וּ בְרָעָֽה׃
יזיז בִּנְפֹ֣ל (אויביך)קקקרי/כתיב · (אויביך) א֭וֹיִבְךָ אַל־תִּשְׂמָ֑ח
וּ֝בִכָּשְׁל֗וֹ אַל־יָגֵ֥ל לִבֶּֽךָ׃
יחיח פֶּן־יִרְאֶ֣ה יְ֭הֹוָה וְרַ֣ע בְּעֵינָ֑יו
וְהֵשִׁ֖יב מֵעָלָ֣יו אַפּֽוֹ׃
יטיט אַל־תִּתְחַ֥ר בַּמְּרֵעִ֑ים
אַל־תְּ֝קַנֵּ֗א בָּֽרְﬞשָׁעִֽיםנבָּֽרְﬞשָׁעִֽים · א=בָּֽרֲשָׁעִֽים (חטף)׃
ככ כִּ֤י ׀ לֹא־תִהְיֶ֣ה אַחֲרִ֣ית לָרָ֑ע
נֵ֖ר רְשָׁעִ֣ים יִדְעָֽךְ׃
כאכא יְרָא־אֶת־יְהֹוָ֣ה בְּנִ֣י וָמֶ֑לֶךְ
עִם־שׁ֝וֹנִ֗ים אַל־תִּתְעָרָֽב׃
כבכב כִּֽי־פִ֭תְאֹם יָק֣וּם אֵידָ֑ם
וּפִ֥יד שְׁ֝נֵיהֶ֗ם מִ֣י יוֹדֵֽעַ׃
{פ}　　
כגכג גַּם־אֵ֥לֶּה לַחֲכָמִ֑ים
הַֽכֵּר־פָּנִ֖ים בְּמִשְׁפָּ֣ט בַּל־טֽוֹב׃
כדכד אֹ֤מֵ֨ר ׀ לְרָשָׁע֮צַדִּ֢יק אָ֥֫תָּהנאָ֥֫תָּה=דפוסים וקורןא=אָ֥תָּה (השמטת סימן העולה)
יִקְּבֻ֥הוּ עַמִּ֑ים
יִזְעָמ֥וּהוּ לְאֻמִּֽים׃
כהכה וְלַמּוֹכִיחִ֥ים יִנְעָ֑ם
וַ֝עֲלֵיהֶ֗ם תָּב֥וֹא בִרְכַּת־טֽוֹב׃
כוכו שְׂפָתַ֥יִם יִשָּׁ֑ק
מֵ֝שִׁ֗יב דְּבָרִ֥ים נְכֹחִֽים׃
כזכז הָ֘כֵ֤ן בַּח֨וּץ ׀ מְלַאכְתֶּ֗ךָ
וְעַתְּדָ֣הּ בַּשָּׂדֶ֣ה לָ֑ךְ
אַ֝חַ֗רוּבָנִ֥יתָ בֵיתֶֽךָ׃
{פ}　　
כחכח אַל־תְּהִ֣י עֵד־חִנָּ֣ם בְּרֵעֶ֑ךָ
וַ֝הֲפִתִּ֗יתָ בִּשְׂפָתֶֽיךָ׃
כטכט אַל־תֹּאמַ֗ר
כַּאֲשֶׁ֣ר עָֽשָׂה־לִ֭יכֵּ֤ן אֶֽעֱשֶׂה־לּ֑וֹ
אָשִׁ֖יב לָאִ֣ישׁ ׃
{פ}　　
לל עַל־שְׂדֵ֣ה אִישׁ־עָצֵ֣ל עָבַ֑רְתִּי
וְעַל־כֶּ֝֗רֶםנוְעַל־כֶּ֝֗רֶם · א=וְעַל־כֶּ֝רֶם (השמטת נקודת הרביע) אָדָ֥ם חֲסַר־לֵֽב׃
לאלא וְהִנֵּ֨ה עָ֘לָ֤ה כֻלּ֨וֹ ׀ קִמְּשֹׂנִ֗ים
כׇּסּ֣וּנכׇּסּ֣וּ · לגבי הקמץ הקטן ראו בהערת הנוסח על תהלים פ,יא. פָנָ֣יו חֲרֻלִּ֑ים
וְגֶ֖דֶר אֲבָנָ֣יו נֶהֱרָֽסָה׃
לבלב וָאֶחֱזֶ֣ה אָ֭נֹכִי אָשִׁ֣ית לִבִּ֑י
רָ֝אִ֗יתִי לָקַ֥חְתִּי מוּסָֽר׃
לגלג מְעַ֣ט שֵׁ֭נוֹת מְעַ֣ט תְּנוּמ֑וֹת
מְעַ֓ט ׀ חִבֻּ֖ק יָדַ֣יִם לִשְׁכָּֽב׃
לדלד וּבָֽא־מִתְהַלֵּ֥ךְ רֵישֶׁ֑ךָ
וּ֝מַחְסֹרֶ֗יךָ כְּאִ֣ישׁ מָגֵֽן׃
