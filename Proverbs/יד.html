מקרא על פי המסורה

ספר משלי

פרק יד

ידאא חַכְמ֣וֹת נָ֭שִׁים בָּנְתָ֣ה בֵיתָ֑הּ
וְ֝אִוֶּ֗לֶת בְּיָדֶ֥יהָ תֶהֶרְסֶֽנּוּ׃
בב הוֹלֵ֣ךְ בְּ֭יׇשְׁרוֹ יְרֵ֣א יְהֹוָ֑ה
וּנְל֖וֹז דְּרָכָ֣יו בּוֹזֵֽהוּ׃
גג בְּֽפִי־אֱ֭וִיל חֹ֣טֶר גַּאֲוָ֑ה
וְשִׂפְתֵ֥י חֲ֝כָמִ֗ים תִּשְׁמוּרֵֽם׃
דד בְּאֵ֣ין אֲ֭לָפִים אֵב֣וּס בָּ֑ר
וְרׇב־תְּ֝בוּא֗וֹת בְּכֹ֣חַ שֽׁוֹר׃
הה עֵ֣ד אֱ֭מוּנִים לֹ֣א יְכַזֵּ֑ב
וְיָפִ֥יחַ כְּ֝זָבִ֗ים עֵ֣ד שָֽׁקֶר׃
וו בִּקֶּשׁ־לֵ֣ץ חׇכְמָ֣ה וָאָ֑יִן
וְדַ֖עַת לְנָב֣וֹן נָקָֽל׃
זז לֵ֣ךְ מִ֭נֶּגֶד לְאִ֣ישׁ כְּסִ֑יל
וּבַל־יָ֝דַ֗עְתָּ שִׂפְתֵי־דָֽעַת׃
חח חׇכְמַ֣ת עָ֭רוּם הָבִ֣ין דַּרְכּ֑וֹ
וְאִוֶּ֖לֶת כְּסִילִ֣ים מִרְמָֽה׃
טט אֱ֭וִלִים יָלִ֣יץ אָשָׁ֑ם
וּבֵ֖ין יְשָׁרִ֣ים רָצֽוֹן׃
יי לֵ֗ב י֭וֹדֵעַ מׇרַּ֣ת נַפְשׁ֑וֹ
וּ֝בְשִׂמְחָת֗וֹ לֹא־יִתְעָ֥רַב זָֽר׃
יאיא בֵּ֣ית רְ֭שָׁעִים יִשָּׁמֵ֑ד
וְאֹ֖הֶל יְשָׁרִ֣ים יַפְרִֽיחַ׃
יביב יֵ֤שׁ דֶּ֣רֶךְ יָ֭שָׁר לִפְנֵי־אִ֑ישׁ
וְ֝אַחֲרִיתָ֗הּ דַּרְכֵי־מָֽוֶת׃
יגיג גַּם־בִּשְׂחֹ֥ק יִכְאַב־לֵ֑ב
וְאַחֲרִיתָ֖הּ שִׂמְחָ֣ה תוּגָֽה׃
ידיד מִדְּרָכָ֣יו יִ֭שְׂבַּע ס֣וּג לֵ֑ב
וּ֝מֵעָלָ֗יו אִ֣ישׁ טֽוֹב׃
טוטו פֶּ֭תִי יַאֲמִ֣ין לְכׇל־דָּבָ֑ר
וְ֝עָר֗וּם יָבִ֥ין לַאֲשֻׁרֽוֹ׃
טזטז חָכָ֣ם יָ֭רֵא וְסָ֣ר מֵרָ֑ע
וּ֝כְסִ֗יל מִתְעַבֵּ֥ר וּבוֹטֵֽחַ׃
יזיז קְֽצַר־אַ֭פַּיִם יַעֲשֶׂ֣ה אִוֶּ֑לֶת
וְאִ֥ישׁ מְ֝זִמּ֗וֹת יִשָּׂנֵֽא׃
יחיח נָחֲל֣וּ פְתָאיִ֣ם אִוֶּ֑לֶת
וַ֝עֲרוּמִ֗ים יַכְתִּ֥רוּ דָֽעַת׃
יטיט שַׁח֣וּ רָ֭עִים לִפְנֵ֣י טוֹבִ֑ים
וּ֝רְשָׁעִ֗ים עַֽל־שַׁעֲרֵ֥י צַדִּֽיק׃
ככ גַּם־לְ֭רֵעֵהוּ יִשָּׂ֣נֵא רָ֑שׁ
וְאֹהֲבֵ֖י עָשִׁ֣יר רַבִּֽים׃
כאכא בׇּז־לְרֵעֵ֥הוּ חוֹטֵ֑א
וּמְחוֹנֵ֖ן עֲנָיוִ֣יםקקקרי/כתיב · (עניים) עֲנָוִ֣ים אַשְׁרָֽיו׃
כבכב הֲֽלוֹא־יִ֭תְעוּ חֹ֣רְשֵׁי רָ֑ע
וְחֶ֥סֶד וֶ֝אֱמֶ֗ת חֹ֣רְשֵׁי טֽוֹב׃
כגכג בְּכׇל־עֶ֭צֶב יִהְיֶ֣ה מוֹתָ֑ר
וּדְבַר־שְׂ֝פָתַ֗יִם אַךְ־לְמַחְסֽוֹר׃
כדכד עֲטֶ֣רֶת חֲכָמִ֣ים עׇשְׁרָ֑ם
אִוֶּ֖לֶת כְּסִילִ֣ים אִוֶּֽלֶת׃
כהכה מַצִּ֣יל נְ֭פָשׁוֹת עֵ֣ד אֱמֶ֑ת
וְיָפִ֖חַ כְּזָבִ֣ים מִרְמָֽה׃
כוכו בְּיִרְאַ֣ת יְ֭הֹוָה מִבְטַח־עֹ֑ז
וּ֝לְבָנָ֗יו יִהְיֶ֥ה מַחְסֶֽה׃
כזכז יִרְאַ֣ת יְ֭הֹוָה מְק֣וֹר חַיִּ֑ים
לָ֝ס֗וּר מִמֹּ֥קְשֵׁי מָֽוֶת׃
כחכח בְּרׇב־עָ֥ם הַדְרַת־מֶ֑לֶךְ
וּבְאֶ֥פֶס לְ֝אֹ֗ם מְחִתַּ֥ת רָזֽוֹן׃
כטכט אֶ֣רֶךְ אַ֭פַּיִם רַב־תְּבוּנָ֑ה
וּקְצַר־ר֝֗וּחַ מֵרִ֥ים אִוֶּֽלֶת׃
לל חַיֵּ֣י בְ֭שָׂרִים לֵ֣ב מַרְפֵּ֑א
וּרְקַ֖ב עֲצָמ֣וֹת קִנְאָֽה׃
לאלא עֹ֣שֵֽׁק דָּ֭ל חֵרֵ֣ף עֹשֵׂ֑הוּ
וּ֝מְכַבְּד֗וֹ חֹנֵ֥ן אֶבְיֽוֹן׃
לבלב בְּֽ֭רָעָתוֹ יִדָּחֶ֣ה רָשָׁ֑ע
וְחֹסֶ֖ה בְמוֹת֣וֹ צַדִּֽיק׃
לגלג בְּלֵ֣ב נָ֭בוֹן תָּנ֣וּחַ חׇכְמָ֑ה
וּבְקֶ֥רֶב כְּ֝סִילִ֗ים תִּוָּדֵֽעַ׃
לדלד צְדָקָ֥ה תְרוֹמֵֽם־גּ֑וֹי
וְחֶ֖סֶד לְאֻמִּ֣ים חַטָּֽאת׃
להלה רְֽצוֹן־מֶ֭לֶךְ לְעֶ֣בֶד מַשְׂכִּ֑יל
וְ֝עֶבְרָת֗וֹ תִּהְיֶ֥ה מֵבִֽישׁ׃
