מקרא על פי המסורה

ספר משלי

פרק כב

כבאא נִבְחָ֣ר שֵׁ֭ם מֵעֹ֣שֶׁר רָ֑ב
מִכֶּ֥סֶף וּ֝מִזָּהָ֗ב חֵ֣ן טֽוֹב׃
בב עָשִׁ֣יר וָרָ֣שׁ נִפְגָּ֑שׁוּ
עֹשֵׂ֖ה כֻלָּ֣ם יְהֹוָֽה׃
גג עָר֤וּם ׀ רָאָ֣ה רָעָ֣ה (ויסתר)קקקרי/כתיב · (ויסתר) וְנִסְתָּ֑ר
וּ֝פְתָיִ֗ים עָבְר֥וּ וְֽנֶעֱנָֽשׁוּ׃
דד עֵ֣קֶב עֲ֭נָוָה יִרְאַ֣ת יְהֹוָ֑ה
עֹ֖שֶׁר וְכָב֣וֹד וְחַיִּֽים׃
הה צִנִּ֣ים פַּ֭חִים בְּדֶ֣רֶךְ עִקֵּ֑שׁ
שׁוֹמֵ֥ר נַ֝פְשׁ֗וֹ יִרְחַ֥ק מֵהֶֽם׃
וו חֲנֹ֣ךְ לַ֭נַּעַר עַל־פִּ֣י דַרְכּ֑וֹ
גַּ֥ם כִּי־יַ֝זְקִ֗ין לֹא־יָס֥וּר מִמֶּֽנָּה׃
זז עָ֭שִׁיר בְּרָשִׁ֣ים יִמְשׁ֑וֹל
וְעֶ֥בֶד לֹ֝וֶ֗ה לְאִ֣ישׁ מַלְוֶֽה׃
חח זוֹרֵ֣עַ עַ֭וְלָה (יקצור)אָ֑וֶןנ(יקצור)אָ֑וֶן · א=יִקְצׇור־אָ֑וֶן
וְשֵׁ֖בֶט עֶבְרָת֣וֹ יִכְלֶֽה׃
טט טֽוֹב־עַ֭יִן ה֣וּא יְבֹרָ֑ךְ
כִּֽי־נָתַ֖ן מִלַּחְמ֣וֹ לַדָּֽל׃
יי גָּ֣רֵֽשׁ לֵ֭ץ וְיֵצֵ֣א מָד֑וֹן
וְ֝יִשְׁבֹּ֗ת דִּ֣ין וְקָלֽוֹן׃
יאיא אֹהֵ֥ב (טהור)לֵ֑בנ(טהור)לֵ֑ב · א=טְהׇור־לֵ֑ב
חֵ֥ן שְׂ֝פָתָ֗יו רֵעֵ֥הוּ מֶֽלֶךְ׃
יביב עֵינֵ֣י יְ֭הֹוָה נָ֣צְרוּ דָ֑עַת
וַ֝יְסַלֵּ֗ף דִּבְרֵ֥י בֹגֵֽד׃
יגיג אָמַ֣ר עָ֭צֵל אֲרִ֣י בַח֑וּץ
בְּת֥וֹךְ רְ֝חֹב֗וֹת אֵרָצֵֽחַ׃
ידיד שׁוּחָ֣ה עֲ֭מֻקָּה פִּ֣י זָר֑וֹת
זְע֥וּם יְ֝הֹוָ֗ה (יפול)שָֽׁםנ(יפול)שָֽׁם · א=יִפׇּול־שָֽׁם׃
טוטו אִ֭וֶּלֶת קְשׁוּרָ֣ה בְלֶב־נָ֑עַר
שֵׁ֥בֶט מ֝וּסָ֗ר יַרְחִיקֶ֥נָּה מִמֶּֽנּוּ׃
טזטז עֹ֣שֵֽׁק דָּ֭ל לְהַרְבּ֣וֹת ל֑וֹ
נֹתֵ֥ן לְ֝עָשִׁ֗יר אַךְ־לְמַחְסֽוֹר׃
יזיז הַ֥ט אׇזְנְךָ֗ וּ֭שְׁמַע דִּבְרֵ֣י חֲכָמִ֑ים
וְ֝לִבְּךָ֗ תָּשִׁ֥ית לְדַעְתִּֽי׃
יחיח כִּֽי־נָ֭עִים כִּֽי־תִשְׁמְרֵ֣ם בְּבִטְנֶ֑ךָ
יִכֹּ֥נוּ יַ֝חְדָּ֗ו עַל־שְׂפָתֶֽיךָ׃
יטיט לִֽהְי֣וֹת בַּ֭יהֹוָה מִבְטַחֶ֑ךָ
הוֹדַעְתִּ֖יךָ הַיּ֣וֹם אַף־אָֽתָּה׃
ככ הֲלֹ֤א כָתַ֣בְתִּֽי לְ֭ךָ (שלשום)קקקרי/כתיב · (שלשום) שָׁלִישִׁ֑ים
בְּמֹ֖עֵצ֣וֹת וָדָֽעַת׃
כאכא לְהֽוֹדִיעֲךָ֗ קֹ֭שְׁטְ אִמְרֵ֣י אֱמֶ֑ת
לְהָשִׁ֥יב אֲמָרִ֥ים אֱ֝מֶ֗ת לְשֹׁלְחֶֽיךָ׃
כבכב אַֽל־תִּגְזׇל־דָּ֭ל כִּ֣י דַל־ה֑וּא
וְאַל־תְּדַכֵּ֖א עָנִ֣י בַשָּֽׁעַר׃
כגכג כִּֽי־יְ֭הֹוָה יָרִ֣יב רִיבָ֑ם
וְקָבַ֖ע אֶת־קֹבְעֵיהֶ֣ם נָֽפֶשׁ׃
כדכד אַל־תִּ֭תְרַע אֶת־בַּ֣עַל אָ֑ף
וְאֶת־אִ֥ישׁ חֵ֝מ֗וֹת לֹ֣א תָבֽוֹא׃
כהכה פֶּן־תֶּאֱלַ֥ף אֹרְחֹתָ֑וקקקרי/כתיב · אֹרְחֹתָ֑ו · אֹרְחֹתָ֑יו
וְלָקַחְתָּ֖ מוֹקֵ֣שׁ לְנַפְשֶֽׁךָ׃
כוכו אַל־תְּהִ֥י בְתֹֽקְעֵי־כָ֑ף
בַּ֝עֹרְבִ֗ים מַשָּׁאֽוֹת׃
כזכז אִם־אֵֽין־לְךָ֥ לְשַׁלֵּ֑ם
לָ֥מָּה יִקַּ֥ח מִ֝שְׁכָּבְךָ֗ מִתַּחְתֶּֽיךָ׃
כחכח אַל־תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם
אֲשֶׁ֖ר עָשׂ֣וּ אֲבוֹתֶֽיךָ׃
כטכט חָזִ֡יתָ אִ֤ישׁ ׀ מָ֘הִ֤יר בִּמְלַאכְתּ֗וֹ
לִֽפְנֵי־מְלָכִ֥ים יִתְיַצָּ֑ב
בַּל־יִ֝תְיַצֵּ֗בלִפְנֵ֥י חֲשֻׁכִּֽים׃
