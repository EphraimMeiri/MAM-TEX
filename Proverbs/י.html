מקרא על פי המסורה

ספר משלי

פרק י


{פ}　　

יאא
בֵּ֣ן חָ֭כָם יְשַׂמַּח־אָ֑ב
וּבֵ֥ן כְּ֝סִ֗יל תּוּגַ֥ת אִמּֽוֹ׃
בב לֹֽא־י֭וֹעִילוּ אוֹצְר֣וֹת רֶ֑שַׁע
וּ֝צְדָקָ֗ה תַּצִּ֥יל מִמָּֽוֶת׃
גג לֹא־יַרְעִ֣יב יְ֭הֹוָה נֶ֣פֶשׁ צַדִּ֑יק
וְהַוַּ֖ת רְשָׁעִ֣ים יֶהְדֹּֽף׃
דד רָ֗אשׁ עֹשֶׂ֥ה כַף־רְמִיָּ֑ה
וְיַ֖ד חָרוּצִ֣ים תַּעֲשִֽׁיר׃
הה אֹגֵ֣ר בַּ֭קַּיִץ בֵּ֣ן מַשְׂכִּ֑יל
נִרְדָּ֥ם בַּ֝קָּצִ֗יר בֵּ֣ן מֵבִֽישׁ׃
וו בְּ֭רָכוֹת לְרֹ֣אשׁ צַדִּ֑יק
וּפִ֥י רְ֝שָׁעִ֗ים יְכַסֶּ֥ה חָמָֽס׃
זז זֵ֣כֶר צַ֭דִּיק לִבְרָכָ֑ה
וְשֵׁ֖ם רְשָׁעִ֣ים יִרְקָֽב׃
חח חֲכַם־לֵ֭ב יִקַּ֣ח מִצְוֺ֑ת
וֶאֱוִ֥יל שְׂ֝פָתַ֗יִם יִלָּבֵֽט׃
טט הוֹלֵ֣ךְ בַּ֭תֹּם יֵ֣לֶךְ בֶּ֑טַח
וּמְעַקֵּ֥שׁ דְּ֝רָכָ֗יו יִוָּדֵֽעַ׃
יי קֹ֣רֵֽץ עַ֭יִן יִתֵּ֣ן עַצָּ֑בֶת
וֶאֱוִ֥יל שְׂ֝פָתַ֗יִם יִלָּבֵֽט׃
יאיא מְק֣וֹר חַ֭יִּים פִּ֣י צַדִּ֑יק
וּפִ֥י רְ֝שָׁעִ֗ים יְכַסֶּ֥ה חָמָֽס׃
יביב שִׂ֭נְאָה תְּעֹרֵ֣ר מְדָנִ֑ים
וְעַ֥ל כׇּל־פְּ֝שָׁעִ֗ים תְּכַסֶּ֥ה אַהֲבָֽה׃
יגיג בְּשִׂפְתֵ֣ינבְּשִׂפְתֵ֣י=לא!=בְּשִׁפְתֵ֣י (נקודת שי"ן ימנית) נָ֭בוֹן תִּמָּצֵ֣א חׇכְמָ֑ה
וְ֝שֵׁ֗בֶט לְגֵ֣ו חֲסַר־לֵֽב׃
ידיד חֲכָמִ֥ים יִצְפְּנוּ־דָ֑עַת
וּפִֽי־אֱ֝וִ֗יל מְחִתָּ֥ה קְרֹבָֽה׃
טוטו ה֣וֹן עָ֭שִׁיר קִרְיַ֣ת עֻזּ֑וֹ
מְחִתַּ֖ת דַּלִּ֣ים רֵישָֽׁם׃
טזטז פְּעֻלַּ֣ת צַדִּ֣יק לְחַיִּ֑ים
תְּבוּאַ֖ת רָשָׁ֣ע לְחַטָּֽאת׃
יזיז אֹ֣רַח לְ֭חַיִּים שׁוֹמֵ֣ר מוּסָ֑ר
וְעֹזֵ֖ב תּוֹכַ֣חַת מַתְעֶֽה׃
יחיח מְכַסֶּ֣ה שִׂ֭נְאָה שִׂפְתֵי־שָׁ֑קֶר
וּמוֹצִ֥א דִ֝בָּ֗ה ה֣וּא כְסִֽיל׃
יטיט בְּרֹ֣ב דְּ֭בָרִים לֹ֣א יֶחְדַּל־פָּ֑שַׁע
וְחוֹשֵׂ֖ךְ שְׂפָתָ֣יו מַשְׂכִּֽיל׃
ככ כֶּ֣סֶף נִ֭בְחָר לְשׁ֣וֹן צַדִּ֑יק
לֵ֖ב רְשָׁעִ֣ים כִּמְעָֽט׃
כאכא שִׂפְתֵ֣י צַ֭דִּיק יִרְע֣וּ רַבִּ֑ים
וֶ֝אֱוִילִ֗ים בַּחֲסַר־לֵ֥ב יָמֽוּתוּ׃
כבכב בִּרְכַּ֣ת יְ֭הֹוָה הִ֣יא תַעֲשִׁ֑יר
וְלֹֽא־יוֹסִ֖ף עֶ֣צֶב עִמָּֽהּ׃
כגכג כִּשְׂח֣וֹק לִ֭כְסִיל עֲשׂ֣וֹת זִמָּ֑ה
וְ֝חׇכְמָ֗ה לְאִ֣ישׁ תְּבוּנָֽה׃
כדכד מְגוֹרַ֣ת רָ֭שָׁע הִ֣יא תְבוֹאֶ֑נּוּ
וְתַאֲוַ֖ת צַדִּיקִ֣ים יִתֵּֽן׃
כהכה כַּעֲב֣וֹר ס֭וּפָה וְאֵ֣ין רָשָׁ֑ע
וְ֝צַדִּ֗יק יְס֣וֹד עוֹלָֽם׃
כוכו כַּחֹ֤מֶץ ׀ לַשִּׁנַּ֗יִם וְכֶעָשָׁ֥ן לָעֵינָ֑יִם
כֵּ֥ן הֶ֝עָצֵ֗ל לְשֹׁלְחָֽיו׃
כזכז יִרְאַ֣ת יְ֭הֹוָה תּוֹסִ֣יף יָמִ֑ים
וּשְׁנ֖וֹת רְשָׁעִ֣ים תִּקְצֹֽרְנָה׃
כחכח תּוֹחֶ֣לֶת צַדִּיקִ֣ים שִׂמְחָ֑ה
וְתִקְוַ֖ת רְשָׁעִ֣ים תֹּאבֵֽד׃
כטכט מָע֣וֹז לַ֭תֹּם דֶּ֣רֶךְ יְהֹוָ֑ה
וּ֝מְחִתָּ֗ה לְפֹ֣עֲלֵי אָֽוֶן׃
לל צַ֭דִּיק לְעוֹלָ֣ם בַּל־יִמּ֑וֹט
וּ֝רְשָׁעִ֗ים לֹ֣א יִשְׁכְּנוּ־אָֽרֶץ׃
לאלא פִּֽי־צַ֭דִּיק יָנ֣וּב חׇכְמָ֑ה
וּלְשׁ֥וֹן תַּ֝הְפֻּכ֗וֹת תִּכָּרֵֽת׃
לבלב שִׂפְתֵ֣י צַ֭דִּיק יֵדְע֣וּן רָצ֑וֹן
וּפִ֥י רְ֝שָׁעִ֗ים תַּהְפֻּכֽוֹת׃
