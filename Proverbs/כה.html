ספר משלי

פרק כה


{פ}　　
כהאא גַּם־אֵ֭לֶּה מִשְׁלֵ֣י שְׁלֹמֹ֑ה
אֲשֶׁ֥ר הֶ֝עְתִּ֗יקוּ אַנְשֵׁ֤י ׀ חִזְקִיָּ֬ה מֶלֶךְ־יְהוּדָֽה׃
בב כְּבֹ֣ד אֱ֭לֹהִים הַסְתֵּ֣ר דָּבָ֑ר
וּכְבֹ֥ד מְ֝לָכִ֗ים חֲקֹ֣ר דָּבָֽר׃
גג שָׁמַ֣יִם לָ֭רוּם וָאָ֣רֶץ לָעֹ֑מֶק
וְלֵ֥ב מְ֝לָכִ֗ים אֵ֣ין חֵֽקֶר׃
דד הָג֣וֹ סִיגִ֣ים מִכָּ֑סֶף
וַיֵּצֵ֖א לַצֹּרֵ֣ף כֶּֽלִי׃
הה הָג֣וֹ רָ֭שָׁע לִפְנֵי־מֶ֑לֶךְ
וְיִכּ֖וֹן בַּצֶּ֣דֶק כִּסְאֽוֹ׃
וו אַל־תִּתְהַדַּ֥ר לִפְנֵי־מֶ֑לֶךְ
וּבִמְק֥וֹם גְּ֝דֹלִ֗ים אַֽל־תַּעֲמֹֽד׃
זז כִּ֤י ט֥וֹב אֲמׇר־לְךָ֗עֲֽלֵ֫ה הֵ֥נָּה
מֵ֭הַשְׁפִּ֣ילְךָ לִפְנֵ֣י נָדִ֑יב
אֲשֶׁ֖ר רָא֣וּ עֵינֶֽיךָ׃
חח אַל־תֵּצֵ֥א לָרִ֗במַ֫הֵ֥ר
פֶּ֣ן מַה־תַּ֭עֲשֶׂה בְּאַחֲרִיתָ֑הּ
בְּהַכְלִ֖ים אֹתְךָ֣ רֵעֶֽךָ׃
טט רִ֭יבְךָ רִ֣יב אֶת־רֵעֶ֑ךָ
וְס֖וֹד אַחֵ֣ר אַל־תְּגָֽל׃
יי פֶּֽן־יְחַסֶּדְךָ֥ שֹׁמֵ֑עַ
וְ֝דִבָּתְךָ֗ לֹ֣א תָשֽׁוּב׃
יאיא תַּפּוּחֵ֣י זָ֭הָב בְּמַשְׂכִּיּ֣וֹת כָּ֑סֶף
דָּ֝בָ֗ר דָּבֻ֥ר עַל־אׇפְנָֽיו׃
יביב נֶ֣זֶם זָ֭הָב וַחֲלִי־כָ֑תֶם
מוֹכִ֥יחַ חָ֝כָ֗ם עַל־אֹ֥זֶן שֹׁמָֽעַת׃
יגיג כְּצִנַּת־שֶׁ֨לֶג ׀ בְּי֬וֹם קָצִ֗יר
צִ֣יר נֶ֭אֱמָן לְשֹׁלְחָ֑יו
וְנֶ֖פֶשׁ אֲדֹנָ֣יו יָשִֽׁיב׃
{פ}　　
ידיד נְשִׂיאִ֣ים וְ֭רוּחַ וְגֶ֣שֶׁם אָ֑יִן
אִ֥ישׁ מִ֝תְהַלֵּ֗ל בְּמַתַּת־שָֽׁקֶר׃
טוטו בְּאֹ֣רֶךְ אַ֭פַּיִם יְפֻתֶּ֣ה קָצִ֑ין
וְלָשׁ֥וֹן רַ֝כָּ֗ה תִּשְׁבׇּר־גָּֽרֶם׃
טזטז דְּבַ֣שׁ מָ֭צָאתָ אֱכֹ֣ל דַּיֶּ֑ךָּ
פֶּן־תִּ֝שְׂבָּעֶ֗נּוּ וַהֲקֵאתֽוֹ׃
יזיז הֹקַ֣ר רַ֭גְלְךָ מִבֵּ֣ית רֵעֶ֑ךָ
פֶּן־יִ֝שְׂבָּעֲךָ֗ וּשְׂנֵאֶֽךָ׃
יחיח מֵפִ֣יץ וְ֭חֶרֶב וְחֵ֣ץ שָׁנ֑וּן
אִ֥ישׁ עֹנֶ֥ה בְ֝רֵעֵ֗הוּ עֵ֣ד שָֽׁקֶר׃
יטיט שֵׁ֣ן רֹ֭עָה וְרֶ֣גֶל מוּעָ֑דֶת
מִבְטָ֥ח בּ֝וֹגֵ֗ד בְּי֣וֹם צָרָֽה׃
ככ מַ֥עֲדֶה־בֶּ֨גֶד ׀ נמַ֥עֲדֶה־בֶּ֨גֶד | · =א ומ"ש (מקף בתיבת משרת)בְּי֣וֹם קָ֭רָה
חֹ֣מֶץ עַל־נָ֑תֶר
וְשָׁ֥ר בַּ֝שִּׁרִ֗יםעַ֣ל לֶב־רָֽע׃
{פ}　　
כאכא אִם־רָעֵ֣ב שֹׂ֭נַאֲךָ הַאֲכִלֵ֣הוּ לָ֑חֶם
וְאִם־צָ֝מֵ֗א הַשְׁקֵ֥הוּ מָֽיִם׃
כבכב כִּ֤י גֶחָלִ֗ים אַ֭תָּה חֹתֶ֣ה עַל־רֹאשׁ֑וֹ
וַ֝יהֹוָ֗ה יְשַׁלֶּם־לָֽךְ׃
כגכג ר֣וּחַ צָ֭פוֹן תְּח֣וֹלֵֽל גָּ֑שֶׁם
וּפָנִ֥ים נִ֝זְעָמִ֗ים לְשׁ֣וֹן סָֽתֶר׃
כדכד ט֗וֹב שֶׁ֥בֶת עַל־פִּנַּת־גָּ֑ג
מֵאֵ֥שֶׁת מִ֝דְויָנִ֗יםקקקרי/כתיב · (מדונים) מִ֝דְיָנִ֗ים וּבֵ֥ית חָֽבֶר׃
כהכה מַ֣יִם קָ֭רִים עַל־נֶ֣פֶשׁ עֲיֵפָ֑ה
וּשְׁמוּעָ֥ה ט֝וֹבָ֗ה מֵאֶ֥רֶץ מֶרְחָֽק׃
כוכו מַעְיָ֣ן נִ֭רְפָּשׂ וּמָק֣וֹר מׇשְׁחָ֑ת
צַ֝דִּ֗יק מָ֣ט לִפְנֵֽי־רָשָֽׁע׃
כזכז אָ֘כֹ֤ל דְּבַ֣שׁ הַרְבּ֣וֹת לֹא־ט֑וֹב
וְחֵ֖קֶר כְּבֹדָ֣ם כָּבֽוֹד׃
כחכח עִ֣יר פְּ֭רוּצָה אֵ֣ין חוֹמָ֑ה
אִ֝֗ישׁ אֲשֶׁ֤ר אֵ֖ין מַעְצָ֣ר לְרוּחֽוֹ׃
