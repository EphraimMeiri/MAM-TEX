מקרא על פי המסורה

ספר משלי

פרק ד


{פ}　　
דאא שִׁמְע֣וּ בָ֭נִים מ֣וּסַר אָ֑ב
וְ֝הַקְשִׁ֗יבוּ לָדַ֥עַת בִּינָֽה׃
בב כִּ֤י לֶ֣קַח ט֭וֹב נָתַ֣תִּי לָכֶ֑ם
תּ֝וֹרָתִ֗י אַֽל־תַּעֲזֹֽבוּ׃
גג כִּֽי־בֵ֭ן הָיִ֣יתִי לְאָבִ֑י
רַ֥ךְ וְ֝יָחִ֗יד לִפְנֵ֥י אִמִּֽי׃
דד וַיֹּרֵ֗נִיוַיֹּ֥אמֶר לִ֗י
יִֽתְמׇךְ־דְּבָרַ֥י לִבֶּ֑ךָ
שְׁמֹ֖ר מִצְוֺתַ֣י וֶחְיֵֽה׃
הה קְנֵ֣ה חׇ֭כְמָה קְנֵ֣ה בִינָ֑ה
אַל־תִּשְׁכַּ֥ח וְאַל־תֵּ֝֗ט מֵאִמְרֵי־פִֽי׃
וו אַל־תַּעַזְבֶ֥הָ וְתִשְׁמְרֶ֑ךָּ
אֱהָבֶ֥הָ וְתִצְּרֶֽךָּ׃
זז רֵאשִׁ֣ית חׇ֭כְמָה קְנֵ֣ה חׇכְמָ֑ה
וּבְכׇל־קִ֝נְיָנְךָ֗ קְנֵ֣ה בִינָֽה׃
חח סַלְסְלֶ֥הָ וּֽתְרוֹמְמֶ֑ךָּ
תְּ֝כַבֵּ֗דְךָ כִּ֣י תְחַבְּקֶֽנָּה׃
טט תִּתֵּ֣ן לְ֭רֹאשְׁךָ לִוְיַת־חֵ֑ן
עֲטֶ֖רֶת תִּפְאֶ֣רֶת תְּמַגְּנֶֽךָּ׃
יי שְׁמַ֣ע בְּ֭נִי וְקַ֣ח אֲמָרָ֑י
וְיִרְבּ֥וּ לְ֝ךָ֗ שְׁנ֣וֹת חַיִּֽים׃
יאיא בְּדֶ֣רֶךְ חׇ֭כְמָה הֹרֵיתִ֑יךָ
הִ֝דְרַכְתִּ֗יךָ בְּמַעְגְּלֵי־יֹֽשֶׁר׃
יביב בְּֽ֭לֶכְתְּךָ לֹא־יֵצַ֣ר צַעֲדֶ֑ךָ
וְאִם־תָּ֝ר֗וּץ לֹ֣א תִכָּשֵֽׁל׃
יגיג הַחֲזֵ֣ק בַּמּוּסָ֣ר אַל־תֶּ֑רֶף
נִ֝צְּרֶ֗הָ כִּי־הִ֥יא חַיֶּֽיךָ׃
ידיד בְּאֹ֣רַח רְ֭שָׁעִים אַל־תָּבֹ֑א
וְאַל־תְּ֝אַשֵּׁ֗ר בְּדֶ֣רֶךְ רָעִֽים׃
טוטו פְּרָעֵ֥הוּ אַל־תַּעֲבׇר־בּ֑וֹ
שְׂטֵ֖ה מֵעָלָ֣יו וַעֲבֹֽר׃
טזטז כִּ֤י לֹ֣א יִֽ֭שְׁנוּ אִם־לֹ֣א יָרֵ֑עוּ
וְֽנִגְזְלָ֥ה שְׁ֝נָתָ֗ם אִם־לֹ֥א יַכְשִֽׁוילוּקקקרי/כתיב · (יכשולו) יַכְשִֽׁילוּ׃
יזיז כִּ֣י לָ֭חֲמוּ לֶ֣חֶם רֶ֑שַׁע
וְיֵ֖ין חֲמָסִ֣ים יִשְׁתּֽוּ׃
יחיח וְאֹ֣רַח צַ֭דִּיקִים כְּא֣וֹר נֹ֑גַהּ
הוֹלֵ֥ךְ וָ֝א֗וֹר עַד־נְכ֥וֹן הַיּֽוֹם׃
יטיט דֶּ֣רֶךְ רְ֭שָׁעִים כָּאֲפֵלָ֑ה
לֹ֥א יָ֝דְע֗וּ בַּמֶּ֥ה יִכָּשֵֽׁלוּ׃
{פ}　　
ככ בְּ֭נִי לִדְבָרַ֣י הַקְשִׁ֑יבָה
לַ֝אֲמָרַ֗י הַט־אׇזְנֶֽךָ׃
כאכא אַל־יַלִּ֥יזוּ מֵעֵינֶ֑יךָ
שׇׁ֝מְרֵ֗ם בְּת֣וֹךְ לְבָבֶֽךָ׃
כבכב כִּֽי־חַיִּ֣ים הֵ֭ם לְמֹצְאֵיהֶ֑ם
וּֽלְכׇל־בְּשָׂר֥וֹ מַרְפֵּֽא׃
כגכג מִֽכׇּל־מִ֭שְׁמָר נְצֹ֣ר לִבֶּ֑ךָ
כִּֽי־מִ֝מֶּ֗נּוּ תּוֹצְא֥וֹת חַיִּֽים׃
כדכד הָסֵ֣ר מִ֭מְּךָ עִקְּשׁ֣וּת פֶּ֑ה
וּלְז֥וּת שְׂ֝פָתַ֗יִם הַרְחֵ֥ק מִמֶּֽךָּ׃
כהכה עֵ֭ינֶיךָ לְנֹ֣כַח יַבִּ֑יטוּ
וְ֝עַפְעַפֶּ֗יךָ יַיְשִׁ֥רוּ נֶגְדֶּֽךָ׃
כוכו פַּ֭לֵּס מַעְגַּ֣ל רַגְלֶ֑ךָ
וְֽכׇל־דְּרָכֶ֥יךָ יִכֹּֽנוּ׃
כזכז אַֽל־תֵּט־יָמִ֥ין וּשְׂמֹ֑אול
הָסֵ֖ר רַגְלְךָ֣ מֵרָֽע׃
