מקרא על פי המסורה

ספר משלי

פרק טז

טזאא לְאָדָ֥ם מַעַרְכֵי־לֵ֑ב
וּ֝מֵיְהֹוָ֗ה מַעֲנֵ֥ה לָשֽׁוֹן׃
בב כׇּֽל־דַּרְכֵי־אִ֭ישׁ זַ֣ךְ בְּעֵינָ֑יו
וְתֹכֵ֖ן רוּח֣וֹת יְהֹוָֽה׃
גג גֹּ֣ל אֶל־יְהֹוָ֣ה מַעֲשֶׂ֑יךָ
וְ֝יִכֹּ֗נוּ מַחְשְׁבֹתֶֽיךָ׃
דד כֹּ֤ל פָּעַ֣ל יְ֭הֹוָה לַֽמַּעֲנֵ֑הוּ
וְגַם־רָ֝שָׁ֗ע לְי֣וֹם רָעָֽה׃
הה תּוֹעֲבַ֣ת יְ֭הֹוָה כׇּל־גְּבַהּ־לֵ֑ב
יָ֥ד לְ֝יָ֗ד לֹ֣א יִנָּקֶֽה׃
וו בְּחֶ֣סֶד וֶ֭אֱמֶת יְכֻפַּ֣ר עָוֺ֑ן
וּבְיִרְאַ֥ת יְ֝הֹוָ֗ה ס֣וּר מֵרָֽע׃
זז בִּרְצ֣וֹת יְ֭הֹוָה דַּרְכֵי־אִ֑ישׁ
גַּם־א֝וֹיְבָ֗יו יַשְׁלִ֥ם אִתּֽוֹ׃
חח טוֹב־מְ֭עַט בִּצְדָקָ֑ה
מֵרֹ֥ב תְּ֝בוּא֗וֹת בְּלֹ֣א מִשְׁפָּֽט׃
טט לֵ֣ב אָ֭דָם יְחַשֵּׁ֣ב דַּרְכּ֑וֹ
וַ֝יהֹוָ֗ה יָכִ֥ין צַעֲדֽוֹ׃
יי קֶ֤סֶם ׀ עַֽל־שִׂפְתֵי־מֶ֑לֶךְ
בְּ֝מִשְׁפָּ֗ט לֹ֣א יִמְעַל־פִּֽיו׃
יאיא פֶּ֤לֶס ׀ וּמֹאזְנֵ֣י מִ֭שְׁפָּט לַיהֹוָ֑ה
מַ֝עֲשֵׂ֗הוּ כׇּל־אַבְנֵי־כִֽיס׃
יביב תּוֹעֲבַ֣ת מְ֭לָכִים עֲשׂ֣וֹת רֶ֑שַׁע
כִּ֥י בִ֝צְדָקָ֗ה יִכּ֥וֹן כִּסֵּֽא׃
יגיג רְצ֣וֹן מְ֭לָכִים שִׂפְתֵי־צֶ֑דֶק
וְדֹבֵ֖ר יְשָׁרִ֣ים יֶאֱהָֽב׃
ידיד חֲמַת־מֶ֥לֶךְ מַלְאֲכֵי־מָ֑וֶת
וְאִ֖ישׁ חָכָ֣ם יְכַפְּרֶֽנָּה׃
טוטו בְּאוֹר־פְּנֵי־מֶ֥לֶךְ חַיִּ֑ים
וּ֝רְצוֹנ֗וֹ כְּעָ֣ב מַלְקֽוֹשׁ׃
טזטז קְֽנֹה־חׇכְמָ֗ה מַה־טּ֥וֹב מֵחָר֑וּץ
וּקְנ֥וֹת בִּ֝ינָ֗ה נִבְחָ֥ר מִכָּֽסֶף׃
יזיז מְסִלַּ֣ת יְ֭שָׁרִים ס֣וּר מֵרָ֑ע
שֹׁמֵ֥ר נַ֝פְשׁ֗וֹ נֹצֵ֥ר דַּרְכּֽוֹ׃
יחיח לִפְנֵי־שֶׁ֥בֶר גָּא֑וֹן
וְלִפְנֵ֥י כִ֝שָּׁל֗וֹן גֹּ֣בַהּ רֽוּחַ׃
יטיט ט֣וֹב שְׁפַל־ר֭וּחַ אֶת־ענוייםקקקרי/כתיב · (עֲנָוִ֑ים) עניים
מֵחַלֵּ֥ק שָׁ֝לָ֗ל אֶת־גֵּאִֽים׃
ככ מַשְׂכִּ֣יל עַל־דָּ֭בָר יִמְצָא־ט֑וֹב
וּבוֹטֵ֖חַ בַּיהֹוָ֣ה אַשְׁרָֽיו׃
כאכא לַֽחֲכַם־לֵ֭ב יִקָּרֵ֣א נָב֑וֹן
וּמֶ֥תֶק שְׂ֝פָתַ֗יִם יֹסִ֥יף לֶֽקַח׃
כבכב מְק֣וֹר חַ֭יִּים שֵׂ֣כֶל בְּעָלָ֑יו
וּמוּסַ֖ר אֱוִלִ֣ים אִוֶּֽלֶת׃
כגכג לֵ֣ב חָ֭כָם יַשְׂכִּ֣יל פִּ֑יהוּ
וְעַל־שְׂ֝פָתָ֗יו יֹסִ֥יף לֶֽקַח׃
כדכד צוּף־דְּ֭בַשׁ אִמְרֵי־נֹ֑עַם
מָת֥וֹק לַ֝נֶּ֗פֶשׁ וּמַרְפֵּ֥א לָעָֽצֶם׃
כהכה יֵ֤שׁ דֶּ֣רֶךְ יָ֭שָׁר לִפְנֵי־אִ֑ישׁ
וְ֝אַחֲרִיתָ֗הּ דַּרְכֵי־מָֽוֶת׃
כוכו נֶ֣פֶשׁ עָ֭מֵל עָ֣מְלָה לּ֑וֹ
כִּֽי־אָכַ֖ף עָלָ֣יו פִּֽיהוּ׃
כזכז אִ֣ישׁ בְּ֭לִיַּעַל כֹּרֶ֣ה רָעָ֑ה
וְעַל־(שפתיו)קקקרי/כתיב · (שְׂ֝פָת֗וֹ) שפתיו כְּאֵ֣שׁ צָרָֽבֶת׃
כחכח אִ֣ישׁ תַּ֭הְפֻּכוֹת יְשַׁלַּ֣ח מָד֑וֹן
וְ֝נִרְגָּ֗ן מַפְרִ֥יד אַלּֽוּף׃
כטכט אִ֣ישׁ חָ֭מָס יְפַתֶּ֣ה רֵעֵ֑הוּ
וְ֝הוֹלִיכ֗וֹ בְּדֶ֣רֶךְ לֹא־טֽוֹב׃
לל עֹצֶ֣ה עֵ֭ינָיו לַחְשֹׁ֣ב תַּהְפֻּכ֑וֹת
קֹרֵ֥ץ שְׂ֝פָתָ֗יו כִּלָּ֥ה רָעָֽה׃
לאלא עֲטֶ֣רֶת תִּפְאֶ֣רֶת שֵׂיבָ֑ה
בְּדֶ֥רֶךְ צְ֝דָקָ֗ה תִּמָּצֵֽא׃
לבלב ט֤וֹב אֶ֣רֶךְ אַ֭פַּיִם מִגִּבּ֑וֹר
וּמֹשֵׁ֥ל בְּ֝רוּח֗וֹ מִלֹּכֵ֥ד עִֽיר׃
לגלג בַּ֭חֵיק יוּטַ֣ל אֶת־הַגּוֹרָ֑ל
וּ֝מֵיְהֹוָ֗ה כׇּל־מִשְׁפָּטֽוֹ׃
