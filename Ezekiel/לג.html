מקרא על פי המסורה

ספר יחזקאל

פרק לג


{פ}　　
לגאא וַיְהִ֥י דְבַר־יְהֹוָ֖ה אֵלַ֥י לֵאמֹֽר׃ בב בֶּן־אָדָ֗ם דַּבֵּ֤ר אֶל־בְּנֵֽי־עַמְּךָ֙ וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם אֶ֕רֶץ כִּֽי־אָבִ֥יא עָלֶ֖יהָ חָ֑רֶב וְלָקְח֨וּ עַם־הָאָ֜רֶץ אִ֤ישׁ אֶחָד֙ מִקְצֵיהֶ֔ם וְנָתְנ֥וּ אֹת֛וֹ לָהֶ֖ם לְצֹפֶֽה׃ גג וְרָאָ֥ה אֶת־הַחֶ֖רֶב בָּאָ֣ה עַל־הָאָ֑רֶץ וְתָקַ֥ע בַּשּׁוֹפָ֖ר וְהִזְהִ֥יר אֶת־הָעָֽם׃ דד וְשָׁמַ֨ע הַשֹּׁמֵ֜עַ אֶת־ק֤וֹל הַשּׁוֹפָר֙ וְלֹ֣א נִזְהָ֔ר וַתָּ֥בוֹא חֶ֖רֶב וַתִּקָּחֵ֑הוּ דָּמ֖וֹ בְּרֹאשׁ֥וֹ יִֽהְיֶֽה׃ הה אֵת֩ ק֨וֹל הַשּׁוֹפָ֤ר שָׁמַע֙ וְלֹ֣א נִזְהָ֔ר דָּמ֖וֹ בּ֣וֹ יִֽהְיֶ֑ה וְה֥וּא נִזְהָ֖ר נַפְשׁ֥וֹ מִלֵּֽט׃ וו וְ֠הַצֹּפֶ֠ה כִּי־יִרְאֶ֨ה אֶת־הַחֶ֜רֶב בָּאָ֗ה וְלֹֽא־תָקַ֤ע בַּשּׁוֹפָר֙ וְהָעָ֣ם לֹֽא־נִזְהָ֔ר וַתָּב֣וֹא חֶ֔רֶב וַתִּקַּ֥ח מֵהֶ֖ם נָ֑פֶשׁ ה֚וּא בַּעֲוֺנ֣וֹ נִלְקָ֔ח וְדָמ֖וֹ מִיַּֽד־הַצֹּפֶ֥ה אֶדְרֹֽשׁ׃
{פ}　　
זז וְאַתָּ֣ה בֶן־אָדָ֔ם צֹפֶ֥ה נְתַתִּ֖יךָ לְבֵ֣ית יִשְׂרָאֵ֑ל וְשָׁמַעְתָּ֤ מִפִּי֙ דָּבָ֔ר וְהִזְהַרְתָּ֥ אֹתָ֖ם מִמֶּֽנִּי׃ חח בְּאׇמְרִ֣י לָרָשָׁ֗ע רָשָׁע֙ מ֣וֹת תָּמ֔וּת וְלֹ֣א דִבַּ֔רְתָּ לְהַזְהִ֥יר רָשָׁ֖ע מִדַּרְכּ֑וֹ ה֤וּא רָשָׁע֙ בַּעֲוֺנ֣וֹ יָמ֔וּת וְדָמ֖וֹ מִיָּדְךָ֥ אֲבַקֵּֽשׁ׃ טט וְ֠אַתָּ֠ה כִּי־הִזְהַ֨רְתָּ רָשָׁ֤ע מִדַּרְכּוֹ֙ לָשׁ֣וּב מִמֶּ֔נָּה וְלֹא־שָׁ֖ב מִדַּרְכּ֑וֹ ה֚וּא בַּעֲוֺנ֣וֹ יָמ֔וּת וְאַתָּ֖ה נַפְשְׁךָ֥ הִצַּֽלְתָּ׃
{פ}　　
יי וְאַתָּ֣ה בֶן־אָדָ֗ם אֱמֹר֙ אֶל־בֵּ֣ית יִשְׂרָאֵ֔ל כֵּ֤ן אֲמַרְתֶּם֙ לֵאמֹ֔ר כִּֽי־פְשָׁעֵ֥ינוּ וְחַטֹּאתֵ֖ינוּ עָלֵ֑ינוּ וּבָ֛ם אֲנַ֥חְנוּ נְמַקִּ֖ים וְאֵ֥יךְ נִֽחְיֶֽה׃ יאיא אֱמֹ֨ר אֲלֵיהֶ֜ם חַי־אָ֣נִי ׀ נְאֻ֣ם ׀ אֲדֹנָ֣י יֱהֹוִ֗ה אִם־אֶחְפֹּץ֙ בְּמ֣וֹת הָרָשָׁ֔ע כִּ֣י אִם־בְּשׁ֥וּב רָשָׁ֛ע מִדַּרְכּ֖וֹ וְחָיָ֑ה שׁ֣וּבוּ שׁ֜וּבוּ מִדַּרְכֵיכֶ֧ם הָרָעִ֛ים וְלָ֥מָּה תָמ֖וּתוּ בֵּ֥ית יִשְׂרָאֵֽל׃
{פ}　　
יביב וְאַתָּ֣ה בֶן־אָדָ֗ם אֱמֹ֤ר אֶל־בְּנֵֽי־עַמְּךָ֙ צִדְקַ֣ת הַצַּדִּ֗יק לֹ֤א תַצִּילֶ֙נּוּ֙ בְּי֣וֹם פִּשְׁע֔וֹ וְרִשְׁעַ֤ת הָֽרָשָׁע֙ לֹא־יִכָּ֣שֶׁל בָּ֔הּ בְּי֖וֹם שׁוּב֣וֹ מֵֽרִשְׁע֑וֹ וְצַדִּ֗יק לֹ֥א יוּכַ֛ל לִֽחְי֥וֹת בָּ֖הּ בְּי֥וֹם חֲטֹאתֽוֹ׃ יגיג בְּאׇמְרִ֤י לַצַּדִּיק֙ חָיֹ֣ה יִֽחְיֶ֔ה וְהוּא־בָטַ֥ח עַל־צִדְקָת֖וֹ וְעָ֣שָׂה עָ֑וֶל כׇּל־צִדְקֹתָו֙קקקרי/כתיב · צִדְקֹתָו֙ · צִדְקֹתָיו֙ לֹ֣א תִזָּכַ֔רְנָה וּבְעַוְל֥וֹ אֲשֶׁר־עָשָׂ֖ה בּ֥וֹ יָמֽוּת׃ ידיד וּבְאׇמְרִ֥י לָרָשָׁ֖ע מ֣וֹת תָּמ֑וּת וְשָׁב֙ מֵֽחַטָּאת֔וֹ וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָֽה׃ טוטו חֲבֹ֨ל יָשִׁ֤יב רָשָׁע֙ גְּזֵלָ֣ה יְשַׁלֵּ֔ם בְּחֻקּ֤וֹת הַֽחַיִּים֙ הָלַ֔ךְ לְבִלְתִּ֖י עֲשׂ֣וֹת עָ֑וֶל חָי֥וֹ יִֽחְיֶ֖ה לֹ֥א יָמֽוּת׃ טזטז כׇּל־חַטֹּאתָו֙קקקרי/כתיב · חַטֹּאתָו֙ · חַטֹּאתָיו֙ אֲשֶׁ֣ר חָטָ֔א לֹ֥א תִזָּכַ֖רְנָה ל֑וֹ מִשְׁפָּ֧ט וּצְדָקָ֛ה עָשָׂ֖ה חָי֥וֹ יִֽחְיֶֽה׃ יזיז וְאָֽמְרוּ֙ בְּנֵ֣י עַמְּךָ֔ לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י וְהֵ֖מָּה דַּרְכָּ֥ם לֹא־יִתָּכֵֽן׃ יחיח בְּשׁוּב־צַדִּ֥יק מִצִּדְקָת֖וֹ וְעָ֣שָׂה עָ֑וֶל וּמֵ֖ת בָּהֶֽם׃ יטיט וּבְשׁ֤וּב רָשָׁע֙ מֵֽרִשְׁעָת֔וֹ וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָ֑ה עֲלֵיהֶ֖ם ה֥וּא יִֽחְיֶֽה׃ ככ וַאֲמַרְתֶּ֕ם לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י אִ֧ישׁ כִּדְרָכָ֛יו אֶשְׁפּ֥וֹט אֶתְכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃
{פ}　　
כאכא וַיְהִ֞י בִּשְׁתֵּ֧י עֶשְׂרֵ֣ה שָׁנָ֗ה בָּעֲשִׂרִ֛י בַּחֲמִשָּׁ֥ה לַחֹ֖דֶשׁ לְגָלוּתֵ֑נוּ בָּא־אֵלַ֨י הַפָּלִ֧יט מִירוּשָׁלַ֛͏ִםם לֵאמֹ֖ר הֻכְּתָ֥ה הָעִֽיר׃ כבכב וְיַד־יְהֹוָה֩ הָיְתָ֨ה אֵלַ֜י בָּעֶ֗רֶב לִפְנֵי֙ בּ֣וֹא הַפָּלִ֔יט וַיִּפְתַּ֣ח אֶת־פִּ֔י עַד־בּ֥וֹא אֵלַ֖י בַּבֹּ֑קֶר וַיִּפָּ֣תַח פִּ֔י וְלֹ֥א נֶאֱלַ֖מְתִּי עֽוֹד׃
{פ}　　
כגכג וַיְהִ֥י דְבַר־יְהֹוָ֖ה אֵלַ֥י לֵאמֹֽר׃ כדכד בֶּן־אָדָ֗ם יֹ֠שְׁבֵ֠י הֶחֳרָב֨וֹת הָאֵ֜לֶּה עַל־אַדְמַ֤ת יִשְׂרָאֵל֙ אֹמְרִ֣ים לֵאמֹ֔ר אֶחָד֙ הָיָ֣ה אַבְרָהָ֔ם וַיִּירַ֖שׁ אֶת־הָאָ֑רֶץ וַאֲנַ֣חְנוּ רַבִּ֔ים לָ֛נוּ נִתְּנָ֥ה הָאָ֖רֶץ לְמוֹרָשָֽׁה׃
{ס}　　　
כהכה לָכֵן֩ אֱמֹ֨ר אֲלֵהֶ֜ם כֹּה־אָמַ֣ר ׀ אֲדֹנָ֣י יֱהֹוִ֗ה עַל־הַדָּ֧ם ׀ תֹּאכֵ֛לוּ וְעֵינֵכֶ֛ם תִּשְׂא֥וּ אֶל־גִּלּוּלֵיכֶ֖ם וְדָ֣ם תִּשְׁפֹּ֑כוּ וְהָאָ֖רֶץ תִּירָֽשׁוּ׃ כוכו עֲמַדְתֶּ֤ם עַֽל־חַרְבְּכֶם֙ עֲשִׂיתֶ֣ן תּוֹעֵבָ֔ה וְאִ֛ישׁ אֶת־אֵ֥שֶׁת רֵעֵ֖הוּ טִמֵּאתֶ֑ם וְהָאָ֖רֶץ תִּירָֽשׁוּ׃
{ס}　　　
כזכז כֹּה־תֹאמַ֨ר אֲלֵהֶ֜ם כֹּה־אָמַ֨ר אֲדֹנָ֣י יֱהֹוִה֮ חַי־אָ֒נִי֒ אִם־לֹ֞א אֲשֶׁ֤ר בֶּֽחֳרָבוֹת֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וַֽאֲשֶׁר֙ עַל־פְּנֵ֣י הַשָּׂדֶ֔ה לַחַיָּ֥ה נְתַתִּ֖יו לְאׇכְל֑וֹ וַאֲשֶׁ֛ר בַּמְּצָד֥וֹת וּבַמְּעָר֖וֹת בַּדֶּ֥בֶר יָמֽוּתוּ׃ כחכח וְנָתַתִּ֤י אֶת־הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה וְנִשְׁבַּ֖ת גְּא֣וֹן עֻזָּ֑הּ וְשָׁ֥מְﬞמ֛וּנוְשָׁ֥מְﬞמ֛וּ · א=וְשָׁ֥מֲמ֛וּ (חטף) הָרֵ֥י יִשְׂרָאֵ֖ל מֵאֵ֥ין עוֹבֵֽר׃ כטכט וְיָדְע֖וּ כִּֽי־אֲנִ֣י יְהֹוָ֑ה בְּתִתִּ֤י אֶת־הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה עַ֥ל כׇּל־תּוֹעֲבֹתָ֖ם אֲשֶׁ֥ר עָשֽׂוּ׃
{פ}　　
לל וְאַתָּ֣ה בֶן־אָדָ֔ם בְּנֵ֣י עַמְּךָ֗ הַנִּדְבָּרִ֤ים בְּךָ֙ אֵ֣צֶל הַקִּיר֔וֹת וּבְפִתְחֵ֖י הַבָּתִּ֑ים וְדִבֶּר־חַ֣ד אֶת־אַחַ֗ד אִ֤ישׁ אֶת־אָחִיו֙ לֵאמֹ֔ר בֹּאוּ־נָ֣א וְשִׁמְע֔וּ מָ֣ה הַדָּבָ֔ר הַיּוֹצֵ֖א מֵאֵ֥ת יְהֹוָֽה׃ לאלא וְיָב֣וֹאוּ אֵ֠לֶ֠יךָ כִּמְבוֹא־עָ֞ם וְיֵשְׁב֤וּ לְפָנֶ֙יךָ֙ עַמִּ֔י וְשָֽׁמְעוּ֙ אֶת־דְּבָרֶ֔יךָ וְאוֹתָ֖ם לֹ֣א יַעֲשׂ֑וּ כִּֽי־עֲגָבִ֤ים בְּפִיהֶם֙ הֵ֣מָּה עֹשִׂ֔ים אַחֲרֵ֥י בִצְעָ֖ם לִבָּ֥ם הֹלֵֽךְ׃ לבלב וְהִנְּךָ֤ לָהֶם֙ כְּשִׁ֣יר עֲגָבִ֔ים יְפֵ֥ה ק֖וֹל וּמֵטִ֣ב נַגֵּ֑ן וְשָֽׁמְעוּ֙ אֶת־דְּבָרֶ֔יךָ וְעֹשִׂ֥ים אֵינָ֖ם אוֹתָֽם׃ לגלג וּבְבֹאָ֑הּ הִנֵּ֣ה בָאָ֔ה וְיָ֣דְע֔וּ כִּ֥י נָבִ֖יא הָיָ֥ה בְתוֹכָֽם׃
