מקרא על פי המסורה

ספר יחזקאל

פרק כג


{פ}　　
כגאא וַיְהִ֥י דְבַר־יְהֹוָ֖ה אֵלַ֥י לֵאמֹֽר׃ בב בֶּן־אָדָ֑ם שְׁתַּ֣יִם נָשִׁ֔ים בְּנ֥וֹת אֵם־אַחַ֖ת הָיֽוּ׃ גג וַתִּזְנֶ֣ינָה בְמִצְרַ֔יִם בִּנְעוּרֵיהֶ֖ן זָנ֑וּ שָׁ֚מָּה מֹעֲכ֣וּ שְׁדֵיהֶ֔ן וְשָׁ֣ם עִשּׂ֔וּ דַּדֵּ֖י בְּתוּלֵיהֶֽן׃ דד וּשְׁמוֹתָ֗ן הַגְּדוֹלָה֙ אֲחוֹתָ֔הּ וַתִּהְיֶ֣ינָה לִ֔י וַתֵּלַ֖דְנָה בָּנִ֣ים וּבָנ֑וֹת וּשְׁמוֹתָ֕ן שֹׁמְר֣וֹן וִירוּשָׁלַ֖͏ִםם ׃ הה וַתִּ֥זֶן תַּחְתָּ֑י וַתֶּעְגַּב֙ עַֽל־מְאַהֲבֶ֔יהָ אֶל־אַשּׁ֖וּר קְרוֹבִֽים׃ וו לְבֻשֵׁ֤י תְכֵ֙לֶת֙ פַּח֣וֹת וּסְגָנִ֔ים בַּח֥וּרֵי חֶ֖מֶד כֻּלָּ֑ם פָּרָשִׁ֕ים רֹכְבֵ֖י סוּסִֽים׃ זז וַתִּתֵּ֤ן תַּזְנוּתֶ֙יהָ֙ עֲלֵיהֶ֔ם מִבְחַ֥ר בְּנֵֽי־אַשּׁ֖וּר כֻּלָּ֑ם וּבְכֹ֧ל אֲשֶׁר־עָגְבָ֛ה בְּכׇל־גִּלּוּלֵיהֶ֖ם נִטְמָֽאָה׃ חח וְאֶת־תַּזְנוּתֶ֤יהָ מִמִּצְרַ֙יִם֙ לֹ֣א עָזָ֔בָה כִּ֤י אוֹתָהּ֙ שָׁכְב֣וּ בִנְעוּרֶ֔יהָ וְהֵ֥מָּה עִשּׂ֖וּ דַּדֵּ֣י בְתוּלֶ֑יהָ וַיִּשְׁפְּכ֥וּ תַזְנוּתָ֖ם עָלֶֽיהָ׃ טט לָכֵ֥ן נְתַתִּ֖יהָ בְּיַד־מְאַהֲבֶ֑יהָ בְּיַד֙ בְּנֵ֣י אַשּׁ֔וּר אֲשֶׁ֥ר עָגְבָ֖ה עֲלֵיהֶֽם׃ יי הֵ֘מָּה֮ גִּלּ֣וּ עֶרְוָתָהּ֒ בָּנֶ֤יהָ וּבְנוֹתֶ֙יהָ֙ לָקָ֔חוּ וְאוֹתָ֖הּ בַּחֶ֣רֶב הָרָ֑גוּ וַתְּהִי־שֵׁם֙ לַנָּשִׁ֔ים וּשְׁפוּטִ֖ים עָ֥שׂוּ בָֽהּ׃
{ס}　　　
יאיא וַתֵּ֙רֶא֙ אֲחוֹתָ֣הּ וַתַּשְׁחֵ֥ת עַגְבָתָ֖הּ מִמֶּ֑נָּה וְאֶ֨ת־תַּזְנוּתֶ֔יהָ מִזְּנוּנֵ֖י אֲחוֹתָֽהּ׃ יביב אֶל־בְּנֵי֩ אַשּׁ֨וּר עָגָ֜בָה פַּח֨וֹת וּסְגָנִ֤ים קְרֹבִים֙ לְבֻשֵׁ֣י מִכְל֔וֹל פָּרָשִׁ֖ים רֹכְבֵ֣י סוּסִ֑ים בַּח֥וּרֵי חֶ֖מֶד כֻּלָּֽם׃ יגיג וָאֵ֖רֶא כִּ֣י נִטְמָ֑אָה דֶּ֥רֶךְ אֶחָ֖ד לִשְׁתֵּיהֶֽן׃ ידיד וַתּ֖וֹסֶף אֶל־תַּזְנוּתֶ֑יהָ וַתֵּ֗רֶא אַנְשֵׁי֙ מְחֻקֶּ֣ה עַל־הַקִּ֔יר צַלְמֵ֣י (כשדיים)קקקרי/כתיב · (כשדיים) כַשְׂדִּ֔ים חֲקֻקִ֖ים בַּשָּׁשַֽׁר׃ טוטו חֲגוֹרֵ֨י אֵז֜וֹר בְּמׇתְנֵיהֶ֗ם סְרוּחֵ֤י טְבוּלִים֙ בְּרָ֣אשֵׁיהֶ֔ם מַרְאֵ֥ה שָׁלִשִׁ֖ים כֻּלָּ֑ם דְּמ֤וּת בְּנֵֽי־בָבֶל֙ כַּשְׂדִּ֔ים אֶ֖רֶץ מוֹלַדְתָּֽם׃ טזטז (ותעגב)קקקרי/כתיב · (ותעגב) וַתַּעְגְּבָ֥ה עֲלֵיהֶ֖ם לְמַרְאֵ֣ה עֵינֶ֑יהָ וַתִּשְׁלַ֧ח מַלְאָכִ֛ים אֲלֵיהֶ֖ם כַּשְׂדִּֽימָה׃ יזיז וַיָּבֹ֨אוּ אֵלֶ֤יהָ בְנֵֽי־בָבֶל֙ לְמִשְׁכַּ֣ב דֹּדִ֔ים וַיְטַמְּא֥וּ אוֹתָ֖הּ בְּתַזְנוּתָ֑ם וַתִּ֨טְמָא־בָ֔ם וַתֵּ֥קַע נַפְשָׁ֖הּ מֵהֶֽם׃ יחיח וַתְּגַל֙ תַּזְנוּתֶ֔יהָ וַתְּגַ֖ל אֶת־עֶרְוָתָ֑הּ וַתֵּ֤קַע נַפְשִׁי֙ מֵעָלֶ֔יהָ כַּאֲשֶׁ֛ר נָקְעָ֥ה נַפְשִׁ֖י מֵעַ֥ל אֲחוֹתָֽהּ׃ יטיט וַתַּרְבֶּ֖ה אֶת־תַּזְנוּתֶ֑יהָ לִזְכֹּר֙ אֶת־יְמֵ֣י נְעוּרֶ֔יהָ אֲשֶׁ֥ר זָנְתָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ ככ וַֽתַּעְגְּבָ֔ה עַ֖ל פִּֽלַגְשֵׁיהֶ֑ם אֲשֶׁ֤ר בְּשַׂר־חֲמוֹרִים֙ בְּשָׂרָ֔ם וְזִרְמַ֥ת סוּסִ֖ים זִרְמָתָֽם׃ כאכא וַֽתִּפְקְדִ֔י אֵ֖ת זִמַּ֣ת נְעוּרָ֑יִךְ בַּעְשׂ֤וֹת מִמִּצְרַ֙יִם֙ דַּדַּ֔יִךְ לְמַ֖עַן שְׁדֵ֥י נְעוּרָֽיִךְ׃
{ס}　　　
כבכב לָכֵ֣ן כֹּה־אָמַר֮ אֲדֹנָ֣י יֱהֹוִה֒ הִנְנִ֨י מֵעִ֤יר אֶת־מְאַהֲבַ֙יִךְ֙ עָלַ֔יִךְ אֵ֛ת אֲשֶׁר־נָקְעָ֥ה נַפְשֵׁ֖ךְ מֵהֶ֑ם וַהֲבֵאתִ֥ים עָלַ֖יִךְ מִסָּבִֽיב׃ כגכג בְּנֵ֧י בָבֶ֣ל וְכׇל־כַּשְׂדִּ֗ים פְּק֤וֹד וְשׁ֙וֹעַ֙ וְק֔וֹעַ כׇּל־בְּנֵ֥י אַשּׁ֖וּר אוֹתָ֑ם בַּח֨וּרֵי חֶ֜מֶד פַּח֤וֹת וּסְגָנִים֙ כֻּלָּ֔ם שָֽׁלִשִׁים֙ וּקְרוּאִ֔ים רֹכְבֵ֥י סוּסִ֖ים כֻּלָּֽם׃ כדכד וּבָ֣אוּ עָלַ֡יִךְ הֹ֠צֶן רֶ֤כֶב וְגַלְגַּל֙ וּבִקְהַ֣ל עַמִּ֔ים צִנָּ֤ה וּמָגֵן֙ וְקוֹבַ֔ע יָשִׂ֥ימוּ עָלַ֖יִךְ סָבִ֑יב וְנָתַתִּ֤י לִפְנֵיהֶם֙ מִשְׁפָּ֔ט וּשְׁפָט֖וּךְ בְּמִשְׁפְּטֵיהֶֽם׃ כהכה וְנָתַתִּ֨י קִנְאָתִ֜י בָּ֗ךְ וְעָשׂ֤וּ אוֹתָךְ֙ בְּחֵמָ֔ה אַפֵּ֤ךְ וְאׇזְנַ֙יִךְ֙ יָסִ֔ירוּ וְאַחֲרִיתֵ֖ךְ בַּחֶ֣רֶב תִּפּ֑וֹל הֵ֗מָּה בָּנַ֤יִךְ וּבְנוֹתַ֙יִךְ֙ יִקָּ֔חוּ וְאַחֲרִיתֵ֖ךְ תֵּאָכֵ֥ל בָּאֵֽשׁ׃ כוכו וְהִפְשִׁיט֖וּךְ אֶת־בְּגָדָ֑יִךְ וְלָקְח֖וּ כְּלֵ֥י תִפְאַרְתֵּֽךְ׃ כזכז וְהִשְׁבַּתִּ֤י זִמָּתֵךְ֙ מִמֵּ֔ךְ וְאֶת־זְנוּתֵ֖ךְ מֵאֶ֣רֶץ מִצְרָ֑יִם וְלֹֽא־תִשְׂאִ֤י עֵינַ֙יִךְ֙ אֲלֵיהֶ֔ם וּמִצְרַ֖יִם לֹ֥א תִזְכְּרִי־עֽוֹד׃
{פ}　　
כחכח כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יֱהֹוִ֔ה הִנְנִי֙ נֹֽתְנָ֔ךְ בְּיַ֖ד אֲשֶׁ֣ר שָׂנֵ֑את בְּיַ֛ד אֲשֶׁר־נָקְעָ֥ה נַפְשֵׁ֖ךְ מֵהֶֽם׃ כטכט וְעָשׂ֨וּ אוֹתָ֜ךְ בְּשִׂנְאָ֗ה וְלָֽקְחוּ֙ כׇּל־יְגִיעֵ֔ךְ וַעֲזָב֖וּךְ עֵירֹ֣ם וְעֶרְיָ֑ה וְנִגְלָה֙ עֶרְוַ֣ת זְנוּנַ֔יִךְ וְזִמָּתֵ֖ךְ וְתַזְנוּתָֽיִךְ׃ לל עָשֹׂ֥ה אֵ֖לֶּה לָ֑ךְ בִּזְנוֹתֵךְ֙ אַחֲרֵ֣י גוֹיִ֔ם עַ֥ל אֲשֶׁר־נִטְמֵ֖את בְּגִלּוּלֵיהֶֽם׃ לאלא בְּדֶ֥רֶךְ אֲחוֹתֵ֖ךְ הָלָ֑כְתְּ וְנָתַתִּ֥י כוֹסָ֖הּ בְּיָדֵֽךְ׃
{ס}　　　
לבלב כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יֱהֹוִ֔ה כּ֤וֹס אֲחוֹתֵךְ֙ תִּשְׁתִּ֔י הָעֲמֻקָּ֖ה וְהָרְחָבָ֑ה תִּהְיֶ֥ה לִצְחֹ֛ק וּלְלַ֖עַג מִרְבָּ֥ה לְהָכִֽיל׃ לגלג שִׁכָּר֥וֹן וְיָג֖וֹן תִּמָּלֵ֑אִי כּ֚וֹס שַׁמָּ֣ה וּשְׁמָמָ֔ה כּ֖וֹס אֲחוֹתֵ֥ךְ שֹׁמְרֽוֹן׃ לדלד וְשָׁתִ֨ית אוֹתָ֜הּ וּמָצִ֗ית וְאֶת־חֲרָשֶׂ֛יהָ תְּגָרֵ֖מִי וְשָׁדַ֣יִךְ תְּנַתֵּ֑קִי כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יֱהֹוִֽה׃
{ס}　　　
להלה לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יֱהֹוִ֔ה יַ֚עַן שָׁכַ֣חַתְּ אוֹתִ֔י וַתַּשְׁלִ֥יכִי אוֹתִ֖י אַחֲרֵ֣י גַוֵּ֑ךְ וְגַם־אַ֛תְּ שְׂאִ֥י זִמָּתֵ֖ךְ וְאֶת־תַּזְנוּתָֽיִךְ׃
{ס}　　　
לולו וַיֹּ֤אמֶר יְהֹוָה֙ אֵלַ֔י בֶּן־אָדָ֕ם הֲתִשְׁפּ֥וֹט וְהַגֵּ֣ד לָהֶ֔ן אֵ֖ת תּוֹעֲבוֹתֵיהֶֽן׃ לזלז כִּ֣י נִאֵ֗פוּ וְדָם֙ בִּֽידֵיהֶ֔ן וְאֶת־גִּלּוּלֵיהֶ֖ן נִאֵ֑פוּ וְגַ֤ם אֶת־בְּנֵיהֶן֙ אֲשֶׁ֣ר יָֽלְדוּ־לִ֔י הֶעֱבִ֥ירוּ לָהֶ֖ם לְאׇכְלָֽה׃ לחלח ע֥וֹד זֹ֖את עָ֣שׂוּ לִ֑י טִמְּא֤וּ אֶת־מִקְדָּשִׁי֙ בַּיּ֣וֹם הַה֔וּא וְאֶת־שַׁבְּתוֹתַ֖י חִלֵּֽלוּ׃ לטלט וּֽבְשַׁחֲטָ֤ם אֶת־בְּנֵיהֶם֙ לְגִלּ֣וּלֵיהֶ֔ם וַיָּבֹ֧אוּ אֶל־מִקְדָּשִׁ֛י בַּיּ֥וֹם הַה֖וּא לְחַלְּל֑וֹ וְהִנֵּה־כֹ֥ה עָשׂ֖וּ בְּת֥וֹךְ בֵּיתִֽי׃ ממ וְאַ֗ף כִּ֤י תִשְׁלַ֙חְנָה֙ לַאֲנָשִׁ֔ים בָּאִ֖ים מִמֶּרְחָ֑ק אֲשֶׁ֨ר מַלְאָ֜ךְ שָׁל֤וּחַ אֲלֵיהֶם֙ וְהִנֵּה־בָ֔אוּ לַאֲשֶׁ֥ר רָחַ֛צְתְּ כָּחַ֥לְתְּ עֵינַ֖יִךְ וְעָדִ֥ית עֶֽדִי׃ מאמא וְיָשַׁבְתְּ֙ עַל־מִטָּ֣ה כְבוּדָּ֔ה וְשֻׁלְחָ֥ן עָר֖וּךְ לְפָנֶ֑יהָ וּקְטׇרְתִּ֥י וְשַׁמְנִ֖י שַׂ֥מְתְּ עָלֶֽיהָ׃ מבמב וְק֣וֹל הָמוֹן֮ שָׁלֵ֣ו בָֿהּ֒ וְאֶל־אֲנָשִׁים֙ מֵרֹ֣ב אָדָ֔ם מוּבָאִ֥ים (סובאים)קקקרי/כתיב · סובאים מִמִּדְבָּ֑ר וַֽיִּתְּנ֤וּ צְמִידִים֙ אֶל־יְדֵיהֶ֔ן וַעֲטֶ֥רֶת תִּפְאֶ֖רֶת עַל־רָאשֵׁיהֶֽן׃ מגמג וָאֹמַ֕ר לַבָּלָ֖ה נִֽאוּפִ֑ים (עת)קקקרי/כתיב · (עת) עַתָּ֛ה (יזנה)קקקרי/כתיב · (יזנה) יִזְנ֥וּ תַזְנוּתֶ֖הָ וָהִֽיא׃ מדמד וַיָּב֣וֹא אֵלֶ֔יהָ כְּב֖וֹא אֶל־אִשָּׁ֣ה זוֹנָ֑ה כֵּ֣ן בָּ֗אוּ אִשֹּׁ֖ת הַזִּמָּֽה׃ מהמה וַאֲנָשִׁ֣ים צַדִּיקִ֗ם הֵ֚מָּה יִשְׁפְּט֣וּ אֽוֹתְהֶ֔ם מִשְׁפַּט֙ נֹֽאֲפ֔וֹת וּמִשְׁפַּ֖ט שֹׁפְכ֣וֹת דָּ֑ם כִּ֤י נֹֽאֲפֹת֙ הֵ֔נָּה וְדָ֖ם בִּידֵיהֶֽן׃

מומו כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יֱהֹוִ֑ה הַעֲלֵ֤ה עֲלֵיהֶם֙ קָהָ֔ל וְנָתֹ֥ן אֶתְהֶ֖ן לְזַעֲוָ֥ה וְלָבַֽז׃ מזמז וְרָגְמ֨וּ עֲלֵיהֶ֥ן אֶ֙בֶן֙ קָהָ֔ל וּבָרֵ֥א אוֹתְהֶ֖ן בְּחַרְבוֹתָ֑ם בְּנֵיהֶ֤ם וּבְנֽוֹתֵיהֶם֙ יַהֲרֹ֔גוּ וּבָתֵּיהֶ֖ן בָּאֵ֥שׁ יִשְׂרֹֽפוּ׃ מחמח וְהִשְׁבַּתִּ֥י זִמָּ֖ה מִן־הָאָ֑רֶץ וְנִוַּסְּרוּ֙ כׇּל־הַנָּשִׁ֔ים וְלֹ֥א תַעֲשֶׂ֖ינָה כְּזִמַּתְכֶֽנָה׃ מטמט וְנָתְנ֤וּ זִמַּתְכֶ֙נָה֙ עֲלֵיכֶ֔ן וַחֲטָאֵ֥י גִלּוּלֵיכֶ֖ן תִּשֶּׂ֑אינָה וִידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יֱהֹוִֽה׃
