ספר דברי הימים

פרק ה


{פ}　　
האא וּבְנֵ֨י רְאוּבֵ֥ן בְּכֽוֹר־יִשְׂרָאֵל֮ כִּ֣י ה֣וּא הַבְּכוֹר֒ וּֽבְחַלְּלוֹ֙ יְצוּעֵ֣י אָבִ֔יו נִתְּנָה֙ בְּכֹ֣רָת֔וֹ לִבְנֵ֥י יוֹסֵ֖ף בֶּן־יִשְׂרָאֵ֑ל וְלֹ֥א לְהִתְיַחֵ֖שׂ לַבְּכֹרָֽה׃ בב כִּ֤י יְהוּדָה֙ גָּבַ֣ר בְּאֶחָ֔יו וּלְנָגִ֖יד מִמֶּ֑נּוּ וְהַבְּכֹרָ֖ה לְיוֹסֵֽף׃
{ס}　　　
גג בְּנֵ֥י רְאוּבֵ֖ן בְּכ֣וֹר יִשְׂרָאֵ֑ל חֲנ֥וֹךְ וּפַלּ֖וּא חֶצְר֥וֹן וְכַרְמִֽי׃ דד בְּנֵ֖י יוֹאֵ֑ל שְׁמַֽעְיָ֥ה בְנ֛וֹ גּ֥וֹג בְּנ֖וֹ שִׁמְעִ֥י בְנֽוֹ׃ הה מִיכָ֥ה בְנ֛וֹ רְאָיָ֥ה בְנ֖וֹ בַּ֥עַל בְּנֽוֹ׃ וו בְּאֵרָ֣ה בְנ֔וֹ אֲשֶׁ֣ר הֶגְלָ֔ה תִּלְּגַ֥ת פִּלְנְאֶ֖סֶר מֶ֣לֶךְ אַשֻּׁ֑ר ה֥וּא נָשִׂ֖יא לָרֽאוּבֵנִֽי׃ זז וְאֶחָיו֙ לְמִשְׁפְּחֹתָ֔יו בְּהִתְיַחֵ֖שׂ לְתֹלְדוֹתָ֑ם הָרֹ֥אשׁ יְעִיאֵ֖ל וּזְכַרְיָֽהוּ׃ חח וּבֶ֙לַע֙ בֶּן־עָזָ֔ז בֶּן־שֶׁ֖מַע בֶּן־יוֹאֵ֑ל ה֚וּא יוֹשֵׁ֣ב בַּעֲרֹעֵ֔ר וְעַד־נְב֖וֹ וּבַ֥עַל מְעֽוֹן׃ טט וְלַמִּזְרָ֗ח יָשַׁב֙ עַד־לְב֣וֹא מִדְבָּ֔רָה לְמִן־הַנָּהָ֖ר פְּרָ֑ת כִּ֧י מִקְנֵיהֶ֛ם רָב֖וּ בְּאֶ֥רֶץ גִּלְעָֽד׃ יי וּבִימֵ֣י שָׁא֗וּל עָשׂ֤וּ מִלְחָמָה֙ עִם־הַֽהַגְרִאִ֔ים וַֽיִּפְּל֖וּ בְּיָדָ֑ם וַיֵּֽשְׁבוּ֙ עַֽל־כׇּל־פְּנֵ֖י מִזְרָ֥ח לַגִּלְעָֽד׃
{ס}　　　
יאיא וּבְנֵי־גָ֣ד לְנֶגְדָּ֗ם יָ֥שְׁב֛וּ בְּאֶ֥רֶץ הַבָּשָׁ֖ן עַד־סַלְכָֽה׃ יביב יוֹאֵ֣ל הָרֹ֔אשׁ וְשָׁפָ֖ם הַמִּשְׁנֶ֑ה וְיַעְנַ֥י וְשָׁפָ֖ט בַּבָּשָֽׁן׃ יגיג וַאֲחֵיהֶ֞ם לְבֵ֣ית אֲבוֹתֵיהֶ֗ם מִיכָאֵ֡ל וּמְשֻׁלָּ֡ם וְ֠שֶׁ֠בַע וְיוֹרַ֧י וְיַעְכָּ֛ן וְזִ֥יעַ וָעֵ֖בֶר שִׁבְעָֽה׃
{ס}　　　
ידיד אֵ֣לֶּה ׀ בְּנֵ֣י אֲבִיחַ֗יִל בֶּן־חוּרִ֡י בֶּן־יָ֠ר֠וֹחַ בֶּן־גִּלְעָ֧ד בֶּן־מִיכָאֵ֛ל בֶּן־יְשִׁישַׁ֥י בֶּן־יַחְדּ֖וֹ בֶּן־בּֽוּז׃ טוטו אֲחִי֙ בֶּן־עַבְדִּיאֵ֣ל בֶּן־גּוּנִ֔י רֹ֖אשׁ לְבֵ֥ית אֲבוֹתָֽם׃ טזטז וַיֵּ֥שְׁב֛וּ בַּגִּלְעָ֥ד בַּבָּשָׁ֖ן וּבִבְנֹתֶ֑יהָ וּבְכׇֽל־מִגְרְﬞשֵׁ֥ינוּבְכׇֽל־מִגְרְﬞשֵׁ֥י · א=וּבְכׇֽל־מִגְרֲשֵׁ֥י (חטף) שָׁר֖וֹן עַל־תּוֹצְאוֹתָֽם׃ יזיז כֻּלָּם֙ הִתְיַחְשׂ֔וּ בִּימֵ֖י יוֹתָ֣ם מֶֽלֶךְ־יְהוּדָ֑ה וּבִימֵ֖י יָרׇבְעָ֥ם מֶלֶךְ־יִשְׂרָאֵֽל׃
{פ}　　
יחיח בְּנֵי־רְאוּבֵ֨ן וְגָדִ֜י וַחֲצִ֥י שֵֽׁבֶט־מְנַשֶּׁה֮ מִן־בְּנֵי־חַ֒יִל֒ אֲ֠נָשִׁ֠ים נֹשְׂאֵ֨י מָגֵ֤ן וְחֶ֙רֶב֙ וְדֹ֣רְכֵי קֶ֔שֶׁת וּלְמוּדֵ֖י מִלְחָמָ֑ה אַרְבָּעִ֨ים וְאַרְבָּעָ֥ה אֶ֛לֶף וּשְׁבַע־מֵא֥וֹת וְשִׁשִּׁ֖ים יֹצְאֵ֥י צָבָֽא׃ יטיט וַיַּעֲשׂ֥וּ מִלְחָמָ֖ה עִם־הַהַגְרִיאִ֑ים וִיט֥וּר וְנָפִ֖ישׁ וְנוֹדָֽב׃ ככ וַיֵּעָזְר֣וּ עֲלֵיהֶ֔ם וַיִּנָּתְנ֤וּ בְיָדָם֙ הַֽהַגְרִיאִ֔ים וְכֹ֖ל שֶׁ֣עִמָּהֶ֑ם כִּ֠י לֵאלֹהִ֤ים זָעֲקוּ֙ בַּמִּלְחָמָ֔ה וְנַעְתּ֥וֹר לָהֶ֖ם כִּי־בָ֥טְחוּ בֽוֹ׃ כאכא וַיִּשְׁבּ֣וּ מִקְנֵיהֶ֗ם גְּֽמַלֵּיהֶ֞ם חֲמִשִּׁ֥ים אֶ֙לֶף֙ וְצֹ֗אן מָאתַ֤יִם וַחֲמִשִּׁים֙ אֶ֔לֶף וַחֲמוֹרִ֖ים אַלְפָּ֑יִם וְנֶ֥פֶשׁ אָדָ֖ם מֵ֥אָה אָֽלֶף׃ כבכב כִּֽי־חֲלָלִ֤ים רַבִּים֙ נָפָ֔לוּ כִּ֥י מֵהָאֱלֹהִ֖ים הַמִּלְחָמָ֑ה וַיֵּשְׁב֥וּ תַחְתֵּיהֶ֖ם עַד־הַגֹּלָֽה׃
{פ}　　
כגכג וּבְנֵ֗י חֲצִי֙ שֵׁ֣בֶט מְנַשֶּׁ֔ה יָשְׁב֖וּ בָּאָ֑רֶץ מִבָּשָׁ֞ן עַד־בַּ֧עַל חֶרְמ֛וֹן וּשְׂנִ֥יר וְהַר־חֶרְמ֖וֹן הֵ֥מָּה רָבֽוּ׃ כדכד וְאֵ֖לֶּה רָאשֵׁ֣י בֵית־אֲבוֹתָ֑ם וְעֵ֡פֶר וְיִשְׁעִ֡י וֶאֱלִיאֵ֡ל וְ֠עַזְרִיאֵ֠ל וְיִרְמְיָ֨ה וְהוֹדַוְיָ֜ה וְיַחְדִּיאֵ֗ל אֲנָשִׁים֙ גִּבּ֣וֹרֵי חַ֔יִל אַנְשֵׁ֣י שֵׁמ֔וֹת רָאשִׁ֖ים לְבֵ֥ית אֲבוֹתָֽם׃ כהכה וַיִּֽמְעֲל֔וּ בֵּאלֹהֵ֖י אֲבֹתֵיהֶ֑ם וַיִּזְנ֗וּ אַֽחֲרֵי֙ אֱלֹהֵ֣י עַמֵּֽי־הָאָ֔רֶץ אֲשֶׁר־הִשְׁמִ֥יד אֱלֹהִ֖ים מִפְּנֵיהֶֽם׃ כוכו וַיָּ֩עַר֩ אֱלֹהֵ֨י יִשְׂרָאֵ֜ל אֶת־ר֣וּחַ ׀ פּ֣וּל מֶלֶךְ־אַשּׁ֗וּר וְאֶת־ר֙וּחַ֙ תִּלְּגַ֤ת פִּלְנֶ֙סֶר֙ מֶ֣לֶךְ אַשּׁ֔וּר וַיַּגְלֵם֙ לָראוּבֵנִ֣י וְלַגָּדִ֔י וְלַחֲצִ֖י שֵׁ֣בֶט מְנַשֶּׁ֑ה וַ֠יְבִיאֵ֠ם לַחְלַ֨ח וְחָב֤וֹר וְהָרָא֙ וּנְהַ֣ר גּוֹזָ֔ן עַ֖ד הַיּ֥וֹם הַזֶּֽה׃
{פ}　　
כזכז בְּנֵ֖י לֵוִ֑י גֵּרְשׁ֕וֹן קְﬞהָ֖תנקְﬞהָ֖ת · א= קֳהָ֖ת (חטף קמץ) וּמְרָרִֽי׃ כחכח וּבְנֵ֖י קְﬞהָ֑תנקְﬞהָ֑ת · א=קֳהָ֑ת (חטף קמץ) עַמְרָ֣ם יִצְהָ֔ר וְחֶבְר֖וֹן וְעֻזִּיאֵֽל׃
{ס}　　　
כטכט וּבְנֵ֣י עַמְרָ֔ם אַהֲרֹ֥ן וּמֹשֶׁ֖ה וּמִרְיָ֑ם וּבְנֵ֣י אַֽהֲרֹ֔ן נָדָב֙ וַאֲבִיה֔וּא אֶלְעָזָ֖ר וְאִיתָמָֽר׃
{ס}　　　
לל אֶלְעָזָר֙ הוֹלִ֣יד אֶת־פִּֽינְחָ֔ס פִּֽינְחָ֖ס הֹלִ֥יד אֶת־אֲבִישֽׁוּעַ׃ לאלא וַאֲבִישׁ֙וּעַ֙ הוֹלִ֣יד אֶת־בֻּקִּ֔י וּבֻקִּ֖י הוֹלִ֥יד אֶת־עֻזִּֽי׃ לבלב וְעֻזִּי֙ הוֹלִ֣יד אֶת־זְרַֽחְיָ֔ה וּֽזְרַֽחְיָ֖ה הוֹלִ֥יד אֶת־מְרָיֽוֹת׃ לגלג מְרָיוֹת֙ הוֹלִ֣יד אֶת־אֲמַרְיָ֔ה וַאֲמַרְיָ֖ה הוֹלִ֥יד אֶת־אֲחִיטֽוּב׃ לדלד וַֽאֲחִיטוּב֙ הוֹלִ֣יד אֶת־צָד֔וֹק וְצָד֖וֹק הוֹלִ֥יד אֶת־אֲחִימָֽעַץ׃ להלה וַאֲחִימַ֙עַץ֙ הוֹלִ֣יד אֶת־עֲזַרְיָ֔ה וַעֲזַרְיָ֖ה הוֹלִ֥יד אֶת־יוֹחָנָֽן׃ לולו וְיוֹחָנָ֖ן הוֹלִ֣יד אֶת־עֲזַרְיָ֑ה ה֚וּא אֲשֶׁ֣ר כִּהֵ֔ן בַּבַּ֕יִת אֲשֶׁר־בָּנָ֥ה שְׁלֹמֹ֖ה בִּירוּשָׁלָֽ͏ִםם׃ לזלז וַיּ֥וֹלֶד עֲזַרְיָ֖ה אֶת־אֲמַרְיָ֑ה וַאֲמַרְיָ֖ה הוֹלִ֥יד אֶת־אֲחִיטֽוּב׃ לחלח וַֽאֲחִיטוּב֙ הוֹלִ֣יד אֶת־צָד֔וֹק וְצָד֖וֹק הוֹלִ֥יד אֶת־שַׁלּֽוּם׃ לטלט וְשַׁלּוּם֙ הוֹלִ֣יד אֶת־חִלְקִיָּ֔ה וְחִלְקִיָּ֖ה הוֹלִ֥יד אֶת־עֲזַרְיָֽה׃ ממ וַעֲזַרְיָה֙ הוֹלִ֣יד אֶת־שְׂרָיָ֔ה וּשְׂרָיָ֖ה הוֹלִ֥יד אֶת־יְהוֹצָדָֽק׃ מאמא וִיהוֹצָדָ֣ק הָלַ֔ךְ בְּהַגְל֣וֹת יְהֹוָ֔ה אֶת־יְהוּדָ֖ה וִירֽוּשָׁלָ֑͏ִםם בְּיַ֖ד נְבֻכַדְנֶאצַּֽרננְבֻכַדְנֶאצַּֽר=ל,ש1,ק13,ק-מ וכך הכריעו ברויאר ומג"ה, וכמו כן בדפוסים וקורן.א!=נְבֻכַדְנֶאצַֽר (חסר דגש באות צד”י)׃
