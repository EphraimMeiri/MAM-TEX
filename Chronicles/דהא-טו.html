מקרא על פי המסורה

ספר דברי הימים

פרק טו

טואא וַיַּעַשׂ־ל֥וֹ בָתִּ֖ים בְּעִ֣יר דָּוִ֑יד וַיָּ֤כֶן מָקוֹם֙ לַאֲר֣וֹן הָאֱלֹהִ֔ים וַיֶּט־ל֖וֹ אֹֽהֶל׃ בב אָ֚ז אָמַ֣ר דָּוִ֔יד לֹ֤א לָשֵׂאת֙ אֶת־אֲר֣וֹן הָאֱלֹהִ֔ים כִּ֖י אִם־הַלְוִיִּ֑ם כִּֽי־בָ֣ם ׀ בָּחַ֣ר יְהֹוָ֗ה לָשֵׂ֞את אֶת־אֲר֧וֹן יְהֹוָ֛ה וּֽלְשָׁרְת֖וֹ עַד־עוֹלָֽם׃
{פ}　　
גג וַיַּקְהֵ֥ל דָּוִ֛יד אֶת־כׇּל־יִשְׂרָאֵ֖ל אֶל־יְרוּשָׁלָ֑͏ִםם לְהַֽעֲלוֹת֙ אֶת־אֲר֣וֹן יְהֹוָ֔ה אֶל־מְקוֹמ֖וֹ אֲשֶׁר־הֵכִ֥ין לֽוֹ׃ דד וַיֶּאֱסֹ֥ף דָּוִ֛יד אֶת־בְּנֵ֥י אַהֲרֹ֖ן וְאֶת־הַלְוִיִּֽם׃
{ס}　　　
הה לִבְנֵ֖י קְהָ֑ת אוּרִיאֵ֣ל הַשָּׂ֔ר וְאֶחָ֖יו מֵאָ֥ה וְעֶשְׂרִֽים׃
{ס}　　　
וו לִבְנֵ֖י מְרָרִ֑י עֲשָׂיָ֣ה הַשָּׂ֔ר וְאֶחָ֖יו מָאתַ֥יִם וְעֶשְׂרִֽים׃
{ס}　　　
זז לִבְנֵ֖י גֵּרְשׁ֑וֹם יוֹאֵ֣ל הַשָּׂ֔ר וְאֶחָ֖יו מֵאָ֥ה וּשְׁלֹשִֽׁים׃
{ס}　　　
חח לִבְנֵ֖י אֱלִיצָפָ֑ן שְׁמַֽעְיָ֥ה הַשָּׂ֖ר וְאֶחָ֥יו מָאתָֽיִם׃
{ס}　　　
טט לִבְנֵ֖י חֶבְר֑וֹן אֱלִיאֵ֥ל הַשָּׂ֖ר וְאֶחָ֥יו שְׁמוֹנִֽים׃
{ס}　　　
יי לִבְנֵ֖י עֻזִּיאֵ֑ל עַמִּינָדָ֣ב הַשָּׂ֔ר וְאֶחָ֕יו מֵאָ֖ה וּשְׁנֵ֥ים עָשָֽׂר׃
{פ}　　
יאיא וַיִּקְרָ֣א דָוִ֔יד לְצָד֥וֹק וּלְאֶבְיָתָ֖ר הַכֹּהֲנִ֑ים וְלַלְוִיִּ֗ם לְאוּרִיאֵ֤ל עֲשָׂיָה֙ וְיוֹאֵ֣ל שְׁמַֽעְיָ֔ה וֶאֱלִיאֵ֖ל וְעַמִּינָדָֽב׃
{ס}　　　
יביב וַיֹּ֣אמֶר לָהֶ֔ם אַתֶּ֛ם רָאשֵׁ֥י הָאָב֖וֹת לַלְוִיִּ֑ם הִֽתְקַדְּשׁוּ֙ אַתֶּ֣ם וַאֲחֵיכֶ֔ם וְהַעֲלִיתֶ֗ם אֵ֣ת אֲר֤וֹן יְהֹוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֶל־הֲכִינ֖וֹתִי לֽוֹ׃ יגיג כִּ֛י לְמַבָּרִ֥אשׁוֹנָ֖ה לֹ֣א אַתֶּ֑ם פָּרַ֨ץ יְהֹוָ֤ה אֱלֹהֵ֙ינוּ֙ בָּ֔נוּ כִּֽי־לֹ֥א דְרַשְׁנֻ֖הוּ כַּמִּשְׁפָּֽט׃ ידיד וַיִּֽתְקַדְּשׁ֔וּ הַכֹּהֲנִ֖ים וְהַלְוִיִּ֑ם לְהַעֲל֕וֹת אֶת־אֲר֥וֹן יְהֹוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ טוטו וַיִּשְׂא֣וּ בְנֵֽי־הַלְוִיִּ֗ם אֵ֚ת אֲר֣וֹן הָאֱלֹהִ֔ים כַּאֲשֶׁ֛ר צִוָּ֥ה מֹשֶׁ֖ה כִּדְבַ֣ר יְהֹוָ֑ה בִּכְתֵפָ֥ם בַּמֹּט֖וֹת עֲלֵיהֶֽם׃
{פ}　　
טזטז וַיֹּ֣אמֶר דָּוִיד֮ לְשָׂרֵ֣י הַלְוִיִּם֒ לְהַעֲמִ֗יד אֶת־אֲחֵיהֶם֙ הַמְשֹׁ֣רְרִ֔ים בִּכְלֵי־שִׁ֛יר נְבָלִ֥ים וְכִנֹּר֖וֹת וּמְצִלְתָּ֑יִם מַשְׁמִיעִ֥ים לְהָרִֽים־בְּק֖וֹל לְשִׂמְחָֽה׃
{פ}　　
יזיז וַיַּעֲמִ֣ידוּ הַלְוִיִּ֗ם אֵ֚ת הֵימָ֣ן בֶּן־יוֹאֵ֔ל וּמִ֨ן־אֶחָ֔יו אָסָ֖ף בֶּן־בֶּרֶכְיָ֑הוּ וּמִן־בְּנֵ֤י מְרָרִי֙ אֲחֵיהֶ֔ם אֵיתָ֖ן בֶּן־קוּשָׁיָֽהוּ׃
{ס}　　　
יחיח וְעִמָּהֶ֖ם אֲחֵיהֶ֣ם הַמִּשְׁנִ֑ים זְכַרְיָ֡הוּ בֵּ֡ן וְיַעֲזִיאֵ֡ל וּשְׁמִירָמ֡וֹת וִיחִיאֵ֣ל ׀ וְעֻנִּ֡י אֱלִיאָ֡ב וּבְנָיָ֡הוּ וּמַעֲשֵׂיָ֡הוּ וּמַתִּתְיָ֩הוּ֩ וֶאֱלִ֨יפְﬞלֵ֜הוּנוֶאֱלִ֨יפְﬞלֵ֜הוּ · א=וֶאֱלִ֨יפֲלֵ֜הוּ (חטף) וּמִקְנֵיָ֨הוּ וְעֹבֵ֥ד אֱדֹ֛ם וִיעִיאֵ֖ל הַשֹּׁעֲרִֽים׃ יטיט וְהַמְשֹׁ֣רְﬞרִ֔יםנוְהַמְשֹׁ֣רְﬞרִ֔ים · א=וְהַמְשֹׁ֣רֲרִ֔ים (חטף) הֵימָ֥ן אָסָ֖ף וְאֵיתָ֑ן בִּמְצִלְתַּ֥יִם נְחֹ֖שֶׁת לְהַשְׁמִֽיעַ׃ ככ וּזְכַרְיָ֨ה וַעֲזִיאֵ֜ל וּשְׁמִירָמ֤וֹת וִֽיחִיאֵל֙ וְעֻנִּ֣י וֶאֱלִיאָ֔ב וּמַעֲשֵׂיָ֖הוּ וּבְנָיָ֑הוּ בִּנְבָלִ֖ים עַל־עֲלָמֽוֹת׃ כאכא וּמַתִּתְיָ֣הוּ וֶאֱלִֽיפְﬞלֵ֗הוּנוֶאֱלִֽיפְﬞלֵ֗הוּ · א=וֶאֱלִֽיפֲלֵ֗הוּ (חטף) וּמִקְנֵיָ֙הוּ֙ וְעֹבֵ֣ד אֱדֹ֔ם וִיעִיאֵ֖ל וַעֲזַזְיָ֑הוּ בְּכִנֹּר֥וֹת עַל־הַשְּׁמִינִ֖ית לְנַצֵּֽחַ׃ כבכב וּכְנַנְיָ֥הוּ שַֽׂר־הַלְוִיִּ֖ם בְּמַשָּׂ֑א יָסֹר֙ בַּמַּשָּׂ֔א כִּ֥י מֵבִ֖ין הֽוּא׃ כגכג וּבֶֽרֶכְיָה֙ וְאֶלְקָנָ֔ה שֹׁעֲרִ֖ים לָאָרֽוֹן׃ כדכד וּשְׁבַנְיָ֡הוּ וְיוֹשָׁפָ֡ט וּנְתַנְאֵ֡ל וַעֲמָשַׂ֡י וּ֠זְכַרְיָ֠הוּ וּבְנָיָ֤הוּ וֶאֱלִיעֶ֙זֶר֙ הַכֹּ֣הֲנִ֔ים (מחצצרים)קקקרי/כתיב · (מחצצרים) מַחְצְרִים֙ בַּחֲצֹ֣צְר֔וֹת לִפְנֵ֖י אֲר֣וֹן הָאֱלֹהִ֑ים וְעֹבֵ֤ד אֱדֹם֙ וִֽיחִיָּ֔ה שֹׁעֲרִ֖ים לָאָרֽוֹן׃ כהכה וַיְהִ֥י דָוִ֛יד וְזִקְנֵ֥י יִשְׂרָאֵ֖ל וְשָׂרֵ֣י הָאֲלָפִ֑ים הַהֹלְכִ֗ים לְֽהַעֲל֞וֹת אֶת־אֲר֧וֹן בְּרִית־יְהֹוָ֛ה מִן־בֵּ֥ית עֹבֵֽד־אֱדֹ֖ם בְּשִׂמְחָֽה׃
{פ}　　
כוכו וַֽיְהִי֙ בֶּעְזֹ֣ר הָאֱלֹהִ֔ים אֶ֨ת־הַלְוִיִּ֔ם נֹשְׂאֵ֖י אֲר֣וֹן בְּרִית־יְהֹוָ֑ה וַיִּזְבְּח֥וּ שִׁבְעָֽה־פָרִ֖ים וְשִׁבְעָ֥ה אֵילִֽים׃ כזכז וְדָוִ֞יד מְכֻרְבָּ֣ל ׀ בִּמְעִ֣יל בּ֗וּץ וְכׇל־הַלְוִיִּם֙ הַנֹּשְׂאִ֣ים אֶת־הָאָר֔וֹן וְהַמְשֹׁ֣רְﬞרִ֔יםנוְהַמְשֹׁ֣רְﬞרִ֔ים · א=וְהַמְשֹׁ֣רֲרִ֔ים (חטף) וּכְנַנְיָ֛ה הַשַּׂ֥ר הַמַּשָּׂ֖א הַמְשֹׁרְﬞרִ֑יםנהַמְשֹׁרְﬞרִ֑ים · א=הַמְשֹׁרֲרִ֑ים (חטף) וְעַל־דָּוִ֖יד אֵפ֥וֹד בָּֽד׃ כחכח וְכׇל־יִשְׂרָאֵ֗ל מַֽעֲלִים֙ אֶת־אֲר֣וֹן בְּרִית־יְהֹוָ֔ה בִּתְרוּעָה֙ וּבְק֣וֹל שׁוֹפָ֔ר וּבַחֲצֹצְר֖וֹת וּבִמְצִלְתָּ֑יִם מַשְׁמִעִ֕ים בִּנְבָלִ֖ים וְכִנֹּרֽוֹת׃ כטכט וַיְהִ֗י אֲרוֹן֙ בְּרִ֣ית יְהֹוָ֔ה בָּ֖א עַד־עִ֣יר דָּוִ֑יד וּמִיכַ֨ל בַּת־שָׁא֜וּל נִשְׁקְפָ֣ה ׀ בְּעַ֣ד הַחַלּ֗וֹן וַתֵּ֨רֶא אֶת־הַמֶּ֤לֶךְ דָּוִיד֙ מְרַקֵּ֣ד וּמְשַׂחֵ֔ק וַתִּ֥בֶז ל֖וֹ בְּלִבָּֽהּ׃
