מקרא על פי המסורה

ספר דברי הימים

פרק כו


{פ}　　
כואא לְמַחְלְק֖וֹת לְשֹׁעֲרִ֑ים לַקׇּרְחִ֕ים מְשֶׁלֶמְיָ֥הוּ בֶן־קֹרֵ֖א מִן־בְּנֵ֥י אָסָֽף׃ בב וְלִמְשֶׁלֶמְיָ֖הוּ בָּנִ֑ים זְכַרְיָ֤הוּ הַבְּכוֹר֙ יְדִיעֲאֵ֣ל הַשֵּׁנִ֔י זְבַדְיָ֙הוּ֙ הַשְּׁלִישִׁ֔י יַתְנִיאֵ֖ל הָרְבִיעִֽי׃ גג עֵילָ֤ם הַחֲמִישִׁי֙ יְהוֹחָנָ֣ן הַשִּׁשִּׁ֔י אֶלְיְהוֹעֵינַ֖י הַשְּׁבִיעִֽי׃ דד וּלְעֹבֵ֥ד אֱדֹ֖ם בָּנִ֑ים שְׁמַעְיָ֤ה הַבְּכוֹר֙ יְהוֹזָבָ֣ד הַשֵּׁנִ֔י יוֹאָ֤ח הַשְּׁלִשִׁי֙ וְשָׂכָ֣ר הָרְבִיעִ֔י וּנְתַנְאֵ֖ל הַחֲמִישִֽׁי׃ הה עַמִּיאֵ֤ל הַשִּׁשִּׁי֙ יִשָּׂשכָ֣ר הַשְּׁבִיעִ֔י פְּעֻלְּתַ֖י הַשְּׁמִינִ֑י כִּ֥י בֵרְﬞכ֖וֹנבֵרְﬞכ֖וֹ · א=בֵרֲכ֖וֹ (חטף) אֱלֹהִֽים׃
{ס}　　　
וו וְלִֽשְׁמַעְיָ֤ה בְנוֹ֙ נוֹלַ֣ד בָּנִ֔ים הַמִּמְשָׁלִ֖ים לְבֵ֣ית אֲבִיהֶ֑ם כִּֽי־גִבּ֥וֹרֵי חַ֖יִל הֵֽמָּה׃ זז בְּנֵ֣י שְׁמַֽעְיָ֗ה עׇ֠תְנִ֠י וּרְפָאֵ֨ל וְעוֹבֵ֧ד אֶלְזָבָ֛ד אֶחָ֖יו בְּנֵי־חָ֑יִל אֱלִיה֖וּ וּסְמַכְיָֽהוּ׃ חח כׇּל־אֵ֜לֶּה מִבְּנֵ֣י ׀ עֹבֵ֣ד אֱדֹ֗ם הֵ֤מָּה וּבְנֵיהֶם֙ וַאֲחֵיהֶ֔ם אִֽישׁ־חַ֥יִל בַּכֹּ֖חַ לַעֲבֹדָ֑ה שִׁשִּׁ֥ים וּשְׁנַ֖יִם לְעֹבֵ֥ד אֱדֹֽם׃ טט וְלִמְשֶׁלֶמְיָ֗הוּ בָּנִ֧ים וְאַחִ֛ים בְּנֵי־חָ֖יִל שְׁמוֹנָ֥ה עָשָֽׂר׃
{ס}　　　
יי וּלְחֹסָ֥ה מִן־בְּנֵי־מְרָרִ֖י בָּנִ֑ים שִׁמְרִ֤י הָרֹאשׁ֙ כִּ֣י לֹֽא־הָיָ֣ה בְכ֔וֹר וַיְשִׂימֵ֥הוּ אָבִ֖יהוּ לְרֹֽאשׁ׃ יאיא חִלְקִיָּ֤הוּ הַשֵּׁנִי֙ טְבַלְיָ֣הוּ הַשְּׁלִשִׁ֔י זְכַרְיָ֖הוּ הָרְבִעִ֑י כׇּל־בָּנִ֧ים וְאַחִ֛ים לְחֹסָ֖ה שְׁלֹשָׁ֥ה עָשָֽׂר׃ יביב לְ֠אֵ֠לֶּה מַחְלְק֨וֹת הַשֹּׁעֲרִ֜ים לְרָאשֵׁ֧י הַגְּבָרִ֛ים מִשְׁמָר֖וֹת לְעֻמַּ֣ת אֲחֵיהֶ֑ם לְשָׁרֵ֖ת בְּבֵ֥ית יְהֹוָֽה׃ יגיג וַיַּפִּ֨ילוּ גוֹרָל֜וֹת כַּקָּטֹ֧ן כַּגָּד֛וֹל לְבֵ֥ית אֲבוֹתָ֖ם לְשַׁ֥עַר וָשָֽׁעַר׃
{פ}　　
ידיד וַיִּפֹּ֧ל הַגּוֹרָ֛ל מִזְרָ֖חָה לְשֶׁלֶמְיָ֑הוּ וּזְכַרְיָ֨הוּ בְנ֜וֹ יוֹעֵ֣ץ ׀ בְּשֶׂ֗כֶל הִפִּ֙ילוּ֙ גּֽוֹרָל֔וֹת וַיֵּצֵ֥א גוֹרָל֖וֹ צָפֽוֹנָה׃ טוטו לְעֹבֵ֥ד אֱדֹ֖ם נֶ֑גְבָּה וּלְבָנָ֖יו בֵּ֥ית הָאֲסֻפִּֽים׃ טזטז לְשֻׁפִּ֤ים וּלְחֹסָה֙ לַֽמַּעֲרָ֔ב עִ֚ם שַׁ֣עַר שַׁלֶּ֔כֶת בַּֽמְﬞסִלָּ֖הנבַּֽמְﬞסִלָּ֖ה · א=בַּֽמֲסִלָּ֖ה (חטף) הָעוֹלָ֑ה מִשְׁמָ֖ר לְעֻמַּ֥ת מִשְׁמָֽר׃
{ס}　　　
יזיז לַמִּזְרָח֮ הַלְוִיִּ֣ם שִׁשָּׁה֒ לַצָּפ֤וֹנָה לַיּוֹם֙ אַרְבָּעָ֔ה לַנֶּ֥גְבָּה לַיּ֖וֹם אַרְבָּעָ֑ה וְלָאֲסֻפִּ֖ים שְׁנַ֥יִם שְׁנָֽיִם׃
{ס}　　　
יחיח לַפַּרְבָּ֖ר לַֽמַּעֲרָ֑ב אַרְבָּעָה֙ לַֽמְﬞסִלָּ֔הנלַֽמְﬞסִלָּ֔ה · א=לַֽמֲסִלָּ֔ה (חטף) שְׁנַ֖יִם לַפַּרְבָּֽר׃ יטיט אֵ֗לֶּה מַחְלְקוֹת֙ הַשֹּׁ֣עֲרִ֔ים לִבְנֵ֥י הַקׇּרְחִ֖י וְלִבְנֵ֥י מְרָרִֽי׃ ככ וְֽהַלְוִיִּ֑ם אֲחִיָּ֗ה עַל־אֽוֹצְרוֹת֙ בֵּ֣ית הָאֱלֹהִ֔ים וּלְאֹצְר֖וֹת הַקֳּדָשִֽׁים׃
{ס}　　　
כאכא בְּנֵ֣י לַ֠עְדָּ֠ן בְּנֵ֨י הַגֵּרְשֻׁנִּ֜י לְלַעְדָּ֗ן רָאשֵׁ֧י הָאָב֛וֹת לְלַעְדָּ֥ן הַגֵּרְשֻׁנִּ֖י יְחִיאֵלִֽי׃
{ס}　　　
כבכב בְּנֵ֖י יְחִיאֵלִ֑י זֵתָם֙ וְיוֹאֵ֣ל אָחִ֔יו עַל־אֹצְר֖וֹת בֵּ֥ית יְהֹוָֽה׃
{ס}　　　
כגכג לַֽעַמְרָמִי֙ לַיִּצְהָרִ֔י לַחֶבְרוֹנִ֖י לָעׇזִּיאֵלִֽי׃ כדכד וּשְׁבֻאֵל֙ בֶּן־גֵּרְשׁ֣וֹם בֶּן־מֹשֶׁ֔ה נָגִ֖יד עַל־הָאֹצָרֽוֹת׃
{ס}　　　
כהכה וְאֶחָ֖יו לֶאֱלִיעֶ֑זֶר רְחַבְיָ֨הוּ בְנ֜וֹ וִישַׁעְיָ֤הֽוּ בְנוֹ֙ וְיֹרָ֣ם בְּנ֔וֹ וְזִכְרִ֥י בְנ֖וֹ וּשְׁלֹמִ֥ויתקקקרי/כתיב · (ושלמות) וּשְׁלֹמִ֥ית בְּנֽוֹ׃ כוכו ה֧וּא שְׁלֹמ֣וֹת וְאֶחָ֗יו עַ֣ל כׇּל־אֹצְר֤וֹת הַקֳּדָשִׁים֙ אֲשֶׁ֨ר הִקְדִּ֜ישׁ דָּוִ֣יד הַמֶּ֗לֶךְ וְרָאשֵׁ֧י הָאָב֛וֹת לְשָׂרֵֽי־הָאֲלָפִ֥ים וְהַמֵּא֖וֹת וְשָׂרֵ֥י הַצָּבָֽא׃ כזכז מִן־הַמִּלְחָמ֥וֹת וּמִן־הַשָּׁלָ֖ל הִקְדִּ֑ישׁוּ לְחַזֵּ֖ק לְבֵ֥ית יְהֹוָֽה׃ כחכח וְכֹ֨ל הַהִקְדִּ֜ישׁ שְׁמוּאֵ֤ל הָרֹאֶה֙ וְשָׁא֣וּל בֶּן־קִ֔ישׁ וְאַבְנֵ֣ר בֶּן־נֵ֔ר וְיוֹאָ֖ב בֶּן־צְרוּיָ֑ה כֹּ֚ל הַמַּקְדִּ֔ישׁ עַ֥ל יַד־שְׁלֹמִ֖ית וְאֶחָֽיו׃
{פ}　　
כטכט לַיִּצְהָרִ֞י כְּנַנְיָ֣הוּ וּבָנָ֗יו לַמְּלָאכָ֤ה הַחִֽיצוֹנָה֙ עַל־יִשְׂרָאֵ֔ל לְשֹׁטְרִ֖ים וּלְשֹׁפְטִֽים׃ לל לַחֶבְרוֹנִ֡י חֲשַׁבְיָ֩הוּ֩ וְאֶחָ֨יו בְּנֵי־חַ֜יִל אֶ֣לֶף וּשְׁבַע־מֵא֗וֹת עַ֚ל פְּקֻדַּ֣ת יִשְׂרָאֵ֔ל מֵעֵ֥בֶר לַיַּרְדֵּ֖ן מַעְרָ֑בָה לְכֹל֙ מְלֶ֣אכֶת יְהֹוָ֔ה וְלַעֲבֹדַ֖ת הַמֶּֽלֶךְ׃ לאלא לַֽחֶבְרוֹנִי֙ יְרִיָּ֣ה הָרֹ֔אשׁ לַֽחֶבְרוֹנִ֥י לְתֹלְדֹתָ֖יו לְאָב֑וֹת בִּשְׁנַ֨ת הָאַרְבָּעִ֜ים לְמַלְכ֤וּת דָּוִיד֙ נִדְרָ֔שׁוּ וַיִּמָּצֵ֥א בָהֶ֛ם גִּבּ֥וֹרֵי חַ֖יִל בְּיַעְזֵ֥יר גִּלְעָֽד׃ לבלב וְאֶחָ֣יו בְּנֵי־חַ֗יִל אַלְפַּ֛יִם וּשְׁבַ֥ע מֵא֖וֹת רָאשֵׁ֣י הָאָב֑וֹת וַֽיַּפְקִידֵ֞ם דָּוִ֣יד הַמֶּ֗לֶךְ עַל־הָראוּבֵנִ֤י וְהַגָּדִי֙ וַחֲצִי֙ שֵׁ֣בֶט הַֽמְﬞנַשִּׁ֔ינהַֽמְﬞנַשִּׁ֔י · א=הַֽמֲנַשִּׁ֔י (חטף) לְכׇל־דְּבַ֥ר הָאֱלֹהִ֖ים וּדְבַ֥ר הַמֶּֽלֶךְ׃
