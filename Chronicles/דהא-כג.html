מקרא על פי המסורה

ספר דברי הימים

פרק כג


{פ}　　
כגאא וְדָוִ֥יד זָקֵ֖ן וְשָׂבַ֣ע יָמִ֑ים וַיַּמְלֵ֛ךְ אֶת־שְׁלֹמֹ֥ה בְנ֖וֹ עַל־יִשְׂרָאֵֽל׃ בב וַיֶּֽאֱסֹף֙ אֶת־כׇּל־שָׂרֵ֣י יִשְׂרָאֵ֔ל וְהַכֹּהֲנִ֖ים וְהַלְוִיִּֽם׃ גג וַיִּסָּֽפְרוּ֙ הַלְוִיִּ֔ם מִבֶּ֛ן שְׁלֹשִׁ֥ים שָׁנָ֖ה וָמָ֑עְלָה וַיְהִ֨י מִסְפָּרָ֤ם לְגֻלְגְּלֹתָם֙ לִגְבָרִ֔ים שְׁלֹשִׁ֥ים וּשְׁמוֹנָ֖ה אָֽלֶף׃ דד מֵאֵ֗לֶּה לְנַצֵּ֙חַ֙ עַל־מְלֶ֣אכֶת בֵּית־יְהֹוָ֔ה עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָ֑לֶף וְשֹׁטְרִ֥ים וְשֹׁפְטִ֖ים שֵׁ֥שֶׁת אֲלָפִֽים׃ הה וְאַרְבַּ֥עַת אֲלָפִ֖ים שֹׁעֲרִ֑ים וְאַרְבַּ֤עַת אֲלָפִים֙ מְהַֽלְלִ֣ים לַיהֹוָ֔ה בַּכֵּלִ֕ים אֲשֶׁ֥ר עָשִׂ֖יתִי לְהַלֵּֽל׃
{ס}　　　
וו וַיֶּחָלְקֵ֥ם דָּוִ֖יד מַחְלְק֑וֹת לִבְנֵ֣י לֵוִ֔י לְגֵרְשׁ֖וֹן קְﬞהָ֥תנקְﬞהָ֥ת · א=קֳהָ֥ת (חטף קמץ) וּמְרָרִֽי׃
{ס}　　　
זז לַגֵּרְשֻׁנִּ֖י לַעְדָּ֥ן וְשִׁמְעִֽי׃
{ס}　　　
חח בְּנֵ֣י לַעְדָּ֗ן הָרֹ֧אשׁ יְחִיאֵ֛ל וְזֵתָ֥ם וְיוֹאֵ֖ל שְׁלֹשָֽׁה׃
{ס}　　　
טט בְּנֵ֣י שִׁמְעִ֗י שְׁלֹמִ֧ויתקקקרי/כתיב · (שלמות) שְׁלֹמִ֧ית וַחֲזִיאֵ֛ל וְהָרָ֖ן שְׁלֹשָׁ֑ה אֵ֛לֶּה רָאשֵׁ֥י הָאָב֖וֹת לְלַעְדָּֽן׃
{ס}　　　
יי וּבְנֵ֣י שִׁמְעִ֔י יַ֣חַת זִינָ֔א וִיע֖וּשׁ וּבְרִיעָ֑ה אֵ֥לֶּה בְנֵֽי־שִׁמְעִ֖י אַרְבָּעָֽה׃ יאיא וַיְהִי־יַ֣חַת הָרֹ֔אשׁ וְזִיזָ֖ה הַשֵּׁנִ֑י וִיע֤וּשׁ וּבְרִיעָה֙ לֹא־הִרְבּ֣וּ בָנִ֔ים וַיִּֽהְיוּ֙ לְבֵ֣ית אָ֔ב לִפְקֻדָּ֖ה אֶחָֽת׃
{ס}　　　
יביב בְּנֵ֣י קְﬞהָ֗תנקְﬞהָ֗ת · א=קֳהָ֗ת (חטף קמץ) עַמְרָ֥ם יִצְהָ֛ר חֶבְר֥וֹן וְעֻזִּיאֵ֖ל אַרְבָּעָֽה׃
{ס}　　　
יגיג בְּנֵ֥י עַמְרָ֖ם אַהֲרֹ֣ן וּמֹשֶׁ֑ה וַיִּבָּדֵ֣ל אַהֲרֹ֡ן לְֽהַקְדִּישׁוֹ֩ קֹ֨דֶשׁ הֽוּא־וּבָנָיו֙ עַד־עוֹלָ֔ם לְהַקְטִיר֩ לִפְנֵ֨י יְהֹוָ֧ה לְשָׁרְת֛וֹ וּלְבָרֵ֥ךְ בִּשְׁמ֖וֹ עַד־עוֹלָֽם׃ ידיד וּמֹשֶׁ֖ה אִ֣ישׁ הָאֱלֹהִ֑ים בָּנָ֕יו יִקָּרְא֖וּ עַל־שֵׁ֥בֶט הַלֵּוִֽי׃
{ס}　　　
טוטו בְּנֵ֣י מֹשֶׁ֔ה גֵּרְשׁ֖וֹם וֶאֱלִיעֶֽזֶר׃ טזטז בְּנֵ֥י גֵרְשׁ֖וֹם שְׁבוּאֵ֥ל הָרֹֽאשׁ׃ יזיז וַיִּהְי֥וּ בְנֵֽי־אֱלִיעֶ֖זֶר רְחַבְיָ֣ה הָרֹ֑אשׁ וְלֹא־הָיָ֤ה לֶאֱלִיעֶ֙זֶר֙ בָּנִ֣ים אֲחֵרִ֔ים וּבְנֵ֥י רְחַבְיָ֖ה רָב֥וּ לְמָֽעְלָה׃
{ס}　　　
יחיח בְּנֵ֥י יִצְהָ֖ר שְׁלֹמִ֥ית הָרֹֽאשׁ׃ יטיט בְּנֵ֖י חֶבְר֑וֹן יְרִיָּ֤הוּ הָרֹאשׁ֙ אֲמַרְיָ֣ה הַשֵּׁנִ֔י יַֽחֲזִיאֵל֙ הַשְּׁלִישִׁ֔י וִֽיקַמְעָ֖ם הָרְבִיעִֽי׃ ככ בְּנֵ֖י עֻזִּיאֵ֑ל מִיכָ֣ה הָרֹ֔אשׁ וְיִשִּׁיָּ֖ה הַשֵּׁנִֽי׃ כאכא בְּנֵ֤י מְרָרִי֙ מַחְלִ֣י וּמוּשִׁ֔י בְּנֵ֥י מַחְלִ֖י אֶלְעָזָ֥ר וְקִֽישׁ׃ כבכב וַיָּ֙מׇת֙ אֶלְעָזָ֔ר וְלֹא־הָ֥יוּ ל֛וֹ בָּנִ֖ים כִּ֣י אִם־בָּנ֑וֹת וַיִּשָּׂא֥וּם בְּנֵי־קִ֖ישׁ אֲחֵיהֶֽם׃ כגכג בְּנֵ֣י מוּשִׁ֗י מַחְלִ֥י וְעֵ֛דֶר וִירֵמ֖וֹת שְׁלוֹשָֽׁה׃
{ס}　　　
כדכד אֵ֣לֶּה בְנֵֽי־לֵוִי֩ לְבֵ֨ית אֲבוֹתֵיהֶ֜ם רָאשֵׁ֧י הָאָב֣וֹת לִפְקוּדֵיהֶ֗ם בְּמִסְפַּ֤ר שֵׁמוֹת֙ לְגֻלְגְּלֹתָ֔ם עֹשֵׂה֙ הַמְּלָאכָ֔ה לַעֲבֹדַ֖ת בֵּ֣ית יְהֹוָ֑ה מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וָמָֽעְלָה׃ כהכה כִּ֚י אָמַ֣ר דָּוִ֔יד הֵנִ֛יחַ יְהֹוָ֥ה אֱלֹהֵֽי־יִשְׂרָאֵ֖ל לְעַמּ֑וֹ וַיִּשְׁכֹּ֥ן בִּירוּשָׁלַ֖͏ִםם עַד־לְעוֹלָֽם׃ כוכו וְגַ֖ם לַלְוִיִּ֑ם אֵין־לָשֵׂ֧את אֶת־הַמִּשְׁכָּ֛ן וְאֶת־כׇּל־כֵּלָ֖יו לַעֲבֹדָתֽוֹ׃ כזכז כִּ֣י בְדִבְרֵ֤י דָוִיד֙ הָאַ֣חֲרוֹנִ֔ים הֵ֖מָּה מִסְפַּ֣ר בְּנֵֽי־לֵוִ֑י מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָֽעְלָה׃ כחכח כִּ֣י מַעֲמָדָ֞ם לְיַד־בְּנֵ֣י אַהֲרֹ֗ן לַֽעֲבֹדַת֙ בֵּ֣ית יְהֹוָ֔ה עַל־הַֽחֲצֵרוֹת֙ וְעַל־הַלְּשָׁכ֔וֹת לְכׇל־קֹ֑דֶשׁ וּמַֽעֲשֵׂ֔ה עֲבֹדַ֖ת בֵּ֥ית הָאֱלֹהִֽים׃ כטכט וּלְלֶ֨חֶם הַֽמַּעֲרֶ֜כֶת וּלְסֹ֤לֶת לְמִנְחָה֙ וְלִרְקִיקֵ֣י הַמַּצּ֔וֹת וְלַֽמַּחֲבַ֖ת וְלַמֻּרְבָּ֑כֶת וּלְכׇל־מְשׂוּרָ֖ה וּמִדָּֽה׃ לל וְלַֽעֲמֹד֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר לְהֹד֥וֹת וּלְהַלֵּ֖ל לַֽיהֹוָ֑ה וְכֵ֖ן לָעָֽרֶב׃ לאלא וּלְכֹ֨ל הַעֲל֤וֹת עֹלוֹת֙ לַֽיהֹוָ֔ה לַשַּׁ֨בָּת֔וֹת לֶחֳדָשִׁ֖ים וְלַמֹּעֲדִ֑ים בְּמִסְפָּ֨ר כְּמִשְׁפָּ֧ט עֲלֵיהֶ֛ם תָּמִ֖יד לִפְנֵ֥י יְהֹוָֽה׃ לבלב וְשָׁמְר֞וּ אֶת־מִשְׁמֶ֣רֶת אֹֽהֶל־מוֹעֵ֗ד וְאֵת֙ מִשְׁמֶ֣רֶת הַקֹּ֔דֶשׁ וּמִשְׁמֶ֕רֶת בְּנֵ֥י אַהֲרֹ֖ן אֲחֵיהֶ֑ם לַעֲבֹדַ֖ת בֵּ֥ית יְהֹוָֽה׃
