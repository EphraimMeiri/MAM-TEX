מקרא על פי המסורה

ספר דברי הימים

פרק כד


{פ}　　
כדאא וְלִבְנֵ֥י אַהֲרֹ֖ן מַחְלְקוֹתָ֑ם בְּנֵ֣י אַהֲרֹ֔ן נָדָב֙ וַאֲבִיה֔וּא אֶלְעָזָ֖ר וְאִיתָמָֽר׃ בב וַיָּ֨מׇת נָדָ֤ב וַֽאֲבִיהוּא֙ לִפְנֵ֣י אֲבִיהֶ֔ם וּבָנִ֖ים לֹא־הָי֣וּ לָהֶ֑ם וַֽיְכַהֲנ֔וּ אֶלְעָזָ֖ר וְאִיתָמָֽר׃ גג וַיֶּחָלְקֵ֣ם דָּוִ֔יד וְצָדוֹק֙ מִן־בְּנֵ֣י אֶלְעָזָ֔ר וַאֲחִימֶ֖לֶךְ מִן־בְּנֵ֣י אִיתָמָ֑ר לִפְקֻדָּתָ֖ם בַּעֲבֹדָתָֽם׃ דד וַיִּמָּצְא֣וּ בְֽנֵי־אֶלְעָזָ֡ר רַבִּ֞ים לְרָאשֵׁ֧י הַגְּבָרִ֛ים מִן־בְּנֵ֥י אִיתָמָ֖ר וַֽיַּחְלְק֑וּם לִבְנֵ֨י אֶלְעָזָ֜ר רָאשִׁ֤ים לְבֵית־אָבוֹת֙ שִׁשָּׁ֣ה עָשָׂ֔ר וְלִבְנֵ֧י אִיתָמָ֛ר לְבֵ֥ית אֲבוֹתָ֖ם שְׁמוֹנָֽה׃ הה וַיַּחְלְק֥וּם בְּגוֹרָל֖וֹת אֵ֣לֶּה עִם־אֵ֑לֶּה כִּֽי־הָי֤וּ שָׂרֵי־קֹ֙דֶשׁ֙ וְשָׂרֵ֣י הָאֱלֹהִ֔ים מִבְּנֵ֥י אֶלְעָזָ֖ר וּבִבְנֵ֥י אִיתָמָֽר׃
{ס}　　　
וו וַֽיִּכְתְּבֵ֡ם שְֽׁמַעְיָה֩ בֶן־נְתַנְאֵ֨ל הַסּוֹפֵ֜ר מִן־הַלֵּוִ֗י לִפְנֵ֨י הַמֶּ֤לֶךְ וְהַשָּׂרִים֙ וְצָד֣וֹק הַכֹּהֵ֗ן וַאֲחִימֶ֙לֶךְ֙ בֶּן־אֶבְיָתָ֔ר וְרָאשֵׁי֙ הָאָב֔וֹת לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם בֵּֽית־אָ֣ב אֶחָ֗ד אָחֻז֙ לְאֶלְעָזָ֔ר וְאָחֻ֥ז ׀ אָחֻ֖ז לְאִיתָמָֽר׃
{פ}　　
זז וַיֵּצֵ֞א הַגּוֹרָ֤ל הָֽרִאשׁוֹן֙ לִיה֣וֹיָרִ֔יבלִידַֽעְיָ֖ה הַשֵּׁנִֽי׃
{ס}　　　
חח לְחָרִם֙ הַשְּׁלִישִׁ֔ילִשְׂעֹרִ֖ים הָרְבִעִֽי׃
{ס}　　　
טט לְמַלְכִּיָּה֙ הַחֲמִישִׁ֔ילְמִיָּמִ֖ן הַשִּׁשִּֽׁי׃
{ס}　　　
יי לְהַקּוֹץ֙ הַשְּׁבִעִ֔ילַאֲבִיָּ֖ה הַשְּׁמִינִֽי׃
{ס}　　　
יאיא לְיֵשׁ֙וּעַ֙ הַתְּשִׁעִ֔ילִשְׁכַנְיָ֖הוּ הָעֲשִׂרִֽי׃
{ס}　　　
יביב לְאֶלְיָשִׁיב֙ עַשְׁתֵּ֣י עָשָׂ֔רלְיָקִ֖ים שְׁנֵ֥ים עָשָֽׂר׃
{ס}　　　
יגיג לְחֻפָּה֙ שְׁלֹשָׁ֣ה עָשָׂ֔רלְיֶשֶׁבְאָ֖ב אַרְבָּעָ֥ה עָשָֽׂר׃
{ס}　　　
ידיד לְבִלְגָּה֙ חֲמִשָּׁ֣ה עָשָׂ֔רלְאִמֵּ֖ר שִׁשָּׁ֥ה עָשָֽׂר׃
{ס}　　　
טוטו לְחֵזִיר֙ שִׁבְעָ֣ה עָשָׂ֔רלְהַפִּצֵּ֖ץ שְׁמוֹנָ֥ה עָשָֽׂר׃
{ס}　　　
טזטז לִֽפְתַחְיָה֙ תִּשְׁעָ֣ה עָשָׂ֔רלִֽיחֶזְקֵ֖אל הָעֶשְׂרִֽים׃
{ס}　　　
יזיז לְיָכִין֙ אֶחָ֣ד וְעֶשְׂרִ֔יםלְגָמ֖וּל שְׁנַ֥יִם וְעֶשְׂרִֽים׃
{ס}　　　
יחיח לִדְלָיָ֙הוּ֙ שְׁלֹשָׁ֣ה וְעֶשְׂרִ֔יםלְמַעַזְיָ֖הוּ אַרְבָּעָ֥ה וְעֶשְׂרִֽים׃
{פ}　　
יטיט אֵ֣לֶּה פְקֻדָּתָ֞ם לַעֲבֹדָתָ֗ם לָב֤וֹא לְבֵית־יְהֹוָה֙ כְּמִשְׁפָּטָ֔ם בְּיַ֖ד אַהֲרֹ֣ן אֲבִיהֶ֑ם כַּאֲשֶׁ֣ר צִוָּ֔הוּ יְהֹוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃
{פ}　　
ככ וְלִבְנֵ֥י לֵוִ֖י הַנּוֹתָרִ֑ים לִבְנֵ֤י עַמְרָם֙ שֽׁוּבָאֵ֔ל לִבְנֵ֥י שׁוּבָאֵ֖ל יֶחְדְּיָֽהוּ׃
{ס}　　　
כאכא לִרְחַבְיָ֑הוּ לִבְנֵ֣י רְחַבְיָ֔הוּ הָרֹ֖אשׁ יִשִּׁיָּֽה׃
{ס}　　　
כבכב לַיִּצְהָרִ֣י שְׁלֹמ֔וֹת לִבְנֵ֥י שְׁלֹמ֖וֹת יָֽחַת׃
{ס}　　　
כגכג וּבְנָ֖י יְרִיָּ֑הוּ אֲמַרְיָ֙הוּ֙ הַשֵּׁנִ֔י יַחֲזִיאֵל֙ הַשְּׁלִישִׁ֔י יְקַמְעָ֖ם הָרְבִיעִֽי׃
{ס}　　　
כדכד בְּנֵ֤י עֻזִּיאֵל֙ מִיכָ֔ה לִבְנֵ֥י מִיכָ֖ה שָׁמִֽוירקקקרי/כתיב · (שמור) שָׁמִֽיר׃ כהכה אֲחִ֤י מִיכָה֙ יִשִּׁיָּ֔הלִבְנֵ֥י יִשִּׁיָּ֖ה זְכַרְיָֽהוּ׃
{ס}　　　
כוכו בְּנֵ֤י מְרָרִי֙ מַחְלִ֣י וּמוּשִׁ֔יבְּנֵ֖י יַעֲזִיָּ֥הוּ בְנֽוֹ׃
{ס}　　　
כזכז בְּנֵ֖י מְרָרִ֑י לְיַעֲזִיָּ֣הוּ בְנ֔וֹ וְשֹׁ֥הַם וְזַכּ֖וּר וְעִבְרִֽי׃
{ס}　　　
כחכח לְמַחְלִי֙ אֶלְעָזָ֔ר וְלֹא־הָ֥יָה ל֖וֹ בָּנִֽים׃
{ס}　　　
כטכט לְקִ֖ישׁ בְּנֵי־קִ֥ישׁ יְרַחְמְאֵֽל׃
{ס}　　　
לל וּבְנֵ֣י מוּשִׁ֔י מַחְלִ֥י וְעֵ֖דֶר וִירִימ֑וֹת אֵ֛לֶּה בְּנֵ֥י הַלְוִיִּ֖ם לְבֵ֥ית אֲבֹתֵיהֶֽם׃ לאלא וַיַּפִּ֩ילוּ֩ גַם־הֵ֨ם גּוֹרָל֜וֹת לְעֻמַּ֣ת ׀ אֲחֵיהֶ֣ם בְּנֵי־אַהֲרֹ֗ן לִפְנֵ֨י דָוִ֤יד הַמֶּ֙לֶךְ֙ וְצָד֣וֹק וַאֲחִימֶ֔לֶךְ וְרָאשֵׁי֙ הָאָב֔וֹת לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם אָב֣וֹת הָרֹ֔אשׁ לְעֻמַּ֖ת אָחִ֥יו הַקָּטָֽן׃
