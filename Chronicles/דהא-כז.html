הה שַׂ֣ר הַצָּבָ֤א הַשְּׁלִישִׁי֙ לַחֹ֣דֶשׁ הַשְּׁלִישִׁ֔י בְּנָיָ֧הוּ בֶן־יְהוֹיָדָ֛ע הַכֹּהֵ֖ן רֹ֑אשׁ וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ וו ה֧וּא בְנָיָ֛הוּ גִּבּ֥וֹר הַשְּׁלֹשִׁ֖ים וְעַל־הַשְּׁלֹשִׁ֑ים וּמַ֣חֲלֻקְתּ֔וֹ עַמִּיזָבָ֖ד בְּנֽוֹ׃

חח הַחֲמִישִׁי֙ לַחֹ֣דֶשׁ הַחֲמִישִׁ֔י הַשַּׂ֖ר שַׁמְה֣וּת הַיִּזְרָ֑ח וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃

טט הַשִּׁשִּׁי֙ לַחֹ֣דֶשׁ הַשִּׁשִּׁ֔י עִירָ֥א בֶן־עִקֵּ֖שׁ הַתְּקוֹעִ֑י וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃
