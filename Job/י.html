מקרא על פי המסורה

ספר איוב

פרק י

יאא נָֽקְטָ֥ה נַפְשִׁ֗י בְּחַ֫יָּ֥י
אֶעֶזְבָ֣ה עָלַ֣י שִׂיחִ֑י
אֲ֝דַבְּרָ֗הבְּמַ֣ר נַפְשִֽׁי׃
בב אֹמַ֣ר אֶל־אֱ֭לוֹהַּ אַל־תַּרְשִׁיעֵ֑נִי
ה֝וֹדִיעֵ֗נִי עַ֣ל מַה־תְּרִיבֵֽנִי׃
גג הֲט֤וֹב לְךָ֨ ׀ כִּֽי־תַעֲשֹׁ֗ק
כִּֽי־תִ֭מְאַס יְגִ֣יעַ כַּפֶּ֑יךָ
וְעַל־עֲצַ֖ת רְשָׁעִ֣ים הוֹפָֽעְתָּ׃
דד הַעֵינֵ֣י בָשָׂ֣ר לָ֑ךְ
אִם־כִּרְא֖וֹת אֱנ֣וֹשׁ תִּרְאֶֽה׃
הה הֲכִימֵ֣י אֱנ֣וֹשׁ יָמֶ֑יךָ
אִם־שְׁ֝נוֹתֶ֗יךָ כִּ֣ימֵי גָֽבֶר׃
וו כִּֽי־תְבַקֵּ֥שׁ לַעֲוֺנִ֑י
וּֽלְחַטָּאתִ֥י תִדְרֽוֹשׁ׃
זז עַֽל־דַּ֭עְתְּךָ כִּֽי־לֹ֣א אֶרְשָׁ֑ע
וְאֵ֖ין מִיָּדְךָ֣ מַצִּֽיל׃
חח יָדֶ֣יךָ עִ֭צְּבוּנִי וַֽיַּעֲשׂ֑וּנִי
יַ֥חַד סָ֝בִ֗יב וַֽתְּבַלְּעֵֽנִי׃
טט זְכׇר־נָ֭א כִּי־כַחֹ֣מֶר עֲשִׂיתָ֑נִי
וְֽאֶל־עָפָ֥ר תְּשִׁיבֵֽנִי׃
יי הֲלֹ֣א כֶ֭חָלָב תַּתִּיכֵ֑נִי
וְ֝כַגְּבִנָּ֗ה תַּקְפִּיאֵֽנִי׃
יאיא ע֣וֹר וּ֭בָשָׂר תַּלְבִּישֵׁ֑נִי
וּֽבַעֲצָמ֥וֹת וְ֝גִידִ֗ים תְּשֹׂכְﬞכֵֽנִינתְּשֹׂכְﬞכֵֽנִי · א=תְּשֹׂכֲכֵֽנִי (חטף)׃
יביב חַיִּ֣ים וָ֭חֶסֶד עָשִׂ֣יתָ עִמָּדִ֑י
וּ֝פְקֻדָּתְךָ֗ שָֽׁמְרָ֥ה רוּחִֽי׃
יגיג וְ֭אֵלֶּה צָפַ֣נְתָּ בִלְבָבֶ֑ךָ
יָ֝דַ֗עְתִּי כִּי־זֹ֥את עִמָּֽךְ׃
ידיד אִם־חָטָ֥אתִי וּשְׁמַרְתָּ֑נִי
וּ֝מֵעֲוֺנִ֗י לֹ֣א תְנַקֵּֽנִי׃
טוטו אִם־רָשַׁ֡עְתִּי אַלְלַ֬י לִ֗י
וְ֭צָדַקְתִּי לֹא־אֶשָּׂ֣א רֹאשִׁ֑י
שְׂבַ֥ע קָ֝ל֗וֹןוּרְאֵ֥ה עׇנְיִֽי׃
טזטז וְ֭יִגְאֶה כַּשַּׁ֣חַל תְּצוּדֵ֑נִי
וְ֝תָשֹׁ֗ב תִּתְפַּלָּא־בִֽי׃
יזיז תְּחַדֵּ֬שׁ עֵדֶ֨יךָ ׀ נֶגְדִּ֗י
וְתֶ֣רֶב כַּ֭עַשְׂךָ עִמָּדִ֑י
חֲלִיפ֖וֹת וְצָבָ֣א עִמִּֽי׃
יחיח וְלָ֣מָּה מֵ֭רֶחֶם הֹצֵאתָ֑נִי
אֶ֝גְוַ֗ע וְעַ֣יִן לֹֽא־תִרְאֵֽנִי׃
יטיט כַּאֲשֶׁ֣ר לֹא־הָיִ֣יתִי אֶהְיֶ֑ה
מִ֝בֶּ֗טֶן לַקֶּ֥בֶר אוּבָֽל׃
ככ הֲלֹֽא־מְעַ֣ט יָמַ֣י יוַחֲדָ֑לקקקרי/כתיב · (יחדל) וַחֲדָ֑ל
יוְשִׁ֥יתקקקרי/כתיב · (ישית) וְשִׁ֥ית מִ֝מֶּ֗נִּי וְאַבְלִ֥יגָה מְּעָֽט׃
כאכא בְּטֶ֣רֶם אֵ֭לֵךְ וְלֹ֣א אָשׁ֑וּב
אֶל־אֶ֖רֶץ חֹ֣שֶׁךְ וְצַלְמָֽוֶת׃
כבכב אֶ֤רֶץ עֵפָ֨תָה ׀ כְּמ֥וֹ אֹ֗פֶל
צַ֭לְמָוֶת וְלֹ֥א סְדָרִ֗ים
וַתֹּ֥פַע כְּמוֹ־אֹֽפֶל׃
