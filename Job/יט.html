מקרא על פי המסורה

ספר איוב

פרק יט


{פ}　　

יטאא
בב עַד־אָ֭נָה תּוֹגְי֣וּן נַפְשִׁ֑י
וּֽתְדַכּאוּנַ֥נִי בְמִלִּֽים׃
גג זֶ֤ה עֶ֣שֶׂר פְּ֭עָמִים תַּכְלִימ֑וּנִי
לֹֽא־תֵ֝בֹ֗שׁוּ תַּהְכְּרוּ־לִֽי׃
דד וְאַף־אׇמְנָ֥ם שָׁגִ֑יתִי
אִ֝תִּ֗י תָּלִ֥ין מְשׁוּגָתִֽי׃
הה אִם־אׇ֭מְנָם עָלַ֣י תַּגְדִּ֑ילוּ
וְתוֹכִ֥יחוּ עָ֝לַ֗י חֶרְפָּתִֽי׃
וו דְּֽעוּ־אֵ֭פוֹ כִּֽי־אֱל֣וֹהַּ עִוְּתָ֑נִי
וּ֝מְצוּד֗וֹ עָלַ֥י הִקִּֽיף׃
זז הֵ֤ן אֶצְעַ֣ק חָ֭מָס וְלֹ֣א אֵעָנֶ֑ה
אֲ֝שַׁוַּ֗ע וְאֵ֣ין מִשְׁפָּֽט׃
חח אׇרְחִ֣י גָ֭דַר וְלֹ֣א אֶעֱב֑וֹר
וְעַ֥ל נְ֝תִיבוֹתַ֗י חֹ֣שֶׁךְ יָשִֽׂים׃
טט כְּ֭בוֹדִי מֵעָלַ֣י הִפְשִׁ֑יט
וַ֝יָּ֗סַר עֲטֶ֣רֶת רֹאשִֽׁי׃
יי יִתְּצֵ֣נִי סָ֭בִיב וָאֵלַ֑ךְ
וַיַּסַּ֥ע כָּ֝עֵ֗ץ תִּקְוָתִֽי׃
יאיא וַיַּ֣חַר עָלַ֣י אַפּ֑וֹ
וַיַּחְשְׁבֵ֖נִי ל֣וֹ כְצָרָֽיו׃
יביב יַ֤חַד ׀ יָ֘בֹ֤אוּ גְדוּדָ֗יו וַיָּסֹ֣לּוּ עָלַ֣י דַּרְכָּ֑ם
וַיַּחֲנ֖וּ סָבִ֣יב ׃
יגיג אַ֭חַי מֵעָלַ֣י הִרְחִ֑יק
וְ֝יֹדְעַ֗י אַךְ־זָ֥רוּ מִמֶּֽנִּי׃
ידיד חָדְל֥וּ קְרוֹבָ֑י
וּֽמְיֻדָּעַ֥י שְׁכֵחֽוּנִי׃
טוטו גָּ֘רֵ֤י בֵיתִ֣י וְ֭אַמְהֹתַי לְזָ֣ר תַּחְשְׁבֻ֑נִי
נׇ֝כְרִ֗י הָיִ֥יתִי בְעֵינֵיהֶֽם׃
טזטז לְעַבְדִּ֣י קָ֭רָאתִי וְלֹ֣א יַעֲנֶ֑ה
בְּמוֹ־פִ֝֗ינבְּמוֹ־פִ֝֗י=ל (גרש בתיבה המוטעמת כדרכו של הכתר) וכן בדפוסים (וכך הכריעו ברויאר, מכון ממרא, סימנים)א!=בְּ֝מוֹ־פִ֗י (גרש בראש התיבה המוקפת שלא כדרכו) וכך הכריעו במג"ה. אֶתְחַנֶּן־לֽוֹ׃
יזיז ר֭וּחִי זָ֣רָה לְאִשְׁתִּ֑י
וְ֝חַנֹּתִ֗י לִבְנֵ֥י בִטְנִֽי׃
יחיח גַּם־עֲ֭וִילִים מָ֣אֲסוּ בִ֑י
אָ֝ק֗וּמָה וַיְדַבְּרוּ־בִֽי׃
יטיט תִּ֭עֲבוּנִי כׇּל־מְתֵ֣י סוֹדִ֑י
וְזֶה־אָ֝הַ֗בְתִּי נֶהְפְּכוּ־בִֽי׃
ככ בְּעוֹרִ֣י וּ֭בִבְשָׂרִי דָּבְקָ֣ה עַצְמִ֑י
וָ֝אֶתְמַלְּטָ֗ה בְּע֣וֹר שִׁנָּֽי׃
כאכא חׇנֻּ֬נִי חׇנֻּ֣נִי אַתֶּ֣ם רֵעָ֑י
כִּ֥י יַד־אֱ֝ל֗וֹהַּ נָ֣גְעָה בִּֽי׃
כבכב לָ֭מָּה תִּרְדְּפֻ֣נִי כְמוֹ־אֵ֑ל
וּ֝מִבְּשָׂרִ֗י לֹ֣א תִשְׂבָּֽעוּ׃
כגכג מִי־יִתֵּ֣ן אֵ֭פוֹ וְיִכָּתְב֣וּן מִלָּ֑י
מִי־יִתֵּ֖ן בַּסֵּ֣פֶר וְיֻחָֽקוּ׃
כדכד בְּעֵט־בַּרְזֶ֥ל וְעֹפָ֑רֶת
לָ֝עַ֗ד בַּצּ֥וּר יֵחָצְבֽוּן׃
כהכה וַאֲנִ֣י יָ֭דַעְתִּי גֹּ֣אֲלִי חָ֑י
וְ֝אַחֲר֗וֹן עַל־עָפָ֥ר יָקֽוּם׃
כוכו וְאַחַ֣ר ע֭וֹרִי נִקְּפוּ־זֹ֑את
וּ֝מִבְּשָׂרִ֗י אֶחֱזֶ֥ה אֱלֽוֹהַּ׃
כזכז אֲשֶׁ֤ר אֲנִ֨י ׀ אֶחֱזֶה־לִּ֗י
וְעֵינַ֣י רָא֣וּ וְלֹא־זָ֑ר
כָּל֖וּ כִלְיֹתַ֣י בְּחֵקִֽי׃
כחכח כִּ֣י תֹ֭אמְרוּ מַה־נִּרְדׇּף־ל֑וֹ
וְשֹׁ֥רֶשׁ דָּ֝בָ֗ר נִמְצָא־בִֽי׃
כטכט גּ֤וּרוּ לָכֶ֨ם ׀ מִפְּנֵי־חֶ֗רֶב
כִּֽי־חֵ֭מָה עֲוֺנ֣וֹת חָ֑רֶב
לְמַ֖עַן תֵּדְע֣וּן שַׁדּֽיוּןקקקרי/כתיב · (שדין) שַׁדּֽוּן׃
