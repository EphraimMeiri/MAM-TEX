פרק יז

יזאא רוּחִ֣י חֻ֭בָּלָה יָמַ֥י נִזְעָ֗כוּ
קְבָרִ֥ים לִֽי׃
בב אִם־לֹ֣א הֲ֭תֻלִים עִמָּדִ֑י
וּ֝בְהַמְּרוֹתָ֗ם תָּלַ֥ן עֵינִֽי׃
גג שִֽׂימָה־נָּ֭א עׇרְבֵ֣נִי עִמָּ֑ךְ
מִ֥י ה֝֗וּאנמִ֥י ה֝֗וּא=ל ("מי" בטעם מרכא) וכן הכריעו ברויאר ומג"הא=<מִ֥י ה֝וּא> (השמטת נקודת הרביע כשיטתו)ממ=מִֽי ה֝֗וּא ("מי" בגעיה ובלתי-מוקפת!)קורן=מִי־ה֝֗וּא ("מי" מוקפת) לְיָדִ֥י יִתָּקֵֽעַ׃
דד כִּֽי־לִ֭בָּם צָפַ֣נְתָּ מִשָּׂ֑כֶל
עַל־כֵּ֝֗ןנעַל־כֵּ֝֗ן · א=עַל־כֵּ֝ן (השמטת נקודת הרביע) לֹ֣א תְרֹמֵֽם׃
הה לְ֭חֵלֶק יַגִּ֣יד רֵעִ֑ים
וְעֵינֵ֖י בָנָ֣יו תִּכְלֶֽנָה׃
וו וְֽ֭הִצִּיגַנִי לִמְשֹׁ֣ל עַמִּ֑ים
וְתֹ֖פֶת לְפָנִ֣ים אֶהְיֶֽה׃
זז וַתֵּ֣כַהּ מִכַּ֣עַשׂ עֵינִ֑י
וִיצֻרַ֖י כַּצֵּ֣ל כֻּלָּֽם׃
חח יָשֹׁ֣מּוּ יְשָׁרִ֣ים עַל־זֹ֑את
וְ֝נָקִ֗י עַל־חָנֵ֥ף יִתְעֹרָֽר׃
טט וְיֹאחֵ֣ז צַדִּ֣יק דַּרְכּ֑וֹ
וּֽטְﬞהׇר־יָ֝דַ֗יִםנוּֽטְﬞהׇר־יָ֝דַ֗יִם · א=וּֽטֳהָר־יָ֝דַ֗יִם (בחטף קמץ) יֹסִ֥יף אֹֽמֶץ׃
יי וְֽאוּלָ֗ם כֻּלָּ֣ם תָּ֭שֻׁבוּ וּבֹ֣אוּ נָ֑א
וְלֹֽא־אֶמְצָ֖א בָכֶ֣ם חָכָֽם׃
יאיא יָמַ֣י עָ֭בְרוּ זִמֹּתַ֣י נִתְּק֑וּ
מ֖וֹרָשֵׁ֣י לְבָבִֽי׃
יביב לַ֭יְלָה לְי֣וֹם יָשִׂ֑ימוּ
א֝֗וֹרנא֝֗וֹר · א=א֝וֹר (השמטת נקודת הרביע) קָר֥וֹב מִפְּנֵי־חֹֽשֶׁךְ׃
יגיג אִם־אֲ֭קַוֶּה שְׁא֣וֹל בֵּיתִ֑י
בַּ֝חֹ֗שֶׁךְ רִפַּ֥דְתִּי יְצוּעָֽי׃
ידיד לַשַּׁ֣חַת קָ֭רָאתִי אָ֣בִי אָ֑תָּה
אִמִּ֥י וַ֝אֲחֹתִ֗י לָרִמָּֽה׃
טוטו וְ֭אַיֵּה אֵפ֣וֹ תִקְוָתִ֑י
וְ֝תִקְוָתִ֗י מִ֣י יְשׁוּרֶֽנָּה׃
טזטז בַּדֵּ֣י שְׁאֹ֣ל תֵּרַ֑דְנָה
אִם־יַ֖חַד עַל־עָפָ֣ר נָֽחַת׃
