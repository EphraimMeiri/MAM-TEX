מקרא על פי המסורה

ספר איוב

פרק לו


{פ}　　

לואא
בב כַּתַּר־לִ֣י זְ֭עֵיר וַאֲחַוֶּ֑ךָּ
כִּ֤י ע֖וֹד לֶאֱל֣וֹהַּ מִלִּֽים׃
גג אֶשָּׂ֣א דֵ֭עִי לְמֵרָח֑וֹק
וּ֝לְפֹעֲלִ֗י אֶֽתֵּֽן־צֶֽדֶק׃
דד כִּֽי־אׇ֭מְנָם לֹא־שֶׁ֣קֶר מִלָּ֑י
תְּמִ֖ים דֵּע֣וֹת עִמָּֽךְ׃
הה הֶן־אֵ֣ל כַּ֭בִּיר וְלֹ֣א יִמְאָ֑ס
כַּ֝בִּ֗יר כֹּ֣חַֽ לֵֽב׃
וו לֹא־יְחַיֶּ֥ה רָשָׁ֑ע
וּמִשְׁפַּ֖ט עֲנִיִּ֣ים יִתֵּֽן׃
זז לֹֽא־יִגְרַ֥ע מִצַּדִּ֗יקעֵ֫ינָ֥יו
וְאֶת־מְלָכִ֥ים לַכִּסֵּ֑א
וַיֹּשִׁיבֵ֥ם לָ֝נֶ֗צַח וַיִּגְבָּֽהוּ׃
חח וְאִם־אֲסוּרִ֥ים בַּזִּקִּ֑ים
יִ֝לָּכְד֗וּן בְּחַבְלֵי־עֹֽנִי׃
טט וַיַּגֵּ֣ד לָהֶ֣ם
וּ֝פִשְׁעֵיהֶ֗ם כִּ֣י יִתְגַּבָּֽרוּ׃
יי וַיִּ֣גֶל אׇ֭זְנָם לַמּוּסָ֑ר
וַ֝יֹּ֗אמֶר כִּֽי־יְשׁוּב֥וּן מֵאָֽוֶן׃
יאיא אִֽם־יִשְׁמְע֗וּוְֽיַ֫עֲבֹ֥דוּ
יְכַלּ֣וּ יְמֵיהֶ֣ם בַּטּ֑וֹב
וּ֝שְׁנֵיהֶ֗ם בַּנְּעִימִֽים׃
יביב וְאִם־לֹ֣א יִ֭שְׁמְעוּ בְּשֶׁ֣לַח יַעֲבֹ֑רוּ
וְ֝יִגְוְע֗וּ בִּבְלִי־דָֽעַת׃
יגיג וְֽחַנְפֵי־לֵ֭ב יָשִׂ֣ימוּ אָ֑ף
לֹ֥א יְ֝שַׁוְּע֗וּ כִּ֣י אֲסָרָֽם׃
ידיד תָּמֹ֣ת בַּנֹּ֣עַר נַפְשָׁ֑ם
וְ֝חַיָּתָ֗ם בַּקְּדֵשִֽׁים׃
טוטו יְחַלֵּ֣ץ עָנִ֣י בְעׇנְי֑וֹ
וְיִ֖גֶל בַּלַּ֣חַץ אׇזְנָֽם׃
טזטז וְאַ֤ף הֲסִֽיתְךָ֨ ׀ מִפִּי־צָ֗ר
רַ֭חַב לֹא־מוּצָ֣ק תַּחְתֶּ֑יהָ
וְנַ֥חַת שֻׁ֝לְחָנְךָ֗מָ֣לֵא דָֽשֶׁן׃
יזיז וְדִין־רָשָׁ֥ע מָלֵ֑אתָ
דִּ֖ין וּמִשְׁפָּ֣ט יִתְמֹֽכוּ׃
יחיח כִּֽי־חֵ֭מָה פֶּן־יְסִֽיתְךָ֣ בְסָ֑פֶק
וְרׇב־כֹּ֝֗פֶרנוְרׇב־כֹּ֝֗פֶר · א=וְרׇב־כֹּ֝פֶר (השמטת נקודת הרביע) אַל־יַטֶּֽךָּ׃
יטיט הֲיַעֲרֹ֣ךְ שׁ֭וּעֲךָ לֹ֣א בְצָ֑ר
וְ֝כֹ֗ל מַאֲמַצֵּי־כֹֽחַ׃
ככ אַל־תִּשְׁאַ֥ף הַלָּ֑יְלָה
לַעֲל֖וֹת עַמִּ֣ים תַּחְתָּֽם׃
כאכא הִ֭שָּׁמֶר אַל־תֵּ֣פֶן אֶל־אָ֑וֶן
כִּֽי־עַל־זֶ֝֗הנכִּֽי־עַל־זֶ֝֗ה · א=כִּֽי־עַל־זֶ֝ה (השמטת נקודת הרביע) בָּחַ֥רְתָּ מֵעֹֽנִי׃
כבכב הֶן־אֵ֭ל יַשְׂגִּ֣יב בְּכֹח֑וֹ
מִ֖י כָמֹ֣הוּ מוֹרֶֽה׃
כגכג מִֽי־פָקַ֣ד עָלָ֣יו דַּרְכּ֑וֹ
וּמִי־אָ֝מַ֗ר פָּעַ֥לְתָּ עַוְלָֽה׃
כדכד זְ֭כֹר כִּֽי־תַשְׂגִּ֣יא
אֲשֶׁ֖ר שֹׁרְﬞר֣וּנשֹׁרְﬞר֣וּ · א=שֹׁרֲר֣וּ (חטף) אֲנָשִֽׁים׃
כהכה כׇּל־אָדָ֥ם חָזוּ־ב֑וֹ
אֱ֝נ֗וֹשׁ יַבִּ֥יט מֵרָחֽוֹק׃
כוכו הֶן־אֵ֣ל שַׂ֭גִּיא וְלֹ֣א נֵדָ֑ע
מִסְפַּ֖ר שָׁנָ֣יו וְלֹא־חֵֽקֶר׃
כזכז כִּ֭י יְגָרַ֣ע נִטְפֵי־מָ֑יִם
יָזֹ֖קּוּ מָטָ֣ר לְאֵדֽוֹ׃
כחכח אֲשֶֽׁר־יִזְּל֥וּ שְׁחָקִ֑ים
יִ֝רְעֲפ֗וּ עֲלֵ֤י ׀ אָדָ֬ם רָֽב׃
כטכט אַ֣ף אִם־יָ֭בִין מִפְרְשֵׂי־עָ֑ב
תְּ֝שֻׁא֗וֹת סֻכָּתֽוֹ׃
לל הֵן־פָּרַ֣שׂ עָלָ֣יו אוֹר֑וֹ
וְשׇׁרְשֵׁ֖י הַיָּ֣ם כִּסָּֽה׃
לאלא כִּֽי־בָ֭ם יָדִ֣ין עַמִּ֑ים
יִֽתֶּן־אֹ֥כֶל לְמַכְבִּֽיר׃
לבלב עַל־כַּפַּ֥יִם כִּסָּה־א֑וֹר
וַיְצַ֖ו עָלֶ֣יהָ בְמַפְגִּֽיעַ׃
לגלג יַגִּ֣יד עָלָ֣יו רֵע֑וֹ
מִ֝קְנֶ֗ה אַ֣ף עַל־עוֹלֶֽה׃
