גג גַּם־לִ֤י לֵבָ֨ב ׀ כְּֽמוֹכֶ֗ם
לֹא־נֹפֵ֣ל אָנֹכִ֣י מִכֶּ֑ם
וְאֶת־מִי־אֵ֥ין כְּמוֹ־אֵֽלֶּה׃
דד שְׂחֹ֤ק לְרֵעֵ֨הוּ ׀ אֶהְיֶ֗ה
קֹרֵ֣א לֶ֭אֱלוֹהַּ וַֽיַּעֲנֵ֑הוּ
שְׂ֝ח֗וֹקצַדִּ֥יק תָּמִֽים׃
הה לַפִּ֣יד בּ֭וּז לְעַשְׁתּ֣וּת שַׁאֲנָ֑ן
נָ֝כ֗וֹן לְמ֣וֹעֲדֵי רָֽגֶל׃
וו יִשְׁלָ֤יוּ אֹהָלִ֨ים ׀ לְשֹׁ֥דְﬞדִ֗יםנלְשֹׁ֥דְﬞדִ֗ים · א=לְשֹׁ֥דֲדִ֗ים (חטף)
וּֽ֭בַטֻּחוֹת לְמַרְגִּ֣יזֵי אֵ֑ל
לַאֲשֶׁ֤ר הֵבִ֖יא אֱל֣וֹהַּ בְּיָדֽוֹ׃
זז וְֽאוּלָ֗ם שְׁאַל־נָ֣א בְהֵמ֣וֹת וְתֹרֶ֑ךָּ
וְע֥וֹף הַ֝שָּׁמַ֗יִם וְיַגֶּד־לָֽךְ׃
חח א֤וֹ שִׂ֣יחַ לָאָ֣רֶץ וְתֹרֶ֑ךָּ
וִיסַפְּר֥וּ לְ֝ךָ֗ דְּגֵ֣י הַיָּֽם׃
טט מִ֭י לֹא־יָדַ֣ע בְּכׇל־אֵ֑לֶּה
כִּ֥י יַד־יְ֝הֹוָ֗ה עָ֣שְׂתָה זֹּֽאת׃
יי אֲשֶׁ֣ר בְּ֭יָדוֹ נֶ֣פֶשׁ כׇּל־חָ֑י
וְ֝ר֗וּחַ כׇּל־בְּשַׂר־אִֽישׁ׃
יאיא הֲלֹא־אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן
וְ֝חֵ֗ךְ אֹ֣כֶל יִטְעַם־לֽוֹ׃
יביב בִּישִׁישִׁ֥ים חׇכְמָ֑ה
וְאֹ֖רֶךְ יָמִ֣ים תְּבוּנָֽה׃
יגיג עִ֭מּוֹ חׇכְמָ֣ה וּגְבוּרָ֑ה
ל֝֗וֹנל֝֗וֹ · א=ל֝וֹ (השמטת נקודת הרביע) עֵצָ֥ה וּתְבוּנָֽה׃
ידיד הֵ֣ן יַ֭הֲרוֹס וְלֹ֣א יִבָּנֶ֑ה
יִסְגֹּ֥ר עַל־אִ֝֗ישׁ וְלֹ֣א יִפָּתֵֽחַ׃
טוטו הֵ֤ן יַעְצֹ֣ר בַּמַּ֣יִם וְיִבָ֑שׁוּ
וִ֝ישַׁלְּחֵ֗ם וְיַ֖הַפְכוּ אָֽרֶץ׃
טזטז עִ֭מּוֹ עֹ֣ז וְתוּשִׁיָּ֑ה
ל֝֗וֹנל֝֗וֹ · א=ל֝וֹ (השמטת נקודת הרביע) שֹׁגֵ֥ג וּמַשְׁגֶּֽה׃
יזיז מוֹלִ֣יךְ יוֹעֲצִ֣ים שׁוֹלָ֑ל
וְֽשֹׁפְטִ֥ים יְהוֹלֵֽל׃
יחיח מוּסַ֣ר מְלָכִ֣ים פִּתֵּ֑חַ
וַיֶּאְסֹ֥ר אֵ֝ז֗וֹר בְּמׇתְנֵיהֶֽם׃

וּמְזִ֖יחַ אֲפִיקִ֣ים רִפָּֽה׃
