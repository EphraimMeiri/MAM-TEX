מקרא על פי המסורה

ספר איוב

פרק יג

יגאא הֶן־כֹּ֭ל רָאֲתָ֣ה עֵינִ֑י
שָֽׁמְעָ֥ה אׇ֝זְנִ֗י וַתָּ֥בֶן לָֽהּ׃
בב כְּֽ֭דַעְתְּכֶם יָדַ֣עְתִּי גַם־אָ֑נִי
לֹֽא־נֹפֵ֖ל אָנֹכִ֣י מִכֶּֽם׃
גג אוּלָ֗ם אֲ֭נִי אֶל־שַׁדַּ֣י אֲדַבֵּ֑ר
וְהוֹכֵ֖חַ אֶל־אֵ֣ל אֶחְפָּֽץ׃
דד וְֽאוּלָ֗ם אַתֶּ֥ם טֹפְלֵי־שָׁ֑קֶר
רֹפְאֵ֖י אֱלִ֣ל כֻּלְּכֶֽם׃
הה מִי־יִ֭תֵּן הַחֲרֵ֣שׁ תַּחֲרִישׁ֑וּן
וּתְהִ֖י לָכֶ֣ם לְחׇכְמָֽה׃
וו שִׁמְעוּ־נָ֥א תוֹכַחְתִּ֑י
וְרִב֖וֹת שְׂפָתַ֣י הַקְשִֽׁיבוּ׃
זז הַ֭לְאֵל תְּדַבְּר֣וּ עַוְלָ֑ה
וְ֝ל֗וֹ תְּֽדַבְּר֥וּ רְמִיָּֽה׃
חח הֲפָנָ֥יו תִּשָּׂא֑וּן
אִם־לָאֵ֥ל תְּרִיבֽוּן׃
טט הֲ֭טוֹב כִּֽי־יַחְקֹ֣ר אֶתְכֶ֑ם
אִם־כְּהָתֵ֥ל בֶּ֝אֱנ֗וֹשׁ תְּהָתֵ֥לּוּ בֽוֹ׃
יי הוֹכֵ֣חַ יוֹכִ֣יחַ אֶתְכֶ֑ם
אִם־בַּ֝סֵּ֗תֶר פָּנִ֥ים תִּשָּׂאֽוּן׃
יאיא הֲלֹ֣א שְׂ֭אֵתוֹ תְּבַעֵ֣ת אֶתְכֶ֑ם
וּ֝פַחְדּ֗וֹ יִפֹּ֥ל עֲלֵיכֶֽם׃
יביב זִֽ֭כְרֹנֵיכֶם מִשְׁלֵי־אֵ֑פֶר
לְגַבֵּי־חֹ֝֗מֶרנלְגַבֵּי־חֹ֝֗מֶר · א=לְגַבֵּי־חֹ֝מֶר (השמטת נקודת הרביע) גַּבֵּיכֶֽם׃
יגיג הַחֲרִ֣ישׁוּ מִ֭מֶּנִּי וַאֲדַבְּרָה־אָ֑נִי
וְיַעֲבֹ֖ר עָלַ֣י מָֽה׃
ידיד עַל־מָ֤ה ׀ אֶשָּׂ֣א בְשָׂרִ֣י בְשִׁנָּ֑י
וְ֝נַפְשִׁ֗י אָשִׂ֥ים בְּכַפִּֽי׃
טוטו הֵ֣ן יִ֭קְטְלֵנִי (לא)קקקרי/כתיב · (לא) ל֣וֹ אֲיַחֵ֑ל
אַךְ־דְּ֝רָכַ֗י אֶל־פָּנָ֥יו אוֹכִֽיחַ׃
טזטז גַּם־הוּא־לִ֥י לִישׁוּעָ֑ה
כִּי־לֹ֥א לְ֝פָנָ֗יו חָנֵ֥ף יָבֽוֹא׃
יזיז שִׁמְע֣וּ שָׁ֭מוֹעַ מִלָּתִ֑י
וְ֝אַחֲוָתִ֗י בְּאׇזְנֵיכֶֽם׃
יחיח הִנֵּה־נָ֭א עָרַ֣כְתִּי מִשְׁפָּ֑ט
יָ֝דַ֗עְתִּי כִּי־אֲנִ֥י אֶצְדָּֽק׃
יטיט מִי־ה֭וּא יָרִ֣יב עִמָּדִ֑י
כִּֽי־עַתָּ֖ה אַחֲרִ֣ישׁ וְאֶגְוָֽע׃
ככ אַךְ־שְׁ֭תַּיִם אַל־תַּ֣עַשׂ עִמָּדִ֑י
אָ֥ז מִ֝פָּנֶ֗יךָ לֹ֣א אֶסָּתֵֽר׃
כאכא כַּ֭פְּךָ מֵעָלַ֣י הַרְחַ֑ק
וְ֝אֵ֥מָתְךָ֗ אַֽל־תְּבַעֲתַֽנִּי׃
כבכב וּ֭קְרָא וְאָנֹכִ֣י אֶעֱנֶ֑ה
אֽוֹ־אֲ֝דַבֵּ֗ר וַהֲשִׁיבֵֽנִי׃
כגכג כַּמָּ֣ה לִ֭י עֲוֺנ֣וֹת וְחַטָּא֑וֹת
פִּֽשְׁעִ֥י וְ֝חַטָּאתִ֗י הֹדִיעֵֽנִי׃
כדכד לָֽמָּה־פָנֶ֥יךָ תַסְתִּ֑יר
וְתַחְשְׁבֵ֖נִי לְאוֹיֵ֣ב לָֽךְ׃
כהכה הֶעָלֶ֣ה נִדָּ֣ף תַּעֲר֑וֹץ
וְאֶת־קַ֖שׁ יָבֵ֣שׁ תִּרְדֹּֽף׃
כוכו כִּֽי־תִכְתֹּ֣ב עָלַ֣י מְרֹר֑וֹת
וְ֝תוֹרִישֵׁ֗נִי עֲוֺנ֥וֹת נְעוּרָֽי׃
כזכז וְתָ֘שֵׂ֤ם בַּסַּ֨ד ׀ רַגְלַ֗י
וְתִשְׁמ֥וֹר כׇּל־אׇרְחוֹתָ֑י
עַל־שׇׁרְשֵׁ֥י רַ֝גְלַ֗יתִּתְחַקֶּֽה׃
כחכח וְ֭הוּא כְּרָקָ֣ב יִבְלֶ֑ה
כְּ֝בֶ֗גֶד אֲכָ֣לוֹ עָֽשׁ׃
