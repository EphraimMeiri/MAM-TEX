פרק כט


{פ}　　

כטאא
בב מִֽי־יִתְּנֵ֥נִי כְיַרְחֵי־קֶ֑דֶם
כִּ֝ימֵ֗י אֱל֣וֹהַּ יִשְׁמְרֵֽנִי׃
גג בְּהִלּ֣וֹ נֵ֭רוֹ עֲלֵ֣י רֹאשִׁ֑י
לְ֝אוֹר֗וֹ אֵ֣לֶךְ חֹֽשֶׁךְ׃
דד כַּאֲשֶׁ֣ר הָ֭יִיתִי בִּימֵ֣י חׇרְפִּ֑י
בְּס֥וֹד אֱ֝ל֗וֹהַּ עֲלֵ֣י ׃
הה בְּע֣וֹד שַׁ֭דַּי עִמָּדִ֑י
סְבִ֖יבוֹתַ֣י נְעָרָֽי׃
וו בִּרְחֹ֣ץ הֲלִיכַ֣י בְּחֵמָ֑ה
וְצ֥וּר יָצ֥וּק עִ֝מָּדִ֗י פַּלְגֵי־שָֽׁמֶן׃
זז בְּצֵ֣אתִי שַׁ֣עַר עֲלֵי־קָ֑רֶת
בָּ֝רְח֗וֹב אָכִ֥ין מוֹשָׁבִֽי׃
חח רָא֣וּנִי נְעָרִ֣ים וְנֶחְבָּ֑אוּ
וִ֝ישִׁישִׁ֗ים קָ֣מוּ עָמָֽדוּ׃
טט שָׂ֭רִים עָצְר֣וּ בְמִלִּ֑ים
וְ֝כַ֗ף יָשִׂ֥ימוּ לְפִיהֶֽם׃
יי קוֹל־נְגִידִ֥ים נֶחְבָּ֑אוּ
וּ֝לְשׁוֹנָ֗ם לְחִכָּ֥ם דָּבֵֽקָה׃
יאיא כִּ֤י אֹ֣זֶן שָׁ֭מְעָה וַֽתְּאַשְּׁרֵ֑נִי
וְעַ֥יִן רָ֝אֲתָ֗ה וַתְּעִידֵֽנִי׃
יביב כִּֽי־אֲ֭מַלֵּט עָנִ֣י מְשַׁוֵּ֑עַ
וְ֝יָת֗וֹם וְֽלֹא־עֹזֵ֥ר לֽוֹ׃
יגיג בִּרְכַּ֣ת אֹ֭בֵד עָלַ֣י תָּבֹ֑א
וְלֵ֖ב אַלְמָנָ֣ה אַרְנִֽן׃
ידיד צֶ֣דֶק לָ֭בַשְׁתִּי וַיִּלְבָּשֵׁ֑נִי
כִּֽמְעִ֥יל וְ֝צָנִ֗יף מִשְׁפָּטִֽי׃
טוטו עֵינַ֣יִם הָ֭יִיתִי לַעִוֵּ֑ר
וְרַגְלַ֖יִם לַפִּסֵּ֣חַ אָֽנִי׃
טזטז אָ֣ב אָ֭נֹכִֽינאָ֭נֹכִֽי=א?,ל (דחי וגעיה); למרות שקו הגעיה קצת נטוי, אין הכרח לראות בו מרכא (וכן הכריעו ברויאר במהדורתו הראשונה ומג"ה)אמנם לפי ברויאר א=אָ֭נֹכִ֥י (דחי ומרכא) בשתי מהדורותיו האחרונות (וכן במכון ממרא)ש1=אָ֭נֹכִי (דחי בלבד) וכן בדפוסים (קורן=אָֽ֭נֹכִי דחי וגעיה באל"ף) לָאֶבְיוֹנִ֑ים
וְרִ֖ב לֹא־יָדַ֣עְתִּי אֶחְקְרֵֽהוּ׃
יזיז וָ֭אֲשַׁבְּרָה מְתַלְּע֣וֹת עַוָּ֑ל
וּ֝מִשִּׁנָּ֗יו אַשְׁלִ֥יךְ טָֽרֶף׃
יחיח וָ֭אֹמַר עִם־קִנִּ֣י אֶגְוָ֑ע
וְ֝כַח֗וֹל אַרְבֶּ֥ה יָמִֽים׃
יטיט שׇׁרְשִׁ֣י פָת֣וּחַ אֱלֵי־מָ֑יִם
וְ֝טַ֗ל יָלִ֥ין בִּקְצִירִֽי׃
ככ כְּ֭בוֹדִי חָדָ֣שׁ עִמָּדִ֑י
וְ֝קַשְׁתִּ֗י בְּיָדִ֥י תַחֲלִֽיף׃
כאכא לִֽי־שָׁמְע֥וּ וְיִחֵ֑לּוּ
וְ֝יִדְּמ֗וּ לְמ֣וֹ עֲצָתִֽי׃
כבכב אַחֲרֵ֣י דְ֭בָרִי לֹ֣א יִשְׁנ֑וּ
וְ֝עָלֵ֗ימוֹ תִּטֹּ֥ף מִלָּתִֽי׃
כגכג וְיִחֲל֣וּ כַמָּטָ֣ר לִ֑י
וּ֝פִיהֶ֗ם פָּעֲר֥וּ לְמַלְקֽוֹשׁ׃
כדכד אֶשְׂחַ֣ק אֲ֭לֵהֶם לֹ֣א יַאֲמִ֑ינוּ
וְא֥וֹר פָּ֝נַ֗י לֹ֣א יַפִּילֽוּן׃

וְ֭אֶשְׁכּוֹן כְּמֶ֣לֶךְ בַּגְּד֑וּד
כַּאֲשֶׁ֖ר אֲבֵלִ֣ים יְנַחֵֽם׃
