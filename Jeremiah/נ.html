ספר ירמיהו

פרק נ


{פ}　　
נאא הַדָּבָ֗ר אֲשֶׁ֨ר דִּבֶּ֧ר יְהֹוָ֛ה אֶל־בָּבֶ֖ל אֶל־אֶ֣רֶץ כַּשְׂדִּ֑ים בְּיַ֖ד יִרְמְיָ֥הוּ הַנָּבִֽיא׃ בב הַגִּ֨ידוּ בַגּוֹיִ֤ם וְהַשְׁמִ֙יעוּ֙ וּֽשְׂאוּ־נֵ֔ס הַשְׁמִ֖יעוּ אַל־תְּכַחֵ֑דוּ אִמְרוּ֩ נִלְכְּדָ֨ה בָבֶ֜ל הֹבִ֥ישׁ בֵּל֙ חַ֣ת מְרֹדָ֔ךְ הֹבִ֣ישׁוּ עֲצַבֶּ֔יהָ חַ֖תּוּ גִּלּוּלֶֽיהָ׃ גג כִּ֣י עָלָה֩ עָלֶ֨יהָ גּ֜וֹי מִצָּפ֗וֹן הֽוּא־יָשִׁ֤ית אֶת־אַרְצָהּ֙ לְשַׁמָּ֔ה וְלֹא־יִהְיֶ֥ה יוֹשֵׁ֖ב בָּ֑הּ מֵאָדָ֥ם וְעַד־בְּהֵמָ֖ה נָ֥דוּ הָלָֽכוּ׃ דד בַּיָּמִ֨ים הָהֵ֜מָּה וּבָעֵ֤ת הַהִיא֙ נְאֻם־יְהֹוָ֔ה יָבֹ֧אוּ בְנֵי־יִשְׂרָאֵ֛ל הֵ֥מָּה וּבְנֵי־יְהוּדָ֖ה יַחְדָּ֑ו הָל֤וֹךְ וּבָכוֹ֙ יֵלֵ֔כוּ וְאֶת־יְהֹוָ֥ה אֱלֹהֵיהֶ֖ם יְבַקֵּֽשׁוּ׃ הה צִיּ֣וֹן יִשְׁאָ֔לוּ דֶּ֖רֶךְ הֵ֣נָּה פְנֵיהֶ֑ם בֹּ֚אוּ וְנִלְו֣וּ אֶל־יְהֹוָ֔ה בְּרִ֥ית עוֹלָ֖ם לֹ֥א תִשָּׁכֵֽחַ׃ וו צֹ֤אן אֹֽבְדוֹת֙ (היה)קקקרי/כתיב · (היה) הָי֣וּ עַמִּ֔י רֹעֵיהֶ֣ם הִתְע֔וּם הָרִ֖ים שׁוֹבְﬞב֑יוּםנ(שובבים) · א-כתיב=שׁוֹבֲב֑יּם (חטף) מֵהַ֤ר אֶל־גִּבְעָה֙ הָלָ֔כוּ שָׁכְח֖וּ רִבְצָֽם׃ זז כׇּל־מוֹצְאֵיהֶ֣ם אֲכָל֔וּם וְצָרֵיהֶ֥ם אָמְר֖וּ לֹ֣א נֶאְשָׁ֑ם תַּ֗חַת אֲשֶׁ֨ר חָטְא֤וּ לַֽיהֹוָה֙ נְוֵה־צֶ֔דֶק וּמִקְוֵ֥ה אֲבוֹתֵיהֶ֖ם יְהֹוָֽה׃
{ס}　　　
חח נֻ֚דוּ מִתּ֣וֹךְ בָּבֶ֔ל וּמֵאֶ֥רֶץ כַּשְׂדִּ֖ים (יצאו)קקקרי/כתיב · (יצאו) צֵ֑אוּ וִֽהְי֕וּ כְּעַתּוּדִ֖ים לִפְנֵי־צֹֽאן׃ טט כִּ֣י הִנֵּ֣ה אָנֹכִ֡י מֵעִיר֩ וּמַעֲלֶ֨ה עַל־בָּבֶ֜ל קְהַל־גּוֹיִ֤ם גְּדֹלִים֙ מֵאֶ֣רֶץ צָפ֔וֹן וְעָ֣רְכוּ לָ֔הּ מִשָּׁ֖ם תִּלָּכֵ֑ד חִצָּיו֙ כְּגִבּ֣וֹר מַשְׁכִּ֔יל לֹ֥א יָשׁ֖וּב רֵיקָֽם׃ יי וְהָיְתָ֥ה כַשְׂדִּ֖ים לְשָׁלָ֑ל כׇּל־שֹׁלְﬞלֶ֥יהָנכׇּל־שֹׁלְﬞלֶ֥יהָ · א=כָּל־שֹׁלֲלֶ֥יהָ (חטף) יִשְׂבָּ֖עוּ נְאֻם־יְהֹוָֽה׃ יאיא כִּ֤י תִשְׂמְחיוּ֙קקקרי/כתיב · (תשמחי) תִשְׂמְחוּ֙ כִּ֣י תַעַלְז֔יוּקקקרי/כתיב · (תעלזי) תַעַלְז֔וּ שֹׁסֵ֖י נַחֲלָתִ֑י כִּ֤י תָפ֙וּשׁיוּ֙קקקרי/כתיב · (תפושי) תָפ֙וּשׁוּ֙ כְּעֶגְלָ֣ה דָשָׁ֔ה וְתִצְהֲל֖יוּקקקרי/כתיב · (ותצהלי) וְתִצְהֲל֖וּ כָּאַבִּרִֽים׃ יביב בּ֤וֹשָׁה אִמְּכֶם֙ מְאֹ֔ד חָפְרָ֖ה יוֹלַדְתְּכֶ֑ם הִנֵּה֙ אַחֲרִ֣ית גּוֹיִ֔ם מִדְבָּ֖ר צִיָּ֥ה וַעֲרָבָֽה׃ יגיג מִקֶּ֤צֶף יְהֹוָה֙ לֹ֣א תֵשֵׁ֔ב וְהָיְתָ֥ה שְׁמָמָ֖ה כֻּלָּ֑הּ כֹּ֚ל עֹבֵ֣ר עַל־בָּבֶ֔ל יִשֹּׁ֥ם וְיִשְׁרֹ֖ק עַל־כׇּל־מַכּוֹתֶֽיהָ׃ ידיד עִרְכ֨וּ עַל־בָּבֶ֤ל ׀ סָבִיב֙ כׇּל־דֹּ֣רְכֵי קֶ֔שֶׁת יְד֣וּ אֵלֶ֔יהָ אַֽל־תַּחְמְל֖וּ אֶל־חֵ֑ץ כִּ֥י לַיהֹוָ֖ה חָטָֽאָה׃ טוטו הָרִ֨יעוּ עָלֶ֤יהָ סָבִיב֙ נָתְנָ֣ה יָדָ֔הּ נָֽפְלוּ֙ (אשויתיה)נ(אשויתיה) · א-כתיב=אָשְׁויֹתֶ֔יהָ נֶהֶרְס֖וּ חוֹמוֹתֶ֑יהָ כִּי֩ נִקְמַ֨ת יְהֹוָ֥ה הִיא֙ הִנָּ֣קְמוּ בָ֔הּ כַּאֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־לָֽהּ׃ טזטז כִּרְת֤וּ זוֹרֵ֙עַ֙ מִבָּבֶ֔ל וְתֹפֵ֥שׂ מַגָּ֖ל בְּעֵ֣ת קָצִ֑יר מִפְּנֵי֙ חֶ֣רֶב הַיּוֹנָ֔ה אִ֤ישׁ אֶל־עַמּוֹ֙ יִפְנ֔וּ וְאִ֥ישׁ לְאַרְצ֖וֹ יָנֻֽסוּ׃
{ס}　　　
יזיז שֶׂ֧ה פְזוּרָ֛ה יִשְׂרָאֵ֖ל אֲרָי֣וֹת הִדִּ֑יחוּ הָרִאשׁ֤וֹן אֲכָלוֹ֙ מֶ֣לֶךְ אַשּׁ֔וּר וְזֶ֤ה הָאַֽחֲרוֹן֙ עִצְּמ֔וֹ נְבוּכַדְרֶאצַּ֖ר מֶ֥לֶךְ בָּבֶֽל׃
{פ}　　
יחיח לָכֵ֗ן כֹּה־אָמַ֞ר יְהֹוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֥י פֹקֵ֛ד אֶל־מֶ֥לֶךְ בָּבֶ֖ל וְאֶל־אַרְצ֑וֹ כַּאֲשֶׁ֥ר פָּקַ֖דְתִּי אֶל־מֶ֥לֶךְ אַשּֽׁוּר׃ יטיט וְשֹׁבַבְתִּ֤י אֶת־יִשְׂרָאֵל֙ אֶל־נָוֵ֔הוּ וְרָעָ֥ה הַכַּרְמֶ֖ל וְהַבָּשָׁ֑ן וּבְהַ֥ר אֶפְרַ֛יִם וְהַגִּלְעָ֖ד תִּשְׂבַּ֥ע נַפְשֽׁוֹ׃ ככ בַּיָּמִ֣ים הָהֵם֩ וּבָעֵ֨ת הַהִ֜יא נְאֻם־יְהֹוָ֗ה יְבֻקַּ֞שׁ אֶת־עֲוֺ֤ן יִשְׂרָאֵל֙ וְאֵינֶ֔נּוּ וְאֶת־חַטֹּ֥את יְהוּדָ֖ה וְלֹ֣א תִמָּצֶ֑אינָה כִּ֥י אֶסְלַ֖ח לַאֲשֶׁ֥ר אַשְׁאִֽיר׃
{פ}　　
כאכא עַל־הָאָ֤רֶץ מְרָתַ֙יִם֙ עֲלֵ֣ה עָלֶ֔יהָ וְאֶל־יוֹשְׁבֵ֖י פְּק֑וֹד חֲרֹ֨ב וְהַחֲרֵ֤ם אַחֲרֵיהֶם֙ נְאֻם־יְהֹוָ֔ה וַעֲשֵׂ֕ה כְּכֹ֖ל אֲשֶׁ֥ר צִוִּיתִֽיךָ׃
{ס}　　　
כבכב ק֥וֹל מִלְחָמָ֖ה בָּאָ֑רֶץ וְשֶׁ֖בֶר גָּדֽוֹל׃ כגכג אֵ֤יךְ נִגְדַּע֙ וַיִּשָּׁבֵ֔ר פַּטִּ֖ישׁ כׇּל־הָאָ֑רֶץ אֵ֣יךְ הָיְתָ֧ה לְשַׁמָּ֛ה בָּבֶ֖ל בַּגּוֹיִֽם׃ כדכד יָקֹ֨שְׁתִּי לָ֤ךְ וְגַם־נִלְכַּדְתְּ֙נוְגַם־נִלְכַּדְתְּ֙=א,לדפוסים=וְגַם־נִלְכַּ֙דְתְּ֙ בטעם כפול, אולי בגלל המרחק מתחילת ההברה. בָּבֶ֔ל וְאַ֖תְּ לֹ֣א יָדָ֑עַתְּ נִמְצֵאת֙ וְגַם־נִתְפַּ֔שְׂתְּ כִּ֥י בַיהֹוָ֖ה הִתְגָּרִֽית׃ כהכה פָּתַ֤ח יְהֹוָה֙ אֶת־א֣וֹצָר֔וֹ וַיּוֹצֵ֖א אֶת־כְּלֵ֣י זַעְמ֑וֹ כִּֽי־מְלָאכָ֣ה הִ֗יא לַאדֹנָ֧י יֱהֹוִ֛ה צְבָא֖וֹת בְּאֶ֥רֶץ כַּשְׂדִּֽים׃ כוכו בֹּֽאוּ־לָ֤הּ מִקֵּץ֙ פִּתְח֣וּ מַאֲבֻסֶ֔יהָ סׇלּ֥וּהָ כְמֽוֹ־עֲרֵמִ֖ים וְהַחֲרִימ֑וּהָ אַל־תְּהִי־לָ֖הּ שְׁאֵרִֽית׃ כזכז חִרְבוּ֙ כׇּל־פָּרֶ֔יהָ יֵרְד֖וּ לַטָּ֑בַח ה֣וֹי עֲלֵיהֶ֔ם כִּי־בָ֥א יוֹמָ֖ם עֵ֥ת פְּקֻדָּתָֽם׃
{ס}　　　
כחכח ק֥וֹל נָסִ֛ים וּפְלֵטִ֖ים מֵאֶ֣רֶץ בָּבֶ֑ל לְהַגִּ֣יד בְּצִיּ֗וֹן אֶת־נִקְמַת֙ יְהֹוָ֣ה אֱלֹהֵ֔ינוּ נִקְמַ֖ת הֵיכָלֽוֹ׃ כטכט הַשְׁמִ֣יעוּ אֶל־בָּבֶ֣ל ׀ רַ֠בִּ֠ים כׇּל־דֹּ֨רְכֵי קֶ֜שֶׁת חֲנ֧וּ עָלֶ֣יהָ סָבִ֗יב אַל־יְהִי־() פְּלֵיטָ֔הנאַל־יְהִי־() פְּלֵיטָ֔הא-כתיב=אַל־יְהִי־[֯] פְּלֵיטָ֔ה (רווח ריק בלי ניקוד וטעמים, ולמעלה עיגול מסורה) שַׁלְּמוּ־לָ֣הּ כְּכֹ֛ל אֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־לָ֑הּ כִּ֧י אֶל־יְהֹוָ֛ה זָ֖דָה אֶל־קְד֥וֹשׁ יִשְׂרָאֵֽל׃ לל לָכֵ֛ן יִפְּל֥וּ בַחוּרֶ֖יהָ בִּרְחֹבֹתֶ֑יהָ וְכׇל־אַנְשֵׁ֨י מִלְחַמְתָּ֥הּ יִדַּ֛מּוּ בַּיּ֥וֹם הַה֖וּא נְאֻם־יְהֹוָֽה׃
{פ}　　
לאלא הִנְנִ֤י אֵלֶ֙יךָ֙ זָד֔וֹן נְאֻם־אֲדֹנָ֥י יֱהֹוִ֖ה צְבָא֑וֹת כִּ֛י בָּ֥א יוֹמְךָ֖ עֵ֥ת פְּקַדְתִּֽיךָ׃ לבלב וְכָשַׁ֤ל זָדוֹן֙ וְנָפַ֔ל וְאֵ֥ין ל֖וֹ מֵקִ֑ים וְהִצַּ֤תִּי אֵשׁ֙ בְּעָרָ֔יו וְאָכְלָ֖ה כׇּל־סְבִיבֹתָֽיו׃
{ס}　　　
לגלג כֹּ֤ה אָמַר֙ יְהֹוָ֣ה צְבָא֔וֹת עֲשׁוּקִ֛ים בְּנֵי־יִשְׂרָאֵ֥ל וּבְנֵי־יְהוּדָ֖ה יַחְדָּ֑ו וְכׇל־שֹֽׁבֵיהֶם֙ הֶחֱזִ֣יקוּ בָ֔ם מֵאֲנ֖וּ שַׁלְּחָֽם׃ לדלד גֹּאֲלָ֣ם ׀ חָזָ֗ק יְהֹוָ֤ה צְבָאוֹת֙ שְׁמ֔וֹ רִ֥יב יָרִ֖יב אֶת־רִיבָ֑ם לְמַ֙עַן֙ הִרְגִּ֣יעַ אֶת־הָאָ֔רֶץ וְהִרְגִּ֖יז לְיֹשְׁבֵ֥י בָבֶֽל׃ להלה חֶ֥רֶב עַל־כַּשְׂדִּ֖ים נְאֻם־יְהֹוָ֑ה וְאֶל־יֹשְׁבֵ֣י בָבֶ֔ל וְאֶל־שָׂרֶ֖יהָ וְאֶל־חֲכָמֶֽיהָ׃ לולו חֶ֥רֶב אֶל־הַבַּדִּ֖ים וְנֹאָ֑לוּ חֶ֥רֶב אֶל־גִּבּוֹרֶ֖יהָ וָחָֽתּוּ׃ לזלז חֶ֜רֶב אֶל־סוּסָ֣יו וְאֶל־רִכְבּ֗וֹ וְאֶל־כׇּל־הָעֶ֛רֶב אֲשֶׁ֥ר בְּתוֹכָ֖הּ וְהָי֣וּ לְנָשִׁ֑ים חֶ֥רֶב אֶל־אוֹצְרֹתֶ֖יהָ וּבֻזָּֽזוּ׃ לחלח חֹ֥רֶב אֶל־מֵימֶ֖יהָ וְיָבֵ֑שׁוּ כִּ֣י אֶ֤רֶץ פְּסִלִים֙ הִ֔יא וּבָאֵימִ֖ים יִתְהֹלָֽלוּ׃ לטלט לָכֵ֗ן יֵשְׁב֤וּ צִיִּים֙ אֶת־אִיִּ֔ים וְיָ֥שְׁבוּ בָ֖הּ בְּנ֣וֹת יַעֲנָ֑ה וְלֹא־תֵשֵׁ֥ב עוֹד֙ לָנֶ֔צַח וְלֹ֥א תִשְׁכּ֖וֹן עַד־דּ֥וֹר וָדֹֽר׃ ממ כְּמַהְפֵּכַ֨ת אֱלֹהִ֜ים אֶת־סְדֹ֧ם וְאֶת־עֲמֹרָ֛ה וְאֶת־שְׁכֵנֶ֖יהָ נְאֻם־יְהֹוָ֑ה לֹא־יֵשֵׁ֥ב שָׁם֙ אִ֔ישׁ וְלֹא־יָג֥וּר בָּ֖הּ בֶּן־אָדָֽם׃ מאמא הִנֵּ֛ה עַ֥ם בָּ֖א מִצָּפ֑וֹן וְג֤וֹי גָּדוֹל֙ וּמְלָכִ֣ים רַבִּ֔ים יֵעֹ֖רוּ מִיַּרְכְּתֵי־אָֽרֶץ׃ מבמב קֶ֣שֶׁת וְכִידֹ֞ן יַחֲזִ֗יקוּ אַכְזָרִ֥י הֵ֙מָּה֙ וְלֹ֣א יְרַחֵ֔מוּ קוֹלָם֙ כַּיָּ֣ם יֶהֱמֶ֔ה וְעַל־סוּסִ֖ים יִרְכָּ֑בוּ עָר֗וּךְ כְּאִישׁ֙ לַמִּלְחָמָ֔ה עָלַ֖יִךְ בַּת־בָּבֶֽל׃ מגמג שָׁמַ֧ע מֶלֶךְ־בָּבֶ֛ל אֶת־שִׁמְעָ֖ם וְרָפ֣וּ יָדָ֑יו צָרָה֙ הֶחֱזִיקַ֔תְהוּ חִ֖יל כַּיּוֹלֵדָֽה׃ מדמד הִ֠נֵּ֠ה כְּאַרְיֵ֞ה יַעֲלֶ֨ה מִגְּא֣וֹן הַיַּרְדֵּן֮ אֶל־נְוֵ֣ה אֵיתָן֒ כִּֽי־אַרְגִּ֤עָה אֲרִויצֵם֙קקקרי/כתיב · (ארוצם) אֲרִיצֵם֙ מֵעָלֶ֔יהָ וּמִ֥י בָח֖וּר אֵלֶ֣יהָ אֶפְקֹ֑ד כִּ֣י מִ֤י כָמ֙וֹנִי֙ וּמִ֣י יוֹעִדֶ֔נִּי וּמִי־זֶ֣ה רֹעֶ֔ה אֲשֶׁ֥ר יַעֲמֹ֖ד לְפָנָֽי׃ מהמה לָכֵ֞ן שִׁמְע֣וּ עֲצַת־יְהֹוָ֗ה אֲשֶׁ֤ר יָעַץ֙ אֶל־בָּבֶ֔ל וּמַ֨חְשְׁבוֹתָ֔יו אֲשֶׁ֥ר חָשַׁ֖ב אֶל־אֶ֣רֶץ כַּשְׂדִּ֑ים אִם־לֹ֤א יִסְחָבוּם֙ צְעִירֵ֣י הַצֹּ֔אן אִם־לֹ֥א יַשִּׁ֛ים עֲלֵיהֶ֖ם נָוֶֽה׃ מומו מִקּוֹל֙ נִתְפְּשָׂ֣ה בָבֶ֔ל נִרְעֲשָׁ֖ה הָאָ֑רֶץ וּזְעָקָ֖ה בַּגּוֹיִ֥ם נִשְׁמָֽע׃
