מקרא על פי המסורה

ספר שמואל

פרק כג


{ס}　　　
כגאא וַיַּגִּ֥דוּ לְדָוִ֖ד לֵאמֹ֑ר הִנֵּ֤ה פְלִשְׁתִּים֙ נִלְחָמִ֣ים בִּקְעִילָ֔ה וְהֵ֖מָּה שֹׁסִ֥ים אֶת־הַגֳּרָנֽוֹת׃ בב וַיִּשְׁאַ֨ל דָּוִ֤ד בַּֽיהֹוָה֙ לֵאמֹ֔ר הַאֵלֵ֣ךְ וְהִכֵּ֔יתִי בַּפְּלִשְׁתִּ֖ים הָאֵ֑לֶּה וַיֹּ֨אמֶר יְהֹוָ֜ה אֶל־דָּוִ֗ד לֵ֚ךְ וְהִכִּ֣יתָ בַפְּלִשְׁתִּ֔ים וְהוֹשַׁעְתָּ֖ אֶת־קְעִילָֽה׃
{ס}　　　
גג וַיֹּ֨אמְר֜וּ אַנְשֵׁ֤י דָוִד֙ אֵלָ֔יו הִנֵּ֨ה אֲנַ֥חְנוּ פֹ֛ה בִּיהוּדָ֖ה יְרֵאִ֑ים וְאַף֙ כִּֽי־נֵלֵ֣ךְ קְעִלָ֔ה אֶל־מַעַרְכ֖וֹת פְּלִשְׁתִּֽים׃
{ס}　　　
דד וַיּ֨וֹסֶף ע֤וֹד דָּוִד֙ לִשְׁא֣וֹל בַּיהֹוָ֔ה וַֽיַּעֲנֵ֖הוּ יְהֹוָ֑ה וַיֹּ֗אמֶר ק֚וּם רֵ֣ד קְעִילָ֔ה כִּי־אֲנִ֥י נֹתֵ֛ן אֶת־פְּלִשְׁתִּ֖ים בְּיָדֶֽךָ׃ הה וַיֵּ֣לֶךְ דָּוִד֩ וַאֲנָשָׁ֨וקקקרי/כתיב · וַאֲנָשָׁ֨ו · וַאֲנָשָׁ֨יו קְעִילָ֜ה וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֗ים וַיִּנְהַג֙ אֶת־מִקְנֵיהֶ֔ם וַיַּ֥ךְ בָּהֶ֖ם מַכָּ֣ה גְדוֹלָ֑ה וַיֹּ֣שַׁע דָּוִ֔ד אֵ֖ת יֹשְׁבֵ֥י קְעִילָֽה׃
{ס}　　　
וו וַיְהִ֗י בִּ֠בְרֹ֠חַ אֶבְיָתָ֧ר בֶּן־אֲחִימֶ֛לֶךְ אֶל־דָּוִ֖ד קְעִילָ֑ה אֵפ֖וֹד יָרַ֥ד בְּיָדֽוֹ׃ זז וַיֻּגַּ֣ד לְשָׁא֔וּל כִּי־בָ֥א דָוִ֖ד קְעִילָ֑ה וַיֹּ֣אמֶר שָׁא֗וּל נִכַּ֨ר אֹת֤וֹ אֱלֹהִים֙ בְּיָדִ֔י כִּ֚י נִסְגַּ֣ר לָב֔וֹא בְּעִ֖יר דְּלָתַ֥יִם וּבְרִֽיחַ׃ חח וַיְשַׁמַּ֥ע שָׁא֛וּל אֶת־כׇּל־הָעָ֖ם לַמִּלְחָמָ֑ה לָרֶ֣דֶת קְעִילָ֔ה לָצ֥וּר אֶל־דָּוִ֖ד וְאֶל־אֲנָשָֽׁיו׃ טט וַיֵּ֣דַע דָּוִ֔ד כִּ֣י עָלָ֔יו שָׁא֖וּל מַחֲרִ֣ישׁ הָרָעָ֑ה וַיֹּ֙אמֶר֙ אֶל־אֶבְיָתָ֣ר הַכֹּהֵ֔ן הַגִּ֖ישָׁה הָאֵפֽוֹד׃
{ס}　　　
יי וַיֹּ֘אמֶר֮ דָּוִד֒ יְהֹוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל שָׁמֹ֤עַ שָׁמַע֙ עַבְדְּךָ֔ כִּֽי־מְבַקֵּ֥שׁ שָׁא֖וּל לָב֣וֹא אֶל־קְעִילָ֑ה לְשַׁחֵ֥ת לָעִ֖יר בַּעֲבוּרִֽי׃ יאיא הֲיַסְגִּרֻ֣נִי בַעֲלֵי֩ קְעִילָ֨ה בְיָד֜וֹ הֲיֵרֵ֣ד שָׁא֗וּל כַּֽאֲשֶׁר֙ שָׁמַ֣ע עַבְדֶּ֔ךָ יְהֹוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַגֶּד־נָ֖א לְעַבְדֶּ֑ךָ וַיֹּ֥אמֶר יְהֹוָ֖ה יֵרֵֽד׃
{ס}　　　
יביב וַיֹּ֣אמֶר דָּוִ֔ד הֲיַסְגִּ֜רוּ בַּעֲלֵ֧י קְעִילָ֛ה אֹתִ֥י וְאֶת־אֲנָשַׁ֖י בְּיַד־שָׁא֑וּל וַיֹּ֥אמֶר יְהֹוָ֖ה יַסְגִּֽירוּ׃
{ס}　　　
יגיג וַיָּ֩קׇם֩ דָּוִ֨ד וַאֲנָשָׁ֜יו כְּשֵׁשׁ־מֵא֣וֹת אִ֗ישׁ וַיֵּֽצְאוּ֙ מִקְּעִלָ֔ה וַיִּֽתְהַלְּכ֖וּ בַּאֲשֶׁ֣ר יִתְהַלָּ֑כוּ וּלְשָׁא֣וּל הֻגַּ֗ד כִּֽי־נִמְלַ֤ט דָּוִד֙ מִקְּעִילָ֔ה וַיֶּחְדַּ֖ל לָצֵֽאת׃ ידיד וַיֵּ֨שֶׁב דָּוִ֤ד בַּמִּדְבָּר֙ בַּמְּצָד֔וֹת וַיֵּ֥שֶׁב בָּהָ֖ר בְּמִדְבַּר־זִ֑יף וַיְבַקְשֵׁ֤הוּ שָׁאוּל֙ כׇּל־הַיָּמִ֔ים וְלֹֽא־נְתָנ֥וֹ אֱלֹהִ֖ים בְּיָדֽוֹ׃ טוטו וַיַּ֣רְא דָּוִ֔ד כִּֽי־יָצָ֥א שָׁא֖וּל לְבַקֵּ֣שׁ אֶת־נַפְשׁ֑וֹ וְדָוִ֥ד בְּמִדְבַּר־זִ֖יף בַּחֹֽרְשָׁה׃
{ס}　　　
טזטז וַיָּ֙קׇם֙ יְהוֹנָתָ֣ן בֶּן־שָׁא֔וּל וַיֵּ֥לֶךְ אֶל־דָּוִ֖ד חֹ֑רְשָׁה וַיְחַזֵּ֥ק אֶת־יָד֖וֹ בֵּאלֹהִֽים׃ יזיז וַיֹּ֨אמֶר אֵלָ֜יו אַל־תִּירָ֗א כִּ֠י לֹ֤א תִֽמְצָאֲךָ֙ יַ֚ד שָׁא֣וּל אָבִ֔י וְאַתָּה֙ תִּמְלֹ֣ךְ עַל־יִשְׂרָאֵ֔ל וְאָנֹכִ֖י אֶהְיֶה־לְּךָ֣נאֶהְיֶה־לְּךָ֣=א! (אין געיה) וכן במג"האמנם לפי חילופי בן־אשר אין געיה, אך משתמע שגם הלמ"ד איננה דגושה (אֶהְיֶה־לְךָ֣), וראו הערת ברויאר במהדורתו הראשונה.ל,ק=אֶֽהְיֶה־לְּךָ֣ (געיה ודגש) וכן בחילופי בן־נפתלי ובדפוסים (וכך הכריע ברויאר נגד הכתר). לְמִשְׁנֶ֑ה וְגַם־שָׁא֥וּל אָבִ֖י יֹדֵ֥עַ כֵּֽן׃ יחיח וַיִּכְרְת֧וּ שְׁנֵיהֶ֛ם בְּרִ֖ית לִפְנֵ֣י יְהֹוָ֑ה וַיֵּ֤שֶׁב דָּוִד֙ בַּחֹ֔רְשָׁה וִיהוֹנָתָ֖ן הָלַ֥ךְ לְבֵיתֽוֹ׃
{ס}　　　
יטיט וַיַּעֲל֤וּ זִפִים֙ אֶל־שָׁא֔וּל הַגִּבְעָ֖תָה לֵאמֹ֑ר הֲל֣וֹא דָ֠וִ֠ד מִסְתַּתֵּ֨ר עִמָּ֤נוּ בַמְּצָדוֹת֙ בַּחֹ֔רְשָׁה בְּגִבְעַת֙ הַחֲכִילָ֔ה אֲשֶׁ֖ר מִימִ֥ין הַיְשִׁימֽוֹן׃ ככ וְ֠עַתָּ֠ה לְכׇל־אַוַּ֨ת נַפְשְׁךָ֥ הַמֶּ֛לֶךְ לָרֶ֖דֶת רֵ֑ד וְלָ֥נוּ הַסְגִּיר֖וֹ בְּיַ֥ד הַמֶּֽלֶךְ׃ כאכא וַיֹּ֣אמֶר שָׁא֔וּל בְּרוּכִ֥ים אַתֶּ֖ם לַֽיהֹוָ֑ה כִּ֥י חֲמַלְתֶּ֖ם עָלָֽי׃ כבכב לְכוּ־נָ֞א הָכִ֣ינוּ ע֗וֹד וּדְע֤וּ וּרְאוּ֙ אֶת־מְקוֹמוֹ֙ אֲשֶׁ֣ר תִּֽהְיֶ֣ה רַגְל֔וֹ מִ֥י רָאָ֖הוּ שָׁ֑ם כִּ֚י אָמַ֣ר אֵלַ֔י עָר֥וֹם יַעְרִ֖ם הֽוּא׃ כגכג וּרְא֣וּ וּדְע֗וּ מִכֹּ֤ל הַמַּֽחֲבֹאִים֙ אֲשֶׁ֣ר יִתְחַבֵּ֣א שָׁ֔ם וְשַׁבְתֶּ֤ם אֵלַי֙ אֶל־נָכ֔וֹן וְהָלַכְתִּ֖י אִתְּכֶ֑ם וְהָיָה֙ אִם־יֶשְׁנ֣וֹ בָאָ֔רֶץ וְחִפַּשְׂתִּ֣י אֹת֔וֹ בְּכֹ֖ל אַלְפֵ֥י יְהוּדָֽה׃ כדכד וַיָּק֛וּמוּ וַיֵּלְכ֥וּ זִ֖יפָה לִפְנֵ֣י שָׁא֑וּל וְדָוִ֨ד וַאֲנָשָׁ֜יו בְּמִדְבַּ֤ר מָעוֹן֙ בָּעֲרָבָ֔ה אֶ֖ל יְמִ֥ין הַיְשִׁימֽוֹן׃ כהכה וַיֵּ֨לֶךְ שָׁא֣וּל וַאֲנָשָׁיו֮ לְבַקֵּשׁ֒ וַיַּגִּ֣דוּ לְדָוִ֔ד וַיֵּ֣רֶד הַסֶּ֔לַע וַיֵּ֖שֶׁב בְּמִדְבַּ֣ר מָע֑וֹן וַיִּשְׁמַ֣ע שָׁא֔וּל וַיִּרְדֹּ֥ף אַחֲרֵֽי־דָוִ֖ד מִדְבַּ֥ר מָעֽוֹן׃ כוכו וַיֵּ֨לֶךְ שָׁא֜וּל מִצַּ֤ד הָהָר֙ מִזֶּ֔ה וְדָוִ֧ד וַאֲנָשָׁ֛יו מִצַּ֥ד הָהָ֖ר מִזֶּ֑ה וַיְהִ֨י דָוִ֜ד נֶחְפָּ֤ז לָלֶ֙כֶת֙ מִפְּנֵ֣י שָׁא֔וּל וְשָׁא֣וּל וַאֲנָשָׁ֗יו עֹ֥טְרִ֛יםנעֹ֥טְרִ֛ים=א (מרכא בתיבת תביר), וכן אצל ברויאר ומג"הממ=עֹֽטְרִ֛ים (כנראה הכריעו כך בגלל שהקו ישר, אבל אין בכך מספיק לומר שאיננו מרכא) אֶל־דָּוִ֥ד וְאֶל־אֲנָשָׁ֖יו לְתׇפְשָֽׂם׃ כזכז וּמַלְאָ֣ךְ בָּ֔א אֶל־שָׁא֖וּל לֵאמֹ֑ר מַהֲרָ֣ה וְלֵ֔כָה כִּֽי־פָשְׁט֥וּ פְלִשְׁתִּ֖ים עַל־הָאָֽרֶץ׃ כחכח וַיָּ֣שׇׁב שָׁא֗וּל מִרְדֹף֙ אַחֲרֵ֣י דָוִ֔ד וַיֵּ֖לֶךְ לִקְרַ֣את פְּלִשְׁתִּ֑ים עַל־כֵּ֗ן קָֽרְאוּ֙ לַמָּק֣וֹם הַה֔וּא סֶ֖לַע הַֽמַּחְלְקֽוֹת׃ כטכט וַיַּ֥עַל דָּוִ֖ד מִשָּׁ֑ם וַיֵּ֖שֶׁב בִּמְצָד֥וֹת עֵֽין־גֶּֽדִי׃
