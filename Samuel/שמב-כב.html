וּמְפַלְטִי־לִֽי׃ גג אֱלֹהֵ֥י צוּרִ֖י אֶחֱסֶה־בּ֑וֹמָגִנִּ֞י וְקֶ֣רֶן יִשְׁעִ֗י
מִשְׂגַּבִּי֙ וּמְנוּסִ֔ימֹשִׁעִ֕י מֵחָמָ֖ס תֹּשִׁעֵֽנִי׃
דד מְהֻלָּ֖ל אֶקְרָ֣א יְהֹוָ֑הוּמֵאֹיְבַ֖י אִוָּשֵֽׁעַ׃
הה כִּ֥י אֲפָפֻ֖נִי מִשְׁבְּרֵי־מָ֑וֶתנַחֲלֵ֥י בְלִיַּ֖עַל יְבַעֲתֻֽנִי׃
וו חֶבְלֵ֥י שְׁא֖וֹל סַבֻּ֑נִיקִדְּמֻ֖נִי מֹ֥קְשֵׁי מָֽוֶתנמֹ֥קְשֵׁי מָֽוֶת=א,ל ("מֹ֥קְשֵׁי" בטעם במרכא), וכן ברויאר ומג"ה (וסימנים)דפוסים=מֹֽקְשֵׁי־מָֽוֶת ("מֹֽקְשֵׁי־" בגעיה ומוקפת) וכן בקורן ומכון ממרא (!)׃
זז בַּצַּר־לִי֙ אֶקְרָ֣א יְהֹוָ֔ה וְאֶל־אֱלֹהַ֖י אֶקְרָ֑אוַיִּשְׁמַ֤ע מֵהֵֽיכָלוֹ֙נמֵהֵֽיכָלוֹ֙=ל,ק ובדפוסיםא!=מֵהֵֽיכָלו֙ (חסרה נקודת החולם)הערת ייבין קוֹלִ֔י וְשַׁוְעָתִ֖י בְּאׇזְנָֽיו׃
חח (ותגעש)קקקרי/כתיב · (ותגעש) וַיִּתְגָּעַ֤שׁ וַתִּרְעַשׁ֙ הָאָ֔רֶץמוֹסְד֥וֹת הַשָּׁמַ֖יִם יִרְגָּ֑זוּ וַיִּֽתְגָּעֲשׁ֖וּ כִּי־חָ֥רָה לֽוֹ׃
טט עָלָ֤ה עָשָׁן֙ בְּאַפּ֔וֹ וְאֵ֥שׁ מִפִּ֖יו תֹּאכֵ֑לגֶּחָלִ֖ים בָּעֲר֥וּ מִמֶּֽנּוּ׃
יי וַיֵּ֥ט שָׁמַ֖יִם וַיֵּרַ֑דוַעֲרָפֶ֖ל תַּ֥חַת רַגְלָֽיו׃
יאיא וַיִּרְכַּ֥ב עַל־כְּר֖וּב וַיָּעֹ֑ףוַיֵּרָ֖א עַל־כַּנְפֵי־רֽוּחַ׃
יביב וַיָּ֥שֶׁת חֹ֛שֶׁךְ סְבִיבֹתָ֖יו סֻכּ֑וֹתחַֽשְׁﬞרַת־מַ֖יִםנחַֽשְׁﬞרַת־מַ֖יִם · א=חַֽשֲׁרַת־מַ֖יִם (חטף) עָבֵ֥י שְׁחָקִֽים׃
יגיג מִנֹּ֖גַהּ נֶגְדּ֑וֹ בָּעֲר֖וּ גַּחֲלֵי־אֵֽשׁ׃ ידיד יַרְעֵ֥ם מִן־שָׁמַ֖יִם יְהֹוָ֑ה
וְעֶלְי֖וֹן יִתֵּ֥ן קוֹלֽוֹ׃ טוטו וַיִּשְׁלַ֥ח חִצִּ֖ים וַיְפִיצֵ֑ם בָּרָ֖ק (ויהמם)קקקרי/כתיב · (ויהמם) וַיָּהֹֽם׃
טזטז וַיֵּֽרָאוּ֙ אֲפִ֣קֵי יָ֔ם יִגָּל֖וּ מֹסְד֣וֹת תֵּבֵ֑לבְּגַעֲרַ֣ת יְהֹוָ֔ה מִנִּשְׁמַ֖ת ר֥וּחַ אַפּֽוֹ׃
יזיז יִשְׁלַ֥ח מִמָּר֖וֹם יִקָּחֵ֑נִייַֽמְשֵׁ֖נִי מִמַּ֥יִם רַבִּֽים׃
יחיח יַצִּילֵ֕נִי מֵאֹיְבִ֖י עָ֑זמִשֹּׂ֣נְאַ֔י כִּ֥י אָמְצ֖וּ מִמֶּֽנִּי׃
יטיט יְקַדְּמֻ֖נִי בְּי֣וֹם אֵידִ֑י וַיְהִ֧י יְהֹוָ֛ה מִשְׁעָ֖ן לִֽי׃ ככ וַיֹּצֵ֥א לַמֶּרְחָ֖ב אֹתִ֑י
יְחַלְּצֵ֖נִי כִּי־חָ֥פֵֽץ בִּֽי׃ כאכא יִגְמְלֵ֥נִי יְהֹוָ֖ה כְּצִדְקָתִ֑י
כְּבֹ֥ר יָדַ֖י יָשִׁ֥יב לִֽי׃ כבכב כִּ֥י שָׁמַ֖רְתִּי דַּרְכֵ֣י יְהֹוָ֑ה
וְלֹ֥א רָשַׁ֖עְתִּי מֵאֱלֹהָֽי׃ כגכג כִּ֥י כׇל־מִשְׁפָּטָ֖וקקקרי/כתיב · מִשְׁפָּטָ֖ו · מִשְׁפָּטָ֖יו לְנֶגְדִּ֑י וְחֻקֹּתָ֖יו לֹא־אָס֥וּר מִמֶּֽנָּה׃
כדכד וָאֶהְיֶ֥ה תָמִ֖ים ל֑וֹוָאֶשְׁתַּמְּרָ֖ה מֵעֲוֺנִֽי׃
כהכה וַיָּ֧שֶׁב יְהֹוָ֛ה לִ֖י כְּצִדְקָתִ֑יכְּבֹרִ֖י לְנֶ֥גֶד עֵינָֽיו׃
כוכו עִם־חָסִ֖יד תִּתְחַסָּ֑דעִם־גִּבּ֥וֹר תָּמִ֖ים תִּתַּמָּֽם׃
כזכז עִם־נָבָ֖ר תִּתָּבָ֑רוְעִם־עִקֵּ֖שׁ תִּתַּפָּֽל׃
כחכח וְאֶת־עַ֥ם עָנִ֖י תּוֹשִׁ֑יעַוְעֵינֶ֥יךָ עַל־רָמִ֖ים תַּשְׁפִּֽיל׃
כטכט כִּֽי־אַתָּ֥ה נֵירִ֖י יְהֹוָ֑הוַיהֹוָ֖ה יַגִּ֥יהַּ חׇשְׁכִּֽי׃
לל כִּ֥י בְכָ֖ה אָר֣וּץ גְּד֑וּדבֵּאלֹהַ֖י אֲדַלֶּג־שֽׁוּר׃
לאלא הָאֵ֖ל תָּמִ֣ים דַּרְכּ֑וֹ אִמְרַ֤ת יְהֹוָה֙ צְרוּפָ֔המָגֵ֣ן ה֔וּא לְכֹ֖ל הַחֹסִ֥ים בּֽוֹ׃
לבלב כִּ֥י מִי־אֵ֖ל מִבַּלְעֲדֵ֣י יְהֹוָ֑הוּמִ֥י צ֖וּר מִֽבַּלְעֲדֵ֥י אֱלֹהֵֽינוּ׃
לגלג הָאֵ֥ל מָעוּזִּ֖י חָ֑יִלוַיַּתֵּ֥ר תָּמִ֖ים דַּרְכִּֽויקקקרי/כתיב · (דרכו) דַּרְכִּֽי׃
לדלד מְשַׁוֶּ֥ה (רגליו)קקקרי/כתיב · (רגליו) רַגְלַ֖י כָּאַיָּל֑וֹתוְעַ֥ל בָּמֹתַ֖י יַעֲמִדֵֽנִי׃
להלה מְלַמֵּ֥ד יָדַ֖י לַמִּלְחָמָ֑הוְנִחַ֥ת קֶשֶׁת־נְחוּשָׁ֖ה זְרֹעֹתָֽי׃
לולו וַתִּתֶּן־לִ֖י מָגֵ֣ן יִשְׁעֶ֑ךָ וַעֲנֹתְךָ֖ תַּרְבֵּֽנִי׃ לזלז תַּרְחִ֥יב צַעֲדִ֖י תַּחְתֵּ֑נִי וְלֹ֥א מָעֲד֖וּ קַרְסֻלָּֽי׃
לחלח אֶרְדְּפָ֥ה אֹיְבַ֖י וָאַשְׁמִידֵ֑םוְלֹ֥א אָשׁ֖וּב עַד־כַּלּוֹתָֽם׃
לטלט וָאֲכַלֵּ֥ם וָאֶמְחָצֵ֖ם וְלֹ֣א יְקוּמ֑וּןוַֽיִּפְּל֖וּ תַּ֥חַת רַגְלָֽי׃
ממ וַתַּזְרֵ֥נִינוַתַּזְרֵ֥נִי=א?,ל,ק ובדפוסים (אין דגש בזי"ן) וכן ברויאר (כתר ירושלים) וסימנים ומ"מ (התיבה נבדקה בתצלום באיכות גבוהה של כתי"ק ע"י החוקרת Elvira Martín-Contreras ותודה לה); וכך גם אצל המדקדקים ר"י אבן חיוג' ור"י אבן ג'אנח.א?=וַתַּזְּרֵ֥נִי (זי"ן דגושה?) וכן בתיגאן, וכך אצל ברויאר (מוסד הרב קוק, חורב) ומג"ה. ברויאר ציין את הדגש כספק בהערותיו; עיון בכתב־היד בהגדלה מראה שהדגש קיים בוודאי אך הוא מטושטש במקום ששאר הטקסט בהיר, כנראה בגלל ניסיון מחיקה. חַ֖יִל לַמִּלְחָמָ֑התַּכְרִ֥יעַ קָמַ֖י תַּחְתֵּֽנִי׃
מאמא וְאֹ֣יְבַ֔י תַּ֥תָּה לִּ֖י עֹ֑רֶף מְשַׂנְאַ֖י וָאַצְמִיתֵֽם׃ מבמב יִשְׁע֖וּ וְאֵ֣ין מֹשִׁ֑יעַ אֶל־יְהֹוָ֖ה וְלֹ֥א עָנָֽם׃
מגמג וְאֶשְׁחָקֵ֖ם כַּעֲפַר־אָ֑רֶץכְּטִיט־חוּצ֥וֹת אֲדִקֵּ֖ם אֶרְקָעֵֽם׃
מדמד וַֽתְּפַלְּטֵ֔נִי מֵרִיבֵ֖י עַמִּ֑י תִּשְׁמְרֵ֙נִי֙ לְרֹ֣אשׁ גּוֹיִ֔םעַ֥ם לֹא־יָדַ֖עְתִּי יַעַבְדֻֽנִי׃
מהמה בְּנֵ֥י נֵכָ֖ר יִתְכַּחֲשׁוּ־לִ֑י לִשְׁמ֥וֹעַ אֹ֖זֶן יִשָּׁ֥מְעוּ לִֽי׃ מומו בְּנֵ֥י נֵכָ֖ר יִבֹּ֑לוּ וְיַחְגְּר֖וּ מִמִּסְגְּרוֹתָֽם׃
מזמז חַי־יְהֹוָ֖ה וּבָר֣וּךְ צוּרִ֑יוְיָרֻ֕ם אֱלֹהֵ֖י צ֥וּר יִשְׁעִֽי׃
מחמח הָאֵ֕ל הַנֹּתֵ֥ן נְקָמֹ֖ת לִ֑יוּמֹרִ֥יד עַמִּ֖ים תַּחְתֵּֽנִי׃
מטמט וּמוֹצִיאִ֖י מֵאֹֽיְבָ֑י וּמִקָּמַי֙ תְּר֣וֹמְמֵ֔נִימֵאִ֥ישׁ חֲמָסִ֖ים תַּצִּילֵֽנִי׃
