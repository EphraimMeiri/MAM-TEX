מקרא על פי המסורה

ספר שמואל

פרק טו


{פ}　　
טואא וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־שָׁא֔וּל אֹתִ֨י שָׁלַ֤ח יְהֹוָה֙ נלגבי קמץ קטן ראו בדוידזון וכך גם בקורן לְמֶ֔לֶךְ עַל־עַמּ֖וֹ עַל־יִשְׂרָאֵ֑ל וְעַתָּ֣ה שְׁמַ֔ע לְק֖וֹל דִּבְרֵ֥י יְהֹוָֽה׃
{ס}　　　
בב כֹּ֤ה אָמַר֙ יְהֹוָ֣ה צְבָא֔וֹת פָּקַ֕דְתִּי אֵ֛ת אֲשֶׁר־עָשָׂ֥ה עֲמָלֵ֖ק לְיִשְׂרָאֵ֑ל אֲשֶׁר־שָׂ֥ם לוֹ֙ בַּדֶּ֔רֶךְ בַּעֲלֹת֖וֹ מִמִּצְרָֽיִם׃ גג עַתָּה֩ לֵ֨ךְ וְהִכִּיתָ֜ה אֶת־עֲמָלֵ֗ק וְהַֽחֲרַמְתֶּם֙ אֶת־כׇּל־אֲשֶׁר־ל֔וֹ וְלֹ֥א תַחְמֹ֖ל עָלָ֑יו וְהֵמַתָּ֞ה מֵאִ֣ישׁ עַד־אִשָּׁ֗ה מֵֽעֹלֵל֙ וְעַד־יוֹנֵ֔ק מִשּׁ֣וֹר וְעַד־שֶׂ֔ה מִגָּמָ֖ל וְעַד־חֲמֽוֹר׃
{ס}　　　
דד וַיְשַׁמַּ֤ע שָׁאוּל֙ אֶת־הָעָ֔ם וַֽיִּפְקְדֵם֙ בַּטְּלָאִ֔ים מָאתַ֥יִם אֶ֖לֶף רַגְלִ֑י וַעֲשֶׂ֥רֶת אֲלָפִ֖ים אֶת־אִ֥ישׁ יְהוּדָֽה׃ הה וַיָּבֹ֥א שָׁא֖וּל עַד־עִ֣יר עֲמָלֵ֑ק וַיָּ֖רֶב בַּנָּֽחַל׃ וו וַיֹּ֣אמֶר שָׁא֣וּל אֶֽל־הַקֵּינִ֡י לְכוּ֩ סֻּ֨רוּ רְד֜וּ מִתּ֣וֹךְ עֲמָלֵקִ֗י פֶּן־אֹֽסִפְךָ֙ עִמּ֔וֹ וְאַתָּ֞ה עָשִׂ֤יתָֽה חֶ֙סֶד֙ עִם־כׇּל־בְּנֵ֣י יִשְׂרָאֵ֔ל בַּעֲלוֹתָ֖ם מִמִּצְרָ֑יִם וַיָּ֥סַר קֵינִ֖י מִתּ֥וֹךְ עֲמָלֵֽק׃ זז וַיַּ֥ךְ שָׁא֖וּל אֶת־עֲמָלֵ֑ק מֵֽחֲוִילָה֙ בּוֹאֲךָ֣ שׁ֔וּר אֲשֶׁ֖ר עַל־פְּנֵ֥י מִצְרָֽיִם׃ חח וַיִּתְפֹּ֛שׂ אֶת־אֲגַ֥ג מֶלֶךְ־עֲמָלֵ֖ק חָ֑י וְאֶת־כׇּל־הָעָ֖ם הֶחֱרִ֥ים לְפִי־חָֽרֶב׃ טט וַיַּחְמֹל֩ שָׁא֨וּל וְהָעָ֜ם עַל־אֲגָ֗ג וְעַל־מֵיטַ֣ב הַצֹּאן֩ וְהַבָּקָ֨ר וְהַמִּשְׁנִ֤ים וְעַל־הַכָּרִים֙ וְעַל־כׇּל־הַטּ֔וֹב וְלֹ֥א אָב֖וּ הַחֲרִימָ֑ם וְכׇל־הַמְּלָאכָ֛ה נְמִבְזָ֥ה וְנָמֵ֖ס אֹתָ֥הּ הֶחֱרִֽימוּ׃
{פ}　　
יי וַֽיְהִי֙ דְּבַר־יְהֹוָ֔ה אֶל־שְׁמוּאֵ֖ל לֵאמֹֽר׃ יאיא נִחַ֗מְתִּי כִּֽי־הִמְלַ֤כְתִּי אֶת־שָׁאוּל֙ לְמֶ֔לֶךְ כִּי־שָׁב֙ מֵֽאַחֲרַ֔י וְאֶת־דְּבָרַ֖י לֹ֣א הֵקִ֑ים וַיִּ֙חַר֙ לִשְׁמוּאֵ֔ל וַיִּזְעַ֥ק אֶל־יְהֹוָ֖ה כׇּל־הַלָּֽיְלָהנכׇּל־הַלָּֽיְלָה=לא!=כׇּל־הַלָּֽילָה (חסר שווא באות יו"ד)׃ יביב וַיַּשְׁכֵּ֧ם שְׁמוּאֵ֛ל לִקְרַ֥את שָׁא֖וּל בַּבֹּ֑קֶר וַיֻּגַּ֨ד לִשְׁמוּאֵ֜ל לֵאמֹ֗ר בָּֽא־שָׁא֤וּל הַכַּרְמֶ֙לָה֙ וְהִנֵּ֨ה מַצִּ֥יב לוֹ֙ יָ֔ד וַיִּסֹּב֙ וַֽיַּעֲבֹ֔ר וַיֵּ֖רֶד הַגִּלְגָּֽל׃ יגיג וַיָּבֹ֥א שְׁמוּאֵ֖ל אֶל־שָׁא֑וּל וַיֹּ֧אמֶר ל֣וֹ שָׁא֗וּל בָּר֤וּךְ אַתָּה֙ לַֽיהֹוָ֔ה הֲקִימֹ֖תִי אֶת־דְּבַ֥ר יְהֹוָֽה׃ ידיד וַיֹּ֣אמֶר שְׁמוּאֵ֔ל וּמֶ֛ה קֽוֹל־הַצֹּ֥אן הַזֶּ֖ה בְּאׇזְנָ֑י וְק֣וֹל הַבָּקָ֔ר אֲשֶׁ֥ר אָנֹכִ֖י שֹׁמֵֽעַ׃ טוטו וַיֹּ֨אמֶר שָׁא֜וּל מֵעֲמָלֵקִ֣י הֱבִיא֗וּם אֲשֶׁ֨ר חָמַ֤ל הָעָם֙ עַל־מֵיטַ֤ב הַצֹּאן֙ וְהַבָּקָ֔ר לְמַ֥עַן זְבֹ֖חַ לַיהֹוָ֣ה אֱלֹהֶ֑יךָ וְאֶת־הַיּוֹתֵ֖ר הֶחֱרַֽמְנוּ׃
{פ}　　
טזטז וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־שָׁא֔וּל הֶ֚רֶף וְאַגִּ֣ידָה לְּךָ֔ אֵת֩ אֲשֶׁ֨ר דִּבֶּ֧ר יְהֹוָ֛ה אֵלַ֖י הַלָּ֑יְלָה (ויאמרו)קקקרי/כתיב · (ויאמרו) וַיֹּ֥אמֶר ל֖וֹ דַּבֵּֽר׃
{ס}　　　
יזיז וַיֹּ֣אמֶר שְׁמוּאֵ֔ל הֲל֗וֹא אִם־קָטֹ֤ן אַתָּה֙ בְּעֵינֶ֔יךָ רֹ֛אשׁ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל אָ֑תָּה וַיִּמְשָׁחֲךָ֧ יְהֹוָ֛ה לְמֶ֖לֶךְ עַל־יִשְׂרָאֵֽל׃ יחיח וַיִּשְׁלָחֲךָ֥ יְהֹוָ֖ה בְּדָ֑רֶךְ וַיֹּ֗אמֶר לֵ֣ךְ וְהַחֲרַמְתָּ֞ה אֶת־הַֽחַטָּאִים֙ אֶת־עֲמָלֵ֔ק וְנִלְחַמְתָּ֣ ב֔וֹ עַ֥ד כַּלּוֹתָ֖ם אֹתָֽם׃ יטיט וְלָ֥מָּה לֹֽא־שָׁמַ֖עְתָּ בְּק֣וֹל יְהֹוָ֑ה וַתַּ֙עַט֙ אֶל־הַשָּׁלָ֔ל וַתַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֥י יְהֹוָֽה׃
{ס}　　　
ככ וַיֹּ֨אמֶר שָׁא֜וּל אֶל־שְׁמוּאֵ֗ל אֲשֶׁ֤ר שָׁמַ֙עְתִּי֙ בְּק֣וֹל יְהֹוָ֔ה וָאֵלֵ֕ךְ בַּדֶּ֖רֶךְ אֲשֶׁר־שְׁלָחַ֣נִי יְהֹוָ֑ה וָאָבִ֗יא אֶת־אֲגַג֙ מֶ֣לֶךְ עֲמָלֵ֔ק וְאֶת־עֲמָלֵ֖ק הֶחֱרַֽמְתִּי׃ כאכא וַיִּקַּ֨ח הָעָ֧ם מֵהַשָּׁלָ֛ל צֹ֥אן וּבָקָ֖ר רֵאשִׁ֣ית הַחֵ֑רֶם לִזְבֹּ֛חַ לַיהֹוָ֥ה אֱלֹהֶ֖יךָ בַּגִּלְגָּֽל׃
{ס}　　　
כבכב וַיֹּ֣אמֶר שְׁמוּאֵ֗ל הַחֵ֤פֶץ לַֽיהֹוָה֙ בְּעֹל֣וֹת וּזְבָחִ֔ים כִּשְׁמֹ֖עַ בְּק֣וֹל יְהֹוָ֑ה הִנֵּ֤ה שְׁמֹ֙עַ֙ מִזֶּ֣בַח ט֔וֹב לְהַקְשִׁ֖יב מֵחֵ֥לֶב אֵילִֽים׃ כגכג כִּ֤י חַטַּאת־קֶ֙סֶם֙ מֶ֔רִי וְאָ֥וֶן וּתְרָפִ֖ים הַפְצַ֑ר יַ֗עַן מָאַ֙סְתָּ֙ אֶת־דְּבַ֣ר יְהֹוָ֔ה וַיִּמְאָסְךָ֖ מִמֶּֽלֶךְ׃
{ס}　　　
כדכד וַיֹּ֨אמֶר שָׁא֤וּל אֶל־שְׁמוּאֵל֙ חָטָ֔אתִי כִּֽי־עָבַ֥רְתִּי אֶת־פִּֽי־יְהֹוָ֖ה וְאֶת־דְּבָרֶ֑יךָ כִּ֤י יָרֵ֙אתִי֙ אֶת־הָעָ֔ם וָאֶשְׁמַ֖ע בְּקוֹלָֽם׃ כהכה וְעַתָּ֕ה שָׂ֥א נָ֖א אֶת־חַטָּאתִ֑י וְשׁ֣וּב עִמִּ֔י וְאֶֽשְׁתַּחֲוֶ֖ה לַיהֹוָֽה׃ כוכו וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־שָׁא֔וּל לֹ֥א אָשׁ֖וּב עִמָּ֑ךְ כִּ֤י מָאַ֙סְתָּה֙ אֶת־דְּבַ֣ר יְהֹוָ֔ה וַיִּמְאָסְךָ֣ יְהֹוָ֔ה מִֽהְי֥וֹת מֶ֖לֶךְ עַל־יִשְׂרָאֵֽל׃
{ס}　　　
כזכז וַיִּסֹּ֥ב שְׁמוּאֵ֖ל לָלֶ֑כֶת וַיַּחֲזֵ֥ק בִּכְנַף־מְעִיל֖וֹ וַיִּקָּרַֽע׃
{ס}　　　
כחכח וַיֹּ֤אמֶר אֵלָיו֙ שְׁמוּאֵ֔ל קָרַ֨ע יְהֹוָ֜ה אֶֽת־מַמְלְכ֧וּת יִשְׂרָאֵ֛ל מֵעָלֶ֖יךָ הַיּ֑וֹם וּנְתָנָ֕הּ לְרֵעֲךָ֖ הַטּ֥וֹב מִמֶּֽךָּ׃
{ס}　　　
כטכט וְגַם֙ נֵ֣צַח יִשְׂרָאֵ֔ל לֹ֥א יְשַׁקֵּ֖ר וְלֹ֣א יִנָּחֵ֑ם כִּ֣י לֹ֥א אָדָ֛ם ה֖וּא לְהִנָּחֵֽם׃ לל וַיֹּ֣אמֶר חָטָ֔אתִי עַתָּ֗ה כַּבְּדֵ֥נִי נָ֛א נֶ֥גֶד זִקְנֵֽי־עַמִּ֖י וְנֶ֣גֶד יִשְׂרָאֵ֑ל וְשׁ֣וּב עִמִּ֔י וְהִֽשְׁתַּחֲוֵ֖יתִי לַיהֹוָ֥ה אֱלֹהֶֽיךָ׃ לאלא וַיָּ֥שׇׁב שְׁמוּאֵ֖ל אַחֲרֵ֣י שָׁא֑וּל וַיִּשְׁתַּ֥חוּ שָׁא֖וּל לַיהֹוָֽה׃
{ס}　　　
לבלב וַיֹּ֣אמֶר שְׁמוּאֵ֗ל הַגִּ֤ישׁוּ אֵלַי֙ אֶת־אֲגַג֙ מֶ֣לֶךְ עֲמָלֵ֔ק וַיֵּ֣לֶךְ אֵלָ֔יו אֲגַ֖ג מַעֲדַנֹּ֑ת וַיֹּ֣אמֶר אֲגָ֔ג אָכֵ֖ן סָ֥ר מַר־הַמָּֽוֶת׃
{ס}　　　
לגלג וַיֹּ֣אמֶר שְׁמוּאֵ֔ל כַּאֲשֶׁ֨ר שִׁכְּלָ֤ה נָשִׁים֙ חַרְבֶּ֔ךָ כֵּן־תִּשְׁכַּ֥ל מִנָּשִׁ֖ים אִמֶּ֑ךָ וַיְשַׁסֵּ֨ף שְׁמוּאֵ֧ל אֶת־אֲגָ֛ג לִפְנֵ֥י יְהֹוָ֖ה בַּגִּלְגָּֽל׃
{ס}　　　
לדלד וַיֵּ֥לֶךְ שְׁמוּאֵ֖ל הָרָמָ֑תָה וְשָׁא֛וּל עָלָ֥ה אֶל־בֵּית֖וֹ גִּבְעַ֥ת שָׁאֽוּל׃ להלה וְלֹא־יָסַ֨ף שְׁמוּאֵ֜ל לִרְא֤וֹת אֶת־שָׁאוּל֙ עַד־י֣וֹם מוֹת֔וֹ כִּֽי־הִתְאַבֵּ֥ל שְׁמוּאֵ֖ל אֶל־שָׁא֑וּל וַיהֹוָ֣ה נִחָ֔ם כִּֽי־הִמְלִ֥יךְ אֶת־שָׁא֖וּל עַל־יִשְׂרָאֵֽל׃
