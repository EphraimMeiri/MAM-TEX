{פ}　　
כגאא וְאֵ֛לֶּה דִּבְרֵ֥י דָוִ֖ד הָאַחֲרֹנִ֑ים נְאֻ֧ם דָּוִ֣ד בֶּן־יִשַׁ֗י וּנְאֻ֤ם הַגֶּ֙בֶר֙ הֻ֣קַם עָ֔ל מְשִׁ֙יחַ֙ אֱלֹהֵ֣י יַעֲקֹ֔ב וּנְעִ֖ים זְמִר֥וֹת יִשְׂרָאֵֽל׃ בב ר֥וּחַ יְהֹוָ֖ה דִּבֶּר־בִּ֑י וּמִלָּת֖וֹ עַל־לְשׁוֹנִֽי׃ גג אָמַר֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל לִ֥י דִבֶּ֖ר צ֣וּר יִשְׂרָאֵ֑ל מוֹשֵׁל֙ בָּאָדָ֔ם צַדִּ֕יק מוֹשֵׁ֖ל יִרְאַ֥ת אֱלֹהִֽים׃ דד וּכְא֥וֹר בֹּ֖קֶר יִזְרַח־שָׁ֑מֶשׁ בֹּ֚קֶר לֹ֣א עָב֔וֹת מִנֹּ֥גַהּ מִמָּטָ֖ר דֶּ֥שֶׁא מֵאָֽרֶץ׃ הה כִּי־לֹא־כֵ֥ן בֵּיתִ֖י עִם־אֵ֑ל כִּי֩ בְרִ֨ית עוֹלָ֜ם שָׂ֣ם לִ֗י עֲרוּכָ֤ה בַכֹּל֙ וּשְׁמֻרָ֔ה כִּי־כׇל־יִשְׁעִ֥י וְכׇל־חֵ֖פֶץ כִּי־לֹ֥א יַצְמִֽיחַ׃ וו וּבְלִיַּ֕עַל כְּק֥וֹץ מֻנָ֖ד כֻּלָּ֑הַם כִּי־לֹ֥א בְיָ֖ד יִקָּֽחוּ׃ זז וְאִישׁ֙ יִגַּ֣ע בָּהֶ֔ם יִמָּלֵ֥א בַרְזֶ֖ל וְעֵ֣ץ חֲנִ֑ית וּבָאֵ֕שׁ שָׂר֥וֹף יִשָּׂרְפ֖וּ בַּשָּֽׁבֶת׃
{פ}　　
חח אֵ֛לֶּה שְׁמ֥וֹת הַגִּבֹּרִ֖ים אֲשֶׁ֣ר לְדָוִ֑ד יֹשֵׁ֨ב בַּשֶּׁ֜בֶת תַּחְכְּמֹנִ֣י ׀ רֹ֣אשׁ הַשָּׁלִשִׁ֗י ה֚וּא עֲדִינ֣וֹ הָֽעֶצְנִ֔ויקקקרי/כתיב · (העצנו) הָֽעֶצְנִ֔י עַל־שְׁמֹנֶ֥ה מֵא֛וֹת חָלָ֖ל בְּפַ֥עַם (אחד)קקקרי/כתיב · (אחד) אֶחָֽת׃
{ס}　　　
טט וְאַחֲרָ֛וקקקרי/כתיב · וְאַחֲרָ֛ו · וְאַחֲרָ֛יו אֶלְעָזָ֥ר בֶּן־דדויקקקרי/כתיב · (דֹּד֖וֹ) דדי בֶּן־אֲחֹחִ֑י בִּשְׁלֹשָׁ֨ה (גברים)קקקרי/כתיב · (גברים) הַגִּבֹּרִ֜ים עִם־דָּוִ֗ד בְּחָרְפָ֤ם בַּפְּלִשְׁתִּים֙ נֶאֶסְפוּ־שָׁ֣ם לַמִּלְחָמָ֔ה וַֽיַּעֲל֖וּ אִ֥ישׁ יִשְׂרָאֵֽל׃ יי ה֣וּא קָם֩ וַיַּ֨ךְ בַּפְּלִשְׁתִּ֜ים עַ֣ד ׀ כִּֽי־יָגְעָ֣ה יָד֗וֹ וַתִּדְבַּ֤ק יָדוֹ֙ אֶל־הַחֶ֔רֶב וַיַּ֧עַשׂ יְהֹוָ֛ה תְּשׁוּעָ֥ה גְדוֹלָ֖ה בַּיּ֣וֹם הַה֑וּא וְהָעָ֛ם יָשֻׁ֥בוּ אַחֲרָ֖יו אַךְ־לְפַשֵּֽׁט׃
{ס}　　　
יאיא וְאַחֲרָ֛יו שַׁמָּ֥ה בֶן־אָגֵ֖א הָרָרִ֑י וַיֵּאָסְפ֨וּ פְלִשְׁתִּ֜ים לַחַיָּ֗ה וַתְּהִי־שָׁ֞ם חֶלְקַ֤ת הַשָּׂדֶה֙ מְלֵאָ֣ה עֲדָשִׁ֔ים וְהָעָ֥ם נָ֖ס מִפְּנֵ֥י פְלִשְׁתִּֽים׃ יביב וַיִּתְיַצֵּ֤ב בְּתוֹךְ־הַחֶלְקָה֙ וַיַּצִּילֶ֔הָ וַיַּ֖ךְ אֶת־פְּלִשְׁתִּ֑ים וַיַּ֥עַשׂ יְהֹוָ֖ה תְּשׁוּעָ֥ה גְדוֹלָֽה׃
{ס}　　　
יגיג וַיֵּרְד֨וּ (שלשים)קקקרי/כתיב · (שלשים) שְׁלֹשָׁ֜ה מֵהַשְּׁלֹשִׁ֣ים רֹ֗אשׁ וַיָּבֹ֤אוּ אֶל־קָצִיר֙ אֶל־דָּוִ֔ד אֶל־מְעָרַ֖ת עֲדֻלָּ֑םנעֲדֻלָּ֑ם=ל,ק ובדפוסים (חטף פתח בעי"ן), וכך אצל ברויאר ומג"הא!=עַדֻלָּ֑ם (חסרות נקודות החטף בעי"ן) וְחַיַּ֣ת פְּלִשְׁתִּ֔ים חֹנָ֖ה בְּעֵ֥מֶק רְפָאִֽים׃ ידיד וְדָוִ֖ד אָ֣ז בַּמְּצוּדָ֑ה וּמַצַּ֣ב פְּלִשְׁתִּ֔ים אָ֖ז בֵּ֥ית לָֽחֶם׃ טוטו וַיִּתְאַוֶּ֥ה דָוִ֖ד וַיֹּאמַ֑ר מִ֚י יַשְׁקֵ֣נִי מַ֔יִם מִבֹּ֥אר בֵּֽית־לֶ֖חֶם אֲשֶׁ֥ר בַּשָּֽׁעַר׃
{ס}　　　
טזטז וַיִּבְקְעוּ֩ שְׁלֹ֨שֶׁת הַגִּבֹּרִ֜ים בְּמַחֲנֵ֣ה פְלִשְׁתִּ֗ים וַיִּֽשְׁאֲבוּ־מַ֙יִם֙ מִבֹּ֤אר בֵּֽית־לֶ֙חֶם֙ אֲשֶׁ֣ר בַּשַּׁ֔עַר וַיִּשְׂא֖וּ וַיָּבִ֣אוּ אֶל־דָּוִ֑ד וְלֹ֤א אָבָה֙ לִשְׁתּוֹתָ֔ם וַיַּסֵּ֥ךְ אֹתָ֖ם לַיהֹוָֽה׃ יזיז וַיֹּ֡אמֶר חָלִ֩ילָה֩ לִּ֨י יְהֹוָ֜ה מֵעֲשֹׂ֣תִי זֹ֗את הֲדַ֤ם הָאֲנָשִׁים֙ הַהֹלְכִ֣ים בְּנַפְשׁוֹתָ֔ם וְלֹ֥א אָבָ֖ה לִשְׁתּוֹתָ֑ם אֵ֣לֶּה עָשׂ֔וּ שְׁלֹ֖שֶׁת הַגִּבֹּרִֽים׃
{ס}　　　
יחיח וַאֲבִישַׁ֞י אֲחִ֣י ׀ יוֹאָ֣ב בֶּן־צְרוּיָ֗ה ה֚וּא רֹ֣אשׁ (השלשי)קקקרי/כתיב · (השלשי) הַשְּׁלֹשָׁ֔ה וְהוּא֙ עוֹרֵ֣ר אֶת־חֲנִית֔וֹ עַל־שְׁלֹ֥שׁ מֵא֖וֹת חָלָ֑ל וְלוֹ־שֵׁ֖ם בַּשְּׁלֹשָֽׁה׃ יטיט מִן־הַשְּׁלֹשָׁה֙ הֲכִ֣י נִכְבָּ֔ד וַיְהִ֥י לָהֶ֖ם לְשָׂ֑ר וְעַד־הַשְּׁלֹשָׁ֖ה לֹא־בָֽא׃
{ס}　　　
ככ וּבְנָיָ֨הוּ בֶן־יְהוֹיָדָ֧ע בֶּן־אִֽישׁ־(חי)קקקרי/כתיב · (חַ֛יִל) חי רַב־פְּעָלִ֖ים מִֽקַּבְצְאֵ֑ל ה֣וּא הִכָּ֗ה אֵ֣ת שְׁנֵ֤י אֲרִאֵל֙ מוֹאָ֔ב וְ֠ה֠וּא יָרַ֞ד וְהִכָּ֧ה אֶֽת־(האריה)קקקרי/כתיב · (הָאֲרִ֛י) האריה בְּת֥וֹךְ הַבֹּ֖אר בְּי֥וֹם הַשָּֽׁלֶג׃ כאכא וְהֽוּא־הִכָּה֩ אֶת־אִ֨ישׁ מִצְרִ֜י (אשר)קקקרי/כתיב · (אשר) אִ֣ישׁ מַרְאֶ֗ה וּבְיַ֤ד הַמִּצְרִי֙ חֲנִ֔ית וַיֵּ֥רֶד אֵלָ֖יו בַּשָּׁ֑בֶט וַיִּגְזֹ֤ל אֶֽת־הַחֲנִית֙ מִיַּ֣ד הַמִּצְרִ֔י וַיַּהַרְגֵ֖הוּ בַּחֲנִיתֽוֹ׃ כבכב אֵ֣לֶּה עָשָׂ֔ה בְּנָיָ֖הוּ בֶּן־יְהוֹיָדָ֑ע וְלוֹ־שֵׁ֖ם בִּשְׁלֹשָׁ֥ה הַגִּבֹּרִֽים׃ כגכג מִן־הַשְּׁלֹשִׁ֣ים נִכְבָּ֔ד וְאֶל־הַשְּׁלֹשָׁ֖ה לֹא־בָ֑א וַיְשִׂמֵ֥הוּ דָוִ֖ד אֶל־מִשְׁמַעְתּֽוֹ׃
{ס}　　　
כדכד עֲשָׂהאֵ֥ל אֲחִֽי־יוֹאָ֖ב בַּשְּׁלֹשִׁ֑ים נ=א (אין פרשה, ונראה שלסופר גם לא היה מספיק מקום בשורה כדי להשאיר רווח ניכר; וכן ברויאר)ל,ק=פרשה סתומה, וכן במג"ה בדפוס למרות שאין מספיק רווח אפילו לאות אחת בלבד בכתראֶלְחָנָ֥ןנאֶלְחָנָ֥ן=ל,ק ובדפוסים (וכן ברויאר ומג"ה)א!=אֶלְחָנָן (חסר טעם המרכא) בֶּן־דֹּד֖וֹ בֵּ֥ית לָֽחֶם׃
{ס}　　　
כהכה שַׁמָּה֙ הַחֲרֹדִ֔יאֱלִיקָ֖א הַחֲרֹדִֽי׃
{ס}　　　
כוכו חֶ֚לֶץ הַפַּלְטִ֔יעִירָ֥א בֶן־עִקֵּ֖שׁנ=אל,ק=אין פרשההַתְּקוֹעִֽי׃
{ס}　　　
כזכז אֲבִיעֶ֙זֶר֙ הָעַנְּתֹתִ֔ימְבֻנַּ֖י הַחֻשָׁתִֽי׃
{ס}　　　
כחכח צַלְמוֹן֙ הָאֲחֹחִ֔ימַהְרַ֖י הַנְּטֹפָתִֽי׃
{ס}　　　
כטכט חֵ֥לֶב בֶּֽן־בַּעֲנָ֖ה הַנְּטֹפָתִ֑יאִתַּי֙ בֶּן־רִיבַ֔י מִגִּבְעַ֖ת בְּנֵ֥י בִנְיָמִֽן׃
{ס}　　　
לל בְּנָיָ֙הוּ֙ פִּרְעָ֣תֹנִ֔יהִדַּ֖י מִנַּ֥חֲלֵי גָֽעַשׁ׃
{ס}　　　
לאלא אֲבִֽי־עַלְבוֹן֙ הָעַרְבָתִ֔יעַזְמָ֖וֶת הַבַּרְחֻמִֽי׃
{ס}　　　
לבלב אֶלְיַחְבָּא֙ הַשַּׁ֣עַלְבֹנִ֔יבְּנֵ֥י יָשֵׁ֖ן יְהוֹנָתָֽן׃

לגלג שַׁמָּה֙ הַהֲרָרִ֔יאֲחִיאָ֥ם בֶּן־שָׁרָ֖ר הָארָרִֽי׃

לדלד אֱלִיפֶ֥לֶט בֶּן־אֲחַסְבַּ֖י בֶּן־הַמַּעֲכָתִ֑יאֱלִיעָ֥ם בֶּן־אֲחִיתֹ֖פֶל הַגִּלֹנִֽי׃
