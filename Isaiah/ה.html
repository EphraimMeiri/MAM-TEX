מקרא על פי המסורה

ספר ישעיהו

פרק ה


{פ}　　
האא אָשִׁ֤ירָה נָּא֙ לִֽידִידִ֔י שִׁירַ֥ת דּוֹדִ֖י לְכַרְמ֑וֹ כֶּ֛רֶם הָיָ֥ה לִידִידִ֖י בְּקֶ֥רֶן בֶּן־שָֽׁמֶן׃ בב וַֽיְעַזְּקֵ֣הוּ וַֽיְסַקְּלֵ֗הוּ וַיִּטָּעֵ֙הוּ֙ שֹׂרֵ֔ק וַיִּ֤בֶן מִגְדָּל֙ בְּתוֹכ֔וֹ וְגַם־יֶ֖קֶב חָצֵ֣ב בּ֑וֹ וַיְקַ֛ו לַעֲשׂ֥וֹת עֲנָבִ֖ים וַיַּ֥עַשׂ בְּאֻשִֽׁים׃ גג וְעַתָּ֛ה יוֹשֵׁ֥ב יְרוּשָׁלַ֖͏ִםם וְאִ֣ישׁ יְהוּדָ֑ה שִׁפְטוּ־נָ֕א בֵּינִ֖י וּבֵ֥ין כַּרְמִֽי׃ דד מַה־לַּעֲשׂ֥וֹת עוֹד֙ לְכַרְמִ֔י וְלֹ֥א עָשִׂ֖יתִי בּ֑וֹ מַדּ֧וּעַ קִוֵּ֛יתִי לַעֲשׂ֥וֹת עֲנָבִ֖ים וַיַּ֥עַשׂ בְּאֻשִֽׁים׃ הה וְעַתָּה֙ אוֹדִיעָה־נָּ֣א אֶתְכֶ֔ם אֵ֛ת אֲשֶׁר־אֲנִ֥י עֹשֶׂ֖ה לְכַרְמִ֑י הָסֵ֤ר מְשׂוּכָּתוֹ֙ וְהָיָ֣ה לְבָעֵ֔ר פָּרֹ֥ץ גְּדֵר֖וֹ וְהָיָ֥ה לְמִרְמָֽס׃ וו וַאֲשִׁיתֵ֣הוּ בָתָ֗ה לֹ֤א יִזָּמֵר֙ וְלֹ֣א יֵעָדֵ֔ר וְעָלָ֥ה שָׁמִ֖יר וָשָׁ֑יִת וְעַ֤ל הֶֽעָבִים֙ אֲצַוֶּ֔ה מֵהַמְטִ֥יר עָלָ֖יו מָטָֽר׃ זז כִּ֣י כֶ֜רֶם יְהֹוָ֤ה צְבָאוֹת֙ בֵּ֣ית יִשְׂרָאֵ֔ל וְאִ֣ישׁ יְהוּדָ֔ה נְטַ֖ע שַׁעֲשׁוּעָ֑יו וַיְקַ֤ו לְמִשְׁפָּט֙ וְהִנֵּ֣ה מִשְׂפָּ֔ח לִצְדָקָ֖ה וְהִנֵּ֥ה צְעָקָֽה׃
{פ}　　
חח ה֗וֹי מַגִּיעֵ֥י בַ֙יִת֙ בְּבַ֔יִת שָׂדֶ֥ה בְשָׂדֶ֖ה יַקְרִ֑יבוּ עַ֚ד אֶ֣פֶס מָק֔וֹם וְהוּשַׁבְתֶּ֥ם לְבַדְּכֶ֖ם בְּקֶ֥רֶב הָאָֽרֶץ׃ טט בְּאׇזְנָ֖י יְהֹוָ֣ה צְבָא֑וֹת אִם־לֹ֞א בָּתִּ֤ים רַבִּים֙ לְשַׁמָּ֣ה יִֽהְי֔וּ גְּדֹלִ֥ים וְטוֹבִ֖ים מֵאֵ֥ין יוֹשֵֽׁב׃ יי כִּ֗י עֲשֶׂ֙רֶת֙ צִמְדֵּי־כֶ֔רֶם יַעֲשׂ֖וּ בַּ֣ת אֶחָ֑ת וְזֶ֥רַע חֹ֖מֶר יַעֲשֶׂ֥ה אֵיפָֽה׃
{ס}　　　
יאיא ה֛וֹי מַשְׁכִּימֵ֥י בַבֹּ֖קֶר שֵׁכָ֣ר יִרְדֹּ֑פוּ מְאַחֲרֵ֣י בַנֶּ֔שֶׁף יַ֖יִן יַדְלִיקֵֽם׃ יביב וְהָיָ֨ה כִנּ֜וֹר וָנֶ֗בֶל תֹּ֧ף וְחָלִ֛יל וָיַ֖יִן מִשְׁתֵּיהֶ֑ם וְאֵ֨ת פֹּ֤עַל יְהֹוָה֙ לֹ֣א יַבִּ֔יטוּ וּמַעֲשֵׂ֥ה יָדָ֖יו לֹ֥א רָאֽוּ׃ יגיג לָכֵ֛ן גָּלָ֥ה עַמִּ֖י מִבְּלִי־דָ֑עַת וּכְבוֹדוֹ֙ מְתֵ֣י רָעָ֔ב וַהֲמוֹנ֖וֹ צִחֵ֥ה צָמָֽא׃ ידיד לָכֵ֗ן הִרְחִ֤יבָה שְּׁאוֹל֙ נַפְשָׁ֔הּ וּפָעֲרָ֥ה פִ֖יהָ לִבְלִי־חֹ֑ק וְיָרַ֨ד הֲדָרָ֧הּ וַהֲמוֹנָ֛הּ וּשְׁאוֹנָ֖הּ וְעָלֵ֥ז בָּֽהּ׃ טוטו וַיִּשַּׁ֥ח אָדָ֖ם וַיִּשְׁפַּל־אִ֑ישׁ וְעֵינֵ֥י גְבֹהִ֖ים תִּשְׁפַּֽלְנָה׃ טזטז וַיִּגְבַּ֛הּ יְהֹוָ֥ה צְבָא֖וֹת בַּמִּשְׁפָּ֑ט וְהָאֵל֙ הַקָּד֔וֹשׁ נִקְדָּ֖שׁ בִּצְדָקָֽה׃ יזיז וְרָע֥וּ כְבָשִׂ֖ים כְּדׇבְרָ֑ם וְחׇרְב֥וֹת מֵחִ֖ים גָּרִ֥ים יֹאכֵֽלוּ׃
{ס}　　　
יחיח ה֛וֹי מֹשְׁכֵ֥י הֶעָוֺ֖ן בְּחַבְלֵ֣י הַשָּׁ֑וְא וְכַעֲב֥וֹת הָעֲגָלָ֖ה חַטָּאָֽה׃ יטיט הָאֹמְרִ֗ים יְמַהֵ֧ר ׀ יָחִ֛ישָׁה מַעֲשֵׂ֖הוּ לְמַ֣עַן נִרְאֶ֑ה וְתִקְרַ֣ב וְתָב֗וֹאָה עֲצַ֛ת קְד֥וֹשׁ יִשְׂרָאֵ֖ל וְנֵדָֽעָה׃
{פ}　　
ככ ה֣וֹי הָאֹמְרִ֥ים לָרַ֛ע ט֖וֹב וְלַטּ֣וֹב רָ֑ע שָׂמִ֨ים חֹ֤שֶׁךְ לְאוֹר֙ וְא֣וֹר לְחֹ֔שֶׁךְ שָׂמִ֥ים מַ֛ר לְמָת֖וֹק וּמָת֥וֹק לְמָֽר׃
{ס}　　　
כאכא ה֖וֹי חֲכָמִ֣ים בְּעֵינֵיהֶ֑ם וְנֶ֥גֶד פְּנֵיהֶ֖ם נְבֹנִֽים׃
{ס}　　　
כבכב ה֕וֹי גִּבּוֹרִ֖ים לִשְׁתּ֣וֹת יָ֑יִן וְאַנְשֵׁי־חַ֖יִל לִמְסֹ֥ךְ שֵׁכָֽר׃ כגכג מַצְדִּיקֵ֥י רָשָׁ֖ע עֵ֣קֶב שֹׁ֑חַד וְצִדְקַ֥ת צַדִּיקִ֖ים יָסִ֥ירוּ מִמֶּֽנּוּ׃
{פ}　　
כדכד לָכֵן֩ כֶּאֱכֹ֨ל קַ֜שׁ לְשׁ֣וֹן אֵ֗שׁ וַחֲשַׁ֤שׁ לֶהָבָה֙ יִרְפֶּ֔ה שׇׁרְשָׁם֙ כַּמָּ֣ק יִֽהְיֶ֔ה וּפִרְחָ֖ם כָּאָבָ֣ק יַעֲלֶ֑ה כִּ֣י מָאֲס֗וּ אֵ֚ת תּוֹרַת֙ יְהֹוָ֣ה צְבָא֔וֹת וְאֵ֛ת אִמְרַ֥ת קְדֽוֹשׁ־יִשְׂרָאֵ֖ל נִאֵֽצוּ׃ כהכה עַל־כֵּ֡ן חָרָה֩ אַף־יְהֹוָ֨ה בְּעַמּ֜וֹ וַיֵּ֣ט יָד֧וֹ עָלָ֣יו וַיַּכֵּ֗הוּ וַֽיִּרְגְּזוּ֙ הֶהָרִ֔ים וַתְּהִ֧י נִבְלָתָ֛ם כַּסּוּחָ֖ה בְּקֶ֣רֶב חוּצ֑וֹת בְּכׇל־זֹאת֙ לֹא־שָׁ֣ב אַפּ֔וֹ וְע֖וֹד יָד֥וֹ נְטוּיָֽה׃ כוכו וְנָֽשָׂא־נֵ֤ס לַגּוֹיִם֙ מֵרָח֔וֹק וְשָׁ֥רַק ל֖וֹ מִקְצֵ֣ה הָאָ֑רֶץ וְהִנֵּ֥ה מְהֵרָ֖ה קַ֥ל יָבֽוֹא׃ כזכז אֵין־עָיֵ֤ף וְאֵין־כּוֹשֵׁל֙ בּ֔וֹ לֹ֥א יָנ֖וּם וְלֹ֣א יִישָׁ֑ן וְלֹ֤א נִפְתַּח֙ אֵז֣וֹר חֲלָצָ֔יו וְלֹ֥א נִתַּ֖ק שְׂר֥וֹךְ נְעָלָֽיו׃ כחכח אֲשֶׁ֤ר חִצָּיו֙ שְׁנוּנִ֔ים וְכׇל־קַשְּׁתֹתָ֖יו דְּרֻכ֑וֹת פַּרְס֤וֹת סוּסָיו֙ כַּצַּ֣ר נֶחְשָׁ֔בוּ וְגַלְגִּלָּ֖יו כַּסּוּפָֽה׃ כטכט שְׁאָגָ֥ה ל֖וֹ כַּלָּבִ֑יא ויִשְׁאַ֨גקקקרי/כתיב · (ושאג) יִשְׁאַ֨ג כַּכְּפִירִ֤ים וְיִנְהֹם֙ וְיֹאחֵ֣ז טֶ֔רֶף וְיַפְלִ֖יט וְאֵ֥ין מַצִּֽיל׃ לל וְיִנְהֹ֥ם עָלָ֛יו בַּיּ֥וֹם הַה֖וּא כְּנַהֲמַת־יָ֑ם וְנִבַּ֤ט לָאָ֙רֶץ֙ וְהִנֵּה־חֹ֔שֶׁךְ צַ֣ר וָא֔וֹר חָשַׁ֖ךְ בַּעֲרִיפֶֽיהָ׃
