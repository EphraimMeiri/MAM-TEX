יאא וּבְכׇל־זֹ֕את אֲנַ֛חְנוּ כֹּרְתִ֥ים אֲמָנָ֖ה וְכֹתְבִ֑ים וְעַל֙ הֶֽחָת֔וּם שָׂרֵ֥ינוּ לְוִיֵּ֖נוּ כֹּהֲנֵֽינוּ׃ בב וְעַ֖ל הַחֲתוּמִ֑ים נְחֶמְיָ֧ה הַתִּרְשָׁ֛תָא בֶּן־חֲכַלְיָ֖ה וְצִדְקִיָּֽה׃ גג שְׂרָיָ֥ה עֲזַרְיָ֖ה יִרְמְיָֽה׃ דד פַּשְׁח֥וּר אֲמַרְיָ֖ה מַלְכִּיָּֽה׃ הה חַטּ֥וּשׁ שְׁבַנְיָ֖ה מַלּֽוּךְ׃ וו חָרִ֥ם מְרֵמ֖וֹת עֹֽבַדְיָֽה׃ זז דָּנִיֵּ֥אל גִּנְּת֖וֹן בָּרֽוּךְ׃ חח מְשֻׁלָּ֥ם אֲבִיָּ֖ה מִיָּמִֽן׃ טט מַֽעַזְיָ֥ה בִלְגַּ֖י שְׁמַֽעְיָ֑ה אֵ֖לֶּה הַכֹּהֲנִֽים׃ נ__ · ל=פרשה סתומה יי וְֽהַלְוִיִּ֑ם וְיֵשׁ֙וּעַ֙ בֶּן־אֲזַנְיָ֔ה בִּנּ֕וּי מִבְּנֵ֥י חֵנָדָ֖ד קַדְמִיאֵֽל׃ יאיא וַאֲחֵיהֶ֑ם שְׁבַנְיָ֧ה הֽוֹדִיָּ֛ה קְלִיטָ֖א פְּלָאיָ֥ה חָנָֽן׃ יביב מִיכָ֥א רְח֖וֹב חֲשַׁבְיָֽה׃ יגיג זַכּ֥וּר שֵׁרֵֽבְיָ֖ה שְׁבַנְיָֽהנשְׁבַנְיָֽה=ש1,ק-מ,ב1 ומסורת-א וטברנית ומ"של?=שְׁכַנְיָֽה (באות כ"ף במקום בי"ת?)ברויאר ציין בי"ת בספק, אבל בהגדלה האות נראית כ"ף ודאית בכתי"ל; אף אחד מהמעתיקים האחרים של כתי"ל העיר לגבי האות וכולם ראו בה כ"ף.׃ ידיד הוֹדִיָּ֥ה בָנִ֖י בְּנִֽינוּ׃

טוטו רָאשֵׁ֖י הָעָ֑ם פַּרְעֹשׁ֙ פַּחַ֣ת מוֹאָ֔ב עֵילָ֥ם זַתּ֖וּא בָּנִֽי׃ טזטז בֻּנִּ֥י עַזְגָּ֖ד בֵּבָֽי׃ יזיז אֲדֹנִיָּ֥ה בִגְוַ֖י עָדִֽין׃ יחיח אָטֵ֥ר חִזְקִיָּ֖ה עַזּֽוּר׃ יטיט הוֹדִיָּ֥ה חָשֻׁ֖ם בֵּצָֽי׃ ככ חָרִ֥יף עֲנָת֖וֹת נֵויבָֽיקקקרי/כתיב · (נובי) נֵיבָֽי׃ כאכא מַגְפִּיעָ֥שׁ מְשֻׁלָּ֖ם חֵזִֽיר׃ כבכב מְשֵׁיזַבְאֵ֥ל צָד֖וֹק יַדּֽוּעַ׃ כגכג פְּלַטְיָ֥ה חָנָ֖ן עֲנָיָֽה׃ כדכד הוֹשֵׁ֥עַ חֲנַנְיָ֖ה חַשּֽׁוּב׃ כהכה הַלּוֹחֵ֥שׁ פִּלְחָ֖א שׁוֹבֵֽק׃ כוכו רְח֥וּם חֲשַׁבְנָ֖ה מַעֲשֵׂיָֽה׃ כזכז וַאֲחִיָּ֥ה חָנָ֖ן עָנָֽן׃ כחכח מַלּ֥וּךְ חָרִ֖ם בַּעֲנָֽה׃ כטכט וּשְׁאָ֣ר הָעָ֡ם הַכֹּהֲנִ֣ים הַ֠לְוִיִּ֠ם הַשּׁוֹעֲרִ֨ים הַמְשֹׁרְרִ֜ים הַנְּתִינִ֗ים וְֽכׇל־הַנִּבְדָּ֞ל מֵעַמֵּ֤י הָאֲרָצוֹת֙ אֶל־תּוֹרַ֣ת הָאֱלֹהִ֔ים נְשֵׁיהֶ֖ם בְּנֵיהֶ֣ם וּבְנֹתֵיהֶ֑ם כֹּ֖ל יוֹדֵ֥עַ מֵבִֽין׃ לל מַחֲזִיקִ֣ים עַל־אֲחֵיהֶם֮ אַדִּירֵיהֶם֒ וּבָאִ֞ים בְּאָלָ֣ה וּבִשְׁבוּעָ֗ה לָלֶ֙כֶת֙ בְּתוֹרַ֣ת הָאֱלֹהִ֔ים אֲשֶׁ֣ר נִתְּנָ֔ה בְּיַ֖ד מֹשֶׁ֣ה עֶֽבֶד־הָֽאֱלֹהִ֑יםנעֶֽבֶד־הָֽאֱלֹהִ֑ים · ל=עֶֽבֶד־הָֽאֱלֹהִ֑ים בגעיה ימנית וְלִשְׁמ֣וֹר וְלַעֲשׂ֗וֹת אֶת־כׇּל־מִצְוֺת֙ יְהֹוָ֣ה אֲדֹנֵ֔ינוּ וּמִשְׁפָּטָ֖יו וְחֻקָּֽיו׃ לאלא וַאֲשֶׁ֛ר לֹא־נִתֵּ֥ן בְּנֹתֵ֖ינוּ לְעַמֵּ֣י הָאָ֑רֶץ וְאֶת־בְּנֹ֣תֵיהֶ֔ם לֹ֥א נִקַּ֖ח לְבָנֵֽינוּ׃ לבלב וְעַמֵּ֣י הָאָ֡רֶץ הַֽמְבִיאִים֩ אֶת־הַמַּקָּח֨וֹת וְכׇל־שֶׁ֜בֶר בְּי֤וֹם הַשַּׁבָּת֙ לִמְכּ֔וֹר לֹא־נִקַּ֥ח מֵהֶ֛ם בַּשַּׁבָּ֖ת וּבְי֣וֹם קֹ֑דֶשׁ וְנִטֹּ֛שׁ אֶת־הַשָּׁנָ֥ה הַשְּׁבִיעִ֖ית וּמַשָּׁ֥א כׇל־יָֽד׃ לגלג וְהֶעֱמַ֤דְנוּ עָלֵ֙ינוּ֙ מִצְוֺ֔ת לָתֵ֥ת עָלֵ֛ינוּ שְׁלִישִׁ֥יתנשְׁלִישִׁ֥ית=ק-מ,ב1 ומסורת-ל וטברנית (כתיב מלא יו"ד ומלא יו"ד)ל!=שְׁלִשִׁ֥ית (כתיב חסר יו"ד ומלא יו"ד) הַשֶּׁ֖קֶל בַּשָּׁנָ֑ה לַעֲבֹדַ֖ת בֵּ֥ית אֱלֹהֵֽינוּ׃ לדלד לְלֶ֣חֶם הַֽמַּעֲרֶ֡כֶת וּמִנְחַ֣ת הַתָּמִ֣יד וּלְעוֹלַ֣ת הַ֠תָּמִ֠יד הַשַּׁבָּת֨וֹת הֶחֳדָשִׁ֜ים לַמּוֹעֲדִ֗ים וְלַקֳּדָשִׁים֙ וְלַ֣חַטָּא֔וֹת לְכַפֵּ֖ר עַל־יִשְׂרָאֵ֑ל וְכֹ֖ל מְלֶ֥אכֶת בֵּית־אֱלֹהֵֽינוּ׃

להלה וְהַגּוֹרָל֨וֹת הִפַּ֜לְנוּ עַל־קֻרְבַּ֣ן הָעֵצִ֗ים הַכֹּהֲנִ֣ים הַלְוִיִּם֮ וְהָעָם֒ לְ֠הָבִ֠יא לְבֵ֨ית אֱלֹהֵ֧ינוּ לְבֵית־אֲבֹתֵ֛ינוּ לְעִתִּ֥ים מְזֻמָּנִ֖ים שָׁנָ֣ה בְשָׁנָ֑ה לְבַעֵ֗ר עַל־מִזְבַּח֙ יְהֹוָ֣ה אֱלֹהֵ֔ינוּ כַּכָּת֖וּב בַּתּוֹרָֽה׃ לולו וּלְהָבִ֞יא אֶת־בִּכּוּרֵ֣י אַדְמָתֵ֗נוּ וּבִכּוּרֵ֛י כׇּל־פְּרִ֥י כׇל־עֵ֖ץ שָׁנָ֣ה בְשָׁנָ֑ה לְבֵ֖ית יְהֹוָֽה׃ לזלז וְאֶת־בְּכֹר֤וֹת בָּנֵ֙ינוּ֙ וּבְהֶמְתֵּ֔נוּנוּבְהֶמְתֵּ֔נוּ=ש1,ק-מ,ב1 ובדפוסים (כתיב חסר יו"ד)ל=וּבְהֶמְתֵּ֔ינוּ (כתיב מלא יו"ד) כַּכָּת֖וּב בַּתּוֹרָ֑ה וְאֶת־בְּכוֹרֵ֨י בְקָרֵ֜ינוּ וְצֹאנֵ֗ינוּ לְהָבִיא֙ לְבֵ֣ית אֱלֹהֵ֔ינוּ לַכֹּ֣הֲנִ֔ים הַמְשָׁרְתִ֖ים בְּבֵ֥ית אֱלֹהֵֽינוּ׃ לחלח וְאֶת־רֵאשִׁ֣ית עֲרִיסֹתֵ֣ינוּ וּ֠תְרוּמֹתֵ֠ינוּ וּפְרִ֨י כׇל־עֵ֜ץ תִּיר֣וֹשׁ וְיִצְהָ֗ר נָבִ֤יא לַכֹּהֲנִים֙ אֶל־לִשְׁכ֣וֹת בֵּית־אֱלֹהֵ֔ינוּ וּמַעְשַׂ֥ר אַדְמָתֵ֖נוּ לַלְוִיִּ֑ם וְהֵם֙ הַלְוִיִּ֔ם הַֽמְעַשְּׂרִ֔ים בְּכֹ֖ל עָרֵ֥י עֲבֹדָתֵֽנוּ׃ לטלט וְהָיָ֨ה הַכֹּהֵ֧ן בֶּֽן־אַהֲרֹ֛ן עִם־הַלְוִיִּ֖ם בַּעְשֵׂ֣ר הַלְוִיִּ֑ם וְהַלְוִיִּ֞ם יַעֲל֨וּ אֶת־מַעֲשַׂ֤ר הַֽמַּעֲשֵׂר֙ לְבֵ֣ית אֱלֹהֵ֔ינוּ אֶל־הַלְּשָׁכ֖וֹת לְבֵ֥ית הָאוֹצָֽר׃ ממ כִּ֣י אֶל־הַ֠לְּשָׁכ֠וֹת יָבִ֨יאוּ בְנֵי־יִשְׂרָאֵ֜ל וּבְנֵ֣י הַלֵּוִ֗י אֶת־תְּרוּמַ֣ת הַדָּגָן֮ הַתִּיר֣וֹשׁ וְהַיִּצְהָר֒ וְשָׁם֙ כְּלֵ֣י הַמִּקְדָּ֔שׁ וְהַכֹּהֲנִים֙ הַמְשָׁ֣רְתִ֔ים וְהַשּׁוֹעֲרִ֖ים וְהַמְשֹׁרְרִ֑ים וְלֹ֥א נַעֲזֹ֖ב אֶת־בֵּ֥ית אֱלֹהֵֽינוּ׃
