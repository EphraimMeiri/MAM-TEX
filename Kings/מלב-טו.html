חח בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וּשְׁמֹנֶה֙ שָׁנָ֔ה לַעֲזַרְיָ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַ֠ךְ זְכַרְיָ֨הוּ בֶן־יָרׇבְעָ֧ם עַל־יִשְׂרָאֵ֛ל בְּשֹׁמְר֖וֹן שִׁשָּׁ֥ה חֳדָשִֽׁים׃ טט וַיַּ֤עַשׂ הָרַע֙ בְּעֵינֵ֣י יְהֹוָ֔ה כַּאֲשֶׁ֥ר עָשׂ֖וּ אֲבֹתָ֑יו לֹ֣א סָ֗ר מֵֽחַטֹּאות֙ יָרׇבְעָ֣ם בֶּן־נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־יִשְׂרָאֵֽל׃ יי וַיִּקְשֹׁ֤ר עָלָיו֙ שַׁלֻּ֣ם בֶּן־יָבֵ֔שׁ וַיַּכֵּ֥הוּ נ=ש1 ובדפוסיםל!=קָֽבָלְ־עָ֖ם (למ"ד שווּאה ומוקפת)הערות דותן וברויאר (וכן הכריעו במג"ה)לגבי הקמצים הקטנים ראו ויינברג 3.6.8.4 שנשאר בספק לגבי תיבה זו, והשוו יחזקאל כו,ט (ויינברג 3.1.1 ושם בקמץ חטוף בדפוסים). הכרענו לגבי הקמץ בבי"ת לפי מסורת הקריאה הספרדית (וכך בסימנים). וַיְמִיתֵ֑הוּ וַיִּמְלֹ֖ךְ תַּחְתָּֽיו׃ יאיא וְיֶ֖תֶר דִּבְרֵ֣י זְכַרְיָ֑ה הִנָּ֣ם כְּתוּבִ֗ים עַל־סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ יביב ה֣וּא דְבַר־יְהֹוָ֗ה אֲשֶׁ֨ר דִּבֶּ֤ר אֶל־יֵהוּא֙ לֵאמֹ֔ר בְּנֵ֣י רְבִיעִ֔ים יֵשְׁב֥וּ לְךָ֖ עַל־כִּסֵּ֣א יִשְׂרָאֵ֑ל וַֽיְהִי־כֵֽןנוַֽיְהִי־כֵֽן · ל=וַֽיְהִי־כֵֽן בגעיה ימנית׃

יגיג שַׁלּ֤וּם בֶּן־יָבֵישׁ֙ מָלַ֔ךְ בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה לְעֻזִּיָּ֖הנלְעֻזִּיָּ֖ה=ק ושיטת-א ובדפוסים (זי"ן דגושהל=לְעֻזִיָּ֖ה (חסר דגש בזי"ן)הערות דותן וברויאר מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֥ךְ יֶֽרַח־יָמִ֖ים בְּשֹׁמְרֽוֹן׃ ידיד וַיַּ֩עַל֩ מְנַחֵ֨ם בֶּן־גָּדִ֜י מִתִּרְצָ֗ה וַיָּבֹא֙ שֹׁמְר֔וֹן וַיַּ֛ךְ אֶת־שַׁלּ֥וּם בֶּן־יָבֵ֖ישׁ בְּשֹׁמְר֑וֹן וַיְמִיתֵ֖הוּ וַיִּמְלֹ֥ךְ תַּחְתָּֽיו׃ טוטו וְיֶ֙תֶר֙ דִּבְרֵ֣י שַׁלּ֔וּם וְקִשְׁר֖וֹ אֲשֶׁ֣ר קָשָׁ֑ר הִנָּ֣ם כְּתוּבִ֗יםנכְּתוּבִ֗ים=ק,ש1 ומסורות-א,ל וטברניות ובדפוסיםל!=כְּתֻבִ֗ים (כתיב חסר וי"ו) עַל־סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ נ__=א(ק) אין פרשהל=פרשה סתומה וכן הוא בכתי"ק, וכן ציין פינפר במקצת ספרים. אבל יהושע קמחי סימן במפורש שאין פרשה בכתר במקום הזה (וכן הכריעו ברויאר ומג"ה). והשוו "וישלח אחז מלאכים" (טז, ז). טזטז אָ֣ז יַכֶּֽה־מְ֠נַחֵ֠ם אֶת־תִּפְסַ֨ח וְאֶת־כׇּל־אֲשֶׁר־בָּ֤הּ וְאֶת־גְּבוּלֶ֙יהָ֙ מִתִּרְצָ֔ה כִּ֛י לֹ֥א פָתַ֖ח וַיַּ֑ךְ אֵ֛ת כׇּל־הֶהָ֥רוֹתֶ֖יהָ בִּקֵּֽעַ׃

יזיז בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה לַעֲזַרְיָ֖ה מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַ֠ךְ מְנַחֵ֨ם בֶּן־גָּדִ֧י עַל־יִשְׂרָאֵ֛ל עֶ֥שֶׂר שָׁנִ֖ים בְּשֹׁמְרֽוֹן׃ יחיח וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהֹוָ֑ה לֹ֣א סָ֠ר מֵעַ֨ל חַטֹּ֜אות יָרׇבְעָ֧ם בֶּן־נְבָ֛ט אֲשֶׁר־הֶחֱטִ֥יא אֶת־יִשְׂרָאֵ֖ל כׇּל־יָמָֽיו׃ יטיט בָּ֣א פ֤וּל מֶֽלֶךְ־אַשּׁוּר֙ עַל־הָאָ֔רֶץ וַיִּתֵּ֤ן מְנַחֵם֙ לְפ֔וּל אֶ֖לֶף כִּכַּר־כָּ֑סֶף לִהְי֤וֹת יָדָיו֙ אִתּ֔וֹ לְהַחֲזִ֥יק הַמַּמְלָכָ֖ה בְּיָדֽוֹ׃ ככ וַיֹּצֵא֩ מְנַחֵ֨ם אֶת־הַכֶּ֜סֶף עַל־יִשְׂרָאֵ֗ל עַ֚ל כׇּל־גִּבּוֹרֵ֣י הַחַ֔יִל לָתֵת֙ לְמֶ֣לֶךְ אַשּׁ֔וּר חֲמִשִּׁ֧ים שְׁקָלִ֛ים כֶּ֖סֶף לְאִ֣ישׁ אֶחָ֑ד וַיָּ֙שׇׁב֙ מֶ֣לֶךְ אַשּׁ֔וּר וְלֹא־עָ֥מַד שָׁ֖ם בָּאָֽרֶץ׃ כאכא וְיֶ֛תֶר דִּבְרֵ֥י מְנַחֵ֖ם וְכׇל־אֲשֶׁ֣ר עָשָׂ֑ה הֲלוֹא־הֵ֣ם כְּתוּבִ֗ים עַל־סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ כבכב וַיִּשְׁכַּ֥ב מְנַחֵ֖ם עִם־אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ פְּקַחְיָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃

לבלב בִּשְׁנַ֣ת שְׁתַּ֔יִם לְפֶ֥קַח בֶּן־רְמַלְיָ֖הוּ מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ יוֹתָ֥ם בֶּן־עֻזִּיָּ֖הוּנבֶּן־עֻזִּיָּ֖הוּ=ק ובדפוסים (זי"ן דגושה)ל=בֶּן־עֻזִיָּ֖הוּ (חסר דגש בזי"ן)הערות דותן וברויאר מֶ֥לֶךְ יְהוּדָֽה׃ לגלג בֶּן־עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ הָיָ֣ה בְמׇלְכ֔וֹ וְשֵׁשׁ־עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלָ֑͏ִםם וְשֵׁ֣ם אִמּ֔וֹ יְרוּשָׁ֖א בַּת־צָדֽוֹק׃ לדלד וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהֹוָ֑ה כְּכֹ֧ל אֲשֶׁר־עָשָׂ֛ה עֻזִּיָּ֥הוּנעֻזִּיָּ֥הוּ=ק ובדפוסים (זי"ן דגושה)ל=עֻזִיָּ֥הוּ (חסר דגש בזי"ן)הערות דותן וברויאר אָבִ֖יו עָשָֽׂה׃ להלה רַ֤ק הַבָּמוֹת֙ לֹ֣א סָ֔רוּ ע֗וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמ֑וֹת ה֗וּא בָּנָ֛ה אֶת־שַׁ֥עַר בֵּית־יְהֹוָ֖ה הָעֶלְיֽוֹן׃ לולו וְיֶ֛תֶר דִּבְרֵ֥י יוֹתָ֖ם אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־הֵ֣ם כְּתוּבִ֗ים עַל־סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ לזלז בַּיָּמִ֣ים הָהֵ֔ם הֵחֵ֣ל יְהֹוָ֗ה לְהַשְׁלִ֙יחַ֙ בִּֽיהוּדָ֔ה רְצִ֖ין מֶ֣לֶךְ אֲרָ֑ם וְאֵ֖ת פֶּ֥קַח בֶּן־רְמַלְיָֽהוּ׃ לחלח וַיִּשְׁכַּ֤ב יוֹתָם֙ עִם־אֲבֹתָ֔יו וַיִּקָּבֵר֙ עִם־אֲבֹתָ֔יו בְּעִ֖יר דָּוִ֣ד אָבִ֑יו וַיִּמְלֹ֛ךְ אָחָ֥ז בְּנ֖וֹ תַּחְתָּֽיו׃
