פרק מה


{פ}　　
מהאא לַמְנַצֵּ֣חַ עַל־שֹׁ֭שַׁנִּים
נפרשה סתומהלִבְנֵי־קֹ֑רַח
מַ֝שְׂכִּ֗יל שִׁ֣יר יְדִידֹֽת׃
בב רָ֘חַ֤שׁ לִבִּ֨י ׀ דָּ֘בָ֤ר ט֗וֹב
אֹמֵ֣ר אָ֭נִי מַעֲשַׂ֣י לְמֶ֑לֶךְ
לְ֝שׁוֹנִ֗יעֵ֤ט ׀ סוֹפֵ֬ר מָהִֽיר׃
גג יׇפְיָפִ֡יתָניׇפְיָפִ֡יתָלגבי הקמץ הקטן ביו"ד הראשונה ראו ויינברג 3.6.8.7, שציין ספק לגבי תיבה זו. הכרענו לפי המסורת הספרדית שבה הקמץ קטן, וכך בסימנים ובתנ"ך קורן החדש.מִבְּנֵ֬י אָדָ֗ם
ה֣וּצַק חֵ֭ן בְּשִׂפְתוֹתֶ֑יךָ
עַל־כֵּ֤ן בֵּרַכְךָ֖ אֱלֹהִ֣ים לְעוֹלָֽם׃
דד חֲגֽוֹר־חַרְבְּךָ֣ עַל־יָרֵ֣ךְ גִּבּ֑וֹר
ה֝וֹדְךָ֗ וַהֲדָרֶֽךָ׃
הה וַהֲדָ֬רְךָ֨ ׀ צְלַ֬ח רְכַ֗ב
עַֽל־דְּבַר־אֱ֭מֶתוְעַנְוָה־צֶ֑דֶק
וְתוֹרְךָ֖ נוֹרָא֣וֹת יְמִינֶֽךָ׃
וו חִצֶּ֗יךָ שְׁנ֫וּנִ֥ים
עַ֭מִּים תַּחְתֶּ֣יךָ יִפְּל֑וּ
בְּ֝לֵ֗באוֹיְבֵ֥י הַמֶּֽלֶךְ׃
זז כִּסְאֲךָ֣ אֱ֭לֹהִים עוֹלָ֣ם וָעֶ֑ד
שֵׁ֥בֶט מִ֝ישֹׁ֗ר שֵׁ֣בֶט מַלְכוּתֶֽךָ׃
חח אָהַ֣בְתָּ צֶּדֶק֮ וַתִּשְׂנָ֫א רֶ֥שַׁע
עַל־כֵּ֤ן ׀ מְשָׁחֲךָ֡
אֱלֹהִ֣ים אֱ֭לֹהֶיךָ שֶׁ֥מֶן שָׂשׂ֗וֹן
מֵחֲבֵרֶֽךָ׃
טט מֹר־וַאֲהָל֣וֹת קְ֭צִיעוֹת כׇּל־בִּגְדֹתֶ֑יךָ
מִֽן־הֵ֥יכְלֵי שֵׁ֝֗ןנשֵׁ֝֗ן · א=שֵׁ֝ן (השמטת נקודת הרביע) מִנִּ֥י שִׂמְּחֽוּךָ׃
יי בְּנ֣וֹת מְ֭לָכִים בְּיִקְּרוֹתֶ֑יךָ
נִצְּבָ֥ה שֵׁגַ֥ל לִ֝ימִינְךָ֗ בְּכֶ֣תֶם אוֹפִֽיר׃
יאיא שִׁמְעִי־בַ֣ת וּ֭רְאִי וְהַטִּ֣י אׇזְנֵ֑ךְ
וְשִׁכְחִ֥י עַ֝מֵּ֗ךְ וּבֵ֥ית אָבִֽיךְ׃
יביב וְיִתְאָ֣ו הַמֶּ֣לֶךְ יׇפְיֵ֑ךְ
כִּי־ה֥וּא אֲ֝דֹנַ֗יִךְ וְהִשְׁתַּֽחֲוִי־לֽוֹ׃
יגיג וּבַֽת־צֹ֨ר ׀
בְּ֭מִנְחָהפָּנַ֥יִךְ יְחַלּ֗וּ
עֲשִׁ֣ירֵי עָֽם׃
ידיד כׇּל־כְּבוּדָּ֣ה בַת־מֶ֣לֶךְ פְּנִ֑ימָה
מִֽמִּשְׁבְּצ֖וֹת זָהָ֣ב לְבוּשָֽׁהּ׃
טוטו לִרְקָמוֹת֮תּוּבַ֢ל לַ֫מֶּ֥לֶךְ
בְּתוּל֣וֹת אַ֭חֲרֶיהָ רֵעוֹתֶ֑יהָ
מ֖וּבָא֣וֹת לָֽךְ׃
טזטז תּ֭וּבַלְנָה בִּשְׂמָחֹ֣ת וָגִ֑יל
תְּ֝בֹאֶ֗ינָה בְּהֵ֣יכַל מֶֽלֶךְ׃
יזיז תַּ֣חַת אֲ֭בֹתֶיךָ יִהְי֣וּ בָנֶ֑יךָ
תְּשִׁיתֵ֥מוֹ לְ֝שָׂרִ֗ים בְּכׇל־הָאָֽרֶץ׃
יחיח אַזְכִּ֣ירָה שִׁ֭מְךָ בְּכׇל־דֹּ֣ר וָדֹ֑ר
עַל־כֵּ֥ן עַמִּ֥ים יְ֝הוֹד֗וּךָ לְעֹלָ֥ם וָעֶֽד׃
