להאא לְדָוִ֨ד ׀
נפרשה סתומהרִיבָ֣ה יְ֭הֹוָה אֶת־יְרִיבַ֑י
לְ֝חַ֗ם אֶת־לֹחֲמָֽי׃
בב הַחֲזֵ֣ק מָגֵ֣ן וְצִנָּ֑ה
וְ֝ק֗וּמָה בְּעֶזְרָתִֽי׃
גג וְהָ֘רֵ֤ק חֲנִ֣ית וּ֭סְגֹר לִקְרַ֣את רֹדְפָ֑י
אֱמֹ֥ר לְ֝נַפְשִׁ֗י יְֽשֻׁעָתֵ֥ךְ אָֽנִי׃
דד יֵבֹ֣שׁוּ וְיִכָּלְמוּ֮מְבַקְשֵׁ֢י נַ֫פְשִׁ֥י
יִסֹּ֣גוּ אָח֣וֹר וְיַחְפְּר֑וּ
חֹ֝שְׁבֵ֗י רָעָתִֽי׃
הה יִֽהְי֗וּ כְּמֹ֥ץ לִפְנֵי־ר֑וּחַ
וּמַלְאַ֖ךְ יְהֹוָ֣ה דּוֹחֶֽה׃
וו יְֽהִי־דַרְכָּ֗ם
חֹ֥שֶׁךְ וַחֲלַקְלַקֹּ֑ת
וּמַלְאַ֥ךְ יְ֝הֹוָ֗הרֹדְפָֽם׃
זז כִּֽי־חִנָּ֣ם טָֽמְנוּ־לִ֭י שַׁ֣חַת רִשְׁתָּ֑ם
חִ֝נָּ֗ם חָפְר֥וּ לְנַפְשִֽׁי׃
חח תְּבוֹאֵ֣הוּ שׁוֹאָה֮לֹֽא־יֵ֫דָ֥ע
וְרִשְׁתּ֣וֹ אֲשֶׁר־טָמַ֣ן תִּלְכְּד֑וֹ
בְּ֝שׁוֹאָ֗ה יִפׇּל־בָּֽהּ׃
טט וְ֭נַפְשִׁי תָּגִ֣יל בַּיהֹוָ֑ה
תָּ֝שִׂ֗ישׂ בִּישׁוּעָתֽוֹ׃
יי נראו במידע על מהדורה זו, פרק שני, על ההכרעה בין קמץ קטן לגדול בתיבה זו. יש עוד "כל" כזה במשלי (יט,ז). עַצְמוֹתַ֨י ׀ תֹּאמַרְנָה֮
יְהֹוָ֗ה מִ֥י כָ֫מ֥וֹךָ
מַצִּ֣יל עָ֭נִי מֵחָזָ֣ק מִמֶּ֑נּוּ
וְעָנִ֥י וְ֝אֶבְי֗וֹן מִגֹּֽזְלֽוֹ׃
יאיא יְ֭קוּמוּן עֵדֵ֣י חָמָ֑ס
אֲשֶׁ֥ר לֹֽא־יָ֝דַ֗עְתִּי יִשְׁאָלֽוּנִי׃
יביב יְשַׁלְּמ֣וּנִי רָ֭עָה תַּ֥חַת טוֹבָ֗ה
שְׁכ֣וֹל לְנַפְשִֽׁי׃
יגיג וַאֲנִ֤י ׀ בַּחֲלוֹתָ֡םלְב֬וּשִׁי שָׂ֗ק
עִנֵּ֣יתִי בַצּ֣וֹם נַפְשִׁ֑י
וּ֝תְפִלָּתִ֗י עַל־חֵיקִ֥י תָשֽׁוּב׃
ידיד כְּרֵעַ־כְּאָ֣ח לִ֭י הִתְהַלָּ֑כְתִּי

טוטו וּבְצַלְעִי֮שָׂמְח֢וּ וְֽנֶ֫אֱסָ֥פוּ
נֶאֶסְפ֬וּ עָלַ֣י נֵ֭כִים וְלֹ֣א יָדַ֑עְתִּי
קָרְע֥וּ וְלֹא־דָֽמּוּ׃
טזטז בְּ֭חַנְפֵי לַעֲגֵ֣י מָע֑וֹג
חָרֹ֖ק עָלַ֣י שִׁנֵּֽימוֹ׃
יזיז אֲדֹנָי֮כַּמָּ֢ה תִ֫רְאֶ֥ה
הָשִׁ֣יבָה נַ֭פְשִׁי מִשֹּׁאֵיהֶ֑ם
מִ֝כְּפִירִ֗ים יְחִידָתִֽי׃

יטיט אַֽל־יִשְׂמְחוּ־לִ֣י אֹיְבַ֣י שֶׁ֑קֶר
שֹׂנְאַ֥י חִ֝נָּ֗ם יִקְרְצוּ־עָֽיִן׃
ככ כִּ֤י לֹ֥א שָׁל֗וֹםיְדַ֫בֵּ֥רוּ
וְעַ֥ל רִגְעֵי־אֶ֑רֶץ
דִּבְרֵ֥י מִ֝רְמ֗וֹת יַחֲשֹׁבֽוּן׃
כאכא וַיַּרְחִ֥יבוּ עָלַ֗יפִּ֫יהֶ֥ם
אָ֭מְרוּ הֶאָ֣ח ׀ הֶאָ֑ח
רָאֲתָ֥ה עֵינֵֽנוּ׃
כבכב רָאִ֣יתָה יְ֭הֹוָה אַֽל־תֶּחֱרַ֑שׁ
אֲ֝דֹנָ֗י אַל־תִּרְחַ֥ק מִמֶּֽנִּי׃
כגכג הָעִ֣ירָה וְ֭הָקִיצָה לְמִשְׁפָּטִ֑י
אֱלֹהַ֖י וַאדֹנָ֣י לְרִיבִֽי׃
כדכד שׇׁפְטֵ֣נִי כְ֭צִדְקְךָ יְהֹוָ֥ה אֱלֹהָ֗י
וְאַל־יִשְׂמְחוּ־לִֽי׃
כהכה אַל־יֹאמְר֣וּ בְ֭לִבָּם הֶאָ֣ח נַפְשֵׁ֑נוּ
אַל־יֹ֝אמְר֗וּ בִּֽלַּעֲנֽוּהוּ׃
כוכו יֵ֘בֹ֤שׁוּ וְיַחְפְּר֨וּ ׀ יַחְדָּו֮שְׂמֵחֵ֢י רָעָ֫תִ֥י
יִֽלְבְּשׁוּ־בֹ֥שֶׁת וּכְלִמָּ֑ה
הַֽמַּגְדִּילִ֥ים עָלָֽי׃
כזכז יָרֹ֣נּוּ וְיִשְׂמְחוּ֮חֲפֵצֵ֢י צִ֫דְקִ֥י
וְיֹאמְר֣וּ תָ֭מִיד יִגְדַּ֣ל יְהֹוָ֑ה
הֶ֝חָפֵ֗ץ שְׁל֣וֹם עַבְדּֽוֹ׃
כחכח וּ֭לְשׁוֹנִי תֶּהְגֶּ֣ה צִדְקֶ֑ךָ
