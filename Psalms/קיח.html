בב יֹאמַר־נָ֥א יִשְׂרָאֵ֑ל
כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃
גג יֹאמְרוּ־נָ֥א בֵֽית־אַהֲרֹ֑ן

דד יֹאמְרוּ־נָ֭א יִרְאֵ֣י יְהֹוָ֑ה
כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃
הה מִֽן־הַ֭מֵּצַר קָרָ֣אתִי יָּ֑הּ
עָנָ֖נִי בַמֶּרְחָ֣ב יָֽהּ׃
וו יְהֹוָ֣ה לִ֭י לֹ֣א אִירָ֑א
מַה־יַּעֲשֶׂ֖ה לִ֣י אָדָֽם׃
זז יְהֹוָ֣ה לִ֭י בְּעֹזְרָ֑י
וַ֝אֲנִ֗י אֶרְאֶ֥ה בְשֹׂנְאָֽי׃
חח ט֗וֹב לַחֲס֥וֹת בַּיהֹוָ֑ה
מִ֝בְּטֹ֗חַ בָּאָדָֽם׃
טט ט֗וֹב לַחֲס֥וֹת בַּיהֹוָ֑ה
מִ֝בְּטֹ֗חַ בִּנְדִיבִֽים׃
יי כׇּל־גּוֹיִ֥ם סְבָב֑וּנִי
בְּשֵׁ֥םנבְּשֵׁ֥ם=ל ובדפוסיםא!=בְּ֭שֵׁ֥ם (קו מיותר של דחי באות בי"ת); הדחי החריג ברור בצילום הכתר. במהדורות ברויאר ומג"ה הורידו אותו. יְ֝הֹוָ֗ה כִּ֣י אֲמִילַֽם׃
יאיא סַבּ֥וּנִי גַם־סְבָב֑וּנִי
בְּשֵׁ֥ם יְ֝הֹוָ֗ה כִּ֣י אֲמִילַֽם׃
יביב סַבּ֤וּנִי כִדְבוֹרִ֗ים
דֹּ֭עֲכוּ כְּאֵ֣שׁ קוֹצִ֑ים
בְּשֵׁ֥ם יְ֝הֹוָ֗הכִּ֣י אֲמִילַֽם׃
יגיג דַּחֹ֣ה דְחִיתַ֣נִי לִנְפֹּ֑ל
וַ֖יהֹוָ֣ה עֲזָרָֽנִי׃
ידיד עׇזִּ֣י וְזִמְרָ֣ת יָ֑הּ
וַֽיְהִי־לִ֝֗ינוַֽיְהִי־לִ֝֗י · א=וַֽיְהִי־לִ֝י (השמטת נקודת הרביע) לִישׁוּעָֽה׃
טוטו ק֤וֹל ׀ רִנָּ֬ה וִישׁוּעָ֗ה צַדִּיקִ֑ים
יְמִ֥ין יְ֝הֹוָ֗ה עֹ֣שָׂה חָֽיִל׃
טזטז יְמִ֣ין יְ֭הֹוָה רוֹמֵמָ֑ה
יְמִ֥ין יְ֝הֹוָ֗ה עֹ֣שָׂה חָֽיִל׃
יזיז לֹא־אָמ֥וּת כִּֽי־אֶחְיֶ֑ה
וַ֝אֲסַפֵּ֗ר מַעֲשֵׂ֥י יָֽהּ׃
יחיח יַסֹּ֣ר יִסְּרַ֣נִּי יָּ֑הּ
וְ֝לַמָּ֗וֶת לֹ֣א נְתָנָֽנִי׃
יטיט פִּתְחוּ־לִ֥י שַׁעֲרֵי־צֶ֑דֶק
אָבֹא־בָ֝֗םנאָבֹא־בָ֝֗ם · א=אָבֹא־בָ֝ם (השמטת נקודת הרביע) אוֹדֶ֥ה יָֽהּ׃
ככ זֶה־הַשַּׁ֥עַר לַיהֹוָ֑ה
צַ֝דִּיקִ֗ים יָבֹ֥אוּ בֽוֹ׃
כאכא א֭וֹדְךָ כִּ֣י עֲנִיתָ֑נִי
וַתְּהִי־לִ֝֗ינוַתְּהִי־לִ֝֗י · א=וַתְּהִי־לִ֝י (השמטת נקודת הרביע) לִישׁוּעָֽה׃
כבכב אֶ֭בֶן מָאֲס֣וּ הַבּוֹנִ֑ים
הָ֝יְתָ֗ה לְרֹ֣אשׁ פִּנָּֽה׃
כגכג מֵאֵ֣ת יְ֭הֹוָה הָ֣יְתָה זֹּ֑את
הִ֖יא נִפְלָ֣את בְּעֵינֵֽינוּ׃
כדכד זֶה־הַ֭יּוֹם עָשָׂ֣ה יְהֹוָ֑ה
נָגִ֖ילָה וְנִשְׂמְחָ֣ה בֽוֹ׃
כהכה אָנָּ֣א יְ֭הֹוָה הוֹשִׁ֘יעָ֥ה נָּ֑א
