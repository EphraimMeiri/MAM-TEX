לאאא לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃
נפרשה פתוחה בב בְּךָֽ־יְהֹוָ֣ה חָ֭סִיתִי
אַל־אֵב֣וֹשָׁה לְעוֹלָ֑ם
בְּצִדְקָתְךָ֥ פַלְּטֵֽנִי׃
גג הַטֵּ֤ה אֵלַ֨י ׀ אׇזְנְךָ֮מְהֵרָ֢ה הַצִּ֫ילֵ֥נִי
הֱיֵ֤ה לִ֨י ׀ לְֽצוּר־מָ֭עוֹז לְבֵ֥ית מְצוּד֗וֹת
לְהוֹשִׁיעֵֽנִי׃
דד כִּֽי־סַלְעִ֣י וּמְצוּדָתִ֣י אָ֑תָּה
וּלְמַ֥עַן שִׁ֝מְךָ֗ תַּֽנְחֵ֥נִי וּֽתְנַהֲלֵֽנִי׃
הה תּוֹצִיאֵ֗נִי מֵרֶ֣שֶׁת ז֭וּ טָ֣מְנוּ לִ֑י
כִּי־אַ֝תָּ֗ה מָעוּזִּֽי׃
וו בְּיָדְךָ֮ אַפְקִ֢יד ר֫וּחִ֥י
פָּדִ֖יתָ אוֹתִ֥י יְהֹוָ֗ה
אֵ֣ל אֱמֶֽת׃
זז שָׂנֵ֗אתִי הַשֹּׁמְרִ֥ים הַבְלֵי־שָׁ֑וְא
וַ֝אֲנִ֗י אֶל־יְהֹוָ֥ה בָּטָֽחְתִּי׃
חח אָגִ֥ילָה וְאֶשְׂמְחָ֗הבְּחַ֫סְדֶּ֥ךָ
אֲשֶׁ֣ר רָ֭אִיתָ אֶת־עׇנְיִ֑י
יָ֝דַ֗עְתָּ בְּצָר֥וֹת נַפְשִֽׁי׃
טט וְלֹ֣א הִ֭סְגַּרְתַּנִי בְּיַד־אוֹיֵ֑ב
הֶעֱמַ֖דְתָּ בַמֶּרְחָ֣ב רַגְלָֽי׃
יי חׇנֵּ֥נִי יְהֹוָה֮כִּ֤י צַ֫ר לִ֥י

נַפְשִׁ֥י וּבִטְנִֽי׃
יאיא כִּ֤י כָל֪וּ בְיָג֡וֹן חַיַּי֮וּשְׁנוֹתַ֢י בַּאֲנָ֫חָ֥ה
כָּשַׁ֣ל בַּעֲוֺנִ֣י כֹחִ֑י
וַעֲצָמַ֥י עָשֵֽׁשׁוּ׃

יגיג נִ֭שְׁכַּחְתִּי כְּמֵ֣ת מִלֵּ֑ב
הָ֝יִ֗יתִי כִּכְלִ֥י אֹבֵֽד׃
ידיד כִּ֤י שָׁמַ֨עְתִּי ׀ דִּבַּ֥ת רַבִּים֮
מָג֢וֹר מִסָּ֫בִ֥יב
בְּהִוָּסְדָ֣ם יַ֣חַד עָלַ֑י
לָקַ֖חַת נַפְשִׁ֣י זָמָֽמוּ׃
טוטו וַאֲנִ֤י ׀ עָלֶ֣יךָ בָטַ֣חְתִּי יְהֹוָ֑ה
אָ֝מַ֗רְתִּי אֱלֹהַ֥י אָֽתָּה׃
טזטז בְּיָדְךָ֥ עִתֹּתָ֑י
הַצִּ֘ילֵ֤נִי מִיַּד־א֝וֹיְבַ֗י וּמֵֽרֹדְפָֽי׃
יזיז הָאִ֣ירָה פָ֭נֶיךָ עַל־עַבְדֶּ֑ךָ
ה֖וֹשִׁיעֵ֣נִי בְחַסְדֶּֽךָ׃
יחיח יְֽהֹוָ֗ה אַל־אֵ֭בוֹשָׁה כִּ֣י קְרָאתִ֑יךָ
יֵבֹ֥שׁוּ רְ֝שָׁעִ֗ים יִדְּמ֥וּ לִשְׁאֽוֹל׃

פָּ֭עַלְתָּ לַחֹסִ֣ים בָּ֑ךְ

כאכא תַּסְתִּירֵ֤ם ׀ בְּסֵ֥תֶר פָּנֶיךָ֮מֵרֻכְסֵ֫י אִ֥ישׁ

כגכג וַאֲנִ֤י ׀ אָ֘מַ֤רְתִּי בְחׇפְזִ֗י
