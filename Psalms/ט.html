פרק ט

טאא לַ֭מְנַצֵּחַ עַל־מ֥וּת לַבֵּ֗ן
נפרשה סתומהמִזְמ֥וֹר לְדָוִֽד׃
בב אוֹדֶ֣ה יְ֭הֹוָה בְּכׇל־לִבִּ֑י
אֲ֝סַפְּרָ֗ה כׇּל־נִפְלְאוֹתֶֽיךָ׃
גג אֶשְׂמְחָ֣ה וְאֶעֶלְצָ֣ה בָ֑ךְ
אֲזַמְּרָ֖ה שִׁמְךָ֣ עֶלְיֽוֹן׃
דד בְּשׁוּב־אוֹיְבַ֥י אָח֑וֹר
יִכָּשְׁל֥וּ וְ֝יֹאבְד֗וּ מִפָּנֶֽיךָ׃
הה כִּֽי־עָ֭שִׂיתָ מִשְׁפָּטִ֣י וְדִינִ֑י
יָשַׁ֥בְתָּ לְ֝כִסֵּ֗א שׁוֹפֵ֥ט צֶֽדֶק׃
וו גָּעַ֣רְתָּ ג֭וֹיִם אִבַּ֣דְתָּ רָשָׁ֑ע
שְׁמָ֥ם מָ֝חִ֗יתָ לְעוֹלָ֥ם וָעֶֽד׃
זז הָֽאוֹיֵ֨ב ׀ תַּ֥מּוּ חֳרָב֗וֹתלָ֫נֶ֥צַח
וְעָרִ֥ים נָתַ֑שְׁתָּ
אָבַ֖ד זִכְרָ֣ם הֵֽמָּה׃
חח וַֽ֭יהֹוָה לְעוֹלָ֣ם יֵשֵׁ֑ב
כּוֹנֵ֖ן לַמִּשְׁפָּ֣ט כִּסְאֽוֹ׃
טט וְה֗וּא יִשְׁפֹּֽט־תֵּבֵ֥ל בְּצֶ֑דֶק
יָדִ֥ין לְ֝אֻמִּ֗ים בְּמֵישָׁרִֽים׃
יי וִ֘יהִ֤י יְהֹוָ֣ה מִשְׂגָּ֣ב לַדָּ֑ךְ
מִ֝שְׂגָּ֗ב לְעִתּ֥וֹת בַּצָּרָֽה׃
יאיא וְיִבְטְח֣וּ בְ֭ךָ יוֹדְעֵ֣י שְׁמֶ֑ךָ
כִּ֤י לֹֽא־עָזַ֖בְתָּ דֹרְשֶׁ֣יךָ יְהֹוָֽה׃
יביב זַמְּר֗וּ לַ֭יהֹוָה יֹשֵׁ֣ב צִיּ֑וֹן
הַגִּ֥ידוּ בָ֝עַמִּ֗ים עֲלִֽילוֹתָֽיו׃
יגיג כִּֽי־דֹרֵ֣שׁ דָּ֭מִים אוֹתָ֣ם זָכָ֑ר
לֹֽא־שָׁ֝כַ֗ח צַעֲקַ֥ת עֲנָיוִֽיםקקקרי/כתיב · (עניים) עֲנָוִֽים׃
ידיד חָֽנְﬞנֵ֬נִינחָֽנְﬞנֵ֬נִיא=חָֽנֲנֵ֬נִי (חטף); בכתר יש לא רק געיה בחי"ת אלא גם חטף בנו"ן הראשונה, כך שהשווא במהדורתנו נע בוודאי. ואילו במהדורת סימנים סימנו קמץ קטן תחת החי"ת, השמיטו את הגעיה בכתר (!) וסימנו שווא נח (!). יְהֹוָ֗ה
רְאֵ֣ה עׇ֭נְיִי מִשֹּׂנְאָ֑י
מְ֝רוֹמְﬞמִ֗ינמְ֝רוֹמְﬞמִ֗י · א=מְ֝רוֹמֲמִ֗י (חטף) מִשַּׁ֥עֲרֵי מָֽוֶת׃
טוטו לְמַ֥עַן אֲסַפְּרָ֗ה כׇּֽל־תְּהִלָּ֫תֶ֥יךָ
בְּשַׁעֲרֵ֥י בַת־צִיּ֑וֹן
אָ֝גִ֗ילָהבִּישׁוּעָתֶֽךָ׃
טזטז טָבְע֣וּ ג֭וֹיִם בְּשַׁ֣חַת עָשׂ֑וּ
בְּרֶֽשֶׁת־ז֥וּ טָ֝מָ֗נוּ נִלְכְּדָ֥ה רַגְלָֽם׃
יזיז נ֤וֹדַ֨ע ׀ יְהֹוָה֮מִשְׁפָּ֢ט עָ֫שָׂ֥ה
בְּפֹ֣עַל כַּ֭פָּיו נוֹקֵ֣שׁ רָשָׁ֑ע
הִגָּי֥וֹן סֶֽלָה׃
יחיח יָשׁ֣וּבוּ רְשָׁעִ֣ים לִשְׁא֑וֹלָה
כׇּל־גּ֝וֹיִ֗ם שְׁכֵחֵ֥י אֱלֹהִֽים׃
יטיט כִּ֤י לֹ֣א לָ֭נֶצַח יִשָּׁכַ֣ח אֶבְי֑וֹן
תִּקְוַ֥ת עֲ֝נִויִּ֗יםנ(ענוים) · א-כתיב=עֲ֝נִויִּ֗ם תֹּאבַ֥ד לָעַֽד׃
ככ קוּמָ֣ה יְ֭הֹוָה אַל־יָעֹ֣ז אֱנ֑וֹשׁ
יִשָּׁפְט֥וּ ג֝וֹיִ֗ם עַל־פָּנֶֽיךָ׃
כאכא שִׁ֘יתָ֤ה יְהֹוָ֨ה ׀ מוֹרָ֗הלָ֫הֶ֥ם
יֵדְע֥וּ גוֹיִ֑ם
אֱנ֖וֹשׁ הֵ֣מָּה סֶּֽלָה׃
