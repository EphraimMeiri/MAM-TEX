ספר תהלים

פרק לח


{פ}　　
לחאא מִזְמ֖וֹר לְדָוִ֣ד לְהַזְכִּֽיר׃
נפרשה פתוחה בב יְֽהֹוָ֗ה אַל־בְּקֶצְפְּךָ֥ תוֹכִיחֵ֑נִי
וּֽבַחֲמָתְךָ֥ תְיַסְּרֵֽנִי׃
גג כִּֽי־חִ֭צֶּיךָ נִ֣חֲתוּ בִ֑י
וַתִּנְחַ֖ת עָלַ֣י יָדֶֽךָ׃
דד אֵין־מְתֹ֣ם בִּ֭בְשָׂרִי מִפְּנֵ֣י זַעְמֶ֑ךָ
אֵין־שָׁל֥וֹם בַּ֝עֲצָמַ֗י מִפְּנֵ֥י חַטָּאתִֽי׃
הה כִּ֣י עֲ֭וֺנֹתַי עָבְר֣וּ רֹאשִׁ֑י
כְּמַשָּׂ֥א כָ֝בֵ֗ד יִכְבְּד֥וּ מִמֶּֽנִּי׃
וו הִבְאִ֣ישׁוּ נָ֭מַקּוּ חַבּוּרֹתָ֑י
מִ֝פְּנֵ֗י אִוַּלְתִּֽי׃
זז נַעֲוֵ֣יתִי שַׁחֹ֣תִי עַד־מְאֹ֑ד
כׇּל־הַ֝יּ֗וֹם קֹדֵ֥ר הִלָּֽכְתִּי׃
חח כִּֽי־כְ֭סָלַי מָלְא֣וּ נִקְלֶ֑ה
וְאֵ֥ין מְ֝תֹ֗ם בִּבְשָׂרִֽי׃
טט נְפוּג֣וֹתִי וְנִדְכֵּ֣יתִי עַד־מְאֹ֑ד
שָׁ֝אַ֗גְתִּי מִֽנַּהֲמַ֥ת לִבִּֽי׃
יי אֲֽדֹנָ֗ינאֲֽדֹנָ֗י · א=געיה ימנית נֶגְדְּךָ֥ כׇל־תַּאֲוָתִ֑י
וְ֝אַנְחָתִ֗י מִמְּךָ֥ לֹֽא־נִסְתָּֽרָה׃
יאיא לִבִּ֣י סְ֭חַרְחַר עֲזָבַ֣נִי כֹחִ֑י
וְֽאוֹר־עֵינַ֥י גַּם־הֵ֝֗םנגַּם־הֵ֝֗ם · א=גַּם־הֵ֝ם (השמטת נקודת הרביע) אֵ֣ין אִתִּֽי׃
יביב אֹֽהֲבַ֨י ׀ וְרֵעַ֗י מִנֶּ֣גֶד נִגְעִ֣י יַעֲמֹ֑דוּ
וּ֝קְרוֹבַ֗י מֵרָחֹ֥ק עָמָֽדוּ׃
יגיג וַיְנַקְשׁ֤וּ ׀ מְבַקְשֵׁ֬י נַפְשִׁ֗י
וְדֹרְשֵׁ֣י רָ֭עָתִי דִּבְּר֣וּ הַוּ֑וֹת
וּ֝מִרְמ֗וֹת כׇּל־הַיּ֥וֹם יֶהְגּֽוּ׃
ידיד וַאֲנִ֣י כְ֭חֵרֵשׁ לֹ֣א אֶשְׁמָ֑ע
וּ֝כְאִלֵּ֗ם לֹ֣א יִפְתַּח־פִּֽיו׃
טוטו וָאֱהִ֗י כְּ֭אִישׁ אֲשֶׁ֣ר לֹֽא־שֹׁמֵ֑עַ
וְאֵ֥ין בְּ֝פִ֗יו תּוֹכָחֽוֹת׃
טזטז כִּֽי־לְךָ֣ יְהֹוָ֣ה הוֹחָ֑לְתִּי
אַתָּ֥ה תַ֝עֲנֶ֗ה אֲדֹנָ֥י אֱלֹהָֽי׃
יזיז כִּֽי־אָ֭מַרְתִּי פֶּן־יִשְׂמְחוּ־לִ֑י
בְּמ֥וֹט רַ֝גְלִ֗י עָלַ֥י הִגְדִּֽילוּ׃
יחיח כִּֽי־אֲ֭נִי לְצֶ֣לַע נָכ֑וֹן
וּמַכְאוֹבִ֖י נֶגְדִּ֣י תָמִֽיד׃
יטיט כִּֽי־עֲוֺנִ֥י אַגִּ֑יד
אֶ֝דְאַ֗ג מֵ֖חַטָּאתִֽי׃
ככ וְֽ֭אֹיְבַי חַיִּ֣ים עָצֵ֑מוּ
וְרַבּ֖וּ שֹׂנְאַ֣י שָֽׁקֶר׃
כאכא וּמְשַׁלְּמֵ֣י רָ֭עָה תַּ֣חַת טוֹבָ֑ה
יִ֝שְׂטְנ֗וּנִי תַּ֣חַת (רדופי)טֽוֹבנ(רדופי)טֽוֹבלגבי קמץ קטן בתיבה זו ראו ויינברג 3.2.1.3; במהדורת סימנים הדפיסו קמץ רחב וגעיה.׃
כבכב אַל־תַּעַזְבֵ֥נִי יְהֹוָ֑ה
אֱ֝לֹהַ֗י אַל־תִּרְחַ֥ק מִמֶּֽנִּי׃
כגכג ח֥וּשָׁה לְעֶזְרָתִ֑י
אֲ֝דֹנָ֗י תְּשׁוּעָתִֽי׃
