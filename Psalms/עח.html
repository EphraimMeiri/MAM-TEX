עחאא מַשְׂכִּ֗יל לְאָ֫סָ֥ף
נפרשה פתוחההַאֲזִ֣ינָה עַ֭מִּי תּוֹרָתִ֑י
הַטּ֥וּ אׇ֝זְנְכֶ֗ם לְאִמְרֵי־פִֽי׃
בב אֶפְתְּחָ֣ה בְמָשָׁ֣ל פִּ֑י
אַבִּ֥יעָה חִ֝יד֗וֹת מִנִּי־קֶֽדֶם׃
גג אֲשֶׁ֣ר שָׁ֭מַעְנוּ וַנֵּדָעֵ֑ם
וַ֝אֲבוֹתֵ֗ינוּ סִפְּרוּ־לָֽנוּ׃
דד לֹ֤א נְכַחֵ֨ד ׀ מִבְּנֵיהֶ֗ם
לְד֥וֹר אַחֲר֗וֹן
מְֽ֭סַפְּרִים תְּהִלּ֣וֹת יְהֹוָ֑ה
וֶעֱזוּז֥וֹ וְ֝נִפְלְאֹתָ֗יו אֲשֶׁ֣ר עָשָֽׂה׃
הה וַיָּ֤קֶם עֵד֨וּת ׀ בְּֽיַעֲקֹ֗ב
וְתוֹרָה֮ שָׂ֤ם בְּיִשְׂרָ֫אֵ֥ל
אֲשֶׁ֣ר צִ֭וָּה אֶת־אֲבוֹתֵ֑ינוּ
לְ֝הוֹדִיעָ֗ם לִבְנֵיהֶֽם׃
וו לְמַ֤עַן יֵדְע֨וּ ׀ דּ֣וֹר אַ֭חֲרוֹן בָּנִ֣ים יִוָּלֵ֑דוּ
יָ֝קֻ֗מוּ וִיסַפְּר֥וּ לִבְנֵיהֶֽם׃
זז וְיָשִׂ֥ימוּ בֵאלֹהִ֗יםכִּ֫סְלָ֥ם
וְלֹ֣א יִ֭שְׁכְּחוּ מַ֥עַלְלֵי־אֵ֑לנמַ֥עַלְלֵי־אֵ֑ל · =א (מקף בתיבת משרת)
וּמִצְוֺתָ֥יו יִנְצֹֽרוּ׃
חח וְלֹ֤א יִהְי֨וּ ׀ כַּאֲבוֹתָ֗ם
דּוֹר֮ סוֹרֵ֢ר וּמֹ֫רֶ֥ה
דּ֭וֹר לֹא־הֵכִ֣ין לִבּ֑וֹ
וְלֹֽא־נֶאֶמְנָ֖ה אֶת־אֵ֣ל רוּחֽוֹ׃
טט בְּֽנֵי־אֶפְרַ֗יִם נוֹשְׁקֵ֥י רֽוֹמֵי־קָ֑שֶׁת
הָ֝פְכ֗וּ בְּי֣וֹם קְרָֽב׃
יי לֹ֣א שָׁ֭מְרוּ בְּרִ֣ית אֱלֹהִ֑ים
וּ֝בְתוֹרָת֗וֹ מֵאֲנ֥וּ לָלֶֽכֶת׃
יאיא וַיִּשְׁכְּח֥וּ עֲלִילוֹתָ֑יו
וְ֝נִפְלְאוֹתָ֗יו אֲשֶׁ֣ר הֶרְאָֽם׃
יביב נֶ֣גֶד אֲ֭בוֹתָם עָ֣שָׂה פֶ֑לֶא
בְּאֶ֖רֶץ מִצְרַ֣יִם שְׂדֵה־צֹֽעַן׃
יגיג בָּ֣קַע יָ֭ם וַיַּעֲבִירֵ֑ם
וַֽיַּצֶּב־מַ֥יִם כְּמוֹ־נֵֽד׃
ידיד וַיַּנְחֵ֣ם בֶּעָנָ֣ן יוֹמָ֑ם
וְכׇל־הַ֝לַּ֗יְלָה בְּא֣וֹר אֵֽשׁ׃
טוטו יְבַקַּ֣ע צֻ֭רִים בַּמִּדְבָּ֑ר
וַ֝יַּ֗שְׁקְ כִּתְהֹמ֥וֹת רַבָּֽה׃
טזטז וַיּוֹצִ֣א נוֹזְלִ֣ים מִסָּ֑לַע
וַיּ֖וֹרֶד כַּנְּהָר֣וֹת מָֽיִם׃
יזיז וַיּוֹסִ֣יפוּ ע֭וֹד לַחֲטֹא־ל֑וֹ
לַֽמְר֥וֹת עֶ֝לְי֗וֹן בַּצִּיָּֽה׃
יחיח וַיְנַסּוּ־אֵ֥ל בִּלְבָבָ֑ם
לִֽשְׁאׇל־אֹ֥כֶל לְנַפְשָֽׁם׃
יטיט וַֽיְדַבְּר֗וּ בֵּאלֹ֫הִ֥ים
אָ֭מְרוּ הֲי֣וּכַל אֵ֑ל
לַעֲרֹ֥ךְ שֻׁ֝לְחָ֗ןבַּמִּדְבָּֽר׃
ככ הֵ֤ן הִכָּה־צ֨וּר ׀ וַיָּז֣וּבוּ מַיִם֮
וּנְחָלִ֢ים יִ֫שְׁטֹ֥פוּ
הֲגַם־לֶ֭חֶם י֣וּכַל תֵּ֑ת
אִם־יָכִ֖ין שְׁאֵ֣ר לְעַמּֽוֹ׃
כאכא לָכֵ֤ן ׀ שָׁמַ֥ע יְהֹוָ֗הוַֽיִּתְעַ֫בָּ֥ר
וְ֭אֵשׁ נִשְּׂקָ֣ה בְיַעֲקֹ֑ב
וְגַם־אַ֝֗ףנוְגַם־אַ֝֗ף · א=וְגַם־אַ֝ף (השמטת נקודת הרביע) עָלָ֥ה בְיִשְׂרָאֵֽל׃
כבכב כִּ֤י לֹ֣א הֶ֭אֱמִינוּ בֵּאלֹהִ֑ים
וְלֹ֥א בָ֝טְח֗וּ בִּישׁוּעָתֽוֹ׃
כגכג וַיְצַ֣ו שְׁחָקִ֣ים מִמָּ֑עַל
וְדַלְתֵ֖י שָׁמַ֣יִם פָּתָֽח׃
כדכד וַיַּמְטֵ֬ר עֲלֵיהֶ֣ם מָ֣ן לֶאֱכֹ֑ל
וּדְגַן־שָׁ֝מַ֗יִם נָ֣תַן לָֽמוֹ׃
כהכה לֶ֣חֶם אַ֭בִּירִים אָ֣כַל אִ֑ישׁ
צֵידָ֬ה שָׁלַ֖ח לָהֶ֣ם לָשֹֽׂבַע׃
כוכו יַסַּ֣ע קָ֭דִים בַּשָּׁמָ֑יִם
וַיְנַהֵ֖ג בְּעֻזּ֣וֹ תֵימָֽן׃
כזכז וַיַּמְטֵ֬ר עֲלֵיהֶ֣ם כֶּעָפָ֣ר שְׁאֵ֑ר
וּֽכְח֥וֹל יַ֝מִּ֗ים ע֣וֹף כָּנָֽף׃
כחכח וַ֭יַּפֵּל בְּקֶ֣רֶב מַחֲנֵ֑הוּ
סָ֝בִ֗יב לְמִשְׁכְּנֹתָֽיו׃
כטכט וַיֹּאכְל֣וּ וַיִּשְׂבְּע֣וּ מְאֹ֑ד
וְ֝תַאֲוָתָ֗ם יָבִ֥א לָהֶֽם׃
לל לֹא־זָ֥רוּ מִתַּאֲוָתָ֑ם
ע֝֗וֹדנע֝֗וֹד · א=ע֝וֹד (השמטת נקודת הרביע) אׇכְלָ֥ם בְּפִיהֶֽם׃
לאלא וְאַ֤ף אֱלֹהִ֨ים ׀ עָ֘לָ֤ה בָהֶ֗ם
וַֽ֭יַּהֲרֹג בְּמִשְׁמַנֵּיהֶ֑ם
וּבַחוּרֵ֖י יִשְׂרָאֵ֣ל הִכְרִֽיעַ׃
לבלב בְּכׇל־זֹ֥את חָֽטְאוּ־ע֑וֹד
וְלֹֽא־הֶ֝אֱמִ֗ינוּ בְּנִפְלְאוֹתָֽיו׃
לגלג וַיְכַל־בַּהֶ֥בֶל יְמֵיהֶ֑ם
וּ֝שְׁנוֹתָ֗ם בַּבֶּהָלָֽה׃
לדלד אִם־הֲרָגָ֥ם וּדְרָשׁ֑וּהוּ
וְ֝שָׁ֗בוּ וְשִֽׁחֲרוּ־אֵֽל׃
להלה וַֽ֭יִּזְכְּרוּ כִּֽי־אֱלֹהִ֣ים צוּרָ֑ם
וְאֵ֥ל עֶ֝לְי֗וֹן גֹּאֲלָֽם׃
לולו וַיְפַתּ֥וּהוּ בְּפִיהֶ֑ם
וּ֝בִלְשׁוֹנָ֗ם יְכַזְּבוּ־לֽוֹ׃
לזלז וְ֭לִבָּם לֹא־נָכ֣וֹן עִמּ֑וֹ
וְלֹ֥א נֶ֝אֶמְנ֗וּ בִּבְרִיתֽוֹ׃
לחלח וְה֤וּא רַח֨וּם ׀ יְכַפֵּ֥ר עָוֺן֮וְֽלֹא־יַֽ֫שְׁחִ֥ית
וְ֭הִרְבָּה לְהָשִׁ֣יב אַפּ֑וֹ
וְלֹא־יָ֝עִ֗יר כׇּל־חֲמָתֽוֹ׃
לטלט וַ֭יִּזְכֹּר כִּֽי־בָשָׂ֣ר הֵ֑מָּה
ר֥וּחַ ה֝וֹלֵ֗ךְ וְלֹ֣א יָשֽׁוּב׃
ממ כַּ֭מָּה יַמְר֣וּהוּ בַמִּדְבָּ֑ר
יַ֝עֲצִיב֗וּהוּ בִּישִׁימֽוֹן׃
מאמא וַיָּשׁ֣וּבוּ וַיְנַסּ֣וּ אֵ֑ל
וּקְד֖וֹשׁ יִשְׂרָאֵ֣ל הִתְוֽוּ׃
מבמב לֹא־זָכְר֥וּ אֶת־יָד֑וֹ
י֝֗וֹםני֝֗וֹם · א=י֝וֹם (השמטת נקודת הרביע) אֲֽשֶׁר־פָּדָ֥ם מִנִּי־צָֽר׃
מגמג אֲשֶׁר־שָׂ֣ם בְּ֭מִצְרַיִם אֹתוֹתָ֑יו
וּ֝מוֹפְתָ֗יו בִּשְׂדֵה־צֹֽעַן׃
מדמד וַיַּהֲפֹ֣ךְ לְ֭דָם יְאֹרֵיהֶ֑ם
וְ֝נֹזְלֵיהֶ֗ם בַּל־יִשְׁתָּיֽוּן׃
מהמה יְשַׁלַּ֬ח בָּהֶ֣ם עָ֭רֹב וַיֹּאכְלֵ֑ם
וּ֝צְפַרְדֵּ֗עַ וַתַּשְׁחִיתֵֽם׃
מומו וַיִּתֵּ֣ן לֶחָסִ֣יל יְבוּלָ֑ם
וִ֝יגִיעָ֗ם לָאַרְבֶּֽה׃
מזמז יַהֲרֹ֣ג בַּבָּרָ֣ד גַּפְנָ֑ם
וְ֝שִׁקְמוֹתָ֗ם בַּחֲנָמַֽל׃
מחמח וַיַּסְגֵּ֣ר לַבָּרָ֣ד בְּעִירָ֑ם
וּ֝מִקְנֵיהֶ֗ם לָרְשָׁפִֽים׃
מטמט יְשַׁלַּח־בָּ֨ם ׀ חֲר֬וֹן אַפּ֗וֹ
עֶבְרָ֣ה וָזַ֣עַם וְצָרָ֑ה
מִ֝שְׁלַ֗חַת מַלְאֲכֵ֥י רָעִֽים׃
ננ יְפַלֵּ֥ס נָתִ֗יבלְאַ֫פּ֥וֹ
לֹא־חָשַׂ֣ךְ מִמָּ֣וֶת נַפְשָׁ֑ם
וְ֝חַיָּתָ֗ם לַדֶּ֥בֶר הִסְגִּֽיר׃
נאנא וַיַּ֣ךְ כׇּל־בְּכ֣וֹר בְּמִצְרָ֑יִם
רֵאשִׁ֥ית א֝וֹנִ֗ים ׃
נבנב וַיַּסַּ֣ע כַּצֹּ֣אן עַמּ֑וֹ
וַֽיְנַהֲגֵ֥ם כַּ֝עֵ֗דֶר בַּמִּדְבָּֽר׃
נגנג וַיַּנְחֵ֣ם לָ֭בֶטַח וְלֹ֣א פָחָ֑דוּ
וְאֶת־א֝וֹיְבֵיהֶ֗ם כִּסָּ֥ה הַיָּֽם׃
נדנד וַ֭יְבִיאֵם אֶל־גְּב֣וּל קׇדְשׁ֑וֹ
הַר־זֶ֝֗הנהַר־זֶ֝֗ה · א=הַר־זֶ֝ה (השמטת נקודת הרביע) קָנְתָ֥ה יְמִינֽוֹ׃
נהנה וַיְגָ֤רֶשׁ מִפְּנֵיהֶ֨ם ׀ גּוֹיִ֗ם
וַֽ֭יַּפִּילֵם בְּחֶ֣בֶל נַחֲלָ֑ה
וַיַּשְׁכֵּ֥ן שִׁבְטֵ֥י יִשְׂרָאֵֽל׃
נונו וַיְנַסּ֣וּ וַ֭יַּמְרוּ אֶת־אֱלֹהִ֣ים עֶלְי֑וֹן
וְ֝עֵדוֹתָ֗יו לֹ֣א שָׁמָֽרוּ׃
נזנז וַיִּסֹּ֣גוּ וַֽ֭יִּבְגְּדוּ כַּאֲבוֹתָ֑ם
נֶ֝הְפְּכ֗וּ כְּקֶ֣שֶׁת רְמִיָּֽה׃
נחנח וַיַּכְעִיס֥וּהוּ בְּבָמוֹתָ֑ם
וּ֝בִפְסִילֵיהֶ֗ם יַקְנִיאֽוּהוּ׃
נטנט שָׁמַ֣ע אֱ֭לֹהִים וַֽיִּתְעַבָּ֑ר
וַיִּמְאַ֥ס מְ֝אֹ֗ד בְּיִשְׂרָאֵֽל׃
סס וַ֭יִּטֹּשׁ מִשְׁכַּ֣ן שִׁל֑וֹ
אֹ֝֗הֶלנאֹ֝֗הֶל · א=אֹ֝הֶל (השמטת נקודת הרביע) שִׁכֵּ֥ן בָּאָדָֽם׃
סאסא וַיִּתֵּ֣ן לַשְּׁבִ֣י עֻזּ֑וֹ
וְֽתִפְאַרְתּ֥וֹ בְיַד־צָֽר׃
סבסב וַיַּסְגֵּ֣ר לַחֶ֣רֶב עַמּ֑וֹ
וּ֝בְנַחֲלָת֗וֹ הִתְעַבָּֽר׃
סגסג בַּחוּרָ֥יו אָכְלָה־אֵ֑שׁ
וּ֝בְתוּלֹתָ֗יו לֹ֣א הוּלָּֽלוּ׃
סדסד כֹּ֭הֲנָיו בַּחֶ֣רֶב נָפָ֑לוּ
וְ֝אַלְמְנֹתָ֗יו לֹ֣א תִבְכֶּֽינָה׃
סהסה וַיִּקַ֖ץ כְּיָשֵׁ֥ן ׀ אֲדֹנָ֑י
כְּ֝גִבּ֗וֹר מִתְרוֹנֵ֥ן מִיָּֽיִן׃
סוסו וַיַּךְ־צָרָ֥יו אָח֑וֹר
חֶרְפַּ֥ת ע֝וֹלָ֗ם נָ֣תַן לָֽמוֹ׃
סזסז וַ֭יִּמְאַס בְּאֹ֣הֶל יוֹסֵ֑ף
וּֽבְשֵׁ֥בֶט אֶ֝פְרַ֗יִם לֹ֣א בָחָֽר׃
סחסח וַ֭יִּבְחַר אֶת־שֵׁ֣בֶט יְהוּדָ֑ה
אֶֽת־הַ֥ר צִ֝יּ֗וֹן אֲשֶׁ֣ר אָהֵֽב׃
סטסט וַיִּ֣בֶן כְּמֽוֹ־רָ֭מִים מִקְדָּשׁ֑וֹ
כְּ֝אֶ֗רֶץ יְסָדָ֥הּ לְעוֹלָֽם׃
עע וַ֭יִּבְחַר בְּדָוִ֣ד עַבְדּ֑וֹ
וַ֝יִּקָּחֵ֗הוּ מִֽמִּכְלְאֹ֥ת צֹֽאן׃
עאעא מֵאַחַ֥ר עָל֗וֹתהֱבִ֫יא֥וֹ
לִ֭רְעוֹת בְּיַעֲקֹ֣ב עַמּ֑וֹ
וּ֝בְיִשְׂרָאֵ֗ל נַחֲלָתֽוֹ׃
עבעב וַ֭יִּרְעֵם כְּתֹ֣ם לְבָב֑וֹ
וּבִתְבוּנ֖וֹת כַּפָּ֣יו יַנְחֵֽם׃
