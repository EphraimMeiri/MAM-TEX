פרק לז


{פ}　　
לזאא לְדָוִ֨ד ׀ נלְדָוִ֨ד | · אין רווח של פרשה אחרי הכותרתאַל־תִּתְחַ֥ר בַּמְּרֵעִ֑ים
אַל־תְּ֝קַנֵּ֗א בְּעֹשֵׂ֥י עַוְלָֽה׃
בב כִּ֣י כֶ֭חָצִיר מְהֵרָ֣ה יִמָּ֑לוּ
וּכְיֶ֥רֶק דֶּ֝֗שֶׁא יִבּוֹלֽוּן׃
גג בְּטַ֣ח בַּ֭יהֹוָה וַעֲשֵׂה־ט֑וֹב
שְׁכׇן־אֶ֝֗רֶץנשְׁכׇן־אֶ֝֗רֶץ · א=שְׁכָן־אֶ֝רֶץ (השמטת נקודת הרביע) וּרְעֵ֥ה אֱמוּנָֽה׃
דד וְהִתְעַנַּ֥ג עַל־יְהֹוָ֑ה
וְיִֽתֶּן־לְ֝ךָ֗ מִשְׁאֲלֹ֥ת לִבֶּֽךָ׃
הה גּ֣וֹל עַל־יְהֹוָ֣ה דַּרְכֶּ֑ךָ
וּבְטַ֥ח עָ֝לָ֗יו וְה֣וּא יַעֲשֶֽׂה׃
וו וְהוֹצִ֣יא כָא֣וֹר צִדְקֶ֑ךָ
וּ֝מִשְׁפָּטֶ֗ךָ ׃
זז דּ֤וֹם ׀ לַיהֹוָה֮וְהִתְח֢וֹלֵֽ֫ל ל֥וֹ
אַל־תִּ֭תְחַר בְּמַצְלִ֣יחַ דַּרְכּ֑וֹ
בְּ֝אִ֗ישׁ עֹשֶׂ֥ה מְזִמּֽוֹת׃
חח הֶ֣רֶף מֵ֭אַף וַעֲזֹ֣ב חֵמָ֑ה
אַל־תִּ֝תְחַ֗ר אַךְ־לְהָרֵֽעַ׃
טט כִּֽי־מְ֭רֵעִים יִכָּרֵת֑וּן
וְקֹוֵ֥י יְ֝הֹוָ֗ה הֵ֣מָּה יִֽירְשׁוּ־אָֽרֶץ׃
יי וְע֣וֹד מְ֭עַט וְאֵ֣ין רָשָׁ֑ע
וְהִתְבּוֹנַ֖נְתָּ עַל־מְקוֹמ֣וֹ וְאֵינֶֽנּוּ׃
יאיא וַעֲנָוִ֥ים יִֽירְשׁוּ־אָ֑רֶץ
וְ֝הִֽתְעַנְּג֗וּ עַל־רֹ֥ב שָׁלֽוֹם׃
יביב זֹמֵ֣ם רָ֭שָׁע לַצַּדִּ֑יק
וְחֹרֵ֖ק עָלָ֣יו שִׁנָּֽיו׃
יגיג אֲדֹנָ֥י יִשְׂחַק־ל֑וֹ
כִּי־רָ֝אָ֗ה כִּֽי־יָבֹ֥א יוֹמֽוֹ׃
ידיד חֶ֤רֶב ׀ פָּ֥תְח֣וּנפָּ֥תְח֣וּ=א? (הקו נטוי קצת כמרכא אבל לא באופן מובהק) וכן הכריעו ברויאר ומג"הממ=פָּֽתְח֣וּ וכן הוא בכתי"ל ובדפוסים רְשָׁעִים֮וְדָרְכ֢וּ קַ֫שְׁתָּ֥ם
לְ֭הַפִּיל עָנִ֣י וְאֶבְי֑וֹן
לִ֝טְב֗וֹחַ יִשְׁרֵי־דָֽרֶךְ׃
טוטו חַ֭רְבָּם תָּב֣וֹא בְלִבָּ֑ם
וְ֝קַשְּׁתוֹתָ֗ם תִּשָּׁבַֽרְנָה׃
טזטז טוֹב־מְ֭עַט לַצַּדִּ֑יק
מֵ֝הֲמ֗וֹן רְשָׁעִ֥ים רַבִּֽים׃
יזיז כִּ֤י זְרוֹע֣וֹת רְ֭שָׁעִים תִּשָּׁבַ֑רְנָה
וְסוֹמֵ֖ךְנוְסוֹמֵ֖ךְא!=וְסוֹמֵ֖ך (חסר שווא בכ"ף סופית)הערת ייבין צַדִּיקִ֣ים יְהֹוָֽה׃
יחיח יוֹדֵ֣עַ יְ֭הֹוָה יְמֵ֣י תְמִימִ֑ם
וְ֝נַחֲלָתָ֗ם לְעוֹלָ֥ם תִּהְיֶֽה׃
יטיט לֹֽא־יֵ֭בֹשׁוּ בְּעֵ֣ת רָעָ֑ה
וּבִימֵ֖י רְעָב֣וֹן יִשְׂבָּֽעוּ׃
ככ כִּ֤י רְשָׁעִ֨ים ׀ יֹאבֵ֗דוּ
וְאֹיְבֵ֣י יְ֭הֹוָה כִּיקַ֣ר כָּרִ֑ים
כָּל֖וּ בֶעָשָׁ֣ן כָּֽלוּ׃
כאכא לֹוֶ֣ה רָ֭שָׁע וְלֹ֣א יְשַׁלֵּ֑ם
וְ֝צַדִּ֗יק חוֹנֵ֥ן וְנוֹתֵֽן׃
כבכב כִּ֣י מְ֭בֹרָכָיו יִ֣ירְשׁוּ אָ֑רֶץ
וּ֝מְקֻלָּלָ֗יו יִכָּרֵֽתוּ׃
כגכג מֵ֭יְהֹוָה מִֽצְעֲדֵי־גֶ֥בֶר כּוֹנָ֗נוּ
וְדַרְכּ֥וֹ יֶחְפָּֽץ׃
כדכד כִּֽי־יִפֹּ֥ל לֹֽא־יוּטָ֑ל
כִּֽי־יְ֝הֹוָ֗ה סוֹמֵ֥ךְ יָדֽוֹ׃
כהכה נַ֤עַר ׀ הָיִ֗יתִיגַּם־זָ֫קַ֥נְתִּי
וְֽלֹא־רָ֭אִיתִי צַדִּ֣יק נֶעֱזָ֑ב
וְ֝זַרְע֗וֹ מְבַקֶּשׁ־לָֽחֶם׃
כוכו כׇּל־הַ֭יּוֹם חוֹנֵ֣ן וּמַלְוֶ֑ה
וְ֝זַרְע֗וֹ לִבְרָכָֽה׃
כזכז ס֣וּר מֵ֭רָע וַעֲשֵׂה־ט֗וֹב
וּשְׁכֹ֥ן לְעוֹלָֽם׃
כחכח כִּ֤י יְהֹוָ֨ה ׀ אֹ֘הֵ֤ב מִשְׁפָּ֗ט
וְלֹא־יַעֲזֹ֣ב אֶת־חֲ֭סִידָיו לְעוֹלָ֣ם נִשְׁמָ֑רוּ
וְזֶ֖רַע רְשָׁעִ֣ים נִכְרָֽת׃
כטכט צַדִּיקִ֥ים יִֽירְשׁוּ־אָ֑רֶץ
וְיִשְׁכְּנ֖וּ לָעַ֣ד עָלֶֽיהָ׃
לל פִּֽי־צַ֭דִּיק יֶהְגֶּ֣ה חׇכְמָ֑ה
וּ֝לְשׁוֹנ֗וֹ תְּדַבֵּ֥ר מִשְׁפָּֽט׃
לאלא תּוֹרַ֣ת אֱלֹהָ֣יו בְּלִבּ֑וֹ
לֹ֖א תִמְעַ֣ד אֲשֻׁרָֽיו׃
לבלב צוֹפֶ֣ה רָ֭שָׁע לַצַּדִּ֑יק
וּ֝מְבַקֵּ֗שׁ לַהֲמִיתֽוֹ׃
לגלג יְ֭הֹוָה לֹא־יַעַזְבֶ֣נּוּ בְיָד֑וֹ
וְלֹ֥א יַ֝רְשִׁיעֶ֗נּוּ בְּהִשָּׁפְטֽוֹ׃
לדלד קַוֵּ֤ה אֶל־יְהֹוָ֨ה ׀ וּשְׁמֹ֬ר דַּרְכּ֗וֹ
וִֽ֭ירוֹמִמְךָ לָרֶ֣שֶׁת אָ֑רֶץ
בְּהִכָּרֵ֖ת רְשָׁעִ֣ים תִּרְאֶֽה׃
להלה רָ֭אִיתִי רָשָׁ֣ע עָרִ֑יץ
וּ֝מִתְעָרֶ֗ה כְּאֶזְרָ֥ח רַעֲנָֽן׃
לולו וַֽ֭יַּעֲבֹרנוַֽ֭יַּעֲבֹר=א,לממ=וַ֭יַּֽעֲבֹר וכן בדפוסים וְהִנֵּ֣ה אֵינֶ֑נּוּ
וָ֝אֲבַקְשֵׁ֗הוּ וְלֹ֣א נִמְצָֽא׃
לזלז שְׁמׇר־תָּ֭ם וּרְאֵ֣ה יָשָׁ֑ר
כִּֽי־אַחֲרִ֖ית לְאִ֣ישׁ שָׁלֽוֹם׃
לחלח וּֽ֭פֹשְׁעִים נִשְׁמְד֣וּ יַחְדָּ֑ו
אַחֲרִ֖ית רְשָׁעִ֣ים נִכְרָֽתָה׃
לטלט וּתְשׁוּעַ֣ת צַ֭דִּיקִים מֵיְהֹוָ֑ה
מָ֝עוּזָּ֗ם בְּעֵ֣ת צָרָֽה׃
ממ וַ֥יַּעְזְרֵ֥ם יְהֹוָ֗ה וַֽיְפַ֫לְּטֵ֥ם
יְפַלְּטֵ֣ם מֵ֭רְשָׁעִים וְיוֹשִׁיעֵ֑ם
כִּי־חָ֥סוּ בֽוֹ׃
