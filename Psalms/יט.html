פרק יט


{פ}　　
יטאא לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃
נפרשה פתוחה בב הַשָּׁמַ֗יִם מְֽסַפְּרִ֥ים כְּבֽוֹד־אֵ֑ל
וּֽמַעֲשֵׂ֥ה יָ֝דָ֗יו מַגִּ֥יד הָרָקִֽיעַ׃
גג י֣וֹם לְ֭יוֹם יַבִּ֣יעַֽ אֹ֑מֶר
וְלַ֥יְלָה לְּ֝לַ֗יְלָה יְחַוֶּה־דָּֽעַת׃
דד אֵֽין־אֹ֭מֶר וְאֵ֣ין דְּבָרִ֑ים
בְּ֝לִ֗י נִשְׁמָ֥ע קוֹלָֽם׃
הה בְּכׇל־הָאָ֨רֶץ ׀ יָ֘צָ֤א קַוָּ֗ם
וּבִקְצֵ֣ה תֵ֭בֵל מִלֵּיהֶ֑ם
לַ֝שֶּׁ֗מֶשׁשָֽׂם־אֹ֥הֶל בָּהֶֽם׃
וו וְה֗וּא כְּ֭חָתָן יֹצֵ֣א מֵחֻפָּת֑וֹ
יָשִׂ֥ישׂ כְּ֝גִבּ֗וֹר לָר֥וּץ אֹֽרַח׃
זז מִקְצֵ֤ה הַשָּׁמַ֨יִם ׀ מֽוֹצָא֗וֹ
וּתְקוּפָת֥וֹ עַל־קְצוֹתָ֑ם
וְאֵ֥ין נִ֝סְתָּ֗רמֵחַמָּתֽוֹנמֵחַמָּתֽוֹ=ש1ל!=מֵֽחַמָּתוֹ (מקום הסילוק, או געיה וחסר סילוק לאחריה)הערות ברויאר ודותן והמקליד׃
חח תּ֘וֹרַ֤ת יְהֹוָ֣ה תְּ֭מִימָה מְשִׁ֣יבַת נָ֑פֶשׁ
עֵד֥וּת יְהֹוָ֥ה נֶ֝אֱמָנָ֗ה מַחְכִּ֥ימַת פֶּֽתִי׃
טט פִּקּ֘וּדֵ֤י יְהֹוָ֣ה יְ֭שָׁרִים מְשַׂמְּחֵי־לֵ֑ב
מִצְוַ֥ת יְהֹוָ֥ה בָּ֝רָ֗ה מְאִירַ֥ת עֵינָֽיִם׃
יי יִרְאַ֤ת יְהֹוָ֨ה ׀ טְהוֹרָה֮ עוֹמֶ֢דֶת לָ֫עַ֥ד
מִֽשְׁפְּטֵי־יְהֹוָ֥ה אֱמֶ֑ת
צָֽדְק֥וּ יַחְדָּֽו׃
יאיא הַֽנֶּחֱמָדִ֗ים מִ֭זָּהָב וּמִפַּ֣ז רָ֑ב
וּמְתוּקִ֥ים מִ֝דְּבַ֗שׁ וְנֹ֣פֶת צוּפִֽים׃
יביב גַּֽם־עַ֭בְדְּךָ נִזְהָ֣ר בָּהֶ֑ם
בְּ֝שׇׁמְרָ֗ם עֵ֣קֶב רָֽב׃
יגיג שְׁגִיא֥וֹת מִֽי־יָבִ֑ין
מִֽנִּסְתָּר֥וֹת נַקֵּֽנִי׃
ידיד גַּ֤ם מִזֵּדִ֨ים ׀ חֲשֹׂ֬ךְ עַבְדֶּ֗ךָ
אַֽל־יִמְשְׁלוּ־בִ֣י אָ֣ז אֵיתָ֑ם
וְ֝נִקֵּ֗יתִימִפֶּ֥שַֽׁע רָֽב׃
טוטו יִ֥הְיֽוּ־לְרָצ֨וֹן ׀ ניִ֥הְיֽוּ־לְרָצ֨וֹן | =ש1 לגבי הטעמים (אך אין בו מקף); וכך קרא ברויאר את התיבה בכתי"ל ולפי כללי משרת של לגרמיה בתחום שלטון פזר (טעמי המקרא עמ' 265; והשוו לעיל יז, יד) וכך גם במג"ה ובמכון ממרא לגבי הטעמים.ל=<יִֽהְי֥וּ־לְרָצ֨וֹן ׀> (געיה ומרכא בתיבה המוקפת) וכן דעתו של המקליד ע"פ BHS; לפי דותן ל=<יִֽהְיֽוּ־לְרָצ֨וֹן ׀> (שתי געיות בתיבה המוקפת) ולפי ברויאר כאמור לעיל ל=<יִ֥הְיֽוּ־לְרָצ֨וֹן ׀>.קורן=יִהְי֥וּ (מרכא מלרע) בדומה למ"ש=יִֽהְי֥וּ, סימנים ע"פ ברויאר במהדורה הראשונה ושינו ל-יִ֣הְיֽוּ (?!) במהדורת התנ"ך השלם.אִמְרֵי־פִ֡י
וְהֶגְי֣וֹן לִבִּ֣י לְפָנֶ֑יךָ
יְ֝הֹוָ֗הצוּרִ֥י וְגֹאֲלִֽי׃
