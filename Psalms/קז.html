פרק קז

קזאא הֹד֣וּ לַיהֹוָ֣ה כִּי־ט֑וֹב
כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃
בב יֹ֭אמְרוּ גְּאוּלֵ֣י יְהֹוָ֑ה
אֲשֶׁ֥ר גְּ֝אָלָ֗ם מִיַּד־צָֽר׃
גג וּֽמֵאֲרָצ֗וֹתקִ֫בְּצָ֥ם
מִמִּזְרָ֥ח וּמִֽמַּעֲרָ֑ב
מִצָּפ֥וֹן וּמִיָּֽם׃
דד תָּע֣וּ בַ֭מִּדְבָּר בִּישִׁימ֣וֹן דָּ֑רֶךְ
עִ֥יר מ֝וֹשָׁ֗ב לֹ֣א מָצָֽאוּ׃
הה רְעֵבִ֥ים גַּם־צְמֵאִ֑ים
נַ֝פְשָׁ֗ם בָּהֶ֥ם תִּתְעַטָּֽףנתִּתְעַטָּֽף=ל,ש1 ובדפוסיםא!=תִתְעַטָּֽף (חסר דגש באות תי"ו הראשונה, וסימן הרפה תלוי בין שתי אותיות התי"ו)׃
וו וַיִּצְעֲק֣וּ אֶל־יְ֭הֹוָה בַּצַּ֣ר לָהֶ֑ם
מִ֝מְּצ֥וּקוֹתֵיהֶ֗ם יַצִּילֵֽם׃
זז וַֽ֭יַּדְרִיכֵם בְּדֶ֣רֶךְ יְשָׁרָ֑ה
לָ֝לֶ֗כֶת אֶל־עִ֥יר מוֹשָֽׁב׃
חח יוֹד֣וּ לַיהֹוָ֣ה חַסְדּ֑וֹ
וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃
טט כִּֽי־הִ֭שְׂבִּיעַ נֶ֣פֶשׁ שֹׁקֵקָ֑ה
וְנֶ֥פֶשׁ רְ֝עֵבָ֗ה מִלֵּא־טֽוֹב׃
יי יֹ֭שְׁבֵי חֹ֣שֶׁךְ וְצַלְמָ֑וֶת
אֲסִירֵ֖י עֳנִ֣י וּבַרְזֶֽל׃
יאיא כִּֽי־הִמְר֥וּ אִמְרֵי־אֵ֑ל
וַעֲצַ֖ת עֶלְי֣וֹן נָאָֽצוּ׃
יביב וַיַּכְנַ֣ע בֶּעָמָ֣ל לִבָּ֑ם
כָּ֝שְׁל֗וּ וְאֵ֣ין עֹזֵֽר׃
יגיג וַיִּזְעֲק֣וּ אֶל־יְ֭הֹוָה בַּצַּ֣ר לָהֶ֑ם
מִ֝מְּצֻ֥קוֹתֵיהֶ֗ם יוֹשִׁיעֵֽם׃
ידיד י֭וֹצִיאֵם מֵחֹ֣שֶׁךְ וְצַלְמָ֑וֶת
וּמוֹסְר֖וֹתֵיהֶ֣ם יְנַתֵּֽק׃
טוטו יוֹד֣וּ לַיהֹוָ֣ה חַסְדּ֑וֹ
וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃
טזטז כִּי־שִׁ֭בַּר דַּלְת֣וֹת נְחֹ֑שֶׁת
וּבְרִיחֵ֖י בַרְזֶ֣ל גִּדֵּֽעַ׃
יזיז אֱ֭וִלִים מִדֶּ֣רֶךְ פִּשְׁעָ֑ם
וּֽ֝מֵעֲוֺ֥נֹתֵיהֶ֗ם יִתְעַנּֽוּניִתְעַנּֽוּ=ל,ש1 וכמו כן בדפוסים וקורןא?!=יִתְעַנּֽו׃ (חסרה נקודת השורוק באות וי"ו?)׃

יטיט וַיִּזְעֲק֣וּ אֶל־יְ֭הֹוָה בַּצַּ֣ר לָהֶ֑ם
מִ֝מְּצֻ֥קוֹתֵיהֶ֗ם יוֹשִׁיעֵֽם׃
ככ יִשְׁלַ֣ח דְּ֭בָרוֹ וְיִרְפָּאֵ֑ם
וִ֝ימַלֵּ֗ט מִשְּׁחִֽיתוֹתָֽם׃
כאכא יוֹד֣וּ לַיהֹוָ֣ה חַסְדּ֑וֹ
וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃
כבכב וְ֭יִזְבְּחוּ זִבְחֵ֣י תוֹדָ֑ה
וִיסַפְּר֖וּ מַעֲשָׂ֣יו בְּרִנָּֽה׃
כגכג ׆יוֹרְדֵ֣י הַ֭יָּם
עֹשֵׂ֥י מְ֝לָאכָ֗ה בְּמַ֣יִם רַבִּֽים׃
כדכד ׆הֵ֣מָּה רָ֭אוּ מַעֲשֵׂ֣י יְהֹוָ֑ה
וְ֝נִפְלְאוֹתָ֗יו בִּמְצוּלָֽה׃
כהכה ׆וַיֹּ֗אמֶר וַֽ֭יַּעֲמֵד ר֣וּחַ סְעָרָ֑ה
וַתְּרוֹמֵ֥ם גַּלָּֽיו׃
כוכו ׆יַעֲל֣וּ שָׁ֭מַיִם יֵרְד֣וּ תְהוֹמ֑וֹת
נַ֝פְשָׁ֗ם בְּרָעָ֥ה תִתְמוֹגָֽג׃
כזכז ׆יָח֣וֹגּוּ וְ֭יָנוּעוּ כַּשִּׁכּ֑וֹר
וְכׇל־חׇ֝כְמָתָ֗ם תִּתְבַּלָּֽע׃
כחכח ׆וַיִּצְעֲק֣וּ אֶל־יְ֭הֹוָה בַּצַּ֣ר לָהֶ֑ם
וּֽ֝מִמְּצ֥וּקֹתֵיהֶ֗ם יוֹצִיאֵֽם׃
כטכט יָקֵ֣ם סְ֭עָרָה לִדְמָמָ֑ה

לאלא יוֹד֣וּ לַיהֹוָ֣ה חַסְדּ֑וֹ
וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃
