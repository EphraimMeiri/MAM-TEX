ספר תהלים

פרק קיט


{פ}　　
קיטאא אַשְׁרֵ֥י תְמִֽימֵי־דָ֑רֶךְ
הַ֝הֹלְכִ֗ים בְּתוֹרַ֥ת יְהֹוָֽה׃
בב אַ֭שְׁרֵי נֹצְרֵ֥י עֵדֹתָ֗יו
בְּכׇל־לֵ֥ב יִדְרְשֽׁוּהוּ׃
גג אַ֭ף לֹא־פָעֲל֣וּ עַוְלָ֑ה
בִּדְרָכָ֥יו הָלָֽכוּ׃
דד אַ֭תָּה צִוִּ֥יתָה פִקֻּדֶ֗יךָ
לִשְׁמֹ֥ר מְאֹֽד׃
הה אַ֭חֲלַי יִכֹּ֥נוּ דְרָכָ֗י
לִשְׁמֹ֥ר חֻקֶּֽיךָ׃
וו אָ֥ז לֹא־אֵב֑וֹשׁ
בְּ֝הַבִּיטִ֗י אֶל־כׇּל־מִצְוֺתֶֽיךָ׃
זז א֭וֹדְךָ בְּיֹ֣שֶׁר לֵבָ֑ב
בְּ֝לׇמְדִ֗י מִשְׁפְּטֵ֥י צִדְקֶֽךָ׃
חח אֶת־חֻקֶּ֥יךָ אֶשְׁמֹ֑ר
אַֽל־תַּעַזְבֵ֥נִי עַד־מְאֹֽד׃
{פ}　　
טט בַּמֶּ֣ה יְזַכֶּה־נַּ֭עַר אֶת־אׇרְח֑וֹ
לִ֝שְׁמֹ֗ר כִּדְבָרֶֽךָ׃
יי בְּכׇל־לִבִּ֥י דְרַשְׁתִּ֑יךָ
אַל־תַּ֝שְׁגֵּ֗נִי מִמִּצְוֺתֶֽיךָ׃
יאיא בְּ֭לִבִּי צָפַ֣נְתִּי אִמְרָתֶ֑ךָ
לְ֝מַ֗עַן לֹ֣א אֶחֱטָא־לָֽךְ׃
יביב בָּר֖וּךְ אַתָּ֥ה יְהֹוָ֗ה
לַמְּדֵ֥נִי חֻקֶּֽיךָ׃
יגיג בִּשְׂפָתַ֥י סִפַּ֑רְתִּי
כֹּ֝֗לנכֹּ֝֗ל · א=כֹּ֝ל (השמטת נקודת הרביע) מִשְׁפְּטֵי־פִֽיךָ׃
ידיד בְּדֶ֖רֶךְ עֵדְוֺתֶ֥יךָ שַּׂ֗שְׂתִּינשַּׂ֗שְׂתִּי=א?,ל,ש1,ב1 (שי"ן דגושה?); הדגש מטושטש בכתר. כך אצל ברויאר ומג"ה ומ"מ וסימנים; מ"ג=שַׂ֗שְׂתִּי (אין דגש בשי"ן).
כְּעַ֣ל כׇּל־הֽוֹן׃
טוטו בְּפִקּוּדֶ֥יךָ אָשִׂ֑יחָה
וְ֝אַבִּ֗יטָה אֹֽרְחֹתֶֽיךָ׃
טזטז בְּחֻקֹּתֶ֥יךָ אֶֽשְׁתַּעֲשָׁ֑ע
לֹ֖א אֶשְׁכַּ֣ח דְּבָרֶֽךָ׃
{פ}　　
יזיז גְּמֹ֖ל עַל־עַבְדְּךָ֥ אֶחְיֶ֗ה
וְאֶשְׁמְרָ֥ה דְבָרֶֽךָ׃
יחיח גַּל־עֵינַ֥י וְאַבִּ֑יטָה
נִ֝פְלָא֗וֹת מִתּוֹרָתֶֽךָ׃
יטיט גֵּ֣ר אָנֹכִ֣י בָאָ֑רֶץ
אַל־תַּסְתֵּ֥ר מִ֝מֶּ֗נִּי מִצְוֺתֶֽיךָ׃
ככ גָּרְסָ֣ה נַפְשִׁ֣י לְתַאֲבָ֑ה
אֶֽל־מִשְׁפָּטֶ֥יךָ בְכׇל־עֵֽת׃
כאכא גָּ֭עַרְתָּ זֵדִ֣ים אֲרוּרִ֑ים
הַ֝שֹּׁגִ֗ים מִמִּצְוֺתֶֽיךָ׃
כבכב גַּ֣ל מֵ֭עָלַי חֶרְפָּ֣ה וָב֑וּז
כִּ֖י עֵדֹתֶ֣יךָ נָצָֽרְתִּי׃
כגכג גַּ֤ם יָשְׁב֣וּ שָׂ֭רִים בִּ֣י נִדְבָּ֑רוּ
עַ֝בְדְּךָ֗ יָשִׂ֥יחַ בְּחֻקֶּֽיךָ׃
כדכד גַּֽם־עֵ֭דֹתֶיךָ שַׁעֲשֻׁעָ֗י
אַנְשֵׁ֥י עֲצָתִֽי׃
{פ}　　
כהכה דָּבְקָ֣ה לֶעָפָ֣ר נַפְשִׁ֑י
חַ֝יֵּ֗נִי כִּדְבָרֶֽךָ׃
כוכו דְּרָכַ֣י סִ֭פַּרְתִּי וַֽתַּעֲנֵ֗נִי
לַמְּדֵ֥נִי חֻקֶּֽיךָ׃
כזכז דֶּרֶךְ־פִּקּוּדֶ֥יךָ הֲבִינֵ֑נִי
וְ֝אָשִׂ֗יחָה בְּנִפְלְאוֹתֶֽיךָ׃
כחכח דָּלְפָ֣ה נַ֭פְשִׁי מִתּוּגָ֑ה
קַ֝יְּמֵ֗נִי כִּדְבָרֶֽךָ׃
כטכט דֶּֽרֶךְ־שֶׁ֭קֶר הָסֵ֣ר מִמֶּ֑נִּי
וְֽתוֹרָתְךָ֥ חׇנֵּֽנִי׃
לל דֶּֽרֶךְ־אֱמוּנָ֥ה בָחָ֑רְתִּי
מִשְׁפָּטֶ֥יךָ שִׁוִּֽיתִי׃
לאלא דָּבַ֥קְתִּי בְעֵדְוֺתֶ֑יךָ
יְ֝הֹוָ֗ה אַל־תְּבִישֵֽׁנִי׃
לבלב דֶּֽרֶךְ־מִצְוֺתֶ֥יךָ אָר֑וּץ
כִּ֖י תַרְחִ֣יב לִבִּֽי׃
{פ}　　
לגלג הוֹרֵ֣נִי יְ֭הֹוָה דֶּ֥רֶךְ חֻקֶּ֗יךָ
וְאֶצְּרֶ֥נָּה עֵֽקֶב׃
לדלד הֲ֭בִינֵנִי וְאֶצְּרָ֥ה תוֹרָתֶ֗ךָ
וְאֶשְׁמְרֶ֥נָּה בְכׇל־לֵֽב׃
להלה הַ֭דְרִיכֵנִי בִּנְתִ֣יב מִצְוֺתֶ֑יךָ
כִּי־ב֥וֹ חָפָֽצְתִּי׃
לולו הַט־לִ֭בִּי אֶל־עֵדְוֺתֶ֗יךָ
וְאַ֣ל אֶל־בָּֽצַע׃
לזלז הַעֲבֵ֣ר עֵ֭ינַי מֵרְא֣וֹת שָׁ֑וְא
בִּדְרָכֶ֥ךָ חַיֵּֽנִי׃
לחלח הָקֵ֣ם לְ֭עַבְדְּךָ אִמְרָתֶ֑ךָ
אֲ֝שֶׁ֗ר לְיִרְאָתֶֽךָ׃
לטלט הַעֲבֵ֣ר חֶ֭רְפָּתִי אֲשֶׁ֣ר יָגֹ֑רְתִּי
כִּ֖י מִשְׁפָּטֶ֣יךָ טוֹבִֽים׃
ממ הִ֭נֵּה תָּאַ֣בְתִּי לְפִקֻּדֶ֑יךָ
בְּצִדְקָתְךָ֥ חַיֵּֽנִי׃
{פ}　　
מאמא וִיבֹאֻ֣נִי חֲסָדֶ֣ךָ יְהֹוָ֑ה
תְּ֝שׁ֥וּעָתְךָ֗ כְּאִמְרָתֶֽךָ׃
מבמב וְאֶעֱנֶ֣ה חֹרְפִ֣י דָבָ֑ר
כִּֽי־בָ֝טַ֗חְתִּי בִּדְבָרֶֽךָ׃
מגמג וְֽאַל־תַּצֵּ֬ל מִפִּ֣י דְבַר־אֱמֶ֣ת עַד־מְאֹ֑ד
כִּ֖י לְמִשְׁפָּטֶ֣ךָ יִחָֽלְתִּי׃
מדמד וְאֶשְׁמְרָ֖ה תוֹרָתְךָ֥ תָמִ֗יד
לְעוֹלָ֥ם וָעֶֽד׃
מהמה וְאֶתְהַלְּכָ֥ה בָרְחָבָ֑ה
כִּ֖י פִקֻּדֶ֣יךָ דָרָֽשְׁתִּי׃
מומו וַאֲדַבְּרָ֣ה בְ֭עֵדֹתֶיךָ נֶ֥גֶד מְלָכִ֗ים
וְלֹ֣א אֵבֽוֹשׁ׃
מזמז וְאֶשְׁתַּעֲשַׁ֥ע בְּמִצְוֺתֶ֗יךָ
אֲשֶׁ֣ר אָהָֽבְתִּי׃
מחמח וְאֶשָּֽׂא־כַפַּ֗י אֶֽל־מִ֭צְוֺתֶיךָ אֲשֶׁ֥ר אָהָ֗בְתִּי
וְאָשִׂ֥יחָה בְחֻקֶּֽיךָ׃
{פ}　　
מטמט זְכֹר־דָּבָ֥ר לְעַבְדֶּ֑ךָ
עַ֝֗לנעַ֝֗ל · א=עַ֝ל (השמטת נקודת הרביע) אֲשֶׁ֣ר יִחַלְתָּֽנִי׃
ננ זֹ֣את נֶחָמָתִ֣י בְעׇנְיִ֑י
כִּ֖י אִמְרָתְךָ֣ חִיָּֽתְנִי׃
נאנא זֵ֭דִים הֱלִיצֻ֣נִי עַד־מְאֹ֑ד
מִ֝תּ֥וֹרָתְךָ֗ לֹ֣א נָטִֽיתִי׃
נבנב זָ֘כַ֤רְתִּי מִשְׁפָּטֶ֖יךָ מֵעוֹלָ֥ם ׀ יְהֹוָ֗ה
וָאֶתְנֶחָֽם׃
נגנג זַלְעָפָ֣ה אֲ֭חָזַתְנִי מֵרְשָׁעִ֑ים
עֹ֝זְבֵ֗י תּוֹרָתֶֽךָ׃
נדנד זְ֭מִרוֹת הָיוּ־לִ֥י חֻקֶּ֗יךָ
בְּבֵ֣ית מְגוּרָֽי׃
נהנה זָ֘כַ֤רְתִּי בַלַּ֣יְלָה שִׁמְךָ֣ יְהֹוָ֑ה
וָ֝אֶשְׁמְרָ֗ה תּוֹרָתֶֽךָ׃
נונו זֹ֥את הָיְתָה־לִּ֑י
כִּ֖י פִקֻּדֶ֣יךָ נָצָֽרְתִּי׃
{פ}　　
נזנז חֶלְקִ֖י יְהֹוָ֥ה אָמַ֗רְתִּי
לִשְׁמֹ֥ר דְּבָרֶֽיךָ׃
נחנח חִלִּ֣יתִי פָנֶ֣יךָ בְכׇל־לֵ֑ב
חׇ֝נֵּ֗נִי כְּאִמְרָתֶֽךָ׃
נטנט חִשַּׁ֥בְתִּי דְרָכָ֑י
וָאָשִׁ֥יבָה רַ֝גְלַ֗י אֶל־עֵדֹתֶֽיךָ׃
סס חַ֭שְׁתִּי וְלֹ֣א הִתְמַהְמָ֑הְתִּי
לִ֝שְׁמֹ֗ר מִצְוֺתֶֽיךָ׃
סאסא חֶבְלֵ֣י רְשָׁעִ֣ים עִוְּדֻ֑נִי
תּ֥֝וֹרָתְךָ֗ לֹ֣א שָׁכָֽחְתִּי׃
סבסב חֲצֽוֹת־לַ֗יְלָה אָ֭קוּם לְהוֹד֣וֹת לָ֑ךְ
עַ֝֗לנעַ֝֗ל · א=עַ֝ל (השמטת נקודת הרביע) מִשְׁפְּטֵ֥י צִדְקֶֽךָ׃
סגסג חָבֵ֣ר אָ֭נִי לְכׇל־אֲשֶׁ֣ר יְרֵא֑וּךָ
וּ֝לְשֹׁמְרֵ֗י פִּקּוּדֶֽיךָ׃
סדסד חַסְדְּךָ֣ יְ֭הֹוָה מָלְאָ֥ה הָאָ֗רֶץ
חֻקֶּ֥יךָ לַמְּדֵֽנִי׃
{פ}　　
סהסה ט֭וֹב עָשִׂ֣יתָ עִֽם־עַבְדְּךָ֑
יְ֝הֹוָ֗ה כִּדְבָרֶֽךָ׃
סוסו ט֤וּב טַ֣עַם וָדַ֣עַת לַמְּדֵ֑נִי
כִּ֖י בְמִצְוֺתֶ֣יךָ הֶאֱמָֽנְתִּי׃
סזסז טֶ֣רֶם אֶ֭עֱנֶה אֲנִ֣י שֹׁגֵ֑ג
וְ֝עַתָּ֗ה אִמְרָתְךָ֥ שָׁמָֽרְתִּי׃
סחסח טוֹב־אַתָּ֥ה וּמֵטִ֗יב
לַמְּדֵ֥נִי חֻקֶּֽיךָ׃
סטסט טָפְל֬וּ עָלַ֣י שֶׁ֣קֶר זֵדִ֑ים
אֲ֝נִ֗י בְּכׇל־לֵ֤ב ׀ אֶצֹּ֬ר פִּקּוּדֶֽיךָ׃
עע טָפַ֣שׁ כַּחֵ֣לֶב לִבָּ֑ם
אֲ֝נִ֗י תּוֹרָתְךָ֥ שִׁעֲשָֽׁעְתִּי׃
עאעא טֽוֹב־לִ֥י כִי־עֻנֵּ֑יתִי
לְ֝מַ֗עַן אֶלְמַ֥ד חֻקֶּֽיךָ׃
עבעב טֽוֹב־לִ֥י תוֹרַת־פִּ֑יךָ
מֵ֝אַלְפֵ֗י זָהָ֥ב וָכָֽסֶף׃
{פ}　　
עגעג יָדֶ֣יךָ עָ֭שׂוּנִי וַֽיְכוֹנְנ֑וּנִי
הֲ֝בִינֵ֗נִי וְאֶלְמְדָ֥ה מִצְוֺתֶֽיךָ׃
עדעד יְ֭רֵאֶיךָ יִרְא֣וּנִי וְיִשְׂמָ֑חוּ
כִּ֖י לִדְבָרְךָ֣ יִחָֽלְתִּי׃
עהעה יָדַ֣עְתִּי יְ֭הֹוָה כִּי־צֶ֣דֶק מִשְׁפָּטֶ֑יךָ
וֶ֝אֱמוּנָ֗ה עִנִּיתָֽנִי׃
עועו יְהִי־נָ֣א חַסְדְּךָ֣ לְנַחֲמֵ֑נִי
כְּאִמְרָתְךָ֥ לְעַבְדֶּֽךָ׃
עזעז יְבֹא֣וּנִי רַחֲמֶ֣יךָ וְאֶחְיֶ֑ה
כִּי־ת֥֝וֹרָתְךָ֗ שַׁעֲשֻׁעָֽי׃
עחעח יֵבֹ֣שׁוּ זֵ֭דִים כִּי־שֶׁ֣קֶר עִוְּת֑וּנִי
אֲ֝נִ֗י אָשִׂ֥יחַ בְּפִקּוּדֶֽיךָ׃

קנוקנו רַחֲמֶ֖יךָ רַבִּ֥ים ׀ יְהֹוָ֑ה
כְּֽמִשְׁפָּטֶ֥יךָ חַיֵּֽנִי׃
קנזקנז רַ֭בִּים רֹדְפַ֣י וְצָרָ֑י
מֵ֝עֵדְוֺתֶ֗יךָ לֹ֣א נָטִֽיתִי׃
קנחקנח רָאִ֣יתִי בֹ֭גְדִים וָאֶתְקוֹטָ֑טָה
אֲשֶׁ֥ר אִ֝מְרָתְךָ֗ לֹ֣א שָׁמָֽרוּ׃
קנטקנט רְ֭אֵה כִּי־פִקּוּדֶ֣יךָ אָהָ֑בְתִּי
יְ֝הֹוָ֗ה כְּֽחַסְדְּךָ֥ חַיֵּֽנִי׃
קסקס רֹאשׁ־דְּבָרְךָ֥ אֱמֶ֑ת
