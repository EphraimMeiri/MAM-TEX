כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃
בב מִ֗י יְ֭מַלֵּל גְּבוּר֣וֹת יְהֹוָ֑ה
יַ֝שְׁמִ֗יעַ כׇּל־תְּהִלָּתֽוֹ׃
גג אַ֭שְׁרֵי שֹׁמְרֵ֣י מִשְׁפָּ֑ט
עֹשֵׂ֖ה צְדָקָ֣ה בְכׇל־עֵֽת׃
דד זׇכְרֵ֣נִי יְ֭הֹוָה בִּרְצ֣וֹן עַמֶּ֑ךָ
פׇּ֝קְדֵ֗נִי בִּישׁוּעָתֶֽךָ׃
הה לִרְא֤וֹת ׀ בְּט֘וֹבַ֤ת בְּחִירֶ֗יךָ
לִ֭שְׂמֹחַ בְּשִׂמְחַ֣ת גּוֹיֶ֑ךָ
לְ֝הִתְהַלֵּ֗לעִם־נַחֲלָתֶֽךָ׃
וו חָטָ֥אנוּ עִם־אֲבוֹתֵ֗ינוּ
הֶעֱוִ֥ינוּ הִרְשָֽׁעְנוּ׃
זז אֲב֘וֹתֵ֤ינוּ בְמִצְרַ֨יִם ׀ לֹא־הִשְׂכִּ֬ילוּ נִפְלְאוֹתֶ֗יךָ
לֹ֣א זָ֭כְרוּ אֶת־רֹ֣ב חֲסָדֶ֑יךָ
וַיַּמְר֖וּ עַל־יָ֣ם בְּיַם־סֽוּף׃
חח וַֽ֭יּוֹשִׁיעֵם לְמַ֣עַן שְׁמ֑וֹ
לְ֝הוֹדִ֗יעַ אֶת־גְּבוּרָתֽוֹ׃
טט וַיִּגְעַ֣ר בְּיַם־ס֭וּף וַֽיֶּחֱרָ֑ב
וַיּוֹלִיכֵ֥ם בַּ֝תְּהֹמ֗וֹת כַּמִּדְבָּֽר׃
יי וַֽ֭יּוֹשִׁיעֵם מִיַּ֣ד שׂוֹנֵ֑א
וַ֝יִּגְאָלֵ֗ם מִיַּ֥ד אוֹיֵֽב׃
יאיא וַיְכַסּוּ־מַ֥יִם צָרֵיהֶ֑ם
אֶחָ֥ד מֵ֝הֶ֗ם לֹ֣א נוֹתָֽר׃
יביב וַיַּאֲמִ֥ינוּ בִדְבָרָ֑יו
יָ֝שִׁ֗ירוּ תְּהִלָּתֽוֹ׃
יגיג מִ֭הֲרוּ שָׁכְח֣וּ מַעֲשָׂ֑יו
לֹא־חִ֝כּ֗וּ לַעֲצָתֽוֹ׃
ידיד וַיִּתְאַוּ֣וּ תַ֭אֲוָה בַּמִּדְבָּ֑ר
וַיְנַסּוּ־אֵ֝֗לנוַיְנַסּוּ־אֵ֝֗ל · א=וַיְנַסּוּ־אֵ֝ל (השמטת נקודת הרביע) בִּישִׁימֽוֹן׃
טוטו וַיִּתֵּ֣ן לָ֭הֶם שֶׁאֱלָתָ֑ם
וַיְשַׁלַּ֖ח רָז֣וֹן בְּנַפְשָֽׁם׃
טזטז וַיְקַנְא֣וּ לְ֭מֹשֶׁה בַּֽמַּחֲנֶ֑ה
לְ֝אַהֲרֹ֗ן קְד֣וֹשׁ יְהֹוָֽה׃
יזיז תִּפְתַּח־אֶ֭רֶץ וַתִּבְלַ֣ע דָּתָ֑ן
וַ֝תְּכַ֗ס עַל־עֲדַ֥ת אֲבִירָֽם׃
יחיח וַתִּבְעַר־אֵ֥שׁ בַּעֲדָתָ֑ם
לֶ֝הָבָ֗ה תְּלַהֵ֥ט רְשָׁעִֽים׃
יטיט יַעֲשׂוּ־עֵ֥גֶל בְּחֹרֵ֑ב
וַ֝יִּשְׁתַּחֲו֗וּנוַ֝יִּשְׁתַּחֲו֗וּ=ל,ש1 ובדפוסים (וכך אצל ברויאר)א=וַ֝יִּשְׁתַּחֲוּ֗וּ (דגש בוי"ו עיצורית) לְמַסֵּכָֽה׃
ככ וַיָּמִ֥ירוּ אֶת־כְּבוֹדָ֑ם
בְּתַבְנִ֥ית שׁ֝֗וֹרנשׁ֝֗וֹר · א=שׁ֝וֹר (השמטת נקודת הרביע) אֹכֵ֥ל עֵֽשֶׂב׃
כאכא שָׁ֭כְחוּ אֵ֣ל מוֹשִׁיעָ֑ם
עֹשֶׂ֖ה גְדֹל֣וֹת בְּמִצְרָֽיִם׃
כבכב נִ֭פְלָאוֹת בְּאֶ֣רֶץ חָ֑ם
נ֝וֹרָא֗וֹת עַל־יַם־סֽוּף׃
כגכג וַיֹּ֗אמֶר לְֽהַשְׁמִ֫ידָ֥ם
לוּלֵ֡י מֹ֘שֶׁ֤ה בְחִיר֗וֹ
עָמַ֣ד בַּפֶּ֣רֶץ לְפָנָ֑יו
לְהָשִׁ֥יב חֲ֝מָת֗וֹ מֵהַשְׁחִֽית׃
כדכד וַֽ֭יִּמְאֲסוּ בְּאֶ֣רֶץ חֶמְדָּ֑ה
לֹא־הֶ֝אֱמִ֗ינוּ לִדְבָרֽוֹ׃
כהכה וַיֵּרָגְנ֥וּ
לֹ֥א שָׁ֝מְע֗וּ בְּק֣וֹל יְהֹוָֽה׃
כוכו וַיִּשָּׂ֣א יָד֣וֹ לָהֶ֑ם
לְהַפִּ֥יל א֝וֹתָ֗ם בַּמִּדְבָּֽר׃
כזכז וּלְהַפִּ֣יל זַ֭רְעָם בַּגּוֹיִ֑ם
וּ֝לְזָרוֹתָ֗ם בָּאֲרָצֽוֹת׃
כחכח וַ֭יִּצָּ֣מְדוּ לְבַ֣עַל פְּע֑וֹר
וַ֝יֹּאכְל֗וּ זִבְחֵ֥י מֵתִֽים׃
כטכט וַ֭יַּכְעִיסוּ בְּמַ֥עַלְלֵיהֶ֑ם
וַתִּפְרׇץ־בָּ֝֗םנוַתִּפְרׇץ־בָּ֝֗ם · א=וַתִּפְרׇץ־בָּ֝ם (השמטת נקודת הרביע) מַגֵּפָֽה׃
לל וַיַּעֲמֹ֣ד פִּֽ֭ינְחָס וַיְפַלֵּ֑ל
וַ֝תֵּעָצַ֗ר הַמַּגֵּפָֽה׃
לאלא וַתֵּחָ֣שֶׁב ל֭וֹ לִצְדָקָ֑ה
לְדֹ֥ר וָ֝דֹ֗ר עַד־עוֹלָֽם׃
לבלב וַ֭יַּקְצִיפוּ עַל־מֵ֥י מְרִיבָ֑ה
וַיֵּ֥רַע לְ֝מֹשֶׁ֗ה בַּעֲבוּרָֽם׃
לגלג כִּי־הִמְר֥וּ אֶת־רוּח֑וֹ
וַ֝יְבַטֵּ֗א בִּשְׂפָתָֽיו׃
לדלד לֹֽא־הִ֭שְׁמִידוּ אֶת־הָעַמִּ֑ים
אֲשֶׁ֤ר אָמַ֖ר יְהֹוָ֣ה לָהֶֽם׃
להלה וַיִּתְעָרְב֥וּ בַגּוֹיִ֑ם
וַֽ֝יִּלְמְד֗וּ מַעֲשֵׂיהֶֽם׃
לולו וַיַּעַבְד֥וּ אֶת־עֲצַבֵּיהֶ֑ם
וַיִּהְי֖וּ לָהֶ֣ם לְמוֹקֵֽשׁ׃
לזלז וַיִּזְבְּח֣וּ אֶת־בְּ֭נֵיהֶם וְאֶת־בְּנוֹתֵיהֶ֗ם
לַשֵּׁדִֽים׃
לחלח וַיִּ֥שְׁפְּכ֨וּ דָ֪ם נָקִ֡י
דַּם־בְּנֵ֘יהֶ֤ם וּֽבְנוֹתֵיהֶ֗ם
אֲשֶׁ֣ר זִ֭בְּחוּ לַעֲצַבֵּ֣י כְנָ֑עַן
וַתֶּחֱנַ֥ף הָ֝אָ֗רֶץ בַּדָּמִֽים׃
לטלט וַיִּטְמְא֥וּ בְמַעֲשֵׂיהֶ֑ם
וַ֝יִּזְנ֗וּ בְּמַ֥עַלְלֵיהֶֽם׃
ממ וַיִּחַר־אַ֣ף יְהֹוָ֣ה בְּעַמּ֑וֹ
וַ֝יְתָעֵ֗ב אֶת־נַחֲלָתֽוֹ׃
מאמא וַיִּתְּנֵ֥ם בְּיַד־גּוֹיִ֑ם
וַֽיִּמְשְׁל֥וּ בָ֝הֶ֗ם שֹׂנְאֵיהֶֽם׃
מבמב וַיִּלְחָצ֥וּם אוֹיְבֵיהֶ֑ם
וַ֝יִּכָּנְע֗וּ תַּ֣חַת יָדָֽם׃

בְּ֝שׇׁמְע֗וֹ אֶת־רִנָּתָֽם׃

לִ֝פְנֵ֗י כׇּל־שׁוֹבֵיהֶֽם׃
