פרק קה


{פ}　　
קהאא הוֹד֣וּ לַ֭יהֹוָה קִרְא֣וּ בִשְׁמ֑וֹ
הוֹדִ֥יעוּ בָ֝עַמִּ֗ים עֲלִילוֹתָֽיו׃
בב שִֽׁירוּ־ל֭וֹ זַמְּרוּ־ל֑וֹ
שִׂ֝֗יחוּנשִׂ֝֗יחוּ · א=שִׂ֝יחוּ (השמטת נקודת הרביע) בְּכׇל־נִפְלְאוֹתָֽיו׃
גג הִֽ֭תְהַלְﬞלוּנהִֽ֭תְהַלְﬞלוּ · א=הִֽ֭תְהַלֲלוּ (חטף) בְּשֵׁ֣ם קׇדְשׁ֑וֹ
יִ֝שְׂמַ֗ח לֵ֤ב ׀ מְבַקְשֵׁ֬י יְהֹוָֽה׃
דד דִּרְשׁ֣וּ יְהֹוָ֣ה וְעֻזּ֑וֹ
בַּקְּשׁ֖וּ פָנָ֣יו תָּמִֽיד׃
הה זִכְר֗וּ נִפְלְאוֹתָ֥יו אֲשֶׁר־עָשָׂ֑ה
מֹ֝פְתָ֗יו וּמִשְׁפְּטֵי־פִֽיו׃
וו זֶ֭רַע אַבְרָהָ֣ם עַבְדּ֑וֹ
בְּנֵ֖י יַעֲקֹ֣ב בְּחִירָֽיו׃
זז ה֭וּא יְהֹוָ֣ה אֱלֹהֵ֑ינוּ
בְּכׇל־הָ֝אָ֗רֶץ מִשְׁפָּטָֽיו׃
חח זָכַ֣ר לְעוֹלָ֣ם בְּרִית֑וֹ
דָּבָ֥ר צִ֝וָּ֗ה לְאֶ֣לֶף דּֽוֹר׃
טט אֲשֶׁ֣ר כָּ֭רַת אֶת־אַבְרָהָ֑ם
וּשְׁב֖וּעָת֣וֹ לְיִשְׂחָֽק׃
יי וַיַּעֲמִידֶ֣הָ לְיַעֲקֹ֣ב לְחֹ֑ק
לְ֝יִשְׂרָאֵ֗ל בְּרִ֣ית עוֹלָֽם׃
יאיא לֵאמֹ֗ר לְךָ֗ אֶתֵּ֥ן אֶת־אֶֽרֶץ־כְּנָ֑עַן
חֶ֝֗בֶלנחֶ֝֗בֶל · א=חֶ֝בֶל (השמטת נקודת הרביע) נַחֲלַתְכֶֽם׃
יביב בִּֽ֭הְיוֹתָם מְתֵ֣י מִסְפָּ֑ר
כִּ֝מְעַ֗ט וְגָרִ֥ים בָּֽהּ׃
יגיג וַ֭יִּֽתְהַלְּכוּ מִגּ֣וֹי אֶל־גּ֑וֹי
מִ֝מַּמְלָכָ֗ה אֶל־עַ֥ם אַחֵֽר׃
ידיד לֹא־הִנִּ֣יחַ אָדָ֣ם לְעׇשְׁקָ֑ם
וַיּ֖וֹכַח עֲלֵיהֶ֣ם מְלָכִֽים׃
טוטו אַֽל־תִּגְּע֥וּ בִמְשִׁיחָ֑י
וְ֝לִנְבִיאַ֗י אַל־תָּרֵֽעוּ׃
טזטז וַיִּקְרָ֣א רָ֭עָב עַל־הָאָ֑רֶץ
כׇּֽל־מַטֵּה־לֶ֥חֶם שָׁבָֽר׃
יזיז שָׁלַ֣ח לִפְנֵיהֶ֣ם אִ֑ישׁ
לְ֝עֶ֗בֶד נִמְכַּ֥ר יוֹסֵֽף׃
יחיח עִנּ֣וּ בַכֶּ֣בֶל (רגליו)קקקרי/כתיב · (רגליו) רַגְל֑וֹ
בַּ֝רְזֶ֗ל בָּ֣אָה נַפְשֽׁוֹ׃
יטיט עַד־עֵ֥ת בֹּא־דְבָר֑וֹ
אִמְרַ֖ת יְהֹוָ֣ה צְרָפָֽתְהוּ׃
ככ שָׁ֣לַח מֶ֭לֶךְ וַיַּתִּירֵ֑הוּ
מֹשֵׁ֥ל עַ֝מִּ֗ים וַֽיְפַתְּחֵֽהוּ׃
כאכא שָׂמ֣וֹ אָד֣וֹן לְבֵית֑וֹ
וּ֝מֹשֵׁ֗ל בְּכׇל־קִנְיָנֽוֹ׃
כבכב לֶאְסֹ֣ר שָׂרָ֣יו בְּנַפְשׁ֑וֹ
וּזְקֵנָ֥יו יְחַכֵּֽם׃
כגכג וַיָּבֹ֣א יִשְׂרָאֵ֣ל מִצְרָ֑יִם
וְ֝יַעֲקֹ֗ב גָּ֣ר בְּאֶֽרֶץ־חָֽם׃
כדכד וַיֶּ֣פֶר אֶת־עַמּ֣וֹ מְאֹ֑ד
וַ֝יַּעֲצִמֵ֗הוּ מִצָּרָֽיו׃
כהכה הָפַ֣ךְ לִ֭בָּם לִשְׂנֹ֣א עַמּ֑וֹ
לְ֝הִתְנַכֵּ֗ל בַּעֲבָדָֽיו׃
כוכו שָׁ֭לַח מֹשֶׁ֣ה עַבְדּ֑וֹ
אַ֝הֲרֹ֗ן אֲשֶׁ֣ר בָּחַר־בּֽוֹ׃
כזכז שָֽׂמוּ־בָ֭ם דִּבְרֵ֣י אֹתוֹתָ֑יו
וּ֝מֹפְתִ֗ים בְּאֶ֣רֶץ חָֽם׃
כחכח שָׁ֣לַֽח חֹ֭שֶׁךְ וַיַּחְשִׁ֑ךְ
וְלֹא־מָ֝ר֗וּ אֶת־(דבריו)קקקרי/כתיב · (דְּבָרֽוֹ) דבריו׃
כטכט הָפַ֣ךְ אֶת־מֵימֵיהֶ֣ם לְדָ֑ם
וַ֝יָּ֗מֶת אֶת־דְּגָתָֽם׃
לל שָׁרַ֣ץ אַרְצָ֣ם צְפַרְדְּעִ֑ים
בְּ֝חַדְרֵ֗י מַלְכֵיהֶֽם׃
לאלא אָ֭מַר וַיָּבֹ֣א עָרֹ֑ב
כִּ֝נִּ֗ים בְּכׇל־גְּבוּלָֽם׃
לבלב נָתַ֣ן גִּשְׁמֵיהֶ֣ם בָּרָ֑ד
אֵ֖שׁ לֶהָב֣וֹת בְּאַרְצָֽם׃
לגלג וַיַּ֣ךְ גַּ֭פְנָם וּתְאֵנָתָ֑ם
וַ֝יְשַׁבֵּ֗ר עֵ֣ץ גְּבוּלָֽם׃
לדלד אָ֭מַר וַיָּבֹ֣א אַרְבֶּ֑ה
וְ֝יֶ֗לֶק וְאֵ֣ין מִסְפָּֽר׃
להלה וַיֹּ֣אכַל כׇּל־עֵ֣שֶׂב בְּאַרְצָ֑ם
וַ֝יֹּ֗אכַל פְּרִ֣י אַדְמָתָֽם׃
לולו וַיַּ֣ךְ כׇּל־בְּכ֣וֹר בְּאַרְצָ֑ם
רֵ֝אשִׁ֗ית לְכׇל־אוֹנָֽם׃
לזלז וַֽ֭יּוֹצִיאֵם בְּכֶ֣סֶף וְזָהָ֑ב
וְאֵ֖ין בִּשְׁבָטָ֣יו כּוֹשֵֽׁל׃
לחלח שָׂמַ֣ח מִצְרַ֣יִם בְּצֵאתָ֑ם
כִּֽי־נָפַ֖ל פַּחְדָּ֣ם עֲלֵיהֶֽם׃
לטלט פָּרַ֣שׂ עָנָ֣ן לְמָסָ֑ךְ
וְ֝אֵ֗שׁ לְהָאִ֥יר לָֽיְלָה׃
ממ שָׁ֭אַל וַיָּבֵ֣א שְׂלָ֑וקקקרי/כתיב · שְׂלָ֑ו · שְׂלָ֑יו
וְלֶ֥חֶם שָׁ֝מַ֗יִם יַשְׂבִּיעֵֽם׃
מאמא פָּ֣תַח צ֭וּר וַיָּז֣וּבוּ מָ֑יִם
הָ֝לְכ֗וּ בַּצִּיּ֥וֹת נָהָֽר׃
מבמב כִּֽי־זָ֭כַר אֶת־דְּבַ֣ר קׇדְשׁ֑וֹ
אֶֽת־אַבְרָהָ֥ם עַבְדּֽוֹ׃
מגמג וַיּוֹצִ֣א עַמּ֣וֹ בְשָׂשׂ֑וֹן
בְּ֝רִנָּ֗ה אֶת־בְּחִירָֽיו׃
מדמד וַיִּתֵּ֣ן לָ֭הֶם אַרְצ֣וֹת גּוֹיִ֑ם
וַעֲמַ֖ל לְאֻמִּ֣ים יִירָֽשׁוּ׃
מהמה בַּעֲב֤וּר ׀ יִשְׁמְר֣וּ חֻ֭קָּיו
וְתוֹרֹתָ֥יו יִנְצֹ֗רוּ
