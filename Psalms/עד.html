עדאא מַשְׂכִּ֗יל לְאָ֫סָ֥ף
נפרשה פתוחהלָמָ֣ה אֱ֭לֹהִים זָנַ֣חְתָּ לָנֶ֑צַח
יֶעְשַׁ֥ן אַ֝פְּךָ֗ בְּצֹ֣אן מַרְעִיתֶֽךָ׃
בב זְכֹ֤ר עֲדָתְךָ֨ ׀ קָ֘נִ֤יתָ קֶּ֗דֶם
גָּ֭אַלְתָּ שֵׁ֣בֶט נַחֲלָתֶ֑ךָ
הַר־צִ֝יּ֗וֹןזֶ֤ה ׀ שָׁכַ֬נְתָּ בּֽוֹ׃
גג הָרִ֣ימָה פְ֭עָמֶיךָ לְמַשֻּׁא֣וֹת נֶ֑צַח
כׇּל־הֵרַ֖ע אוֹיֵ֣ב בַּקֹּֽדֶשׁ׃
דד שָׁאֲג֣וּ צֹ֭רְﬞרֶיךָנצֹ֭רְﬞרֶיךָ · א=צֹ֭רֲרֶיךָ (חטף) בְּקֶ֣רֶב מוֹעֲדֶ֑ךָ
שָׂ֖מוּ אוֹתֹתָ֣ם אֹתֽוֹת׃
הה יִ֭וָּדַע כְּמֵבִ֣יא לְמָ֑עְלָה
בִּסְﬞבׇךְ־עֵ֝֗ץנבִּסְﬞבׇךְ־עֵ֝֗ץ · א=בִּסֲבָךְ־עֵ֝ץ (חטף; השמטת נקודת הרביע) קַרְדֻּמּֽוֹת׃
וו (ועת)קקקרי/כתיב · (ועת) וְ֭עַתָּה פִּתּוּחֶ֣יהָ יָּ֑חַד
בְּכַשִּׁ֥יל וְ֝כֵילַפּ֗וֹת יַהֲלֹמֽוּן׃
זז שִׁלְח֣וּ בָ֭אֵשׁ מִקְדָּשֶׁ֑ךָ
לָ֝אָ֗רֶץ חִלְּל֥וּ מִֽשְׁכַּן־שְׁמֶֽךָ׃
חח אָמְר֣וּ בְ֭לִבָּם נִינָ֣ם יָ֑חַד
שָׂרְפ֖וּ כׇל־מוֹעֲדֵי־אֵ֣ל בָּאָֽרֶץ׃
טט אֽוֹתֹתֵ֗ינוּ לֹ֥א רָ֫אִ֥ינוּ
אֵֽין־ע֥וֹד נָבִ֑יא
וְלֹֽא־אִ֝תָּ֗נוּיֹדֵ֥עַ עַד־מָֽה׃
יי עַד־מָתַ֣י אֱ֭לֹהִים יְחָ֣רֶף צָ֑ר
יְנָ֘אֵ֤ץ אוֹיֵ֖ב שִׁמְךָ֣ לָנֶֽצַח׃
יאיא לָ֤מָּה תָשִׁ֣יב יָ֭דְךָ וִימִינֶ֑ךָ
מִקֶּ֖רֶב חֵויקְךָ֣קקקרי/כתיב · (חוקך) חֵיקְךָ֣ כַלֵּֽה׃
יביב וֵ֭אלֹהִים מַלְכִּ֣י מִקֶּ֑דֶם
פֹּעֵ֥ל יְ֝שׁוּע֗וֹת בְּקֶ֣רֶב הָאָֽרֶץ׃
יגיג אַתָּ֤ה פוֹרַ֣רְתָּ בְעׇזְּךָ֣ יָ֑ם
שִׁבַּ֥רְתָּ רָאשֵׁ֥י תַ֝נִּינִ֗ים עַל־הַמָּֽיִם׃
ידיד אַתָּ֣ה רִ֭צַּצְתָּ רָאשֵׁ֣י לִוְיָתָ֑ן
תִּתְּנֶ֥נּוּ מַ֝אֲכָ֗ל לְעָ֣ם לְצִיִּֽים׃
טוטו אַתָּ֣ה בָ֭קַעְתָּ מַעְיָ֣ן וָנָ֑חַל
אַתָּ֥ה ה֝וֹבַ֗שְׁתָּ נַהֲר֥וֹת אֵיתָֽן׃
טזטז לְךָ֣ י֭וֹם אַף־לְךָ֥ לָ֑יְלָה
אַתָּ֥ה הֲ֝כִינ֗וֹתָ מָא֥וֹר וָשָֽׁמֶשׁ׃
יזיז אַתָּ֣ה הִ֭צַּבְתָּ כׇּל־גְּבוּל֣וֹת אָ֑רֶץ
קַ֥יִץ וָ֝חֹ֗רֶף אַתָּ֥ה יְצַרְתָּֽם׃
יחיח זְכׇר־זֹ֗את א֭וֹיֵב חֵרֵ֣ף ׀ יְהֹוָ֑ה
וְעַ֥ם נָ֝בָ֗ל נִאֲצ֥וּ שְׁמֶֽךָ׃
יטיט אַל־תִּתֵּ֣ן לְ֭חַיַּת נֶ֣פֶשׁ תּוֹרֶ֑ךָ
חַיַּ֥ת עֲ֝נִיֶּ֗יךָ אַל־תִּשְׁכַּ֥ח לָנֶֽצַח׃
ככ הַבֵּ֥ט לַבְּרִ֑ית
כִּ֥י מָלְא֥וּ מַחֲשַׁכֵּי־אֶ֝֗רֶץנמַחֲשַׁכֵּי־אֶ֝֗רֶץ · א=מַחֲשַׁכֵּי־אֶ֝רֶץ (השמטת נקודת הרביע) נְא֣וֹת חָמָֽס׃
כאכא אַל־יָשֹׁ֣ב דַּ֣ךְ נִכְלָ֑ם
עָנִ֥י וְ֝אֶבְי֗וֹן יְֽהַלְל֥וּ שְׁמֶֽךָ׃
כבכב קוּמָ֣ה אֱ֭לֹהִים רִיבָ֣ה רִיבֶ֑ךָ
זְכֹ֥ר חֶרְפָּתְךָ֥ מִנִּי־נָ֝בָ֗ל כׇּל־הַיּֽוֹם׃
כגכג אַל־תִּ֭שְׁכַּח ק֣וֹל צֹרְﬞרֶ֑יךָנצֹרְﬞרֶ֑יךָ · א=צֹרֲרֶ֑יךָ (חטף)
