פרק נה


{פ}　　
נהאא לַמְנַצֵּ֥חַ בִּנְגִינֹ֗ת מַשְׂכִּ֥יל לְדָוִֽד׃
נפרשה פתוחה בב הַאֲזִ֣ינָה אֱ֭לֹהִים תְּפִלָּתִ֑י
וְאַל־תִּ֝תְעַלַּ֗ם מִתְּחִנָּתִֽי׃
גג הַקְשִׁ֣יבָה לִּ֣י וַעֲנֵ֑נִי
אָרִ֖יד בְּשִׂיחִ֣י וְאָהִֽימָה׃
דד מִקּ֤וֹל אוֹיֵ֗ב מִ֭פְּנֵי עָקַ֣ת רָשָׁ֑ע
כִּֽי־יָמִ֥יטוּ עָלַ֥י אָ֝֗וֶןנאָ֝֗וֶן · א=אָ֝וֶן (השמטת נקודת הרביע) וּבְאַ֥ף יִשְׂטְמֽוּנִי׃
הה לִ֭בִּי יָחִ֣יל בְּקִרְבִּ֑י
וְאֵימ֥וֹת מָ֝֗וֶתנמָ֝֗וֶת · א=מָ֝וֶת (השמטת נקודת הרביע) נָפְל֥וּ עָלָֽי׃
וו יִרְאָ֣ה וָ֭רַעַד יָ֣בֹא בִ֑י
וַ֝תְּכַסֵּ֗נִי פַּלָּצֽוּת׃
זז וָאֹמַ֗ר מִֽי־יִתֶּן־לִ֣י אֵ֭בֶר כַּיּוֹנָ֗ה
אָע֥וּפָה וְאֶשְׁכֹּֽנָה׃
חח הִ֭נֵּה אַרְחִ֣יק נְדֹ֑ד
אָלִ֖ין בַּמִּדְבָּ֣ר סֶֽלָה׃
טט אָחִ֣ישָׁה מִפְלָ֣ט לִ֑י
מֵר֖וּחַ סֹעָ֣ה מִסָּֽעַר׃
יי בַּלַּ֣ע אֲ֭דֹנָי פַּלַּ֣ג לְשׁוֹנָ֑ם
כִּֽי־רָאִ֨יתִי חָמָ֖ס וְרִ֣יב בָּעִֽיר׃
יאיא יוֹמָ֤ם וָלַ֗יְלָה יְסוֹבְﬞבֻ֥הָניְסוֹבְﬞבֻ֥הָ · א=יְסוֹבֲבֻ֥הָ (חטף) עַל־חוֹמֹתֶ֑יהָ
וְאָ֖וֶן וְעָמָ֣ל בְּקִרְבָּֽהּ׃
יביב הַוּ֥וֹת בְּקִרְבָּ֑הּ
וְֽלֹא־יָמִ֥ישׁ מֵ֝רְחֹבָ֗הּ תֹּ֣ךְ וּמִרְמָֽה׃
יגיג כִּ֤י לֹֽא־אוֹיֵ֥ב יְחָֽרְפֵ֗נִיוְאֶ֫שָּׂ֥א
לֹא־מְ֭שַׂנְאִי עָלַ֣י הִגְדִּ֑יל
וְאֶסָּתֵ֥ר מִמֶּֽנּוּ׃
ידיד וְאַתָּ֣ה אֱנ֣וֹשׁ כְּעֶרְכִּ֑י
אַ֝לּוּפִ֗י וּמְיֻדָּעִֽי׃
טוטו אֲשֶׁ֣ר יַ֭חְדָּו נַמְתִּ֣יק ס֑וֹד
בְּבֵ֥ית אֱ֝לֹהִ֗ים נְהַלֵּ֥ךְ בְּרָֽגֶשׁ׃
טזטז ישימותקקקרי/כתיב · ישימות · כןעָלֵ֗ימוֹ
יֵרְד֣וּ שְׁא֣וֹל חַיִּ֑ים
כִּֽי־רָע֖וֹת בִּמְגוּרָ֣ם בְּקִרְבָּֽם׃
יזיז אֲ֭נִי אֶל־אֱלֹהִ֣ים אֶקְרָ֑א
וַ֝יהֹוָ֗ה יוֹשִׁיעֵֽנִי׃
יחיח עֶ֤רֶב וָבֹ֣קֶר אָשִׂ֣יחָה וְאֶהֱמֶ֑ה
וַיִּשְׁמַ֥ע קוֹלִֽי׃
יטיט פָּ֘דָ֤ה בְשָׁל֣וֹם נַ֭פְשִׁי מִקְּרׇב־לִ֑י
כִּֽי־בְ֝רַבִּ֗ים הָי֥וּ עִמָּדִֽי׃
ככ יִשְׁמַ֤ע ׀ אֵ֨ל ׀ וְֽיַעֲנֵם֮וְיֹ֤שֵׁ֥ב קֶ֗דֶם סֶ֥֫לָהנסֶ֥֫לָה · א=סֶ֥לָה (המשטת סימן העולה)
אֲשֶׁ֤ר אֵ֣ין חֲלִיפ֣וֹת לָ֑מוֹ
וְלֹ֖א יָרְא֣וּ אֱלֹהִֽים׃
כאכא שָׁלַ֣ח יָ֭דָיו בִּשְׁלֹמָ֗יו
חִלֵּ֥ל בְּרִיתֽוֹ׃
כבכב חָלְק֤וּ ׀ מַחְמָאֹ֣ת פִּיו֮וּֽקְﬞרָב־לִ֫בּ֥וֹנוּֽקְﬞרָב־לִ֫בּ֥וֹ · א=וּֽקֲרָב־לִ֫בּ֥וֹ (חטף)
רַכּ֖וּ דְבָרָ֥יו מִשֶּׁ֗מֶן
וְהֵ֣מָּה פְתִחֽוֹת׃
כגכג הַשְׁלֵ֤ךְ עַל־יְהֹוָ֨ה ׀ יְהָבְךָ֮וְה֢וּא יְכַ֫לְכְּלֶ֥ךָ
לֹא־יִתֵּ֖ן לְעוֹלָ֥ם מ֗וֹט
לַצַּדִּֽיק׃
כדכד וְאַתָּ֤ה אֱלֹהִ֨ים ׀ תּוֹרִדֵ֬ם ׀ לִבְאֵ֬ר שַׁ֗חַת
אַנְשֵׁ֤י דָמִ֣ים וּ֭מִרְמָה לֹא־יֶחֱצ֣וּ יְמֵיהֶ֑ם
וַ֝אֲנִ֗י אֶבְטַח־בָּֽךְ׃
