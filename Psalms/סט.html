פרק סט


{פ}　　
סטאא לַמְנַצֵּ֬חַ ׀ עַֽל־שׁוֹשַׁנִּ֬ים לְדָוִֽד׃
נפרשה פתוחה בב הוֹשִׁיעֵ֥נִי אֱלֹהִ֑ים
כִּ֤י בָ֖אוּ מַ֣יִם עַד־נָֽפֶשׁ׃
גג טָבַ֤עְתִּי ׀ בִּיוֵ֣ן מְ֭צוּלָה וְאֵ֣ין
בָּ֥אתִי בְמַעֲמַקֵּי־מַ֝֗יִםנבְמַעֲמַקֵּי־מַ֝֗יִם · א=בְמַעֲמַקֵּי־מַ֝יִם (השמטת נקודת הרביע)
וְשִׁבֹּ֥לֶת שְׁטָפָֽתְנִי׃
דד יָגַ֣עְתִּי בְקׇרְאִי֮נִחַ֢ר גְּר֫וֹנִ֥י
כָּל֥וּ עֵינַ֑י
מְ֝יַחֵ֗ל לֵאלֹהָֽי׃
הה רַבּ֤וּ ׀ מִשַּׂעֲר֣וֹת רֹאשִׁי֮שֹׂנְאַ֢י חִ֫נָּ֥ם
עָצְמ֣וּ מַ֭צְמִיתַי אֹיְבַ֣י שֶׁ֑קֶר
אֲשֶׁ֥ר לֹֽא־גָ֝זַ֗לְתִּי אָ֣ז אָשִֽׁיב׃
וו אֱֽלֹהִ֗ים אַתָּ֣ה יָ֭דַעְתָּ לְאִוַּלְתִּ֑י
וְ֝אַשְׁמוֹתַ֗י מִמְּךָ֥ לֹֽא־נִכְחָֽדוּ׃
זז אַל־יֵ֘בֹ֤שׁוּ בִ֨י ׀ קֹוֶיךָ֮
אֲדֹנָ֥י יֱהֹוִ֗ה צְבָ֫א֥וֹת
אַל־יִכָּ֣לְמֽוּ בִ֣י מְבַקְשֶׁ֑יךָ
אֱ֝לֹהֵ֗י יִשְׂרָאֵֽל׃
חח כִּֽי־עָ֭לֶיךָ נָשָׂ֣אתִי חֶרְפָּ֑ה
כִּסְּתָ֖ה כְלִמָּ֣ה פָנָֽי׃
טט מ֭וּזָר הָיִ֣יתִי לְאֶחָ֑י
וְ֝נׇכְרִ֗י לִבְנֵ֥י אִמִּֽי׃
יי כִּֽי־קִנְאַ֣ת בֵּיתְךָ֣ אֲכָלָ֑תְנִי
וְחֶרְפּ֥וֹת ח֝וֹרְפֶ֗יךָ נָפְל֥וּ עָלָֽי׃
יאיא וָאֶבְכֶּ֣ה בַצּ֣וֹם נַפְשִׁ֑י
וַתְּהִ֖י לַחֲרָפ֣וֹת לִֽי׃
יביב וָאֶתְּנָ֣ה לְבוּשִׁ֣י שָׂ֑ק
וָאֱהִ֖י לָהֶ֣ם לְמָשָֽׁל׃
יגיג יָשִׂ֣יחוּ בִ֭י יֹ֣שְׁבֵי שָׁ֑עַר
וּ֝נְגִינ֗וֹת שׁוֹתֵ֥י שֵׁכָֽר׃
ידיד וַאֲנִ֤י תְפִלָּֽתִי־לְךָ֨ ׀ יְהֹוָ֡העֵ֤ת רָצ֗וֹן
אֱלֹהִ֥ים בְּרׇב־חַסְדֶּ֑ךָ
עֲ֝נֵ֗נִי בֶּאֱמֶ֥ת יִשְׁעֶֽךָ׃
טוטו הַצִּילֵ֣נִי מִ֭טִּיט וְאַל־אֶטְבָּ֑עָה
אִנָּצְלָ֥ה מִ֝שֹּׂנְאַ֗י וּמִמַּ֖עֲמַקֵּי מָֽיִם׃
טזטז אַל־תִּשְׁטְפֵ֤נִי ׀ שִׁבֹּ֣לֶת מַ֭יִם
וְאַל־תִּבְלָעֵ֣נִי מְצוּלָ֑ה
וְאַל־תֶּאְטַר־עָלַ֖י בְּאֵ֣ר פִּֽיהָ׃
יזיז עֲנֵ֣נִי יְ֭הֹוָה כִּי־ט֣וֹב חַסְדֶּ֑ךָ
כְּרֹ֥ב רַ֝חֲמֶ֗יךָ פְּנֵ֣ה אֵלָֽי׃
יחיח וְאַל־תַּסְתֵּ֣ר פָּ֭נֶיךָ מֵעַבְדֶּ֑ךָ
כִּי־צַר־לִ֝֗ינכִּי־צַר־לִ֝֗י · א=כִּי־צַר־לִ֝י (השמטת נקודת הרביע) מַהֵ֥ר עֲנֵֽנִי׃
יטיט קׇרְבָ֣ה אֶל־נַפְשִׁ֣י גְאָלָ֑הּ
לְמַ֖עַן אֹיְבַ֣י פְּדֵֽנִי׃
ככ אַתָּ֤ה יָדַ֗עְתָּ חֶרְפָּתִ֣י וּ֭בׇשְׁתִּי וּכְלִמָּתִ֑י
נֶ֝גְדְּךָ֗ כׇּל־צוֹרְﬞרָֽינכׇּל־צוֹרְﬞרָֽי · א=כׇּל־צוֹרֲרָֽי (חטף)׃
כאכא חֶרְפָּ֤ה ׀ שָׁ֥בְרָ֥ה לִבִּ֗יוָאָ֫נ֥וּשָׁה
וָאֲקַוֶּ֣ה לָנ֣וּד וָאַ֑יִן
וְ֝לַמְנַחֲמִ֗ים וְלֹ֣א מָצָֽאתִי׃
כבכב וַיִּתְּנ֣וּ בְּבָרוּתִ֣י רֹ֑אשׁ
וְ֝לִצְמָאִ֗י יַשְׁק֥וּנִי חֹֽמֶץ׃
כגכג יְהִי־שֻׁלְחָנָ֣ם לִפְנֵיהֶ֣ם לְפָ֑ח
וְלִשְׁלוֹמִ֥ים לְמוֹקֵֽשׁ׃
כדכד תֶּחְשַׁ֣כְנָה עֵ֭ינֵיהֶם מֵרְא֑וֹת
וּ֝מׇתְנֵיהֶ֗ם תָּמִ֥יד הַמְעַֽד׃
כהכה שְׁפׇךְ־עֲלֵיהֶ֥ם זַעְמֶ֑ךָ
וַחֲר֥וֹן אַ֝פְּךָ֗ יַשִּׂיגֵֽם׃
כוכו תְּהִי־טִירָתָ֥ם נְשַׁמָּ֑ה
אַל־יְהִ֥י יֹשֵֽׁב׃
כזכז כִּי־אַתָּ֣ה אֲשֶׁר־הִכִּ֣יתָ רָדָ֑פוּ
וְאֶל־מַכְא֖וֹב חֲלָלֶ֣יךָ יְסַפֵּֽרוּ׃
כחכח תְּֽנָה־עָ֭וֺן עַל־עֲוֺנָ֑ם
וְאַל־יָ֝בֹ֗אוּ בְּצִדְקָתֶֽךָ׃

וְעִ֥ם צַ֝דִּיקִ֗ים אַל־יִכָּתֵֽבוּ׃
לל וַ֭אֲנִי עָנִ֣י וְכוֹאֵ֑ב
יְשׁוּעָתְךָ֖ אֱלֹהִ֣ים תְּשַׂגְּבֵֽנִי׃

וַאֲגַדְּלֶ֥נּוּ בְתוֹדָֽה׃
לבלב וְתִיטַ֣ב לַ֭יהֹוָה מִשּׁ֥וֹר פָּ֗ר
מַקְרִ֥ן מַפְרִֽיס׃
לגלג רָא֣וּ עֲנָוִ֣ים יִשְׂמָ֑חוּ
דֹּרְשֵׁ֥י אֱ֝לֹהִ֗ים וִיחִ֥י לְבַבְכֶֽם׃
לדלד כִּי־שֹׁמֵ֣עַ אֶל־אֶבְיוֹנִ֣ים יְהֹוָ֑ה
וְאֶת־אֲ֝סִירָ֗יו לֹ֣א בָזָֽה׃

יַ֝מִּ֗ים וְֽכׇל־רֹמֵ֥שׂ בָּֽם׃
לולו כִּ֤י אֱלֹהִ֨ים ׀ י֘וֹשִׁ֤יעַ צִיּ֗וֹן
וְ֭יִבְנֶה עָרֵ֣י יְהוּדָ֑ה

וְאֹהֲבֵ֥י שְׁ֝מ֗וֹ יִשְׁכְּנוּ־בָֽהּ׃
