עבאא לִשְׁלֹמֹ֨ה ׀
נפרשה סתומהאֱֽלֹהִ֗יםמִ֭שְׁפָּטֶיךָ לְמֶ֣לֶךְ תֵּ֑ן
וְצִדְקָתְךָ֥ לְבֶן־מֶֽלֶךְ׃
בב יָדִ֣ין עַמְּךָ֣ בְצֶ֑דֶק
וַעֲנִיֶּ֥יךָ בְמִשְׁפָּֽט׃
גג יִשְׂא֤וּ הָרִ֓ים שָׁ֘ל֥וֹם לָעָ֑ם
וּ֝גְבָע֗וֹת בִּצְדָקָֽה׃
דד יִשְׁפֹּ֤ט ׀ עֲֽנִיֵּי־עָ֗ם
י֭וֹשִׁיעַ לִבְנֵ֣י אֶבְי֑וֹן
וִ֖ידַכֵּ֣א עוֹשֵֽׁק׃
הה יִירָא֥וּךָ עִם־שָׁ֑מֶשׁ
וְלִפְנֵ֥י יָ֝רֵ֗חַ דּ֣וֹר דּוֹרִֽים׃
וו יֵ֭רֵד כְּמָטָ֣ר עַל־גֵּ֑ז
כִּ֝רְבִיבִ֗ים זַרְזִ֥יף אָֽרֶץ׃
זז יִֽפְרַח־בְּיָמָ֥יו צַדִּ֑יק
וְרֹ֥ב שָׁ֝ל֗וֹם עַד־בְּלִ֥י יָרֵֽחַ׃
חח וְ֭יֵרְדְּ מִיָּ֣ם עַד־יָ֑ם
וּ֝מִנָּהָ֗ר עַד־אַפְסֵי־אָֽרֶץ׃
טט לְ֭פָנָיו יִכְרְע֣וּ צִיִּ֑ים
וְ֝אֹיְבָ֗יו עָפָ֥ר יְלַחֵֽכוּ׃
יי מַלְכֵ֬י תַרְשִׁ֣ישׁ וְ֭אִיִּים
מִנְחָ֣ה יָשִׁ֑יבוּ
מַלְכֵ֥י שְׁבָ֥א וּ֝סְבָ֗א
אֶשְׁכָּ֥ר יַקְרִֽיבוּ׃
יאיא וְיִשְׁתַּחֲווּ־ל֥וֹ כׇל־מְלָכִ֑ים
כׇּל־גּוֹיִ֥ם יַעַבְדֽוּהוּ׃
יביב כִּֽי־יַ֭צִּיל אֶבְי֣וֹן מְשַׁוֵּ֑עַ
וְ֝עָנִ֗י וְֽאֵין־עֹזֵ֥ר לֽוֹ׃
יגיג יָ֭חֹס עַל־דַּ֣ל וְאֶבְי֑וֹן
וְנַפְשׁ֖וֹת אֶבְיוֹנִ֣ים יוֹשִֽׁיעַ׃
ידיד מִתּ֣וֹךְ וּ֭מֵחָמָס יִגְאַ֣ל נַפְשָׁ֑ם
וְיֵיקַ֖ר דָּמָ֣ם בְּעֵינָֽיו׃
טוטו וִיחִ֗יוְיִתֶּן־לוֹ֮ מִזְּהַ֢ב שְׁ֫בָ֥א
וְיִתְפַּלֵּ֣ל בַּעֲד֣וֹ תָמִ֑יד
כׇּל־הַ֝יּ֗וֹם יְבָרְﬞכֶֽנְהוּניְבָרְﬞכֶֽנְהוּ · א=יְבָרֲכֶֽנְהוּ (חטף)׃
טזטז יְהִ֤י פִסַּת־בַּ֨ר ׀ בָּאָרֶץ֮בְּרֹ֢אשׁ הָ֫רִ֥ים
יִרְעַ֣שׁ כַּלְּבָנ֣וֹן פִּרְי֑וֹ
וְיָצִ֥יצוּ מֵ֝עִ֗יר כְּעֵ֣שֶׂב הָאָֽרֶץ׃
יזיז יְהִ֤י שְׁמ֨וֹ ׀ לְֽעוֹלָ֗ם
לִפְנֵי־שֶׁמֶשׁ֮ יִנּ֢יוֹןקקקרי/כתיב · (ינין) יִנּ֢וֹן שְׁ֫מ֥וֹ
וְיִתְבָּ֥רְכוּ ב֑וֹ
כׇּל־גּוֹיִ֥ם יְאַשְּׁרֽוּהוּ׃
יחיח בָּר֤וּךְ ׀ יְהֹוָ֣ה אֱ֭לֹהִים אֱלֹהֵ֣י יִשְׂרָאֵ֑ל
עֹשֵׂ֖ה נִפְלָא֣וֹת לְבַדּֽוֹ׃
יטיט וּבָר֤וּךְ ׀ שֵׁ֥ם כְּבוֹד֗וֹלְע֫וֹלָ֥ם
וְיִמָּלֵ֣א כְ֭בוֹדוֹ אֶת־כֹּ֥ל הָאָ֗רֶץ
אָ֘מֵ֥ן ׀ וְאָמֵֽן׃
ככ כׇּלּ֥וּנכׇּלּ֥וּהקמץ קטן; ראו בהערה לדעת מקרא כאן: <'כלו' עבר לנסתרים של בניין פועל (הסביל של הקל) של שורש 'כלי' הלמ"ד דגושה והכ"ף בקמץ קטן תמורת קיבוץ. ואפשר שניקדוהו בבניין פועל ולא בבניין קל. לפי שבבניין קל יש לו משמעות של אבדון, כמו: כלו בעשן כלו (לעיל לז כ) ובבניין הסביל יש לו משמעות של סיום היצירה, כמו ויכלו השמים (בר' ב א).> והשוו תהלים פ,יא; משלי כד,יא, וראו שם בהערות לדעת מקרא. תְפִלּ֑וֹת
דָּ֝וִ֗ד בֶּן־יִשָֽׁי׃
